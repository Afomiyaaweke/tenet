   187→function fmtMoney(amount: number, currency: string): string {
   188→  if (!amount) return '—';
   189→  try {
   190→    return new Intl.NumberFormat('en-US', {
   191→      style: 'currency',
   192→      currency: currency || 'USD',
   193→      maximumFractionDigits: 0,
   194→    }).format(amount);
   195→  } catch {
   196→    return `${currency} ${amount.toLocaleString()}`;
   197→  }
   198→}
   199→
   200→function daysUntil(dateStr: string): number {
   201→  return Math.ceil((new Date(dateStr).getTime() - Date.now()) / (1000 * 60 * 60 * 24));
   202→}
   203→
   204→function deadlineBadge(days: number): string {
   205→  if (days <= 0) return 'bg-rose-50 text-rose-700 dark:bg-rose-950/40 dark:text-rose-300';
   206→  if (days <= 7) return 'bg-amber-50 text-amber-700 dark:bg-amber-950/40 dark:text-amber-300';
   207→  if (days <= 21) return 'bg-sky-50 text-sky-700 dark:bg-sky-950/40 dark:text-sky-300';
   208→  return 'bg-emerald-50 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300';
   209→}
   210→
   211→function relativeTime(dateStr: string): string {
   212→  const diff = Date.now() - new Date(dateStr).getTime();
   213→  const day = 86400000;
   214→  if (diff < day) return 'today';
   215→  if (diff < 2 * day) return 'yesterday';
   216→  if (diff < 30 * day) return `${Math.floor(diff / day)}d ago`;
   217→  return new Date(dateStr).toLocaleDateString();
   218→}

   220→/* ─────────────────────────────────────────────────────────────────────
   221→ * Inline Document Viewer Component
   222→ * ───────────────────────────────────────────────────────────────────── */
   223→
   224→function InlineDocumentViewer({ doc, onClose }: { doc: InlineDocument; onClose: () => void }) {
   225→  const scrollRef = useRef<HTMLDivElement>(null);
   226→
   227→  const copyContent = () => {
   228→    const text = doc.sections
   229→      ? doc.sections.map((s) => `${s.heading}\n${s.content}`).join('\n\n')
   230→      : doc.content;
   231→    navigator.clipboard.writeText(text);
   232→    toast.success('Content copied to clipboard');
   233→  };
   234→
   235→  return (
   236→    <motion.div
   237→      initial={{ opacity: 0, height: 0 }}
   238→      animate={{ opacity: 1, height: 'auto' }}
   239→      exit={{ opacity: 0, height: 0 }}
   240→      transition={{ duration: 0.3, ease: 'easeInOut' }}
   241→      className="overflow-hidden"
   242→    >
   243→      <div className="border-t border-border bg-gradient-to-b from-muted/30 to-background">
   244→        <div className="p-4 md:p-6 space-y-4">
   245→          {/* Document header */}
   246→          <div className="flex items-start justify-between gap-3">
   247→            <div className="min-w-0 flex-1">
   248→              <div className="flex items-center gap-2 text-xs text-muted-foreground mb-1">
   249→                <BookOpen className="h-3.5 w-3.5" />
   250→                <span>Document loaded from {new URL(doc.url).hostname}</span>
   251→                <span>·</span>
   252→                <span>{new Date(doc.fetchedAt).toLocaleTimeString()}</span>
   253→              </div>
   254→              <h3 className="text-lg font-semibold text-foreground leading-snug">
   255→                {doc.title || 'Tender Document'}
   256→              </h3>
   257→              {doc.metaDescription && (
   258→                <p className="text-sm text-muted-foreground mt-1 line-clamp-2">
   259→                  {doc.metaDescription}
   260→                </p>
   261→              )}
   262→            </div>
   263→            <div className="flex items-center gap-2 shrink-0">
   264→              <Button variant="ghost" size="sm" className="gap-1.5 text-xs" onClick={copyContent}>
   265→                <Copy className="h-3.5 w-3.5" />
   266→                Copy
   267→              </Button>
   268→              <a
   269→                href={doc.url}
   270→                target="_blank"
   271→                rel="noopener noreferrer"
   272→                className="inline-flex items-center gap-1.5 text-xs font-medium text-primary hover:underline"
   273→              >
   274→                <ExternalLink className="h-3.5 w-3.5" />
   275→                Open original
   276→              </a>
   277→              <Button variant="ghost" size="sm" onClick={onClose} className="gap-1.5 text-xs text-muted-foreground">
   278→                <ChevronUp className="h-3.5 w-3.5" />
   279→                Collapse
   280→              </Button>
   281→            </div>
   282→          </div>
   283→
   284→          {/* Extracted metadata pills */}
   285→          {(doc.deadlines && doc.deadlines.length > 0) || (doc.budgets && doc.budgets.length > 0) ? (
   286→            <div className="flex flex-wrap gap-2">
   287→              {doc.deadlines?.map((d, i) => (
   288→                <Badge key={`dl-${i}`} variant="outline" className="gap-1.5 text-xs border-amber-300 text-amber-700 dark:border-amber-800 dark:text-amber-300">
   289→                  <Calendar className="h-3 w-3" />
   290→                  {d}
   291→                </Badge>
   292→              ))}
   293→              {doc.budgets?.map((b, i) => (
   294→                <Badge key={`bg-${i}`} variant="outline" className="gap-1.5 text-xs border-emerald-300 text-emerald-700 dark:border-emerald-800 dark:text-emerald-300">
   295→                  <DollarSign className="h-3 w-3" />
   296→                  {b}
   297→                </Badge>
   298→              ))}
   299→            </div>
   300→          ) : null}
   301→
   302→          {/* Sections view */}
   303→          {doc.sections && doc.sections.length > 0 ? (
   304→            <div ref={scrollRef} className="max-h-80 overflow-y-auto rounded-lg border border-border bg-background p-4 space-y-4 scrollbar-thin">
   305→              {doc.sections.map((section, i) => (
   306→                <div key={i}>
   307→                  <h4 className="text-sm font-semibold text-foreground mb-1.5">{section.heading}</h4>
   308→                  <p className="text-sm text-muted-foreground leading-relaxed whitespace-pre-line">
   309→                    {section.content}
   310→                  </p>
   311→                  {i < doc.sections!.length - 1 && <Separator className="mt-4 bg-border" />}
   312→                </div>
   313→              ))}
   314→            </div>
   315→          ) : (
   316→            <div ref={scrollRef} className="max-h-80 overflow-y-auto rounded-lg border border-border bg-background p-4 scrollbar-thin">
   317→              <p className="text-sm text-muted-foreground leading-relaxed whitespace-pre-line">
   318→                {doc.content || 'No content could be extracted from this page.'}
   319→              </p>
   320→            </div>
   321→          )}
   322→        </div>
   323→      </div>
   324→    </motion.div>
   325→  );
   326→}

   328→/* ─────────────────────────────────────────────────────────────────────
   329→ * Main LiveTendersView component — Notion-style

   331→
   332→export function LiveTendersView() {
   333→  const [tenders, setTenders] = useState<LiveTender[]>([]);
   334→  const [dataSources, setDataSources] = useState<DataSource[]>([]);
   335→  const [sourceMeta, setSourceMeta] = useState<
   336→    { id: string; name: string; live: boolean; ok: boolean; count: number }[]
   337→  >([]);
   338→  const [loading, setLoading] = useState(true);
   339→  const [refreshing, setRefreshing] = useState(false);
   340→  const [fallback, setFallback] = useState(false);
   341→  const [search, setSearch] = useState('');
   342→  const [sourceFilter, setSourceFilter] = useState('all');
   343→  const [sectorFilter, setSectorFilter] = useState('');
   344→  const [sectorCounts, setSectorCounts] = useState<{ id: string; label: string; count: number }[]>([]);
   345→
   346→  // Inline document state
   347→  const [expandedId, setExpandedId] = useState<string | null>(null);
   348→  const [docLoading, setDocLoading] = useState<string | null>(null);
   349→  const [docData, setDocData] = useState<Record<string, InlineDocument>>({});
   350→  const [docError, setDocError] = useState<Record<string, string>>({});
   351→
   352→  // Import state
   353→  const [importing, setImporting] = useState<string | null>(null);
   354→
   355→  const load = useCallback(
   356→    async (isRefresh = false) => {
   357→      if (isRefresh) setRefreshing(true);
   358→      else setLoading(true);
   359→      const params: Record<string, string> = { rows: '50' };
   360→      if (search) params.search = search;
   361→      if (sourceFilter && sourceFilter !== 'all') params.source = sourceFilter;
   362→      if (sectorFilter) params.sector = sectorFilter;
   363→      const res = await api.get('/tenders/live', params);
   364→      if (res.success) {
   365→        setTenders(res.data as LiveTender[]);
   366→        setSourceMeta(res.meta?.sources || []);
   367→        setFallback(Boolean(res.meta?.fallback));
   368→        if (Array.isArray(res.meta?.dataSources)) setDataSources(res.meta.dataSources);
   369→        if (Array.isArray(res.meta?.sectors)) setSectorCounts(res.meta.sectors);
   370→      } else {
   371→        toast.error(res.error || 'Failed to load live tenders');
   372→      }
   373→      setLoading(false);
   374→      setRefreshing(false);
   375→    },
   376→    [search, sourceFilter, sectorFilter],
   377→  );
   378→
   379→  useEffect(() => {
   380→    const t = setTimeout(() => load(), 250);
   381→    return () => clearTimeout(t);
   382→  }, [load]);
   383→
   384→  const liveSourcesCount = useMemo(
   385→    () => sourceMeta.filter((s) => s.live && s.ok).length,
   386→    [sourceMeta],
   387→  );
   388→
   389→  const totalTenderValue = useMemo(() => {
   390→    return tenders.reduce((sum, t) => sum + (t.budgetMax || 0), 0);
   391→  }, [tenders]);
   392→
   393→  /* ────── Inline document loading ────── */
   394→  const loadDocument = useCallback(async (tender: LiveTender) => {
   395→    const id = tender.id;
   396→    if (expandedId === id) {
   397→      setExpandedId(null);
   398→      return;
   399→    }
   400→
   401→    // Already fetched?
   402→    if (docData[id]) {
   403→      setExpandedId(id);
   404→      return;
   405→    }
   406→
   407→    setExpandedId(id);
   408→    setDocLoading(id);
   409→    setDocError((prev) => { const n = { ...prev }; delete n[id]; return n; });
   410→
   411→    try {
   412→      const res = await api.get(`/tenders/${encodeURIComponent(id)}/documents`, { url: tender.externalUrl });
   413→      if (res.success && res.data) {
   414→        setDocData((prev) => ({ ...prev, [id]: res.data as InlineDocument }));
   415→      } else {
   416→        setDocError((prev) => ({ ...prev, [id]: res.error || 'Failed to load document' }));
   417→      }
   418→    } catch {
   419→      setDocError((prev) => ({ ...prev, [id]: 'Network error' }));
   420→    }
   421→    setDocLoading(null);
   422→  }, [expandedId, docData]);
   423→
   424→  /* ────── Import to local tenders ────── */
   425→  const importTender = useCallback(async (tender: LiveTender) => {

   429→        title: tender.title,
   430→        scope: tender.scope,
   431→        budgetMin: tender.budgetMin,
   432→        budgetMax: tender.budgetMax,
   433→        deadline: tender.deadline,
   434→        location: tender.location,
   435→        categoryTags: tender.categoryTags,
   436→        requiredDocs: `Source: ${SOURCE_LABELS[tender.source] || tender.source} | External ID: ${tender.externalId} | URL: ${tender.externalUrl}`,
   437→        status: 'open',
   438→        currency: tender.currency,
   439→      });
   440→      if (res.success) {
   441→        toast.success('Tender imported successfully', {
   442→          description: `"${tender.title}" is now in your tenders list.`,
   443→        });
   444→      } else {
   445→        toast.error(res.error || 'Failed to import tender');
   446→      }
   447→    } catch {
   448→      toast.error('Failed to import tender');
   449→    }
   450→    setImporting(null);
   451→  }, []);
   452→
   453→  /* ────── Render ────── */
   454→  return (
   455→    <div className="view-enter">
   456→      {/* ───────────────── Notion-style Cover ───────────────── */}
   457→      <div className="relative h-32 md:h-40 bg-gradient-to-br from-emerald-600 via-teal-700 to-cyan-800 overflow-hidden">
   458→        <div className="absolute inset-0 bg-[url('data:image/svg+xml;base64,PHN2ZyB3aWR0aD0iNDAiIGhlaWdodD0iNDAiIHZpZXdCb3g9IjAgMCA0MCA0MCIgeG1sbnM9Imh0dHA6Ly93d3cudzMub3JnLzIwMDAvc3ZnIj48Y2lyY2xlIGN4PSIyMCIgY3k9IjIwIiByPSIxIiBmaWxsPSJyZ2JhKDI1NSwyNTUsMjU1LDAuMDUpIi8+PC9zdmc+')] opacity-60" />
   459→        <div className="absolute bottom-0 left-0 right-0 h-16 bg-gradient-to-t from-background to-transparent" />
   460→        <div className="absolute top-4 right-4 md:right-6">
   461→          <Button
   462→            variant="secondary"
   463→            size="sm"
   464→            onClick={() => load(true)}
   465→            disabled={refreshing}
   466→            className="gap-1.5 shadow-lg bg-white/90 dark:bg-gray-800/90 backdrop-blur-sm"
   467→          >
   468→            <RefreshCw className={`h-3.5 w-3.5 ${refreshing ? 'animate-spin' : ''}`} />
   469→            {refreshing ? 'Refreshing…' : 'Refresh'}
   470→          </Button>
   471→        </div>
   472→      </div>
   473→
   474→      <div className="px-4 md:px-6 max-w-5xl mx-auto -mt-8 relative z-10 space-y-6">
   475→        {/* ───────────────── Notion-style Icon + Title ───────────────── */}
   476→        <div className="flex items-end gap-4">
   477→          <div className="h-14 w-14 md:h-16 md:w-16 rounded-xl bg-gradient-to-br from-emerald-500 to-teal-600 flex items-center justify-center shadow-lg shadow-emerald-500/20 border-4 border-background shrink-0">
