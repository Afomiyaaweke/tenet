   135→function getInitials(name: string | null | undefined, email: string): string {
   136→  if (name) {
   137→    const parts = name.split(' ').filter(Boolean);
   138→    return parts.length >= 2 ? (parts[0][0] + parts[parts.length - 1][0]).toUpperCase() : name[0].toUpperCase();
   139→  }
   140→  return email[0].toUpperCase();
   141→}
   142→
   143→function formatDate(d: string | null): string {
   144→  if (!d) return '—';
   145→  return new Date(d).toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
   146→}
   147→
   148→function parsePerms(str: string): Set<string> {
   149→  return new Set(str.split(',').map((s) => s.trim()).filter(Boolean));
   150→}
   151→
   152→function permsToString(set: Set<string>): string {
   153→  return Array.from(set).join(',');
   154→}

   322→function MembersTab({
   323→  members,
   324→  activeMembers,
   325→  onRefresh,
   326→  currentUserId,
   327→}: {
   328→  members: TeamMember[];
   329→  activeMembers: TeamMember[];
   330→  onRefresh: () => Promise<void>;
   331→  currentUserId: string;
   332→}) {
   333→  const [showAdd, setShowAdd] = useState(false);
   334→  const [searchQ, setSearchQ] = useState('');
   335→  const [editingMember, setEditingMember] = useState<TeamMember | null>(null);
   336→  const [editRole, setEditRole] = useState('');
   337→
   338→  const filtered = activeMembers.filter((m) => {
   339→    if (!searchQ) return true;
   340→    const name = m.user.profile?.fullName || '';
   341→    return name.toLowerCase().includes(searchQ.toLowerCase()) || m.user.email.toLowerCase().includes(searchQ.toLowerCase());
   342→  });
   343→
   344→  const handleRemove = async (id: string) => {
   345→    if (!confirm('Remove this team member?')) return;
   346→    const res = await api.delete(`/team/members/${id}`);
   347→    if (res.success) onRefresh();
   348→  };
   349→
   350→  const handleRoleChange = async () => {
   351→    if (!editingMember) return;
   352→    const res = await api.patch(`/team/members/${editingMember.id}`, { role: editRole });
   353→    if (res.success) {
   354→      setEditingMember(null);
   355→      onRefresh();
   356→    }
   357→  };
   358→
   359→  return (
   360→    <div className="space-y-4">
   361→      {/* ── Actions ── */}
   362→      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
   363→        <div className="relative flex-1 max-w-sm">
   364→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   365→          <Input
   366→            placeholder="Search members..."
   367→            value={searchQ}
   368→            onChange={(e) => setSearchQ(e.target.value)}
   369→            className="pl-9"
   370→          />
   371→        </div>
   372→        <Button onClick={() => setShowAdd(true)} className="gap-1.5">
   373→          <UserPlus className="h-4 w-4" /> Add Member
   374→        </Button>
   375→      </div>
   376→
   377→      {/* ── Member List ── */}
   378→      <div className="grid gap-3">
   379→        {filtered.length === 0 ? (
   380→          <Card className="border-dashed">
   381→            <CardContent className="py-12 text-center">
   382→              <Users className="h-10 w-10 mx-auto text-muted-foreground/40 mb-3" />
   383→              <p className="text-sm font-medium text-muted-foreground">No team members found</p>
   384→              <p className="text-xs text-muted-foreground/60 mt-1">Add members to get started</p>
   385→            </CardContent>
   386→          </Card>
   387→        ) : (
   388→          filtered.map((member) => (
   389→            <Card key={member.id} className="border border-border/50 hover:border-border transition-colors">
   390→              <CardContent className="p-4 flex items-center gap-4">
   391→                {/* Avatar */}
   392→                <div className="w-10 h-10 rounded-full bg-primary/10 flex items-center justify-center flex-shrink-0">
   393→                  <span className="text-sm font-bold text-primary">
   394→                    {getInitials(member.user.profile?.fullName, member.user.email)}
   395→                  </span>
   396→                </div>
   397→
   398→                {/* Info */}
   399→                <div className="flex-1 min-w-0">
   400→                  <div className="flex items-center gap-2">
   401→                    <p className="text-sm font-semibold truncate text-foreground">
   402→                      {member.user.profile?.fullName || member.user.email}
   403→                    </p>
   404→                    {member.userId === currentUserId && (
   405→                      <Badge variant="secondary" className="text-[9px] px-1.5 py-0 bg-primary/10 text-primary border-0">You</Badge>
   406→                    )}
   407→                  </div>
   408→                  <p className="text-xs text-muted-foreground truncate">{member.user.email}</p>
   409→                  {member.user.profile?.jobTitle && (
   410→                    <p className="text-[11px] text-muted-foreground/70 truncate">{member.user.profile.jobTitle}</p>
   411→                  )}
   412→                </div>
   413→
   414→                {/* Role Badge */}
   415→                <Badge className={`text-[10px] px-2 py-0.5 font-semibold border-0 ${ROLE_COLORS[member.role] || ROLE_COLORS.member}`}>
   416→                  {member.role === 'owner' && <Crown className="h-3 w-3 mr-1" />}
   417→                  {ROLE_LABELS[member.role] || member.role}
   418→                </Badge>
   419→
   420→                {/* Joined Date */}
   421→                <span className="text-[11px] text-muted-foreground/60 hidden lg:block">
   422→                  Joined {formatDate(member.joinedAt)}
   423→                </span>
   424→
   425→                {/* Actions */}
   426→                {member.role !== 'owner' && (
   427→                  <div className="flex items-center gap-1">
   428→                    <Button
   429→                      variant="ghost"
   430→                      size="icon"
   431→                      className="h-8 w-8 hover:bg-primary/10"
   432→                      onClick={() => {
   433→                        setEditingMember(member);
   434→                        setEditRole(member.role);
   435→                      }}
   436→                    >
   437→                      <Edit3 className="h-3.5 w-3.5 text-muted-foreground" />
   438→                    </Button>
   439→                    <Button
   440→                      variant="ghost"
   441→                      size="icon"
   442→                      className="h-8 w-8 hover:bg-red-50 dark:hover:bg-red-950/30"
   443→                      onClick={() => handleRemove(member.id)}
   444→                    >
   445→                      <Trash2 className="h-3.5 w-3.5 text-red-500" />
   446→                    </Button>
   447→                  </div>
   448→                )}
   449→              </CardContent>
   450→            </Card>
   451→          ))
   452→        )}
   453→      </div>
   454→
   455→      {/* ── Add Member Dialog ── */}
   456→      <AddMemberDialog open={showAdd} onOpenChange={setShowAdd} onAdded={onRefresh} />
   457→
   458→      {/* ── Edit Role Dialog ── */}
   459→      <Dialog open={!!editingMember} onOpenChange={(open) => !open && setEditingMember(null)}>
   460→        <DialogContent className="sm:max-w-md">
   461→          <DialogHeader>
   462→            <DialogTitle>Change Role</DialogTitle>
   463→          </DialogHeader>
   464→          <div className="space-y-4 py-2">
   465→            <p className="text-sm text-muted-foreground">
   466→              Change role for <span className="font-semibold text-foreground">{editingMember?.user.profile?.fullName || editingMember?.user.email}</span>
   467→            </p>
   468→            <Select value={editRole} onValueChange={setEditRole}>
   469→              <SelectTrigger>
   470→                <SelectValue placeholder="Select role" />
   471→              </SelectTrigger>
   472→              <SelectContent>
   473→                {ROLES.filter((r) => r !== 'owner').map((role) => (
   474→                  <SelectItem key={role} value={role}>
   475→                    <span className="flex items-center gap-2">
   476→                      {ROLE_LABELS[role]}
   477→                      <span className="text-[10px] text-muted-foreground">({role})</span>
   478→                    </span>
   479→                  </SelectItem>
   480→                ))}
   481→              </SelectContent>
   482→            </Select>
   483→          </div>
   484→          <DialogFooter>
   485→            <Button variant="outline" onClick={() => setEditingMember(null)}>Cancel</Button>
   486→            <Button onClick={handleRoleChange}>Save</Button>
   487→          </DialogFooter>
   488→        </DialogContent>
   489→      </Dialog>
   490→    </div>
   491→  );
   492→}
