   195→function relativeTime(dateStr: string): string {
   196→  const now = Date.now();
   197→  const then = new Date(dateStr).getTime();
   198→  const diff = Math.max(0, now - then);
   199→  const seconds = Math.floor(diff / 1000);
   200→  if (seconds < 60) return 'just now';
   201→  const minutes = Math.floor(seconds / 60);
   202→  if (minutes < 60) return `${minutes}m ago`;
   203→  const hours = Math.floor(minutes / 60);
   204→  if (hours < 24) return `${hours}h ago`;
   205→  const days = Math.floor(hours / 24);
   206→  if (days < 7) return `${days}d ago`;
   207→  return new Date(dateStr).toLocaleDateString();
   208→}

   370→/* ──────────────────── NotificationDropdown ──────────────────── */
   371→
   372→function NotificationDropdown({
   373→  notifications,
   374→  onMarkAsRead,
   375→  onMarkAllRead,
   376→}: {
   377→  notifications: { id: string; title: string; message: string; type: string; read: boolean; createdAt: string }[];
   378→  onMarkAsRead: (id: string) => void;
   379→  onMarkAllRead: () => void;
   380→}) {
   381→  const [open, setOpen] = useState(false);
   382→  const unread = useMemo(() => notifications.filter((n) => !n.read), [notifications]);
   383→  const displayItems = useMemo(() => unread.slice(0, 6), [unread]);
   384→
   385→  return (
   386→    <Popover open={open} onOpenChange={setOpen}>
   387→      <PopoverTrigger asChild>
   388→        <Button variant="ghost" size="icon" className="relative h-9 w-9 hover:bg-muted/80 transition-colors">
   389→          <Bell className="h-[18px] w-[18px] text-muted-foreground" />
   390→          {unread.length > 0 && (
   391→            <span className="absolute -top-0.5 -right-0.5 min-w-[18px] h-[18px] px-1 bg-red-500 text-white text-[10px] font-bold rounded-full flex items-center justify-center shadow-sm shadow-red-200 animate-in zoom-in-50 duration-200">
   392→              {unread.length > 99 ? '99+' : unread.length}
   393→            </span>
   394→          )}
   395→        </Button>
   396→      </PopoverTrigger>
   397→      <PopoverContent
   398→        className="w-80 p-0 glass-card rounded-xl overflow-hidden animate-in slide-in-from-top-2 fade-in-0 duration-200"
   399→        align="end"
   400→        sideOffset={8}
   401→      >
   402→        {/* Header */}
   403→        <div className="flex items-center justify-between px-4 py-3 border-b border-border/50">
   404→          <div className="flex items-center gap-2">
   405→            <h4 className="text-sm font-bold text-foreground">Notifications</h4>
   406→            {unread.length > 0 && (
   407→              <Badge className="text-[10px] px-1.5 py-0 bg-red-500 text-white font-bold border-0">
   408→                {unread.length}
   409→              </Badge>
   410→            )}
   411→          </div>
   412→          {unread.length > 0 && (
   413→            <Button
   414→              variant="ghost"
   415→              size="sm"
   416→              className="text-[11px] h-7 px-2 text-primary hover:text-primary/80 hover:bg-primary/10 font-semibold"
   417→              onClick={onMarkAllRead}
   418→            >
   419→              <Check className="h-3 w-3 mr-1" /> Mark all read
   420→            </Button>
   421→          )}
   422→        </div>
   423→
   424→        {/* Items */}
   425→        <ScrollArea className="max-h-[340px]">
   426→          {displayItems.length === 0 ? (
   427→            <div className="py-8 text-center">
   428→              <div className="w-12 h-12 mx-auto mb-3 rounded-full bg-orange-50 dark:bg-orange-950/30 flex items-center justify-center">
   429→                <CheckCircle className="h-6 w-6 text-orange-400" />
   430→              </div>
   431→              <p className="text-sm font-medium text-muted-foreground">All caught up!</p>
   432→              <p className="text-xs text-muted-foreground/60 mt-0.5">No new notifications</p>
   433→            </div>
   434→          ) : (
   435→            <div className="divide-y divide-border/40">
   436→              {displayItems.map((n, i) => {
   437→                const Icon = NOTIFICATION_ICONS[n.type] || Info;
   438→                const colorClass = NOTIFICATION_COLORS[n.type] || 'text-gray-500';
   439→                const bgClass = NOTIFICATION_BG[n.type] || 'bg-gray-50';
   440→                return (
   441→                  <button
   442→                    key={n.id}
   443→                    onClick={() => {
   444→                      onMarkAsRead(n.id);
   445→                      setOpen(false);
   446→                    }}
   447→                    className="w-full flex items-start gap-3 px-4 py-3 hover:bg-muted/50 transition-colors text-left group"
   448→                    style={{ animationDelay: `${i * 40}ms` }}
   449→                  >
   450→                    <div
   451→                      className={`w-8 h-8 rounded-lg ${bgClass} flex items-center justify-center flex-shrink-0 mt-0.5 transition-transform group-hover:scale-110`}
   452→                    >
   453→                      <Icon className={`h-4 w-4 ${colorClass}`} />
   454→                    </div>
   455→                    <div className="min-w-0 flex-1">
   456→                      <p className="text-[13px] font-semibold truncate text-foreground">{n.title}</p>
   457→                      <p className="text-[11px] text-muted-foreground truncate mt-0.5 leading-relaxed">{n.message}</p>
   458→                      <p className="text-[10px] text-muted-foreground/50 mt-1 font-medium">{relativeTime(n.createdAt)}</p>
   459→                    </div>
   460→                    <div className="w-2 h-2 rounded-full bg-orange-500 flex-shrink-0 mt-2 ring-2 ring-orange-100 dark:ring-orange-900/40" />
   461→                  </button>
   462→                );
   463→              })}
   464→            </div>
   465→          )}
   466→        </ScrollArea>
   467→
   468→        {/* Footer */}
   469→        {notifications.length > 6 && (
   470→          <div className="px-3 py-2 border-t border-border/40">
   471→            <Button
   472→              variant="ghost"
   473→              size="sm"
   474→              className="w-full text-[11px] text-muted-foreground hover:text-foreground font-medium"
   475→              onClick={() => setOpen(false)}
   476→            >
   477→              View all notifications
   478→            </Button>
   479→          </div>
   480→        )}
   481→      </PopoverContent>
   482→    </Popover>
   483→  );
   484→}

   487→
   488→export function AppShell() {
   489→  const { user, company, logout } = useAuthStore();
   490→  const { view, viewParams, setView } = useNavStore();
   491→  const { fetchNotifications, notifications } = useDataStore();
   492→  const role = user?.role || 'user';
   493→  const navSections = useMemo(() => getNavItemsForRole(role), [role]);
   494→  const unreadCount = notifications.filter((n) => !n.read).length;
   495→
   496→  // Flat list for title lookup
   497→  const allNavItems = useMemo(
   498→    () => navSections.flatMap((s) => s.items),
   499→    [navSections]
   500→  );
   501→
   502→  useEffect(() => {
   503→    fetchNotifications();
   504→  }, [fetchNotifications]);
   505→
   506→  const handleMarkAsRead = async (id: string) => {
   507→    await api.patch(`/notifications/${id}`, { read: true });
   508→    fetchNotifications();
   509→  };
   510→
   511→  const handleMarkAllRead = async () => {
   512→    const unread = notifications.filter((n) => !n.read);
   513→    await Promise.all(unread.map((n) => api.patch(`/notifications/${n.id}`, { read: true })));
   514→    fetchNotifications();
   515→  };
   516→
   517→  const renderView = () => {
   518→    switch (view) {
   519→      case 'dashboard':
   520→        return <DashboardView />;
   521→      case 'tenders':
   522→        return <TendersView />;
   523→      case 'live-tenders':
   524→        return <LiveTendersView />;
   525→      case 'tender-detail':
   526→        return <TenderDetailView tenderId={viewParams.id} />;
   527→      case 'tender-compare':
   528→        return <TenderCompareView tenderIds={viewParams.ids} />;
   529→      case 'bid-compare':
   530→        return <BidCompareView tenderId={viewParams.tenderId} />;
   531→      case 'bid-analysis':
   532→        return <BidsView />;
   533→      case 'bids':
   534→        return <BidsView />;
   535→      case 'projects':
   536→        return <ProjectsView />;
   537→      case 'project-detail':
   538→        return <ProjectDetailView projectId={viewParams.id} />;
   539→      case 'chat':
   540→        return <ChatView chatId={viewParams.id} />;
   541→      case 'finance':
   542→        return <ProjectDetailView projectId={viewParams.id} />;
   543→      case 'events':
   544→        return <EventsView />;
   545→      case 'profile':
   546→        return <ProfileView />;
   547→      case 'documents':
   548→        return <DocumentsView />;
   549→      case 'admin':
   550→        return <AdminView />;
   551→      case 'agent':
   552→        return <AgentView />;
   553→      case 'staff':
   554→        return <StaffView />;
   555→      case 'contact-us':
   556→        return <ContactUsView />;
   557→      case 'privacy-policy':
   558→        return <PrivacyPolicyView />;
   559→      case 'audit':
   560→        return <AuditView />;
   561→      default:
   562→        return <DashboardView />;
   563→    }
   564→  };
   565→
   566→  const pageTitle = allNavItems.find((i) => i.id === view)?.label
   567→    || (view === 'tender-compare' ? 'Compare Tenders' : view === 'bid-compare' ? 'Compare Bids' : view === 'bid-analysis' ? 'Bid Analysis' : 'Dashboard');
   568→  const breadcrumb = view === 'tender-detail' || view === 'bid-compare' || view === 'bid-analysis' ? 'Tenders' : view === 'tender-compare' ? 'Tenders' : view === 'project-detail' ? 'Projects' : null;
   569→
   570→  const badgeConfig = ROLE_BADGE_CONFIG[role] || ROLE_BADGE_CONFIG.user;
   571→  const companyName = company?.name || user?.company?.name;
   572→
   573→  const sidebarProps = { user, role, company, navSections, view, setView, unreadCount, logout };
   574→
   575→  return (
   576→    <div className="min-h-screen flex bg-background">
   577→      {/* ── Desktop Sidebar ── */}
   578→      <aside className="hidden md:flex w-[260px] bg-card border-r border-border/50 flex-col flex-shrink-0 sidebar-shadow">
   579→        <SidebarContent {...sidebarProps} />
   580→      </aside>
   581→
   582→      {/* ── Main Area ── */}
   583→      <div className="flex-1 flex flex-col min-w-0">
   584→        {/* ── Top Bar ── */}
   585→        <header className="h-14 bg-background/80 backdrop-blur-md border-b border-border/50 flex items-center px-4 gap-3 flex-shrink-0 sticky top-0 z-30">
   586→          {/* Mobile hamburger */}
   587→          <Sheet>
   588→            <SheetTrigger asChild>
   589→              <Button variant="ghost" size="icon" className="md:hidden h-9 w-9 hover:bg-muted/80">
   590→                <Menu className="h-5 w-5 text-muted-foreground" />
   591→              </Button>
   592→            </SheetTrigger>
   593→            <SheetContent side="left" className="p-0 w-[260px]">
   594→              <SidebarContent {...sidebarProps} />
   595→            </SheetContent>
   596→          </Sheet>
   597→
   598→          {/* Page Title & Breadcrumb */}
   599→          <div className="flex-1 min-w-0 flex items-center gap-1.5">
   600→            {breadcrumb && (
   601→              <>
   602→                <span className="text-sm text-muted-foreground/60 font-medium hidden sm:inline">
   603→                  {breadcrumb}
   604→                </span>
   605→                <ChevronRight className="h-3 w-3 text-muted-foreground/40 hidden sm:inline" />
   606→              </>
   607→            )}
   608→            <h1 className="text-base font-bold truncate text-foreground">{pageTitle}</h1>
   609→          </div>
   610→
   611→          {/* Right Actions */}
   612→          <div className="flex items-center gap-1">
   613→            {/* Search */}
   614→            <Button variant="ghost" size="icon" className="h-9 w-9 hover:bg-muted/80 transition-colors">
   615→              <Search className="h-[18px] w-[18px] text-muted-foreground" />
   616→            </Button>
   617→
   618→            {/* Theme Toggle */}
   619→            <ThemeToggle />
   620→
   621→            {/* Notifications */}
   622→            <NotificationDropdown
   623→              notifications={notifications}
   624→              onMarkAsRead={handleMarkAsRead}
   625→              onMarkAllRead={handleMarkAllRead}
   626→            />
   627→
   628→            {/* User Avatar Dropdown */}
   629→            <DropdownMenu>
   630→              <DropdownMenuTrigger asChild>
   631→                <Button variant="ghost" size="icon" className="h-9 w-9 hover:bg-muted/80 transition-colors ml-0.5">
   632→                  <div className="w-7 h-7 rounded-full bg-slate-700 flex items-center justify-center shadow-sm shadow-slate-200">
   633→                    <span className="text-white font-bold text-[11px]">{getUserInitial(user)}</span>
   634→                  </div>
   635→                </Button>
   636→              </DropdownMenuTrigger>
   637→              <DropdownMenuContent align="end" className="w-56">
   638→                <DropdownMenuLabel className="font-normal">
   639→                  <div className="flex flex-col space-y-1">
   640→                    <div className="flex items-center gap-2">
   641→                      <p className="text-sm font-medium leading-none">{user?.profile?.fullName || user?.email || 'User'}</p>
   642→                      <Badge variant="secondary" className={`text-[9px] px-1.5 py-0 font-semibold border-0 ${badgeConfig.className}`}>
   643→                        {badgeConfig.label}
   644→                      </Badge>
   645→                    </div>
   646→                    <p className="text-xs leading-none text-muted-foreground">{user?.email}</p>
   647→                    {companyName && (
   648→                      <p className="text-[11px] leading-none text-muted-foreground/70 flex items-center gap-1">
   649→                        <Building2 className="w-3 h-3" /> {companyName}
   650→                      </p>
   651→                    )}
   652→                  </div>
   653→                </DropdownMenuLabel>
   654→                <DropdownMenuSeparator />
   655→                <DropdownMenuItem onClick={() => setView('profile' as View)} className="cursor-pointer">
   656→                  <User className="mr-2 h-4 w-4" />
   657→                  <span>Profile</span>
   658→                </DropdownMenuItem>
   659→                <DropdownMenuItem onClick={() => setView('documents' as View)} className="cursor-pointer">
   660→                  <FileText className="mr-2 h-4 w-4" />
   661→                  <span>Documents</span>
   662→                </DropdownMenuItem>
   663→                {role === 'super_admin' && (
   664→                  <DropdownMenuItem onClick={() => setView('admin' as View)} className="cursor-pointer">
   665→                    <Shield className="mr-2 h-4 w-4" />
   666→                    <span>Administration</span>
   667→                  </DropdownMenuItem>
   668→                )}
   669→                <DropdownMenuSeparator />
   670→                <DropdownMenuItem onClick={logout} className="cursor-pointer text-red-600 focus:text-red-600 focus:bg-red-50 dark:focus:bg-red-950/30">
   671→                  <LogOut className="mr-2 h-4 w-4" />
   672→                  <span>Sign Out</span>
   673→                </DropdownMenuItem>
   674→              </DropdownMenuContent>
   675→            </DropdownMenu>
   676→          </div>
   677→        </header>
   678→
   679→        {/* ── Content Area ── */}
   680→        <main className="flex-1 overflow-auto">
   681→          <div key={view} className="view-enter">
   682→            <Suspense fallback={<ViewLoader />}>
   683→              {renderView()}
   684→            </Suspense>
   685→          </div>
   686→        </main>
   687→      </div>
   688→    </div>
   689→  );
   690→}
