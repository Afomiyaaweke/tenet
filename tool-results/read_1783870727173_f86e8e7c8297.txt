   168→function relativeTime(dateStr: string): string {
   169→  const now = Date.now();
   170→  const then = new Date(dateStr).getTime();
   171→  const diff = Math.max(0, now - then);
   172→  const seconds = Math.floor(diff / 1000);
   173→  if (seconds < 60) return 'just now';
   174→  const minutes = Math.floor(seconds / 60);
   175→  if (minutes < 60) return `${minutes}m ago`;
   176→  const hours = Math.floor(minutes / 60);
   177→  if (hours < 24) return `${hours}h ago`;
   178→  const days = Math.floor(hours / 24);
   179→  if (days < 7) return `${days}d ago`;
   180→  return new Date(dateStr).toLocaleDateString();
   181→}

   461→export function AppShell() {
   462→  const { user, company, logout } = useAuthStore();
   463→  const { view, viewParams, setView } = useNavStore();
   464→  const { fetchNotifications, notifications } = useDataStore();
   465→  const role = user?.role || 'user';
   466→  const navSections = useMemo(() => getNavItemsForRole(role), [role]);
   467→  const unreadCount = notifications.filter((n) => !n.read).length;
   468→
   469→  // Flat list for title lookup
   470→  const allNavItems = useMemo(
   471→    () => navSections.flatMap((s) => s.items),
   472→    [navSections]
   473→  );
   474→
   475→  useEffect(() => {
   476→    fetchNotifications();
   477→  }, [fetchNotifications]);
   478→
