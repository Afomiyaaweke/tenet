    59→function daysUntil(dateStr: string): number {
    60→  return Math.ceil((new Date(dateStr).getTime() - Date.now()) / (1000 * 60 * 60 * 24));
    61→}
    62→
    63→function deadlineColor(days: number) {
    64→  if (days <= 0) return 'rose';
    65→  if (days <= 3) return 'rose';
    66→  if (days <= 7) return 'amber';
    67→  return 'emerald';
    68→}
    69→
    70→function deadlineBg(days: number) {
    71→  const c = deadlineColor(days);
    72→  const map: Record<string, string> = {
    73→    rose: 'bg-rose-50 text-rose-700',
    74→    amber: 'bg-amber-50 text-amber-700',
    75→    emerald: 'bg-emerald-50 text-emerald-700',
    76→  };
    77→  return map[c] || map.emerald;
    78→}
    79→
    80→function formatBudget(min: number, max: number) {
    81→  const fmt = (n: number) => n >= 1_000_000 ? `${(n / 1_000_000).toFixed(1)}M` : n >= 1_000 ? `${(n / 1_000).toFixed(0)}K` : n.toLocaleString();
    82→  return `ETB ${fmt(min)} – ${fmt(max)}`;
    83→}

  1411→function statusColor(status: string) {
  1412→  switch (status) {
  1413→    case 'open': return 'bg-emerald-100 text-emerald-700 hover:bg-emerald-100';
  1414→    case 'closed': return 'bg-rose-100 text-rose-700 hover:bg-rose-100';
  1415→    case 'awarded': return 'bg-teal-100 text-teal-700 hover:bg-teal-100';
  1416→    case 'cancelled': return 'bg-muted text-muted-foreground hover:bg-muted';
  1417→    default: return 'bg-muted text-muted-foreground hover:bg-muted';
  1418→  }
  1419→}
  1420→
  1421→function statusAccent(status: string) {
  1422→  switch (status) {
  1423→    case 'open': return 'from-emerald-400 to-emerald-600';
  1424→    case 'closed': return 'from-rose-400 to-rose-600';
  1425→    case 'awarded': return 'from-teal-400 to-teal-600';
  1426→    case 'cancelled': return 'from-gray-300 to-gray-400';
  1427→    default: return 'from-gray-300 to-gray-400';
  1428→  }
  1429→}
  1430→
  1431→function matchBarColor(score: number) {
  1432→  if (score >= 70) return 'from-emerald-400 to-emerald-600';
  1433→  if (score >= 40) return 'from-amber-400 to-amber-600';
  1434→  return 'from-gray-300 to-gray-400';
  1435→}
