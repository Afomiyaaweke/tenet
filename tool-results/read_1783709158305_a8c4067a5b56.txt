   179→function fmtMoney(amount: number, currency: string): string {
   180→  if (!amount) return '—';
   181→  try {
   182→    return new Intl.NumberFormat('en-US', {
   183→      style: 'currency',
   184→      currency: currency || 'USD',
   185→      maximumFractionDigits: 0,
   186→    }).format(amount);
   187→  } catch {
   188→    return `${currency} ${amount.toLocaleString()}`;
   189→  }
   190→}
   191→
   192→function daysUntil(dateStr: string): number {
   193→  return Math.ceil((new Date(dateStr).getTime() - Date.now()) / (1000 * 60 * 60 * 24));
   194→}
   195→
   196→function deadlineBadge(days: number): string {
   197→  if (days <= 0) return 'bg-rose-50 text-rose-700 dark:bg-rose-950/40 dark:text-rose-300';
   198→  if (days <= 7) return 'bg-amber-50 text-amber-700 dark:bg-amber-950/40 dark:text-amber-300';
   199→  if (days <= 21) return 'bg-sky-50 text-sky-700 dark:bg-sky-950/40 dark:text-sky-300';
   200→  return 'bg-emerald-50 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300';
   201→}
   202→
   203→function relativeTime(dateStr: string): string {
   204→  const diff = Date.now() - new Date(dateStr).getTime();
   205→  const day = 86400000;
   206→  if (diff < day) return 'today';
   207→  if (diff < 2 * day) return 'yesterday';
   208→  if (diff < 30 * day) return `${Math.floor(diff / day)}d ago`;
   209→  return new Date(dateStr).toLocaleDateString();
   210→}

   212→/* ─────────────────────────────────────────────────────────────────────
   213→ * Inline Document Viewer Component

   215→
   216→function InlineDocumentViewer({ doc, onClose }: { doc: InlineDocument; onClose: () => void }) {
   217→  const scrollRef = useRef<HTMLDivElement>(null);
   218→
   219→  const copyContent = () => {
   220→    const text = doc.sections
   221→      ? doc.sections.map((s) => `${s.heading}\n${s.content}`).join('\n\n')
   222→      : doc.content;
   223→    navigator.clipboard.writeText(text);
   224→    toast.success('Content copied to clipboard');
   225→  };
   226→
   227→  return (
   228→    <div
   229→ className="overflow-hidden animate-[fadeIn_0.3s_ease-out]"
   230→ >
   231→      <div className="border-t border-border bg-gradient-to-b from-muted/30 to-background">
   232→        <div className="p-4 md:p-6 space-y-4">
   233→          {/* Document header */}
   234→          <div className="flex items-start justify-between gap-3">
   235→            <div className="min-w-0 flex-1">
   236→              <div className="flex items-center gap-2 text-xs text-muted-foreground mb-1">
   237→                <BookOpen className="h-3.5 w-3.5" />
   238→                <span>Document loaded from {new URL(doc.url).hostname}</span>
   239→                <span>·</span>
   240→                <span>{new Date(doc.fetchedAt).toLocaleTimeString()}</span>
   241→              </div>
   242→              <h3 className="text-lg font-semibold text-foreground leading-snug">
   243→                {doc.title || 'Tender Document'}
   244→              </h3>
   245→              {doc.metaDescription && (
   246→                <p className="text-sm text-muted-foreground mt-1 line-clamp-2">
   247→                  {doc.metaDescription}
   248→                </p>
   249→              )}
   250→            </div>
   251→            <div className="flex items-center gap-2 shrink-0">
   252→              <Button variant="ghost" size="sm" className="gap-1.5 text-xs" onClick={copyContent}>
   253→                <Copy className="h-3.5 w-3.5" />
   254→                Copy
   255→              </Button>
   256→              <a
   257→                href={doc.url}
   258→                target="_blank"
   259→                rel="noopener noreferrer"
   260→                className="inline-flex items-center gap-1.5 text-xs font-medium text-primary hover:underline"
   261→              >
   262→                <ExternalLink className="h-3.5 w-3.5" />
   263→                Open original
   264→              </a>
   265→              <Button variant="ghost" size="sm" onClick={onClose} className="gap-1.5 text-xs text-muted-foreground">
   266→                <ChevronUp className="h-3.5 w-3.5" />
   267→                Collapse
   268→              </Button>
   269→            </div>
   270→          </div>
   271→
   272→          {/* Extracted metadata pills */}
   273→          {(doc.deadlines && doc.deadlines.length > 0) || (doc.budgets && doc.budgets.length > 0) ? (
   274→            <div className="flex flex-wrap gap-2">
   275→              {doc.deadlines?.map((d, i) => (
   276→                <Badge key={`dl-${i}`} variant="outline" className="gap-1.5 text-xs border-amber-300 text-amber-700 dark:border-amber-800 dark:text-amber-300">
   277→                  <Calendar className="h-3 w-3" />
   278→                  {d}
   279→                </Badge>
   280→              ))}
   281→              {doc.budgets?.map((b, i) => (
   282→                <Badge key={`bg-${i}`} variant="outline" className="gap-1.5 text-xs border-emerald-300 text-emerald-700 dark:border-emerald-800 dark:text-emerald-300">
   283→                  <DollarSign className="h-3 w-3" />
   284→                  {b}
   285→                </Badge>
   286→              ))}
   287→            </div>
   288→          ) : null}
   289→
   290→          {/* Sections view */}
   291→          {doc.sections && doc.sections.length > 0 ? (
   292→            <div ref={scrollRef} className="max-h-80 overflow-y-auto rounded-lg border border-border bg-background p-4 space-y-4 scrollbar-thin">
   293→              {doc.sections.map((section, i) => (
   294→                <div key={i}>
   295→                  <h4 className="text-sm font-semibold text-foreground mb-1.5">{section.heading}</h4>
   296→                  <p className="text-sm text-muted-foreground leading-relaxed whitespace-pre-line">
   297→                    {section.content}
   298→                  </p>
   299→                  {i < doc.sections!.length - 1 && <Separator className="mt-4 bg-border" />}
   300→                </div>
   301→              ))}
   302→            </div>
   303→          ) : (
   304→            <div ref={scrollRef} className="max-h-80 overflow-y-auto rounded-lg border border-border bg-background p-4 scrollbar-thin">
   305→              <p className="text-sm text-muted-foreground leading-relaxed whitespace-pre-line">
   306→                {doc.content || 'No content could be extracted from this page.'}
   307→              </p>
   308→            </div>
   309→          )}
   310→        </div>
   311→      </div>
   312→    </div>
   313→  );
   314→}

   319→
   320→export function LiveTendersView() {
   321→  const [tenders, setTenders] = useState<LiveTender[]>([]);
   322→  const [dataSources, setDataSources] = useState<DataSource[]>([]);
   323→  const [sourceMeta, setSourceMeta] = useState<
   324→    { id: string; name: string; live: boolean; ok: boolean; count: number; error?: string }[]
   325→  >([]);
   326→  const [loading, setLoading] = useState(true);
   327→  const [refreshing, setRefreshing] = useState(false);
   328→  const [fallback, setFallback] = useState(false);
   329→  const [search, setSearch] = useState('');
   330→  const [sourceFilter, setSourceFilter] = useState('all');
   331→  const [sectorFilter, setSectorFilter] = useState('');
   332→  const [sectorCounts, setSectorCounts] = useState<{ id: string; label: string; count: number }[]>([]);
   333→
   334→  // Pagination state
   335→  const [visibleCount, setVisibleCount] = useState(20);
   336→  const PAGE_SIZE = 20;
   337→
   338→  // Inline document state
   339→  const [expandedId, setExpandedId] = useState<string | null>(null);
   340→  const [docLoading, setDocLoading] = useState<string | null>(null);
   341→  const [docData, setDocData] = useState<Record<string, InlineDocument>>({});
   342→  const [docError, setDocError] = useState<Record<string, string>>({});
   343→
   344→  // Import state
   345→  const [importing, setImporting] = useState<string | null>(null);
   346→
   347→  const load = useCallback(
   348→    async (isRefresh = false) => {
   349→      if (isRefresh) setRefreshing(true);
   350→      else setLoading(true);
   351→      const params: Record<string, string> = { rows: '500' };
   352→      if (search) params.search = search;
   353→      if (sourceFilter && sourceFilter !== 'all') params.source = sourceFilter;
   354→      if (sectorFilter) params.sector = sectorFilter;
   355→      const res = await api.get('/tenders/live', params);
   356→      if (res.success) {
   357→        setTenders(res.data as LiveTender[]);
   358→        setSourceMeta(res.meta?.sources || []);
   359→        setFallback(Boolean(res.meta?.fallback));
   360→        if (Array.isArray(res.meta?.dataSources)) setDataSources(res.meta.dataSources);
   361→        if (Array.isArray(res.meta?.sectors)) setSectorCounts(res.meta.sectors);
   362→      } else {
   363→        toast.error(res.error || 'Failed to load live tenders');
   364→      }
   365→      setLoading(false);
   366→      setRefreshing(false);
   367→    },
   368→    [search, sourceFilter, sectorFilter],
   369→  );
   370→
   371→  useEffect(() => {
   372→    const t = setTimeout(() => load(), 250);
   373→    return () => clearTimeout(t);
   374→  }, [load]);
   375→
   376→  const liveSourcesCount = useMemo(
   377→    () => sourceMeta.filter((s) => s.live && s.ok).length,
   378→    [sourceMeta],
   379→  );
   380→
   381→  const totalTenderValue = useMemo(() => {
   382→    return tenders.reduce((sum, t) => sum + (t.budgetMax || 0), 0);
   383→  }, [tenders]);
   384→
   385→  /* ────── Inline document loading ────── */
   386→  const loadDocument = useCallback(async (tender: LiveTender) => {
   387→    const id = tender.id;
   388→    if (expandedId === id) {
   389→      setExpandedId(null);
   390→      return;
   391→    }
   392→
   393→    // Already fetched?
   394→    if (docData[id]) {
   395→      setExpandedId(id);
   396→      return;
   397→    }
   398→
   399→    setExpandedId(id);
   400→    setDocLoading(id);
   401→    setDocError((prev) => { const n = { ...prev }; delete n[id]; return n; });
   402→
   403→    try {
   404→      const res = await api.get(`/tenders/${encodeURIComponent(id)}/documents`, { url: tender.externalUrl });
   405→      if (res.success && res.data) {
   406→        setDocData((prev) => ({ ...prev, [id]: res.data as InlineDocument }));
   407→      } else {
   408→        setDocError((prev) => ({ ...prev, [id]: res.error || 'Failed to load document' }));
   409→      }
   410→    } catch {
   411→      setDocError((prev) => ({ ...prev, [id]: 'Network error' }));
   412→    }
   413→    setDocLoading(null);
   414→  }, [expandedId, docData]);

   429→        status: 'open',
   430→        currency: tender.currency,
   431→      });
   432→      if (res.success) {
   433→        toast.success('Tender imported successfully', {
   434→          description: `"${tender.title}" is now in your tenders list.`,
   435→        });
   436→      } else {
   437→        toast.error(res.error || 'Failed to import tender');
   438→      }
   439→    } catch {
   440→      toast.error('Failed to import tender');
   441→    }
   442→    setImporting(null);
   443→  }, []);
   444→
   445→  /* ────── Render ────── */
   446→  return (

   452→        <div className="absolute top-4 right-4 md:right-6">
   453→          <Button
   454→            variant="secondary"
   455→            size="sm"
   456→            onClick={() => load(true)}
   457→            disabled={refreshing}
   458→            className="gap-1.5 shadow-lg bg-white/90 dark:bg-gray-800/90 backdrop-blur-sm"
   459→          >
   460→            <RefreshCw className={`h-3.5 w-3.5 ${refreshing ? 'animate-spin' : ''}`} />

   467→        {/* ───────────────── Notion-style Icon + Title ───────────────── */}
