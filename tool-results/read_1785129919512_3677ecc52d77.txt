   187→function relativeTime(dateStr: string): string {
   188→  const now = Date.now();
   189→  const then = new Date(dateStr).getTime();
   190→  const diff = Math.max(0, now - then);
   191→  const seconds = Math.floor(diff / 1000);
   192→  if (seconds < 60) return 'just now';
   193→  const minutes = Math.floor(seconds / 60);
   194→  if (minutes < 60) return `${minutes}m ago`;
   195→  const hours = Math.floor(minutes / 60);
   196→  if (hours < 24) return `${hours}h ago`;
   197→  const days = Math.floor(hours / 24);
   198→  if (days < 7) return `${days}d ago`;
   199→  return new Date(dateStr).toLocaleDateString();
   200→}

   361→/* ──────────────────── NotificationDropdown ──────────────────── */

   363→function NotificationDropdown({
   364→  notifications,
   365→  onMarkAsRead,
   366→  onMarkAllRead,
   367→}: {
   368→  notifications: { id: string; title: string; message: string; type: string; read: boolean; createdAt: string }[];

   370→  onMarkAllRead: () => void;

   372→  const [open, setOpen] = useState(false);
   373→  const unread = useMemo(() => notifications.filter((n) => !n.read), [notifications]);
   374→  const displayItems = useMemo(() => unread.slice(0, 6), [unread]);
   375→
   376→  return (
   377→    <Popover open={open} onOpenChange={setOpen}>
   378→      <PopoverTrigger asChild>
   379→        <Button variant="ghost" size="icon" className="relative h-9 w-9 hover:bg-muted/80 transition-colors">
   380→          <Bell className="h-[18px] w-[18px] text-muted-foreground" />

   382→            <span className="absolute -top-0.5 -right-0.5 min-w-[18px] h-[18px] px-1 bg-red-500 text-white text-[10px] font-bold rounded-full flex items-center justify-center shadow-sm shadow-red-200 animate-in zoom-in-50 duration-200">
   383→              {unread.length > 99 ? '99+' : unread.length}
   384→            </span>
   385→          )}
   386→        </Button>
   387→      </PopoverTrigger>
   388→      <PopoverContent
   389→        className="w-80 p-0 glass-card rounded-xl overflow-hidden animate-in slide-in-from-top-2 fade-in-0 duration-200"
   390→        align="end"
   391→        sideOffset={8}

   394→        <div className="flex items-center justify-between px-4 py-3 border-b border-border/50">
   395→          <div className="flex items-center gap-2">
   396→            <h4 className="text-sm font-bold text-foreground">Notifications</h4>
   397→            {unread.length > 0 && (
   398→              <Badge className="text-[10px] px-1.5 py-0 bg-red-500 text-white font-bold border-0">
   399→                {unread.length}
   400→              </Badge>

   405→              variant="ghost"

   412→          )}
   413→        </div>

   417→          {displayItems.length === 0 ? (
   418→            <div className="py-8 text-center">
   419→              <div className="w-12 h-12 mx-auto mb-3 rounded-full bg-orange-50 dark:bg-orange-950/30 flex items-center justify-center">
   420→                <CheckCircle className="h-6 w-6 text-orange-400" />
   421→              </div>
   422→              <p className="text-sm font-medium text-muted-foreground">All caught up!</p>
   423→              <p className="text-xs text-muted-foreground/60 mt-0.5">No new notifications</p>
   424→            </div>
   425→          ) : (
   426→            <div className="divide-y divide-border/40">
   427→              {displayItems.map((n, i) => {
   428→                const Icon = NOTIFICATION_ICONS[n.type] || Info;
   429→                const colorClass = NOTIFICATION_COLORS[n.type] || 'text-gray-500';
   430→                const bgClass = NOTIFICATION_BG[n.type] || 'bg-gray-50';
   431→                return (
   432→                  <button
   433→                    key={n.id}
   434→                    onClick={() => {
   435→                      onMarkAsRead(n.id);
   436→                      setOpen(false);
   437→                    }}
   438→                    className="w-full flex items-start gap-3 px-4 py-3 hover:bg-muted/50 transition-colors text-left group"
   439→                    style={{ animationDelay: `${i * 40}ms` }}
   440→                  >
   441→                    <div
   442→                      className={`w-8 h-8 rounded-lg ${bgClass} flex items-center justify-center flex-shrink-0 mt-0.5 transition-transform group-hover:scale-110`}
   443→                    >
   444→                      <Icon className={`h-4 w-4 ${colorClass}`} />
   445→                    </div>
   446→                    <div className="min-w-0 flex-1">
   447→                      <p className="text-[13px] font-semibold truncate text-foreground">{n.title}</p>
   448→                      <p className="text-[11px] text-muted-foreground truncate mt-0.5 leading-relaxed">{n.message}</p>
   449→                      <p className="text-[10px] text-muted-foreground/50 mt-1 font-medium">{relativeTime(n.createdAt)}</p>
   450→                    </div>
   451→                    <div className="w-2 h-2 rounded-full bg-orange-500 flex-shrink-0 mt-2 ring-2 ring-orange-100 dark:ring-orange-900/40" />
   452→                  </button>
   453→                );
   454→              })}
   455→            </div>
   456→          )}
   457→        </ScrollArea>
