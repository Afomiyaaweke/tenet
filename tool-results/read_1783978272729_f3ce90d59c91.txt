   364→/* ─────────────────────────────────────────────────────────────────────
   365→ * Utility helpers

   367→
   368→function parseAmount(value: number | string | undefined): number {
   369→  if (value === undefined || value === null) return 0;
   370→  if (typeof value === 'number') return value;
   371→  const n = Number(String(value).replace(/[^0-9.-]/g, ''));
   372→  return Number.isFinite(n) ? n : 0;
   373→}
   374→
   375→function truncate(text: string, max = 280): string {
   376→  if (!text) return '';
   377→  return text.length > max ? text.slice(0, max).trimEnd() + '…' : text;
   378→}

   380→/* ─────────────────────────────────────────────────────────────────────
   381→ * World Bank adapter (live, public, no auth)
   382→ * ───────────────────────────────────────────────────────────────────── */
   383→
   384→interface WorldBankProcurementRow {
   385→  project_id?: string;
   386→  procurement_group_id?: string;
   387→  project_name?: string;
   388→  borrower?: string;
   389→  country?: string;
   390→  region?: string;
   391→  contract_description?: string;
   392→  contract_type?: string;
   393→  supplier?: string;
   394→  supplier_country?: string | string[];
   395→  contract_signing_date?: string;
   396→  total_contract_amount?: number | string;
   397→  wb_contract_number?: string;
   398→}
   399→
   400→interface WorldBankResponse {
   401→  total?: number;
   402→  rows?: WorldBankProcurementRow[];
   403→  procurements?: WorldBankProcurementRow[];
   404→  [k: string]: unknown;
   405→}
   406→
   407→function inferCategory(row: WorldBankProcurementRow): string {
   408→  const t = (row.contract_type || '').toLowerCase();
   409→  if (t.includes('work') || t.includes('civil') || t.includes('construction')) return 'Construction';
   410→  if (t.includes('consult')) return 'Consulting';
   411→  if (t.includes('good') || t.includes('supply') || t.includes('equipment')) return 'Supply';
   412→  if (t.includes('service')) return 'Consulting';
   413→  const p = (row.project_name || '').toLowerCase();
   414→  if (p.includes('road') || p.includes('bridge') || p.includes('infrastructure')) return 'Construction';
   415→  if (p.includes('agric') || p.includes('farm') || p.includes('irrigat')) return 'Agriculture';
   416→  if (p.includes('health') || p.includes('clinic') || p.includes('hospital')) return 'Healthcare';
   417→  if (p.includes('energy') || p.includes('power') || p.includes('electric')) return 'Engineering';
   418→  if (p.includes('edu') || p.includes('school')) return 'Education';
   419→  if (p.includes('water') || p.includes('sanitation')) return 'Engineering';
   420→  return 'Consulting';
   421→}
   422→
   423→function buildDeadline(signingDate?: string): string {
   424→  if (!signingDate) return new Date(Date.now() + 45 * 86400000).toISOString();
   425→  const base = new Date(signingDate);
   426→  if (Number.isNaN(base.getTime())) return new Date(Date.now() + 45 * 86400000).toISOString();
   427→  return new Date(base.getTime() + 45 * 86400000).toISOString();
   428→}

   430→export async function fetchWorldBankTenders(opts: {
   431→  search?: string;
   432→  rows?: number;
   433→  offset?: number;
   434→}): Promise<{ tenders: LiveTender[]; total: number; ok: boolean }> {
   435→  const rows = Math.min(Math.max(opts.rows ?? 20, 1), 500);
   436→  const offset = opts.offset || 0;
   437→  const params = new URLSearchParams({
   438→    format: 'json',
   439→    rows: String(rows),
   440→    fl: [
   441→      'project_id',
   442→      'procurement_group_id',
   443→      'project_name',
   444→      'borrower',
   445→      'country',
   446→      'region',
   447→      'contract_description',
   448→      'contract_type',
   449→      'supplier',
   450→      'supplier_country',
   451→      'contract_signing_date',
   452→      'total_contract_amount',
   453→      'wb_contract_number',
   454→    ].join(','),
   455→  });
   456→  if (opts.search) params.set('q', opts.search);
   457→  if (offset > 0) params.set('osstart', String(offset + 1));
   458→
   459→  const url = `https://search.worldbank.org/api/v2/procurement?${params.toString()}`;
   460→  const ctrl = new AbortController();
   461→  const timer = setTimeout(() => ctrl.abort(), 8000);
   462→
   463→  try {
   464→    const res = await fetch(url, {
   465→      signal: ctrl.signal,
   466→      headers: { Accept: 'application/json', 'User-Agent': 'Tenet-Tender-Ecosystem/1.0' },
   467→      cache: 'no-store',
   468→    });
   469→    clearTimeout(timer);
   470→    if (!res.ok) {
   471→      return { tenders: [], total: 0, ok: false };
   472→    }
   473→    const json = (await res.json()) as WorldBankResponse;
   474→    const rowsArr = json.rows || json.procurements || [];
   475→    const total = typeof json.total === 'number' ? json.total : rowsArr.length;
   476→
   477→    const tenders: LiveTender[] = rowsArr
   478→      .filter((r) => r && (r.contract_description || r.project_name))
   479→      .map((r, idx) => {
   480→        const externalId = r.procurement_group_id || r.project_id || `wb-${idx}`;
   481→        const title = truncate(r.contract_description || r.project_name || 'World Bank Contract Award', 160);
   482→        const amount = parseAmount(r.total_contract_amount);
   483→        const signingDate = r.contract_signing_date;
   484→        return {
   485→          id: `worldbank-${externalId}`,
   486→          title,
   487→          scope: truncate(r.contract_description || r.project_name || '', 400),
   488→          budgetMin: amount,
   489→          budgetMax: amount,
   490→          deadline: buildDeadline(signingDate),
   491→          location: r.country || r.region || 'Multiple',
   492→          categoryTags: inferCategory(r),
   493→          requiredDocs: r.project_id
   494→            ? `https://projects.worldbank.org/en/projects-operations/project-detail/${r.project_id}`
   495→            : 'https://financesone.worldbank.org',
   496→          status: 'awarded',
   497→          createdBy: 'worldbank',
   498→          createdAt: signingDate || new Date().toISOString(),
   499→          updatedAt: signingDate || new Date().toISOString(),
   500→          source: 'worldbank',
   501→          externalId,
   502→          externalUrl: r.project_id
   503→            ? `https://projects.worldbank.org/en/projects-operations/project-detail/${r.project_id}`
   504→            : 'https://financesone.worldbank.org',
   505→          currency: 'USD',
   506→          borrower: r.borrower || undefined,
   507→          supplier: r.supplier || undefined,
   508→          contractType: r.contract_type || undefined,
   509→          signingDate: signingDate || undefined,
   510→          region: r.region || undefined,
   511→          documentUrl: r.project_id ? `https://projects.worldbank.org/en/projects-operations/project-detail/${r.project_id}` : undefined,
   512→        } satisfies LiveTender;
   513→      });
   514→
   515→    return { tenders, total, ok: true };
   516→  } catch {
   517→    clearTimeout(timer);
   518→    return { tenders: [], total: 0, ok: false };
   519→  }
   520→}

   522→/* ─────────────────────────────────────────────────────────────────────
   523→ * EU TED adapter (live, public RSS-style feed)
   524→ * ───────────────────────────────────────────────────────────────────── */
   525→
   526→export async function fetchEuTedTenders(opts: {
   527→  search?: string;
   528→  rows?: number;
   529→  offset?: number;
   530→}): Promise<{ tenders: LiveTender[]; total: number; ok: boolean }> {
   531→  const rows = Math.min(Math.max(opts.rows ?? 20, 1), 500);
   532→  const offset = opts.offset || 0;
   533→  const base = 'https://api.ted.europa.eu/v3/notices/search';
   534→  const body = {
   535→    query: opts.search ? `(${opts.search})` : '*',
