    24→function formatFileSize(bytes: number): string {
    25→  if (bytes === 0) return '0 B';
    26→  const k = 1024;
    27→  const sizes = ['B', 'KB', 'MB', 'GB'];
    28→  const i = Math.floor(Math.log(bytes) / Math.log(k));
    29→  return `${parseFloat((bytes / Math.pow(k, i)).toFixed(1))} ${sizes[i]}`;
    30→}
    31→
    32→function getFileExtension(fileName: string): string {
    33→  const parts = fileName.split('.');
    34→  return parts.length > 1 ? parts[parts.length - 1].toUpperCase() : 'FILE';
    35→}
