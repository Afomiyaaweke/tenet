   167→
   168→function parseAmount(value: number | string | undefined): number {
   169→  if (value === undefined || value === null) return 0;
   170→  if (typeof value === 'number') return value;
   171→  const n = Number(String(value).replace(/[^0-9.-]/g, ''));
   172→  return Number.isFinite(n) ? n : 0;
   173→}
   174→
   175→function truncate(text: string, max = 280): string {
   176→  if (!text) return '';
   177→  return text.length > max ? text.slice(0, max).trimEnd() + '…' : text;
   178→}

   200→interface WorldBankResponse {
   201→  total?: number;

   206→
   207→function inferCategory(row: WorldBankProcurementRow): string {
   208→  const t = (row.contract_type || '').toLowerCase();
   209→  if (t.includes('work') || t.includes('civil') || t.includes('construction')) return 'Construction';
   210→  if (t.includes('consult')) return 'Consulting';
   211→  if (t.includes('good') || t.includes('supply') || t.includes('equipment')) return 'Supply';
   212→  if (t.includes('service')) return 'Consulting';
   213→  const p = (row.project_name || '').toLowerCase();
   214→  if (p.includes('road') || p.includes('bridge') || p.includes('infrastructure')) return 'Construction';
   215→  if (p.includes('agric') || p.includes('farm') || p.includes('irrigat')) return 'Agriculture';
   216→  if (p.includes('health') || p.includes('clinic') || p.includes('hospital')) return 'Healthcare';
   217→  if (p.includes('energy') || p.includes('power') || p.includes('electric')) return 'Engineering';
   218→  if (p.includes('edu') || p.includes('school')) return 'Education';
   219→  if (p.includes('water') || p.includes('sanitation')) return 'Engineering';
   220→  return 'Consulting';
   221→}
   222→
   223→function buildDeadline(signingDate?: string): string {
   224→  if (!signingDate) return new Date(Date.now() + 45 * 86400000).toISOString();
   225→  const base = new Date(signingDate);
   226→  if (Number.isNaN(base.getTime())) return new Date(Date.now() + 45 * 86400000).toISOString();
   227→  return new Date(base.getTime() + 45 * 86400000).toISOString();
   228→}

   230→export async function fetchWorldBankTenders(opts: {
   231→  search?: string;
   232→  rows?: number;
   233→}): Promise<{ tenders: LiveTender[]; total: number; ok: boolean }> {
   234→  const rows = Math.min(Math.max(opts.rows ?? 20, 1), 50);
   235→  const params = new URLSearchParams({
   236→    format: 'json',
   237→    rows: String(rows),
   238→    fl: [
   239→      'project_id',
   240→      'procurement_group_id',
   241→      'project_name',
   242→      'borrower',
   243→      'country',
   244→      'region',
   245→      'contract_description',
   246→      'contract_type',
   247→      'supplier',
   248→      'supplier_country',
   249→      'contract_signing_date',
   250→      'total_contract_amount',
   251→      'wb_contract_number',
   252→    ].join(','),
   253→  });
   254→  if (opts.search) params.set('q', opts.search);
   255→
   256→  const url = `https://search.worldbank.org/api/v2/procurement?${params.toString()}`;
   257→  const ctrl = new AbortController();
   258→  const timer = setTimeout(() => ctrl.abort(), 8000);
   259→
   260→  try {
   261→    const res = await fetch(url, {
   262→      signal: ctrl.signal,
   263→      headers: { Accept: 'application/json', 'User-Agent': 'Tenets-Tender-Ecosystem/1.0' },
   264→      cache: 'no-store',
   265→    });
   266→    clearTimeout(timer);
   267→    if (!res.ok) {
   268→      return { tenders: [], total: 0, ok: false };
   269→    }
   270→    const json = (await res.json()) as WorldBankResponse;
   271→    const rowsArr = json.rows || json.procurements || [];
   272→    const total = typeof json.total === 'number' ? json.total : rowsArr.length;
   273→
   274→    const tenders: LiveTender[] = rowsArr
   275→      .filter((r) => r && (r.contract_description || r.project_name))
   276→      .map((r, idx) => {
   277→        const externalId = r.procurement_group_id || r.project_id || `wb-${idx}`;
   278→        const title = truncate(r.contract_description || r.project_name || 'World Bank Contract Award', 160);
   279→        const amount = parseAmount(r.total_contract_amount);
   280→        const signingDate = r.contract_signing_date;
   281→        return {
   282→          id: `worldbank-${externalId}`,
   283→          title,
   284→          scope: truncate(r.contract_description || r.project_name || '', 400),
   285→          budgetMin: amount,
   286→          budgetMax: amount,
   287→          deadline: buildDeadline(signingDate),
   288→          location: r.country || r.region || 'Multiple',
   289→          categoryTags: inferCategory(r),
   290→          requiredDocs: '',
   291→          status: 'awarded',
   292→          createdBy: 'worldbank',
   293→          createdAt: signingDate || new Date().toISOString(),
   294→          updatedAt: signingDate || new Date().toISOString(),
   295→          source: 'worldbank',
   296→          externalId,
   297→          externalUrl: r.project_id
   298→            ? `https://projects.worldbank.org/en/projects-operations/project-detail/${r.project_id}`
   299→            : 'https://financesone.worldbank.org',
   300→          currency: 'USD',
   301→          borrower: r.borrower || undefined,
   302→          supplier: r.supplier || undefined,
   303→          contractType: r.contract_type || undefined,
   304→          signingDate: signingDate || undefined,
   305→          region: r.region || undefined,
   306→        } satisfies LiveTender;
   307→      });
   308→
   309→    return { tenders, total, ok: true };
   310→  } catch {
   311→    clearTimeout(timer);
   312→    return { tenders: [], total: 0, ok: false };
   313→  }
   314→}

   319→
   320→export async function fetchEuTedTenders(opts: {
   321→  search?: string;
   322→  rows?: number;
   323→}): Promise<{ tenders: LiveTender[]; total: number; ok: boolean }> {
   324→  const rows = Math.min(Math.max(opts.rows ?? 20, 1), 50);
   325→  const base = 'https://api.ted.europa.eu/v3/notices/search';
   326→  const body = {
   327→    query: opts.search ? `(${opts.search})` : '*',
   328→    fields: ['ND', 'TI_DOC', 'DS_DATE_PUB', 'DEADLINE_DATE', 'PLACE_CONTRACT', 'VALUE_CONTRACT', 'CA_NAME', 'TITLE'],
   329→    limit: rows,
   330→    offset: 0,
   331→    pagination: { pageNumber: 1, pageSize: rows },
   332→    sort: { field: 'DS_DATE_PUB', order: 'DESC' },
   333→  };
   334→  const ctrl = new AbortController();
   335→  const timer = setTimeout(() => ctrl.abort(), 8000);
   336→
   337→  try {
   338→    const res = await fetch(base, {
   339→      method: 'POST',
   340→      signal: ctrl.signal,
   341→      headers: { 'Content-Type': 'application/json', Accept: 'application/json' },
   342→      body: JSON.stringify(body),
   343→      cache: 'no-store',
   344→    });
   345→    clearTimeout(timer);
   346→    if (!res.ok) return { tenders: [], total: 0, ok: false };
   347→    const json = (await res.json()) as { notices?: unknown[]; total?: number };
   348→    const notices = Array.isArray(json.notices) ? json.notices : [];
   349→    const total = typeof json.total === 'number' ? json.total : notices.length;
   350→
   351→    const tenders: LiveTender[] = notices.map((n, idx) => {
   352→      const row = (n || {}) as Record<string, unknown>;
   353→      const nd = String(row.ND || row['ND'] || `ted-${idx}`);
   354→      const title = String(row.TI_DOC || row.TITLE || 'EU TED Notice');
   355→      const deadlineRaw = row.DEADLINE_DATE as string | undefined;
   356→      const amount = Number(row.VALUE_CONTRACT) || 0;
   357→      return {
   358→        id: `eu_ted-${nd}`,
   359→        title: truncate(title, 160),
   360→        scope: truncate(title, 400),
   361→        budgetMin: amount,
   362→        budgetMax: amount,
   363→        deadline: deadlineRaw || new Date(Date.now() + 30 * 86400000).toISOString(),
   364→        location: String(row.PLACE_CONTRACT || 'EU'),
   365→        categoryTags: 'Consulting',
   366→        requiredDocs: '',
   367→        status: 'open',
   368→        createdBy: 'eu_ted',
   369→        createdAt: String(row.DS_DATE_PUB || new Date().toISOString()),
   370→        updatedAt: String(row.DS_DATE_PUB || new Date().toISOString()),
   371→        source: 'eu_ted',
   372→        externalId: nd,
   373→        externalUrl: `https://ted.europa.eu/udl?uri=TED:NOTICE:${nd}:TEXT:EN:HTML`,
   374→        currency: 'EUR',
   375→        borrower: String(row.CA_NAME || '') || undefined,
   376→      } satisfies LiveTender;
   377→    });
   378→
   379→    return { tenders, total, ok: true };
   380→  } catch {
   381→    clearTimeout(timer);
   382→    return { tenders: [], total: 0, ok: false };
   383→  }
   384→}
