    77→// ─── Helpers ────────────────────────────────────────────────────────
    78→function formatETB(amount: number): string {
    79→  if (amount >= 1_000_000) return `ETB ${(amount / 1_000_000).toFixed(1)}M`;
    80→  if (amount >= 1_000) return `ETB ${(amount / 1_000).toFixed(0)}K`;
    81→  return `ETB ${amount.toLocaleString()}`;
    82→}
    83→
    84→function daysUntil(dateStr: string): number {
    85→  return Math.ceil((new Date(dateStr).getTime() - Date.now()) / 86400000);
    86→}
    87→
    88→function getBudgetBucket(avgBudget: number): string {
    89→  if (avgBudget < 500_000) return '<500K';
    90→  if (avgBudget < 1_000_000) return '500K-1M';
    91→  if (avgBudget < 5_000_000) return '1M-5M';
    92→  if (avgBudget < 10_000_000) return '5M-10M';
    93→  return '>10M';
    94→}

   265→// ─── Main Component ─────────────────────────────────────────────────
   266→export function TenderAnalyzerView() {
   267→  const { user } = useAuthStore();
   268→  const { setView } = useNavStore();
   269→
   270→  const [tenders, setTenders] = useState<Tender[]>([]);
   271→  const [bids, setBids] = useState<Bid[]>([]);
   272→  const [loading, setLoading] = useState(true);
   273→  const [exporting, setExporting] = useState(false);
   274→
   275→  // ── Data Loading ──
   276→  const loadData = useCallback(async () => {
   277→    setLoading(true);
   278→    try {
   279→      const [tendersRes, bidsRes] = await Promise.all([
   280→        api.get('/tenders', { limit: '100' }),
   281→        api.get('/bids'),
   282→      ]);
   283→      if (tendersRes.success) setTenders(tendersRes.data);
   284→      if (bidsRes.success) setBids(bidsRes.data);
   285→    } catch {
   286→      toast.error('Failed to load analyzer data');
   287→    }
   288→    setLoading(false);
   289→  }, []);
   290→
   291→  useEffect(() => {
   292→    loadData();
   293→  }, [loadData]);
   294→
   295→  // ── Computed Stats ──
   296→  const stats = useMemo(() => {
   297→    const total = tenders.length;
   298→    const open = tenders.filter(t => t.status === 'open').length;
   299→    const closed = tenders.filter(t => t.status === 'closed').length;
   300→    const awarded = tenders.filter(t => t.status === 'awarded').length;
   301→    const cancelled = tenders.filter(t => t.status === 'cancelled').length;
   302→
   303→    const avgBudgetMin = total > 0 ? tenders.reduce((s, t) => s + t.budgetMin, 0) / total : 0;
   304→    const avgBudgetMax = total > 0 ? tenders.reduce((s, t) => s + t.budgetMax, 0) / total : 0;
   305→
   306→    const totalBidVolume = bids.length;
   307→
   308→    return { total, open, closed, awarded, cancelled, avgBudgetMin, avgBudgetMax, totalBidVolume };
   309→  }, [tenders, bids]);
   310→
   311→  // ── Category Distribution Data ──
   312→  const categoryData = useMemo(() => {
   313→    const catMap: Record<string, number> = {};
   314→    tenders.forEach(t => {
   315→      if (!t.categoryTags) return;
   316→      t.categoryTags.split(',').map(c => c.trim()).filter(Boolean).forEach(cat => {
   317→        catMap[cat] = (catMap[cat] || 0) + 1;
   318→      });
   319→    });
   320→    return Object.entries(catMap)
   321→      .map(([name, count]) => ({ name, count }))
   322→      .sort((a, b) => b.count - a.count);
   323→  }, [tenders]);
   324→
   325→  // ── Budget Distribution Data ──
   326→  const budgetData = useMemo(() => {
   327→    const buckets: Record<string, number> = {
   328→      '<500K': 0,
   329→      '500K-1M': 0,
   330→      '1M-5M': 0,
   331→      '5M-10M': 0,
   332→      '>10M': 0,
   333→    };
   334→    tenders.forEach(t => {
   335→      const avg = (t.budgetMin + t.budgetMax) / 2;
   336→      const bucket = getBudgetBucket(avg);
   337→      buckets[bucket]++;
   338→    });
   339→    const order = ['<500K', '500K-1M', '1M-5M', '5M-10M', '>10M'];
   340→    return order.map(name => ({
   341→      name,
   342→      count: buckets[name],
   343→      fill: BUDGET_COLORS[name],
   344→    }));
   345→  }, [tenders]);
   346→
   347→  // ── Timeline Urgency Data ──
   348→  const timelineData = useMemo(() => {
   349→    const openTenders = tenders.filter(t => t.status === 'open');
   350→    const now = Date.now();
   351→
   352→    const within7 = openTenders.filter(t => {
   353→      const d = daysUntil(t.deadline);
   354→      return d >= 0 && d <= 7;
   355→    }).length;
   356→
   357→    const within14 = openTenders.filter(t => {
   358→      const d = daysUntil(t.deadline);
   359→      return d > 7 && d <= 14;
   360→    }).length;
   361→
   362→    const within30 = openTenders.filter(t => {
   363→      const d = daysUntil(t.deadline);
   364→      return d > 14 && d <= 30;
   365→    }).length;
   366→
   367→    const beyond30 = openTenders.filter(t => {
   368→      const d = daysUntil(t.deadline);
   369→      return d > 30;
   370→    }).length;
   371→
   372→    const overdue = openTenders.filter(t => new Date(t.deadline).getTime() < now).length;
   373→
   374→    return {
   375→      within7, within14, within30, beyond30, overdue,
   376→      totalOpen: openTenders.length,
   377→      chartData: [
   378→        { name: '≤ 7 days', count: within7, fill: CHART_COLORS.rose },
   379→        { name: '8-14 days', count: within14, fill: CHART_COLORS.amber },
   380→        { name: '15-30 days', count: within30, fill: CHART_COLORS.teal },
   381→        { name: '30+ days', count: beyond30, fill: CHART_COLORS.emerald },
   382→      ],
   383→    };
   384→  }, [tenders]);
   385→
   386→  // ── Bid Competition Data ──
   387→  const competitionData = useMemo(() => {
   388→    const tendersWithBids = tenders
   389→      .map(t => ({
   390→        ...t,
   391→        bidCount: t._count?.bids ?? 0,
   392→      }))
   393→      .filter(t => t.bidCount > 0)
   394→      .sort((a, b) => b.bidCount - a.bidCount);
   395→
   396→    const maxBids = tendersWithBids.length > 0 ? tendersWithBids[0].bidCount : 0;
   397→
   398→    return { tendersWithBids, maxBids };
   399→  }, [tenders]);
   400→
   401→  // ── XLSX Export ──
   402→  const handleExport = async () => {
   403→    setExporting(true);
   404→    try {
   405→      const token = localStorage.getItem('tenet_token');
   406→      const params = new URLSearchParams();
   407→      if (user?.role === 'team_admin') params.set('ownerOnly', 'true');
   408→      const res = await fetch(`/api/tenders/export?${params.toString()}`, {
   409→        headers: { Authorization: `Bearer ${token}` },
   410→      });
   411→      if (res.ok) {
   412→        const blob = await res.blob();
   413→        const url = window.URL.createObjectURL(blob);
   414→        const a = document.createElement('a');
   415→        a.href = url;
   416→        a.download = `tenet-tenders-${new Date().toISOString().slice(0, 10)}.xlsx`;
   417→        a.click();
   418→        window.URL.revokeObjectURL(url);
   419→        toast.success('Excel report downloaded successfully!');
   420→      } else {
   421→        toast.error('Failed to export report');
   422→      }
   423→    } catch {
   424→      toast.error('Export failed');
   425→    }
   426→    setExporting(false);
   427→  };
   428→
   429→  // ── Loading State ──
   430→  if (loading) {
   431→    return <AnalyzerSkeleton />;
   432→  }
   433→
   434→  return (
   435→    <div className="p-4 md:p-6 space-y-6 max-w-7xl mx-auto view-enter">
   436→      {/* ── 1. Header ──────────────────────────────────────────── */}
   437→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   438→        <div className="flex items-center gap-4">
   439→          <div className="p-3 rounded-2xl gradient-emerald flex-shrink-0">
   440→            <BarChart3 className="h-6 w-6 text-white" />
   441→          </div>
   442→          <div>
   443→            <h2 className="text-2xl md:text-3xl font-bold tracking-tight">
   444→              <span className="text-gradient-emerald">Tender</span> Analyzer
   445→            </h2>
   446→            <p className="text-muted-foreground text-sm mt-0.5">Insights, trends & competitive intelligence</p>
   447→          </div>
   448→        </div>
   449→        <Button
   450→          className="gradient-emerald hover:opacity-90 text-white rounded-xl px-5 premium-shadow transition-all hover:-translate-y-0.5"
   451→          onClick={handleExport}
   452→          disabled={exporting}
   453→        >
   454→          {exporting ? (
   455→            <Loader2 className="h-4 w-4 mr-2 animate-spin" />
   456→          ) : (
   457→            <Download className="h-4 w-4 mr-2" />
   458→          )}
   459→          {exporting ? 'Exporting...' : 'Export XLSX'}
   460→        </Button>
   461→      </div>
   462→
   463→      {/* ── 2. Stats Cards Row ──────────────────────────────────── */}
   464→      <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-5 gap-4">
   465→        <AnalyzerStatCard
   466→          icon={FileSearch}
   467→          label="Total Tenders"
   468→          value={stats.total}
   469→          subtext="All time"
   470→          gradientClass="gradient-emerald"
   471→        />
   472→        <AnalyzerStatCard
   473→          icon={CheckCircle2}
   474→          label="Open"
   475→          value={stats.open}
   476→          subtext={`${stats.total > 0 ? Math.round((stats.open / stats.total) * 100) : 0}% of total`}
   477→          gradientClass="gradient-teal"
   478→        />
   479→        <AnalyzerStatCard
   480→          icon={XCircle}
   481→          label="Closed"
   482→          value={stats.closed}
   483→          subtext="Submission ended"
   484→          gradientClass="gradient-amber"
   485→        />
   486→        <AnalyzerStatCard
   487→          icon={Target}
   488→          label="Awarded"
   489→          value={stats.awarded}
   490→          subtext="Contract assigned"
   491→          gradientClass="gradient-rose"
   492→        />
   493→        <AnalyzerStatCard
   494→          icon={Ban}
   495→          label="Cancelled"
   496→          value={stats.cancelled}
   497→          subtext="Withdrawn"
   498→          gradientClass="bg-gradient-to-br from-gray-400 to-gray-500"
   499→        />
   500→      </div>
   501→
   502→      {/* ── 3. Avg Budget + Total Bids Summary Cards ───────────── */}
   503→      <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   504→        <Card className="premium-shadow rounded-xl border-0 bg-white">
   505→          <CardContent className="p-5 flex items-center gap-4">
   506→            <div className="p-3 rounded-xl gradient-emerald flex-shrink-0">
   507→              <DollarSign className="h-5 w-5 text-white" />
   508→            </div>
   509→            <div>
   510→              <p className="text-sm text-muted-foreground font-medium">Average Budget Range</p>
   511→              <p className="text-xl font-bold mt-0.5">
   512→                {formatETB(stats.avgBudgetMin)} – {formatETB(stats.avgBudgetMax)}
   513→              </p>
   514→            </div>
   515→          </CardContent>
   516→        </Card>
   517→        <Card className="premium-shadow rounded-xl border-0 bg-white">
   518→          <CardContent className="p-5 flex items-center gap-4">
   519→            <div className="p-3 rounded-xl gradient-amber flex-shrink-0">
   520→              <Gavel className="h-5 w-5 text-white" />
   521→            </div>
   522→            <div>
