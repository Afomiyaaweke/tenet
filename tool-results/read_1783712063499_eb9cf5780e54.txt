    25→function daysUntil(dateStr: string): number {
    26→  return Math.ceil((new Date(dateStr).getTime() - Date.now()) / (1000 * 60 * 60 * 24));
    27→}
    28→
    29→function deadlineColor(days: number) {
    30→  if (days <= 0) return 'rose';
    31→  if (days <= 3) return 'rose';
    32→  if (days <= 7) return 'amber';
    33→  return 'emerald';
    34→}
    35→
    36→function deadlineBg(days: number) {
    37→  const c = deadlineColor(days);
    38→  const map: Record<string, string> = {
    39→    rose: 'bg-rose-50 text-rose-700',
    40→    amber: 'bg-amber-50 text-amber-700',
    41→    emerald: 'bg-emerald-50 text-emerald-700',
    42→  };
    43→  return map[c] || map.emerald;
    44→}
