   295→function fmtMoney(amount: number, currency: string): string {
   296→  if (!amount) return '—';
   297→  try {
   298→    return new Intl.NumberFormat('en-US', {
   299→      style: 'currency',
   300→      currency: currency || 'USD',
   301→      maximumFractionDigits: 0,
   302→    }).format(amount);
   303→  } catch {
   304→    return `${currency} ${amount.toLocaleString()}`;
   305→  }
   306→}
   307→
   308→function daysUntil(dateStr: string): number {
   309→  if (!dateStr) return 999;
   310→  return Math.ceil((new Date(dateStr).getTime() - Date.now()) / (1000 * 60 * 60 * 24));
   311→}
   312→
   313→function deadlineBadge(days: number): string {
   314→  if (days <= 0) return 'bg-rose-50 text-rose-700 dark:bg-rose-950/40 dark:text-rose-300';
   315→  if (days <= 7) return 'bg-amber-50 text-amber-700 dark:bg-amber-950/40 dark:text-amber-300';
   316→  if (days <= 21) return 'bg-sky-50 text-sky-700 dark:bg-sky-950/40 dark:text-sky-300';
   317→  return 'bg-emerald-50 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300';
   318→}
   319→
   320→function deadlineLabel(days: number): string {
   321→  if (days <= 0) return 'Closed';
   322→  if (days <= 7) return `${days}d left`;
   323→  if (days <= 30) return `${days}d left`;
   324→  return `${days}d left`;
   325→}
   326→
   327→function relativeTime(dateStr: string): string {
   328→  if (!dateStr) return '';
   329→  const diff = Date.now() - new Date(dateStr).getTime();
   330→  const day = 86400000;
   331→  if (diff < day) return 'today';
   332→  if (diff < 2 * day) return 'yesterday';
   333→  if (diff < 30 * day) return `${Math.floor(diff / day)}d ago`;
   334→  return new Date(dateStr).toLocaleDateString();
   335→}
   336→
   337→function riskBadgeColor(level: string): string {
   338→  if (level === 'low') return 'bg-emerald-50 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300';
   339→  if (level === 'medium') return 'bg-amber-50 text-amber-700 dark:bg-amber-950/40 dark:text-amber-300';
   340→  return 'bg-rose-50 text-rose-700 dark:bg-rose-950/40 dark:text-rose-300';
   341→}
   342→
   343→function readinessColor(score: number): string {
   344→  if (score >= 8) return 'bg-emerald-500';
   345→  if (score >= 5) return 'bg-amber-500';
   346→  return 'bg-rose-500';
   347→}

   349→/* ─────────────────────────────────────────────────────────────────────
   350→ * Inline Document Viewer Component
   351→ * ───────────────────────────────────────────────────────────────────── */
   352→
   353→function InlineDocumentViewer({ doc, onClose }: { doc: InlineDocument; onClose: () => void }) {
   354→  const copyContent = () => {
   355→    const text = doc.sections
   356→      ? doc.sections.map((s) => `${s.heading}\n${s.content}`).join('\n\n')
   357→      : doc.content;
   358→    navigator.clipboard.writeText(text);
   359→    toast.success('Content copied to clipboard');
   360→  };
   361→
   362→  return (
   363→    <div className="overflow-hidden animate-[fadeIn_0.3s_ease-out]">
   364→      <div className="border-t border-border bg-gradient-to-b from-muted/30 to-background">
   365→        <div className="p-4 md:p-6 space-y-4">
   366→          {/* Document header */}
   367→          <div className="flex items-start justify-between gap-3">
   368→            <div className="min-w-0 flex-1">
   369→              <div className="flex items-center gap-2 text-xs text-muted-foreground mb-1">
   370→                <BookOpen className="h-3.5 w-3.5" />
   371→                <span>Document loaded from {new URL(doc.url).hostname}</span>
   372→                <span>·</span>
   373→                <span>{new Date(doc.fetchedAt).toLocaleTimeString()}</span>
   374→              </div>
   375→              <h3 className="text-lg font-semibold text-foreground leading-snug">
   376→                {doc.title || 'Tender Document'}
   377→              </h3>
   378→              {doc.metaDescription && (
   379→                <p className="text-sm text-muted-foreground mt-1 line-clamp-2">
   380→                  {doc.metaDescription}
   381→                </p>
   382→              )}
   383→            </div>
   384→            <div className="flex items-center gap-2 shrink-0">
   385→              <Button variant="ghost" size="sm" className="gap-1.5 text-xs" onClick={copyContent}>
   386→                <Copy className="h-3.5 w-3.5" />
   387→                Copy
   388→              </Button>
   389→              <a
   390→                href={doc.url}
   391→                target="_blank"
   392→                rel="noopener noreferrer"
   393→                className="inline-flex items-center gap-1.5 text-xs font-medium text-primary hover:underline"
   394→              >
   395→                <ExternalLink className="h-3.5 w-3.5" />
   396→                Open original
   397→              </a>
   398→              <Button variant="ghost" size="sm" onClick={onClose} className="gap-1.5 text-xs text-muted-foreground">
   399→                <ChevronUp className="h-3.5 w-3.5" />
   400→                Collapse
   401→              </Button>
   402→            </div>
   403→          </div>
   404→
   405→          {/* Extracted metadata pills */}
   406→          {(doc.deadlines && doc.deadlines.length > 0) || (doc.budgets && doc.budgets.length > 0) ? (
   407→            <div className="flex flex-wrap gap-2">
   408→              {doc.deadlines?.map((d, i) => (
   409→                <Badge key={`dl-${i}`} variant="outline" className="gap-1.5 text-xs border-amber-300 text-amber-700 dark:border-amber-800 dark:text-amber-300">
   410→                  <Calendar className="h-3 w-3" />
   411→                  {d}
   412→                </Badge>
   413→              ))}
   414→              {doc.budgets?.map((b, i) => (
   415→                <Badge key={`bg-${i}`} variant="outline" className="gap-1.5 text-xs border-emerald-300 text-emerald-700 dark:border-emerald-800 dark:text-emerald-300">
   416→                  <DollarSign className="h-3 w-3" />
   417→                  {b}
   418→                </Badge>
   419→              ))}
   420→            </div>
   421→          ) : null}
   422→
   423→          {/* Sections view */}
   424→          {doc.sections && doc.sections.length > 0 ? (
   425→            <div className="max-h-80 overflow-y-auto rounded-lg border border-border bg-background p-4 space-y-4 scrollbar-thin">
   426→              {doc.sections.map((section, i) => (
   427→                <div key={i}>
   428→                  <h4 className="text-sm font-semibold text-foreground mb-1.5">{section.heading}</h4>
   429→                  <p className="text-sm text-muted-foreground leading-relaxed whitespace-pre-line">
   430→                    {section.content}
   431→                  </p>
   432→                  {i < doc.sections!.length - 1 && <Separator className="mt-4 bg-border" />}
   433→                </div>
   434→              ))}
   435→            </div>
   436→          ) : (
   437→            <div className="max-h-80 overflow-y-auto rounded-lg border border-border bg-background p-4 scrollbar-thin">
   438→              <p className="text-sm text-muted-foreground leading-relaxed whitespace-pre-line">
   439→                {doc.content || 'No content could be extracted from this page.'}
   440→              </p>
   441→            </div>
   442→          )}
   443→        </div>
   444→      </div>
   445→    </div>
   446→  );
   447→}

   731→export function LiveTendersView() {
   732→  const [tenders, setTenders] = useState<LiveTender[]>([]);
   733→  const [dataSources, setDataSources] = useState<DataSource[]>([]);
   734→  const [sourceMeta, setSourceMeta] = useState<
   735→    { id: string; name: string; live: boolean; ok: boolean; count: number; error?: string }[]
   736→  >([]);
   737→  const [loading, setLoading] = useState(true);
   738→  const [refreshing, setRefreshing] = useState(false);
   739→  const [fallback, setFallback] = useState(false);
   740→  const [search, setSearch] = useState('');
   741→  const [sourceFilter, setSourceFilter] = useState('all');
   742→  const [sectorFilter, setSectorFilter] = useState('');
   743→  const [sectorCounts, setSectorCounts] = useState<{ id: string; label: string; count: number }[]>([]);
   744→
   745→  // Pagination state
   746→  const [visibleCount, setVisibleCount] = useState(20);
   747→  const PAGE_SIZE = 20;
   748→
   749→  // Inline document state
   750→  const [expandedId, setExpandedId] = useState<string | null>(null);
   751→  const [docLoading, setDocLoading] = useState<string | null>(null);
   752→  const [docData, setDocData] = useState<Record<string, InlineDocument>>({});
   753→  const [docError, setDocError] = useState<Record<string, string>>({});
   754→
   755→  // AI Review state
   756→  const [aiReviewData, setAiReviewData] = useState<Record<string, AIReview>>({});
   757→  const [aiReviewLoading, setAiReviewLoading] = useState<Record<string, boolean>>({});
   758→  const [aiReviewExpanded, setAiReviewExpanded] = useState<Record<string, boolean>>({});
   759→
   760→  // Saved/bookmark state
   761→  const [savedTenders, setSavedTenders] = useState<Record<string, boolean>>({});
   762→  const [savingTender, setSavingTender] = useState<Record<string, boolean>>({});
   763→
   764→  // Import state
   765→  const [importing, setImporting] = useState<string | null>(null);
   766→
   767→  // Credential dialog state
   768→  const [credDialogSource, setCredDialogSource] = useState<GatedSource | null>(null);
   769→  const [credDialogOpen, setCredDialogOpen] = useState(false);
   770→
   771→  const load = useCallback(
   772→    async (isRefresh = false) => {
   773→      if (isRefresh) setRefreshing(true);
   774→      else setLoading(true);
   775→      const params: Record<string, string> = { rows: '500' };
   776→      if (search) params.search = search;
   777→      if (sourceFilter && sourceFilter !== 'all') params.source = sourceFilter;
   778→      if (sectorFilter) params.sector = sectorFilter;
   779→      const res = await api.get('/tenders/live', params);
   780→      if (res.success) {
   781→        setTenders(res.data as LiveTender[]);
   782→        setSourceMeta(res.meta?.sources || []);
   783→        setFallback(Boolean(res.meta?.fallback));
   784→        if (Array.isArray(res.meta?.dataSources)) setDataSources(res.meta.dataSources);
   785→        if (Array.isArray(res.meta?.sectors)) setSectorCounts(res.meta.sectors);
   786→      } else {
   787→        toast.error(res.error || 'Failed to load live tenders');
   788→      }
   789→      setLoading(false);
   790→      setRefreshing(false);
   791→    },
   792→    [search, sourceFilter, sectorFilter],
   793→  );
   794→
   795→  useEffect(() => {
   796→    const t = setTimeout(() => load(), 250);
   797→    return () => clearTimeout(t);
   798→  }, [load]);
   799→
   800→  // Check saved status for visible tenders
   801→  useEffect(() => {
   802→    const checkSaved = async () => {
   803→      const visibleTenders = tenders.slice(0, visibleCount);
   804→      for (const t of visibleTenders) {
   805→        if (savedTenders[t.id] !== undefined) continue;
   806→        try {
   807→          const res = await api.get('/tenders/saved/check', {
   808→            tenderId: t.externalId || t.id,
   809→            source: t.source,
   810→          });
   811→          if (res.success) {
   812→            setSavedTenders((prev) => ({ ...prev, [t.id]: res.saved }));
   813→          }
   814→        } catch {
   815→          // Ignore errors for saved check
   816→        }
   817→      }
   818→    };
   819→    if (tenders.length > 0) checkSaved();
   820→  }, [tenders, visibleCount, savedTenders]);
   821→
   822→  const liveSourcesCount = useMemo(
   823→    () => sourceMeta.filter((s) => s.live && s.ok).length,
   824→    [sourceMeta],
   825→  );
   826→
   827→  const totalTenderValue = useMemo(() => {

   832→  const loadDocument = useCallback(async (tender: LiveTender) => {

   835→      setExpandedId(null);

   841→      setExpandedId(id);

   852→        setDocData((prev) => ({ ...prev, [id]: res.data as InlineDocument }));

   854→        setDocError((prev) => ({ ...prev, [id]: res.error || 'Failed to load document' }));

   856→    } catch {
   857→      setDocError((prev) => ({ ...prev, [id]: 'Network error' }));

   879→        scope: tender.scope,
   880→        budgetMin: tender.budgetMin,
   881→        budgetMax: tender.budgetMax,
   882→        deadline: tender.deadline,
   883→        location: tender.location,
   884→        categoryTags: tender.categoryTags,
   885→        source: tender.source,
   886→        currency: tender.currency,
   887→        borrower: tender.borrower,
   888→        contractType: tender.contractType,
   889→        region: tender.region,
   890→      });

   892→        setAiReviewData((prev) => ({ ...prev, [id]: res.data as AIReview }));

   896→      }
   897→    } catch {
   898→      toast.error('Failed to generate AI review');
   899→      setAiReviewExpanded((prev) => ({ ...prev, [id]: false }));

   915→          source: tender.source,

   925→          tenderId: tender.externalId || tender.id,
   926→          source: tender.source,
   927→          title: tender.title,
   928→          scope: tender.scope,
   929→          budgetMin: tender.budgetMin,
   930→          budgetMax: tender.budgetMax,
   931→          deadline: tender.deadline,
   932→          location: tender.location,
   933→          categoryTags: tender.categoryTags,
   934→          externalUrl: tender.externalUrl,
   935→          currency: tender.currency,
   936→        });

   938→        toast.success('Tender saved to your bids list', { description: `"${tender.title}" has been bookmarked.` });

   940→    } catch {
   941→      toast.error(isCurrentlySaved ? 'Failed to remove from saved' : 'Failed to save tender');
   942→    }
   943→    setSavingTender((prev) => ({ ...prev, [id]: false }));

   952→        scope: tender.scope,
   953→        budgetMin: tender.budgetMin,
   954→        budgetMax: tender.budgetMax,

   960→        currency: tender.currency,
