    25→function formatETB(amount: number): string {
    26→  if (amount >= 1_000_000) return `ETB ${(amount / 1_000_000).toFixed(1)}M`;
    27→  if (amount >= 1_000) return `ETB ${(amount / 1_000).toFixed(0)}K`;
    28→  return `ETB ${amount.toLocaleString()}`;
    29→}

    41→function statusColorMap(status: string) {
    42→  switch (status) {
    43→    case 'active': return { bg: 'bg-emerald-100 dark:bg-emerald-900/30', text: 'text-emerald-700 dark:text-emerald-300', dot: 'bg-emerald-500', bar: '#10b981', barBg: 'bg-emerald-100 dark:bg-emerald-900/30' };
    44→    case 'completed': return { bg: 'bg-teal-100 dark:bg-teal-900/30', text: 'text-teal-700 dark:text-teal-300', dot: 'bg-teal-500', bar: '#14b8a6', barBg: 'bg-teal-100 dark:bg-teal-900/30' };
    45→    case 'on_hold': return { bg: 'bg-amber-100 dark:bg-amber-900/30', text: 'text-amber-700 dark:text-amber-300', dot: 'bg-amber-500', bar: '#f59e0b', barBg: 'bg-amber-100 dark:bg-amber-900/30' };
    46→    case 'cancelled': return { bg: 'bg-rose-100 dark:bg-rose-900/30', text: 'text-rose-700 dark:text-rose-300', dot: 'bg-rose-500', bar: '#f43f5e', barBg: 'bg-rose-100 dark:bg-rose-900/30' };
    47→    default: return { bg: 'bg-muted', text: 'text-muted-foreground', dot: 'bg-muted-foreground/50', bar: '#94a3b8', barBg: 'bg-muted' };
    48→  }

    51→function computeProgress(project: Project) {
    52→  const tasks = project.tasks || [];
    53→  if (tasks.length === 0) return 0;
    54→  const done = tasks.filter(t => t.status === 'done').length;
    55→  return Math.round((done / tasks.length) * 100);
    56→}
    57→
    58→function getNextMilestone(project: Project) {
    59→  return (project.milestones || [])
    60→    .filter(m => !m.completed)
    61→    .sort((a, b) => new Date(a.dueDate).getTime() - new Date(b.dueDate).getTime())[0];
    62→}

    65→function getTimelineRange(projects: Project[]) {
    66→  const now = new Date();
    67→  const start = new Date(now.getFullYear(), now.getMonth() - 2, 1);
    68→  const end = new Date(now.getFullYear(), now.getMonth() + 6, 0);
    69→  return { start, end };
    70→}

    72→function monthRange(start: Date, end: Date): Date[] {
    73→  const months: Date[] = [];
    74→  const d = new Date(start.getFullYear(), start.getMonth(), 1);
    75→  while (d <= end) {
    76→    months.push(new Date(d));
    77→    d.setMonth(d.getMonth() + 1);
    78→  }
    79→  return months;
    80→}
    81→
    82→function dateToX(date: Date, start: Date, end: Date, width: number): number {
    83→  const total = end.getTime() - start.getTime();
    84→  if (total === 0) return 0;
    85→  return ((date.getTime() - start.getTime()) / total) * width;
    86→}

   199→function SortIcon({ col, sortKey, sortDir }: { col: SortKey; sortKey: SortKey; sortDir: SortDir }) {
   200→  if (sortKey !== col) return <ArrowUpDown className="h-3 w-3 ml-1 text-muted-foreground/40" />;
   201→  return sortDir === 'asc'
   202→    ? <ArrowUp className="h-3 w-3 ml-1 text-emerald-500" />
   203→    : <ArrowDown className="h-3 w-3 ml-1 text-emerald-500" />;

   206→// ─── List View ───────────────────────────────────────────────────────
   207→function ListView({
   208→  projects,
   209→  onRowClick,
   210→  sortKey,
   211→  sortDir,
   212→  onSort,
   213→}: {
   214→  projects: Project[];
   215→  onRowClick: (id: string) => void;
   216→  sortKey: SortKey;
   217→  sortDir: SortDir;
   218→  onSort: (key: SortKey) => void;
   219→}) {
   220→  const sorted = useMemo(() => {
   221→    const copy = [...projects];
   222→    copy.sort((a, b) => {
   223→      let cmp = 0;
   224→      switch (sortKey) {
   225→        case 'title':
   226→          cmp = (a.tender?.title || '').localeCompare(b.tender?.title || '');
   227→          break;
   228→        case 'contractValue':
   229→          cmp = a.contractValue - b.contractValue;
   230→          break;
   231→        case 'status':
   232→          cmp = a.status.localeCompare(b.status);
   233→          break;
   234→        case 'progress':
   235→          cmp = computeProgress(a) - computeProgress(b);
   236→          break;
   237→      }
   238→      return sortDir === 'asc' ? cmp : -cmp;
   239→    });
   240→    return copy;
   241→  }, [projects, sortKey, sortDir]);
   242→
   243→  return (
   244→    <Card className="rounded-xl border border-border/60 bg-card shadow-sm overflow-hidden">
   245→      <Table>
   246→        <TableHeader>
   247→          <TableRow className="hover:bg-transparent">
   248→            <TableHead className="cursor-pointer select-none" onClick={() => onSort('title')}>
   249→              <span className="flex items-center">Project <SortIcon col="title" sortKey={sortKey} sortDir={sortDir} /></span>
   250→            </TableHead>
   251→            <TableHead className="hidden md:table-cell">Tender</TableHead>
   252→            <TableHead className="hidden sm:table-cell">Contractor</TableHead>
   253→            <TableHead className="cursor-pointer select-none" onClick={() => onSort('contractValue')}>

   257→              <span className="flex items-center">Status <SortIcon col="status" sortKey={sortKey} sortDir={sortDir} /></span>
   258→            </TableHead>
   259→            <TableHead className="cursor-pointer select-none" onClick={() => onSort('progress')}>
   260→              <span className="flex items-center">Progress <SortIcon col="progress" sortKey={sortKey} sortDir={sortDir} /></span>
   261→            </TableHead>
   262→            <TableHead className="hidden lg:table-cell">Next Milestone</TableHead>
   263→          </TableRow>
   264→        </TableHeader>
   265→        <TableBody>
   266→          {sorted.map(project => {
   267→            const sc = statusColorMap(project.status);
   268→            const progress = computeProgress(project);
   269→            const totalTasks = project.tasks?.length || 0;
   270→            const doneTasks = project.tasks?.filter(t => t.status === 'done').length || 0;
   271→            const nextMilestone = getNextMilestone(project);

   275→                key={project.id}
   276→                className="cursor-pointer hover:bg-emerald-50/50 dark:hover:bg-emerald-900/10 transition-colors"
   277→                onClick={() => onRowClick(project.id)}
   278→              >
   279→                <TableCell>
   280→                  <div className="flex items-center gap-2">
   281→                    <FolderKanban className="h-4 w-4 text-emerald-500 flex-shrink-0" />
   282→                    <span className="font-medium text-sm truncate max-w-[180px]">
   283→                      {project.tender?.title || 'Project'}
   284→                    </span>
   285→                  </div>
   286→                </TableCell>
   287→                <TableCell className="hidden md:table-cell text-xs text-muted-foreground truncate max-w-[150px]">
   288→                  {project.tender?.title || '-'}
   289→                </TableCell>
   290→                <TableCell className="hidden sm:table-cell text-xs text-muted-foreground truncate max-w-[120px]">
   291→                  {project.bid?.user?.profile?.fullName || '-'}
   292→                </TableCell>
   293→                <TableCell className="text-sm font-semibold">{formatETB(project.contractValue)}</TableCell>
   294→                <TableCell>
   295→                  <Badge className={`text-[10px] px-2 py-0 border-0 font-semibold ${sc.bg} ${sc.text}`}>

   302→                    <div className={`h-1.5 flex-1 ${sc.barBg} rounded-full overflow-hidden`}>
   303→                      <div className="h-full rounded-full" style={{ width: `${progress}%`, backgroundColor: sc.bar }} />
   304→                    </div>
   305→                    <span className="text-[10px] font-medium text-muted-foreground whitespace-nowrap">
   306→                      {doneTasks}/{totalTasks}

   329→// ─── Timeline View (SVG Gantt) ───────────────────────────────────────
   330→function TimelineView({ projects, onBarClick }: { projects: Project[]; onBarClick: (id: string) => void }) {
   331→  const timelineWidth = 900;
   332→  const rowHeight = 44;
   333→  const headerHeight = 36;
   334→  const { start, end } = useMemo(() => getTimelineRange(projects), [projects]);
   335→  const months = useMemo(() => monthRange(start, end), [start, end]);
   336→
   337→  const barData = useMemo(() => {
   338→    return projects.map(p => {
   339→      const createdAt = new Date(p.createdAt);
   340→      // Estimate end: use last milestone or +3 months
   341→      const lastMilestone = (p.milestones || []).sort((a, b) => new Date(b.dueDate).getTime() - new Date(a.dueDate).getTime())[0];
   342→      const endDate = lastMilestone
   343→        ? new Date(lastMilestone.dueDate)
   344→        : new Date(createdAt.getTime() + 90 * 86400000);
   345→      const sc = statusColorMap(p.status);
   346→      return { project: p, start: createdAt, end: endDate, color: sc.bar };
   347→    });
   348→  }, [projects]);
   349→
   350→  const milestoneMarkers = useMemo(() => {
   351→    const markers: { x: number; y: number; color: string; completed: boolean; title: string; projectId: string }[] = [];
   352→    barData.forEach((bd, rowIdx) => {
   353→      (bd.project.milestones || []).forEach(m => {
   354→        const mx = dateToX(new Date(m.dueDate), start, end, timelineWidth);
   355→        markers.push({
   356→          x: mx, y: headerHeight + rowIdx * rowHeight + rowHeight / 2,
   357→          color: bd.color, completed: m.completed, title: m.title, projectId: bd.project.id,
   358→        });
   359→      });
   360→    });
   361→    return markers;
   362→  }, [barData, start, end]);
   363→
   364→  const nowX = dateToX(new Date(), start, end, timelineWidth);
   365→  const svgHeight = headerHeight + barData.length * rowHeight + 20;
   366→
   367→  return (
   368→    <Card className="rounded-xl border border-border/60 bg-card shadow-sm overflow-hidden">
   369→      <div className="overflow-x-auto">
   370→        <svg width={timelineWidth + 200} height={svgHeight} className="min-w-full">
   371→          {/* Month columns */}
   372→          {months.map((m, i) => {
   373→            const x = dateToX(m, start, end, timelineWidth) + 200;
   374→            const nextM = new Date(m);
   375→            nextM.setMonth(nextM.getMonth() + 1);
   376→            const x2 = dateToX(nextM, start, end, timelineWidth) + 200;
   377→            return (
   378→              <g key={i}>
   379→                <rect x={x} y={0} width={x2 - x} height={svgHeight} fill={i % 2 === 0 ? 'transparent' : 'rgba(128,128,128,0.03)'} />
   380→                <text x={x + (x2 - x) / 2} y={22} textAnchor="middle" className="text-[10px] fill-muted-foreground font-medium">
   381→                  {m.toLocaleDateString('en-US', { month: 'short', year: '2-digit' })}
   382→                </text>
   383→                <line x1={x} y1={headerHeight} x2={x} y2={svgHeight} stroke="currentColor" strokeOpacity={0.06} />
   384→              </g>
   385→            );
   386→          })}
   387→
   388→          {/* Today line */}
   389→          <line x1={nowX + 200} y1={0} x2={nowX + 200} y2={svgHeight} stroke="#f43f5e" strokeWidth={1.5} strokeDasharray="4 3" />
   390→          <text x={nowX + 200} y={12} textAnchor="middle" className="text-[9px] fill-rose-500 font-semibold">TODAY</text>
   391→
   392→          {/* Project label column */}
   393→          {barData.map((bd, i) => {
   394→            const y = headerHeight + i * rowHeight + rowHeight / 2;
   395→            return (
   396→              <g key={bd.project.id}>
   397→                <text x={8} y={y + 4} className="text-[10px] fill-foreground font-medium cursor-pointer" onClick={() => onBarClick(bd.project.id)}>
   398→                  {(bd.project.tender?.title || 'Project').slice(0, 22)}
   399→                </text>
   400→              </g>
   401→            );
   402→          })}
   403→
   404→          {/* Project bars */}
   405→          {barData.map((bd, i) => {
   406→            const x1 = dateToX(bd.start, start, end, timelineWidth) + 200;
   407→            const x2 = dateToX(bd.end, start, end, timelineWidth) + 200;
   408→            const y = headerHeight + i * rowHeight + 12;
   409→            const barH = 20;
   410→            const barW = Math.max(x2 - x1, 8);
   411→            return (
   412→              <g key={bd.project.id} className="cursor-pointer" onClick={() => onBarClick(bd.project.id)}>
   413→                <rect x={x1} y={y} width={barW} height={barH} rx={4} fill={bd.color} fillOpacity={0.2} />
   414→                <rect x={x1} y={y} width={barW} height={barH} rx={4} fill={bd.color} fillOpacity={0.7} />
   415→                <rect x={x1} y={y} width={barW * (computeProgress(bd.project) / 100)} height={barH} rx={4} fill={bd.color} />
   416→              </g>
   417→            );
   418→          })}
   419→
   420→          {/* Milestone diamonds */}
   421→          {milestoneMarkers.map((m, i) => (
   422→            <g key={i}>
   423→              <polygon
   424→                points={`${m.x + 200},${m.y - 5} ${m.x + 205},${m.y} ${m.x + 200},${m.y + 5} ${m.x + 195},${m.y}`}
   425→                fill={m.completed ? '#10b981' : m.color}
   426→                stroke="white"
   427→                strokeWidth={1}
   428→              />
   429→            </g>
   430→          ))}
   431→        </svg>
   432→      </div>
   433→    </Card>
   434→  );
   435→}
