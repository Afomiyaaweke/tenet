   130→function formatAIResultToHTML(data: Record<string, unknown>): string {
   131→  let html = '';
   132→  for (const [key, value] of Object.entries(data)) {
   133→    const title = key.replace(/([A-Z])/g, ' $1').replace(/^./, s => s.toUpperCase());
   134→    const content = typeof value === 'string' ? value : JSON.stringify(value, null, 2);
   135→    const formattedContent = content
   136→      .replace(/&/g, '&amp;')
   137→      .replace(/</g, '&lt;')
   138→      .replace(/>/g, '&gt;')
   139→      .replace(/\*\*(.+?)\*\*/g, '<strong>$1</strong>')
   140→      .replace(/^[-*•]\s+(.*)/gm, '<li>$1</li>')
   141→      .replace(/^(\d+)[.)]\s+(.*)/gm, '<li>$2</li>')
   142→      .replace(/^#{1,3}\s+(.*)/gm, '<strong>$1</strong>')
   143→      .replace(/\n\n/g, '</p><p>')
   144→      .replace(/\n/g, '<br/>');
   145→    html += `<h2 style="font-size:16px;font-weight:700;margin:16px 0 8px;color:#059669;border-bottom:1px solid #d1fae5;padding-bottom:4px;">${title}</h2>`;
   146→    html += `<p>${formattedContent}</p>`;
   147→  }
   148→  return html;
   149→}

   151→function getFileExtBadge(fileName: string): { color: string; label: string } {
   152→  const ext = fileName.toLowerCase().split('.').pop() || '';
   153→  const map: Record<string, { color: string; label: string }> = {
   154→    pdf: { color: 'bg-rose-100 text-rose-700', label: 'PDF' },
   155→    docx: { color: 'bg-blue-100 text-blue-700', label: 'DOCX' },
   156→    doc: { color: 'bg-blue-100 text-blue-700', label: 'DOC' },
   157→    txt: { color: 'bg-gray-100 text-gray-700', label: 'TXT' },
   158→    jpg: { color: 'bg-amber-100 text-amber-700', label: 'JPG' },
   159→    jpeg: { color: 'bg-amber-100 text-amber-700', label: 'JPEG' },
   160→    png: { color: 'bg-emerald-100 text-emerald-700', label: 'PNG' },
   161→  };
   162→  return map[ext] || { color: 'bg-gray-100 text-gray-600', label: ext.toUpperCase() };
   163→}

   171→function SkillTagSelector({ selected, onChange }: { selected: string[]; onChange: (s: string[]) => void }) {
   172→  const toggle = (skill: string) => {
   173→    if (selected.includes(skill)) onChange(selected.filter(s => s !== skill));
   174→    else onChange([...selected, skill]);
   175→  };
   176→  return (
   177→    <div className="flex flex-wrap gap-1">
   178→      {COMMON_SKILLS.map(skill => (
   179→        <button key={skill} type="button" onClick={() => toggle(skill)}
   180→          className={`px-2 py-0.5 rounded-full text-[10px] font-medium transition-all border ${
   181→            selected.includes(skill)
   182→              ? 'bg-emerald-50 text-emerald-700 border-emerald-300'
   183→              : 'bg-muted/50 text-muted-foreground border-border hover:border-emerald-300'
   184→          }`}>
   185→          {skill}
   186→        </button>
   187→      ))}
   188→    </div>
   189→  );
   190→}

   511→    try {
   512→      const res = await api.post(`/document-ocr/${docId}`);
   513→      if (res.success) {
   514→        toast.success('OCR processing started');
   515→        // Poll for completion
   516→        pollOCRStatus(docId);
   517→      } else {
   518→        toast.error(res.error || 'OCR failed');
   519→        setOcrStatusMap(prev => ({ ...prev, [docId]: { loading: false } }));
   520→      }
   521→    } catch {
   522→      toast.error('OCR request failed');
   523→      setOcrStatusMap(prev => ({ ...prev, [docId]: { loading: false } }));
   524→    }
   525→  };

   527→  const pollOCRStatus = async (docId: string) => {
   528→    let attempts = 0;
   529→    const maxAttempts = 30;
   530→    const poll = async () => {
   531→      try {
   532→        const res = await api.get(`/document-ocr/${docId}`);
   533→        if (res.success && res.data) {
   534→          const status = res.data.ocrStatus;
   535→          if (status === 'completed') {
   536→            setOcrStatusMap(prev => ({ ...prev, [docId]: { loading: false, text: res.data.ocrText } }));
   537→            toast.success('OCR completed');
   538→            // Refresh document list
   539→            loadDocuments();
   540→            return;
   541→          } else if (status === 'failed') {
   542→            setOcrStatusMap(prev => ({ ...prev, [docId]: { loading: false } }));
   543→            toast.error('OCR failed');
   544→            return;
   545→          }
   546→        }
   547→        attempts++;
   548→        if (attempts < maxAttempts) {
   549→          setTimeout(poll, 2000);
   550→        } else {
   551→          setOcrStatusMap(prev => ({ ...prev, [docId]: { loading: false } }));
   552→          toast.info('OCR is taking longer than expected. Check back later.');
   553→        }
   554→      } catch {
   555→        attempts++;
   556→        if (attempts < maxAttempts) {
   557→          setTimeout(poll, 3000);
   558→        }
   559→      }
   560→    };
   561→    setTimeout(poll, 2000);
   562→  };
