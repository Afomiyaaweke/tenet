    44→function formatETB(amount: number): string {
    45→  if (amount >= 1_000_000) return `ETB ${(amount / 1_000_000).toFixed(1)}M`;
    46→  if (amount >= 1_000) return `ETB ${(amount / 1_000).toFixed(0)}K`;
    47→  return `ETB ${amount.toLocaleString()}`;
    48→}

    50→function statusColorMap(status: string) {
    51→  switch (status) {
    52→    case 'active': return { bg: 'bg-emerald-100 dark:bg-emerald-900/30', text: 'text-emerald-700 dark:text-emerald-300', dot: 'bg-emerald-500', bar: '#10b981', label: 'Active' };
    53→    case 'completed': return { bg: 'bg-teal-100 dark:bg-teal-900/30', text: 'text-teal-700 dark:text-teal-300', dot: 'bg-teal-500', bar: '#14b8a6', label: 'Completed' };
    54→    case 'on_hold': return { bg: 'bg-amber-100 dark:bg-amber-900/30', text: 'text-amber-700 dark:text-amber-300', dot: 'bg-amber-500', bar: '#f59e0b', label: 'On Hold' };
    55→    case 'cancelled': return { bg: 'bg-rose-100 dark:bg-rose-900/30', text: 'text-rose-700 dark:text-rose-300', dot: 'bg-rose-500', bar: '#f43f5e', label: 'Cancelled' };
    56→    default: return { bg: 'bg-muted', text: 'text-muted-foreground', dot: 'bg-muted-foreground/50', bar: '#94a3b8', label: status };
    57→  }

    60→// ─── Timeline (Gantt) helpers ────────────────────────────────────────
    61→function getTimelineRange(project: Project) {
    62→  const now = new Date();
    63→  const start = new Date(now.getFullYear(), now.getMonth() - 2, 1);
    64→  const end = new Date(now.getFullYear(), now.getMonth() + 8, 0);
    65→  return { start, end };
    66→}
    67→
    68→function monthRange(start: Date, end: Date): Date[] {
    69→  const months: Date[] = [];
    70→  const d = new Date(start.getFullYear(), start.getMonth(), 1);
    71→  while (d <= end) {
    72→    months.push(new Date(d));
    73→    d.setMonth(d.getMonth() + 1);
    74→  }
    75→  return months;
    76→}
    77→
    78→function dateToX(date: Date, start: Date, end: Date, width: number): number {
    79→  const total = end.getTime() - start.getTime();
    80→  if (total === 0) return 0;
    81→  return ((date.getTime() - start.getTime()) / total) * width;
    82→}

   215→    setLoading(false);

   239→      toast.success('Task updated');
   240→      loadProject();
   241→    } else toast.error(res.error || 'Failed to update task');

   243→

   261→    if (res.success) {
   262→      toast.success('Task updated');
   263→      setShowEditTask(false);
   264→      setEditingTask(null);
   265→      loadProject();
   266→    } else toast.error(res.error || 'Failed to update task');
   267→  };
   268→
   269→  // ── Milestone Handlers ──
   270→  const handleAddMilestone = async () => {
   271→    if (!newMilestone.title.trim()) { toast.error('Milestone title is required'); return; }
   272→    const res = await api.post(`/projects/${projectId}/milestones`, newMilestone);
   273→    if (res.success) {
   274→      toast.success('Milestone added');
   275→      setShowAddMilestone(false);

   871→// ─── Project Timeline Sub-Component ──────────────────────────────────
   872→function ProjectTimeline({
   873→  project,
   874→  onToggleMilestone,
   875→}: {
   876→  project: Project;
   877→  onToggleMilestone: (id: string, completed: boolean) => void;
   878→}) {
   879→  const timelineWidth = 800;
   880→  const headerHeight = 36;
   881→  const milestones = project.milestones || [];
   882→  const rowHeight = 44;
   883→
   884→  const { start, end } = useMemo(() => getTimelineRange(project), [project]);
   885→  const months = useMemo(() => monthRange(start, end), [start, end]);
   886→
   887→  const createdAt = new Date(project.createdAt);
   888→  const lastMilestone = milestones.sort((a, b) => new Date(b.dueDate).getTime() - new Date(a.dueDate).getTime())[0];
   889→  const projectEnd = lastMilestone ? new Date(lastMilestone.dueDate) : new Date(createdAt.getTime() + 90 * 86400000);
   890→
   891→  const sc = statusColorMap(project.status);
   892→  const taskProgress = (project.tasks || []).length > 0
   893→    ? Math.round((project.tasks!.filter(t => t.status === 'done').length / project.tasks!.length) * 100)
   894→    : 0;
   895→
   896→  const nowX = dateToX(new Date(), start, end, timelineWidth);
   897→
   898→  const milestoneData = milestones.map(m => ({
   899→    ...m,
   900→    x: dateToX(new Date(m.dueDate), start, end, timelineWidth),
   901→  }));
   902→
   903→  const svgHeight = headerHeight + rowHeight + 20;
   904→
   905→  return (
   906→    <div className="space-y-3">
   907→      <Card className="rounded-xl border border-border/60 bg-card overflow-hidden">
   908→        <div className="overflow-x-auto">
   909→          <svg width={timelineWidth + 180} height={svgHeight} className="min-w-full">
   910→            {/* Month columns */}
   911→            {months.map((m, i) => {
   912→              const x = dateToX(m, start, end, timelineWidth) + 180;
   913→              const nextM = new Date(m);
   914→              nextM.setMonth(nextM.getMonth() + 1);
   915→              const x2 = dateToX(nextM, start, end, timelineWidth) + 180;
   916→              return (
   917→                <g key={i}>
   918→                  <rect x={x} y={0} width={x2 - x} height={svgHeight} fill={i % 2 === 0 ? 'transparent' : 'rgba(128,128,128,0.03)'} />
   919→                  <text x={x + (x2 - x) / 2} y={22} textAnchor="middle" className="text-[10px] fill-muted-foreground font-medium">
   920→                    {m.toLocaleDateString('en-US', { month: 'short', year: '2-digit' })}
   921→                  </text>
   922→                  <line x1={x} y1={headerHeight} x2={x} y2={svgHeight} stroke="currentColor" strokeOpacity={0.06} />
   923→                </g>
   924→              );
   925→            })}
   926→
   927→            {/* Today line */}
   928→            <line x1={nowX + 180} y1={0} x2={nowX + 180} y2={svgHeight} stroke="#f43f5e" strokeWidth={1.5} strokeDasharray="4 3" />
   929→            <text x={nowX + 180} y={12} textAnchor="middle" className="text-[9px] fill-rose-500 font-semibold">TODAY</text>
   930→
   931→            {/* Project bar */}
   932→            {(() => {
   933→              const x1 = dateToX(createdAt, start, end, timelineWidth) + 180;
   934→              const x2 = dateToX(projectEnd, start, end, timelineWidth) + 180;
