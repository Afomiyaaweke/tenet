   168→   168→function relativeTime(dateStr: string): string {
   169→   169→  const now = Date.now();
   170→   170→  const then = new Date(dateStr).getTime();
   171→   171→  const diff = Math.max(0, now - then);
   172→   172→  const seconds = Math.floor(diff / 1000);
   173→   173→  if (seconds < 60) return 'just now';
   174→   174→  const minutes = Math.floor(seconds / 60);
   175→   175→  if (minutes < 60) return `${minutes}m ago`;
   176→   176→  const hours = Math.floor(minutes / 60);
   177→   177→  if (hours < 24) return `${hours}h ago`;
   178→   178→  const days = Math.floor(hours / 24);
   179→   179→  if (days < 7) return `${days}d ago`;
   180→   180→  return new Date(dateStr).toLocaleDateString();
   181→   181→}
