    34→function daysUntil(dateStr: string): number {
    35→  return Math.ceil((new Date(dateStr).getTime() - Date.now()) / (1000 * 60 * 60 * 24));
    36→}
    37→
    38→function deadlineColor(days: number) {
    39→  if (days <= 0) return 'rose';
    40→  if (days <= 3) return 'rose';
    41→  if (days <= 7) return 'amber';
    42→  return 'emerald';
    43→}
    44→
    45→function deadlineBg(days: number) {
    46→  const c = deadlineColor(days);
    47→  return `bg-${c}-50 text-${c}-700`;
    48→}
