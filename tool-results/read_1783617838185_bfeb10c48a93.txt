   116→              {isAdminOrOwner ? 'Manage and evaluate submitted bids' : 'Track your bid submissions'}

   218→          </Card>

   221→        <div className="animate-[fadeIn_0.3s_ease-out]">

   229→            </CardContent>
   230→          </Card>
   231→        </div>

   233→        <div className="space-y-4 animate-[fadeIn_0.3s_ease-out]">
   234→          {filteredBids.slice(0, visibleCount).map(bid => {
   235→              const sInfo = statusIcon(bid.status);
   236→              const SIcon = sInfo.icon;
   237→              const isExpanded = expandedId === bid.id;
   238→              const companyName = bid.user?.company?.name;
   239→              const jobTitle = bid.user?.profile?.jobTitle;
   240→
   241→              return (
   242→                <div className="hover:-translate-y-[2px] transition-all duration-200"
   243→ key={bid.id}
   244→ >
   245→                  <Card className="premium-shadow rounded-xl border-0 bg-card overflow-hidden">
   246→                    {/* Status accent strip */}
   247→                    <div className={`h-1 ${
   248→                      bid.status === 'pending_review' ? 'bg-gradient-to-r from-amber-400 to-amber-500' :
   249→                      bid.status === 'shortlisted' ? 'bg-gradient-to-r from-teal-400 to-teal-600' :
   250→                      bid.status === 'awarded' ? 'bg-gradient-to-r from-emerald-400 to-emerald-600' :

   265→                            <div className="min-w-0">
   266→                              <p className="font-semibold text-sm truncate hover:text-emerald-700 transition-colors">

   274→                                      {jobTitle && <span className="text-muted-foreground/60">&middot; {jobTitle}</span>}
   275→                                      {companyName && (
   276→                                        <span className="inline-flex items-center gap-0.5">
   277→                                          <span className="text-muted-foreground/60">&middot;</span>
   278→                                          <Building2 className="h-3 w-3" /> {companyName}
   279→                                        </span>
   280→                                      )}
   281→                                      <span className="text-muted-foreground/60">&middot; {new Date(bid.createdAt).toLocaleDateString()}</span>
   282→                                    </span>
   283→                                  )
   284→                                  : (
   285→                                    <span className="flex items-center gap-1.5 flex-wrap">
   286→                                      <span>{bid.user?.profile?.fullName || 'Contractor'}</span>
   287→                                      {companyName && (
   288→                                        <span className="inline-flex items-center gap-0.5">
   289→                                          <span className="text-muted-foreground/60">&middot;</span>
   290→                                          <Building2 className="h-3 w-3" /> {companyName}
   291→                                        </span>
   292→                                      )}
   293→                                      <span className="text-muted-foreground/60">&middot; {new Date(bid.createdAt).toLocaleDateString()}</span>
   294→                                    </span>
   295→                                  )
   296→                                }
   297→                              </p>

   299→                          </div>

   301→                            {/* Financial badge */}
