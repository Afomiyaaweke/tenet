   224→/* ─────────────────────────────────────────────────────────────────────

   227→
   228→function parseAmount(value: number | string | undefined): number {
   229→  if (value === undefined || value === null) return 0;
   230→  if (typeof value === 'number') return value;
   231→  const n = Number(String(value).replace(/[^0-9.-]/g, ''));
   232→  return Number.isFinite(n) ? n : 0;
   233→}
   234→
   235→function truncate(text: string, max = 280): string {
   236→  if (!text) return '';
   237→  return text.length > max ? text.slice(0, max).trimEnd() + '…' : text;
   238→}

   241→ * World Bank adapter (live, public, no auth)

   243→
   244→interface WorldBankProcurementRow {
   245→  project_id?: string;
   246→  procurement_group_id?: string;
   247→  project_name?: string;
   248→  borrower?: string;
   249→  country?: string;
   250→  region?: string;
   251→  contract_description?: string;
   252→  contract_type?: string;
   253→  supplier?: string;
   254→  supplier_country?: string | string[];
   255→  contract_signing_date?: string;
   256→  total_contract_amount?: number | string;
   257→  wb_contract_number?: string;
   258→}
   259→
   260→interface WorldBankResponse {
   261→  total?: number;
   262→  rows?: WorldBankProcurementRow[];
   263→  procurements?: WorldBankProcurementRow[];
   264→  [k: string]: unknown;
   265→}
   266→
   267→function inferCategory(row: WorldBankProcurementRow): string {
   268→  const t = (row.contract_type || '').toLowerCase();
   269→  if (t.includes('work') || t.includes('civil') || t.includes('construction')) return 'Construction';
   270→  if (t.includes('consult')) return 'Consulting';
   271→  if (t.includes('good') || t.includes('supply') || t.includes('equipment')) return 'Supply';
   272→  if (t.includes('service')) return 'Consulting';
   273→  const p = (row.project_name || '').toLowerCase();
   274→  if (p.includes('road') || p.includes('bridge') || p.includes('infrastructure')) return 'Construction';
   275→  if (p.includes('agric') || p.includes('farm') || p.includes('irrigat')) return 'Agriculture';
   276→  if (p.includes('health') || p.includes('clinic') || p.includes('hospital')) return 'Healthcare';
   277→  if (p.includes('energy') || p.includes('power') || p.includes('electric')) return 'Engineering';
   278→  if (p.includes('edu') || p.includes('school')) return 'Education';
   279→  if (p.includes('water') || p.includes('sanitation')) return 'Engineering';
   280→  return 'Consulting';
   281→}
   282→
   283→function buildDeadline(signingDate?: string): string {
   284→  if (!signingDate) return new Date(Date.now() + 45 * 86400000).toISOString();
   285→  const base = new Date(signingDate);
   286→  if (Number.isNaN(base.getTime())) return new Date(Date.now() + 45 * 86400000).toISOString();
   287→  return new Date(base.getTime() + 45 * 86400000).toISOString();
   288→}

   290→export async function fetchWorldBankTenders(opts: {
   291→  search?: string;
   292→  rows?: number;
   293→}): Promise<{ tenders: LiveTender[]; total: number; ok: boolean }> {
   294→  const rows = Math.min(Math.max(opts.rows ?? 20, 1), 500);
   295→  const params = new URLSearchParams({
   296→    format: 'json',
   297→    rows: String(rows),
   298→    fl: [
   299→      'project_id',
   300→      'procurement_group_id',
   301→      'project_name',
   302→      'borrower',
   303→      'country',
   304→      'region',
   305→      'contract_description',
   306→      'contract_type',
   307→      'supplier',
   308→      'supplier_country',
   309→      'contract_signing_date',
   310→      'total_contract_amount',
   311→      'wb_contract_number',
   312→    ].join(','),
   313→  });
   314→  if (opts.search) params.set('q', opts.search);
   315→
   316→  const url = `https://search.worldbank.org/api/v2/procurement?${params.toString()}`;
   317→  const ctrl = new AbortController();
   318→  const timer = setTimeout(() => ctrl.abort(), 8000);
   319→
   320→  try {
   321→    const res = await fetch(url, {
   322→      signal: ctrl.signal,
   323→      headers: { Accept: 'application/json', 'User-Agent': 'Tenet-Tender-Ecosystem/1.0' },
   324→      cache: 'no-store',
   325→    });
   326→    clearTimeout(timer);
   327→    if (!res.ok) {
   328→      return { tenders: [], total: 0, ok: false };
   329→    }
   330→    const json = (await res.json()) as WorldBankResponse;
   331→    const rowsArr = json.rows || json.procurements || [];
   332→    const total = typeof json.total === 'number' ? json.total : rowsArr.length;
   333→
   334→    const tenders: LiveTender[] = rowsArr
   335→      .filter((r) => r && (r.contract_description || r.project_name))
   336→      .map((r, idx) => {
   337→        const externalId = r.procurement_group_id || r.project_id || `wb-${idx}`;
   338→        const title = truncate(r.contract_description || r.project_name || 'World Bank Contract Award', 160);
   339→        const amount = parseAmount(r.total_contract_amount);
   340→        const signingDate = r.contract_signing_date;
   341→        return {
   342→          id: `worldbank-${externalId}`,
   343→          title,
   344→          scope: truncate(r.contract_description || r.project_name || '', 400),
   345→          budgetMin: amount,
   346→          budgetMax: amount,
   347→          deadline: buildDeadline(signingDate),
   348→          location: r.country || r.region || 'Multiple',
   349→          categoryTags: inferCategory(r),
   350→          requiredDocs: r.project_id
   351→            ? `https://projects.worldbank.org/en/projects-operations/project-detail/${r.project_id}`
   352→            : 'https://financesone.worldbank.org',
   353→          status: 'awarded',
   354→          createdBy: 'worldbank',
   355→          createdAt: signingDate || new Date().toISOString(),
   356→          updatedAt: signingDate || new Date().toISOString(),
   357→          source: 'worldbank',
   358→          externalId,
   359→          externalUrl: r.project_id
   360→            ? `https://projects.worldbank.org/en/projects-operations/project-detail/${r.project_id}`
   361→            : 'https://financesone.worldbank.org',
   362→          currency: 'USD',
   363→          borrower: r.borrower || undefined,
   364→          supplier: r.supplier || undefined,
   365→          contractType: r.contract_type || undefined,
   366→          signingDate: signingDate || undefined,
   367→          region: r.region || undefined,
   368→        } satisfies LiveTender;
   369→      });
   370→
   371→    return { tenders, total, ok: true };
   372→  } catch {
   373→    clearTimeout(timer);
   374→    return { tenders: [], total: 0, ok: false };
   375→  }
   376→}

   379→ * EU TED adapter (live, public RSS-style feed)

   381→
   382→export async function fetchEuTedTenders(opts: {
   383→  search?: string;
   384→  rows?: number;
   385→}): Promise<{ tenders: LiveTender[]; total: number; ok: boolean }> {
   386→  const rows = Math.min(Math.max(opts.rows ?? 20, 1), 500);
   387→  const base = 'https://api.ted.europa.eu/v3/notices/search';
   388→  const body = {
   389→    query: opts.search ? `(${opts.search})` : '*',
   390→    fields: ['ND', 'TI_DOC', 'DS_DATE_PUB', 'DEADLINE_DATE', 'PLACE_CONTRACT', 'VALUE_CONTRACT', 'CA_NAME', 'TITLE'],
   391→    limit: rows,
   392→    offset: 0,
   393→    pagination: { pageNumber: 1, pageSize: rows },
   394→    sort: { field: 'DS_DATE_PUB', order: 'DESC' },
   395→  };
   396→  const ctrl = new AbortController();
   397→  const timer = setTimeout(() => ctrl.abort(), 8000);
   398→
   399→  try {
   400→    const res = await fetch(base, {
   401→      method: 'POST',
   402→      signal: ctrl.signal,
   403→      headers: { 'Content-Type': 'application/json', Accept: 'application/json' },
   404→      body: JSON.stringify(body),
   405→      cache: 'no-store',
   406→    });
   407→    clearTimeout(timer);
   408→    if (!res.ok) return { tenders: [], total: 0, ok: false };
   409→    const json = (await res.json()) as { notices?: unknown[]; total?: number };
   410→    const notices = Array.isArray(json.notices) ? json.notices : [];
   411→    const total = typeof json.total === 'number' ? json.total : notices.length;
   412→
   413→    const tenders: LiveTender[] = notices.map((n, idx) => {
   414→      const row = (n || {}) as Record<string, unknown>;
   415→      const nd = String(row.ND || row['ND'] || `ted-${idx}`);
   416→      const title = String(row.TI_DOC || row.TITLE || 'EU TED Notice');
   417→      const deadlineRaw = row.DEADLINE_DATE as string | undefined;
   418→      const amount = Number(row.VALUE_CONTRACT) || 0;
   419→      return {
   420→        id: `eu_ted-${nd}`,
   421→        title: truncate(title, 160),
   422→        scope: truncate(title, 400),
   423→        budgetMin: amount,
   424→        budgetMax: amount,
   425→        deadline: deadlineRaw || new Date(Date.now() + 30 * 86400000).toISOString(),
   426→        location: String(row.PLACE_CONTRACT || 'EU'),
   427→        categoryTags: 'Consulting',
   428→        requiredDocs: `https://ted.europa.eu/udl?uri=TED:NOTICE:${nd}:TEXT:EN:HTML`,
   429→        status: 'open',
   430→        createdBy: 'eu_ted',
   431→        createdAt: String(row.DS_DATE_PUB || new Date().toISOString()),
   432→        updatedAt: String(row.DS_DATE_PUB || new Date().toISOString()),
   433→        source: 'eu_ted',
   434→        externalId: nd,
   435→        externalUrl: `https://ted.europa.eu/udl?uri=TED:NOTICE:${nd}:TEXT:EN:HTML`,
   436→        currency: 'EUR',
   437→        borrower: String(row.CA_NAME || '') || undefined,
   438→      } satisfies LiveTender;
   439→    });
   440→
   441→    return { tenders, total, ok: true };
   442→  } catch {
   443→    clearTimeout(timer);
   444→    return { tenders: [], total: 0, ok: false };
   445→  }
   446→}

   448→/* ─────────────────────────────────────────────────────────────────────
   449→ * UNGM — UN Global Procurement adapter (live via RSS)
   450→ * ───────────────────────────────────────────────────────────────────── */
   451→
   452→export async function fetchUngmTenders(opts: {
   453→  search?: string;
   454→  rows?: number;
   455→}): Promise<{ tenders: LiveTender[]; total: number; ok: boolean }> {
   456→  const ctrl = new AbortController();
   457→  const timer = setTimeout(() => ctrl.abort(), 8000);
   458→
   459→  try {
   460→    const res = await fetch('https://www.ungm.org/Public/Notice/RSS', {
   461→      signal: ctrl.signal,
   462→      headers: { 'User-Agent': 'Tenet-Tender-Ecosystem/1.0', Accept: 'application/rss+xml, text/xml, */*' },
   463→      cache: 'no-store',
   464→    });
   465→    clearTimeout(timer);
   466→
   467→    if (!res.ok) return { tenders: [], total: 0, ok: false };
   468→
   469→    const xml = await res.text();
   470→    const itemRegex = /<item[\s\S]*?<title>([\s\S]*?)<\/title>[\s\S]*?<link>([\s\S]*?)<\/link>[\s\S]*?<description>([\s\S]*?)<\/description>[\s\S]*?<\/item>/gi;
   471→    const matches = [...xml.matchAll(itemRegex)];
   472→
   473→    if (matches.length === 0) return { tenders: [], total: 0, ok: false };
   474→
   475→    const tenders: LiveTender[] = matches.slice(0, opts.rows ?? 10).map((m, idx) => {
   476→      const rawTitle = m[1]?.replace(/<!\[CDATA\[([\s\S]*?)\]\]>/, '$1').trim() || `UNGM Notice ${idx + 1}`;
   477→      const rawLink = m[2]?.trim() || 'https://www.ungm.org';
   478→      const rawDesc = m[3]?.replace(/<!\[CDATA\[([\s\S]*?)\]\]>/, '$1').trim() || '';
   479→      return {
   480→        id: `ungm-rss-${idx}`,
   481→        title: truncate(rawTitle, 160),
   482→        scope: truncate(rawDesc, 400),
   483→        budgetMin: 0,
   484→        budgetMax: 0,
   485→        deadline: new Date(Date.now() + 30 * 86400000).toISOString(),
   486→        location: 'International',
   487→        categoryTags: 'Consulting',
   488→        requiredDocs: rawLink,
   489→        status: 'open',
   490→        createdBy: 'ungm',
   491→        createdAt: new Date().toISOString(),
   492→        updatedAt: new Date().toISOString(),
   493→        source: 'ungm',
   494→        externalId: `ungm-rss-${idx}`,
   495→        externalUrl: rawLink,
   496→        currency: 'USD',
   497→        region: 'Global',
   498→      } satisfies LiveTender;
   499→    });
   500→
   501→    return { tenders, total: matches.length, ok: true };
   502→  } catch {
   503→    clearTimeout(timer);
   504→    return { tenders: [], total: 0, ok: false };
   505→  }
   506→}

   512→export async function fetchSamGovTenders(opts: {

   516→  const rows = Math.min(Math.max(opts.rows ?? 10, 1), 50);
   517→  const params = new URLSearchParams({
   518→    limit: String(rows),
   519→    offset: '0',
   520→    postedFrom: new Date(Date.now() - 30 * 86400000).toISOString().split('T')[0],
   521→    postedTo: new Date().toISOString().split('T')[0],
   522→  });
   523→  if (opts.search) params.set('q', opts.search);
   524→
   525→  const url = `https://api.sam.gov/opportunities/v2/search?${params.toString()}`;
   526→  const ctrl = new AbortController();
   527→  const timer = setTimeout(() => ctrl.abort(), 8000);
   528→
