    95→/* ───────────────────────── Password Strength Meter ───────────────────────── */
    96→function getPasswordStrength(password: string): { score: number; label: string; color: string } {
    97→  let score = 0;
    98→  if (password.length >= 12) score += 20;
    99→  if (password.length >= 16) score += 10;
   100→  if (/[A-Z]/.test(password)) score += 15;
   101→  if (/[a-z]/.test(password)) score += 15;
   102→  if (/[0-9]/.test(password)) score += 15;
   103→  if (/[^A-Za-z0-9]/.test(password)) score += 15;
   104→  if (new Set(password.toLowerCase()).size >= 8) score += 10;
   105→
   106→  if (score >= 80) return { score, label: 'Strong', color: 'bg-green-500' };
   107→  if (score >= 60) return { score, label: 'Good', color: 'bg-blue-500' };
   108→  if (score >= 40) return { score, label: 'Fair', color: 'bg-yellow-500' };
   109→  return { score, label: 'Weak', color: 'bg-red-500' };
   110→}
   111→
   112→function PasswordStrengthMeter({ password }: { password: string }) {
   113→  if (!password) return null;
   114→  const { score, label, color } = getPasswordStrength(password);
   115→  return (
   116→    <div className="space-y-1.5 mt-2">
   117→      <div className="flex gap-1">
   118→        {[25, 50, 75, 100].map(threshold => (
   119→          <div key={threshold} className="h-1.5 flex-1 rounded-full bg-muted">
   120→            <div
   121→              className={`h-full rounded-full transition-all duration-300 ${score >= threshold ? color : 'bg-transparent'}`}
   122→              style={{ width: score >= threshold ? '100%' : '0%' }}
   123→            />
   124→          </div>
   125→        ))}
   126→      </div>
   127→      <div className="flex justify-between items-center">
   128→        <span className="text-xs text-muted-foreground">Strength:</span>
   129→        <span className={`text-xs font-medium ${score >= 80 ? 'text-green-600' : score >= 60 ? 'text-blue-600' : score >= 40 ? 'text-yellow-600' : 'text-red-600'}`}>{label}</span>
   130→      </div>
   131→    </div>
   132→  );
   133→}

   135→function PasswordRequirements({ password }: { password: string }) {
   136→  const checks = [
   137→    { label: 'At least 12 characters', met: password.length >= 12 },
   138→    { label: 'One uppercase letter', met: /[A-Z]/.test(password) },
   139→    { label: 'One lowercase letter', met: /[a-z]/.test(password) },
   140→    { label: 'One number', met: /[0-9]/.test(password) },
   141→    { label: 'One special character', met: /[^A-Za-z0-9]/.test(password) },
   142→  ];
   143→  return (
   144→    <div className="space-y-1 mt-2">
   145→      {checks.map(({ label, met }) => (
   146→        <div key={label} className="flex items-center gap-1.5">
   147→          {met ? (
   148→            <CheckCircle2 className="w-3 h-3 text-green-500 flex-shrink-0" />
   149→          ) : (
   150→            <div className="w-3 h-3 rounded-full border border-muted-foreground/30 flex-shrink-0" />
   151→          )}
   152→          <span className={`text-xs ${met ? 'text-green-600' : 'text-muted-foreground'}`}>{label}</span>
   153→        </div>
   154→      ))}
   155→    </div>
   156→  );

   159→/* ───────────────────────── Step Indicator ───────────────────────── */
   160→function StepIndicator({ currentStep, totalSteps = 4 }: { currentStep: RegStep; totalSteps?: number }) {
   161→  return (
   162→    <div className="flex items-center justify-center gap-1 mb-6">
   163→      {Array.from({ length: totalSteps }, (_, i) => (i + 1) as RegStep).map((step, i) => {
   164→        const meta = REG_STEP_META[step];
   165→        const isCompleted = currentStep > step;
   166→        const isCurrent = currentStep === step;
   167→        const Icon = meta.icon;
   168→        return (
   169→          <React.Fragment key={step}>
   170→            <div className="flex items-center gap-1.5">
   171→              <div
   172→                className={`h-7 w-7 rounded-full flex items-center justify-center text-xs font-bold transition-all ${
   173→                  isCompleted || isCurrent
   174→                    ? 'gradient-orange text-white shadow-md shadow-orange-500/30'
   175→                    : 'bg-muted text-muted-foreground'
   176→                }`}
   177→              >
   178→                {isCompleted ? <CheckCircle2 className="w-4 h-4" /> : <Icon className="w-3.5 h-3.5" />}
   179→              </div>
   180→              <span className={`text-xs font-medium hidden sm:inline ${isCompleted || isCurrent ? 'text-foreground' : 'text-muted-foreground'}`}>
   181→                {meta.label}
   182→              </span>
   183→            </div>
   184→            {i < totalSteps - 1 && (
   185→              <div className={`h-0.5 w-4 sm:w-8 rounded-full ${currentStep > step ? 'bg-orange-500' : 'bg-muted'}`} />
   186→            )}
   187→          </React.Fragment>
   188→        );
   189→      })}
   190→    </div>
   191→  );
   192→}

   224→
   225→  // Handle ?token=xxx in URL for email reset links
   226→  useEffect(() => {
   227→    const params = new URLSearchParams(window.location.search);
   228→    const tokenFromUrl = params.get('token');
   229→    if (tokenFromUrl) {
   230→      setResetToken(tokenFromUrl);
   231→      setAuthMode('reset-password');

   246→    } catch {
   247→      toast.error('Login failed. Please check your connection and try again.');
   248→    }
   249→    setLoading(false);

   251→
   252→  const handleForgotPassword = async (e: React.FormEvent) => {

   255→    setLoading(true);
   256→    try {
   257→      const { api } = await import('@/lib/api');
   258→      const res = await api.post('/auth/forgot-password', { email: forgotEmail });
   259→      if (res.success) {
   260→        // Always show the confirmation screen to prevent email enumeration
   261→        // The reset token is sent ONLY to the user's email — never in the API response
   262→        setResetToken(''); // Clear any previous token
   263→        setAuthMode('forgot-sent');
   264→      } else {
   265→        toast.error(res.error || 'Failed to process request');
   266→      }
   267→    } catch {
   268→      toast.error('Network error. Please try again.');
   269→    }
   270→    setLoading(false);
   271→  };
   272→
   273→  const handleResetPassword = async (e: React.FormEvent) => {
   274→    e.preventDefault();

   276→    if (newPassword !== confirmPassword) {
   277→      toast.error('Passwords do not match');
   278→      return;
   279→    }
   280→    if (newPassword.length < 12) {
   281→      toast.error('Password must be at least 12 characters');
   282→      return;
   283→    }
   284→    if (!/[A-Z]/.test(newPassword) || !/[a-z]/.test(newPassword) || !/[0-9]/.test(newPassword) || !/[^A-Za-z0-9]/.test(newPassword)) {
   285→      toast.error('Password must include uppercase, lowercase, number, and special character');
   286→      return;
   287→    }
   288→    setLoading(true);
   289→    try {
   290→      const { api } = await import('@/lib/api');
   291→      const res = await api.post('/auth/reset-password', {
   292→        token: resetToken,
   293→        newPassword,
   294→        confirmPassword,
   295→      });

   297→        setResetSuccess(true);
   298→        toast.success('Password reset successfully! You can now sign in.');

   300→        toast.error(res.error || 'Failed to reset password');
