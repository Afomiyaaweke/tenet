    96→    96→/* ───────────────────────── Password Strength Meter ───────────────────────── */
    97→    97→function getPasswordStrength(password: string): { score: number; label: string; color: string } {
    98→    98→  let score = 0;
    99→    99→  if (password.length >= 8) score += 25;
   100→   100→  if (password.length >= 12) score += 10;
   101→   101→  if (/[A-Z]/.test(password)) score += 20;
   102→   102→  if (/[a-z]/.test(password)) score += 20;
   103→   103→  if (/[0-9]/.test(password)) score += 20;
   104→   104→  if (/[^A-Za-z0-9]/.test(password)) score += 5;  // bonus only
   105→   105→  if (new Set(password.toLowerCase()).size >= 8) score += 5;
   106→   106→
   107→   107→  if (score >= 70) return { score, label: 'Strong', color: 'bg-green-500' };
   108→   108→  if (score >= 50) return { score, label: 'Good', color: 'bg-blue-500' };
   109→   109→  if (score >= 30) return { score, label: 'Fair', color: 'bg-yellow-500' };
   110→   110→  return { score, label: 'Weak', color: 'bg-red-500' };
   111→   111→}
   112→   112→
   113→   113→function PasswordStrengthMeter({ password }: { password: string }) {
   114→   114→  if (!password) return null;
   115→   115→  const { score, label, color } = getPasswordStrength(password);
   116→   116→  return (
   117→   117→    <div className="space-y-1.5 mt-2">
   118→   118→      <div className="flex gap-1">
   119→   119→        {[25, 50, 75, 100].map(threshold => (
   120→   120→          <div key={threshold} className="h-1.5 flex-1 rounded-full bg-muted">
   121→   121→            <div
   122→   122→              className={`h-full rounded-full transition-all duration-300 ${score >= threshold ? color : 'bg-transparent'}`}
   123→   123→              style={{ width: score >= threshold ? '100%' : '0%' }}
   124→   124→            />
   125→   125→          </div>
   126→   126→        ))}
   127→   127→      </div>
   128→   128→      <div className="flex justify-between items-center">
   129→   129→        <span className="text-xs text-muted-foreground">Strength:</span>
   130→   130→        <span className={`text-xs font-medium ${score >= 80 ? 'text-green-600' : score >= 60 ? 'text-blue-600' : score >= 40 ? 'text-yellow-600' : 'text-red-600'}`}>{label}</span>
   131→   131→      </div>
   132→   132→    </div>
   133→   133→  );
   134→   134→}

   165→   165→        const isCompleted = currentStep > step;

   170→   170→            <div className="flex items-center gap-1.5">

   251→   251→      } catch {
   252→   252→        setTokenValidation('idle'); // Network error — don't block
   253→   253→      }
   254→   254→    }, 600); // 600ms debounce
   255→   255→
