   119→/* ───────────────────────── Password Strength Meter ───────────────────────── */
   120→function scorePassword(pw: string): { score: number; label: string; color: string } {
   121→  let score = 0;
   122→  if (pw.length >= 8) score++;
   123→  if (pw.length >= 12) score++;
   124→  if (/[a-z]/.test(pw) && /[A-Z]/.test(pw)) score++;
   125→  if (/[0-9]/.test(pw)) score++;
   126→  if (/[^A-Za-z0-9]/.test(pw)) score++;
   127→  const labels = ['Too weak', 'Weak', 'Fair', 'Good', 'Strong', 'Very strong'];
   128→  const colors = ['bg-rose-500', 'bg-rose-500', 'bg-amber-500', 'bg-yellow-500', 'bg-orange-500', 'bg-orange-600'];
   129→  return { score, label: labels[score], color: colors[score] };
   130→}

   132→/* ───────────────────────── Slide-to-Verify Captcha ───────────────────────── */
   133→function SlideCaptcha({ onVerified }: { onVerified: () => void }) {
   134→  const [offset, setOffset] = useState(0);
   135→  const [status, setStatus] = useState<'idle' | 'dragging' | 'success' | 'fail'>('idle');
   136→  const [target] = useState(() => 180 + Math.floor(Math.random() * 80));
   137→  const trackWidth = 320;
   138→  const handleWidth = 44;
   139→  const tolerance = 8;
   140→  const dragRef = React.useRef<{ start: number; startOffset: number } | null>(null);
   141→
   142→  const onPointerDown = (e: React.PointerEvent) => {
   143→    if (status === 'success') return;
   144→    setStatus('dragging');
   145→    dragRef.current = { start: e.clientX, startOffset: offset };
   146→    (e.target as HTMLElement).setPointerCapture(e.pointerId);
   147→  };
   148→
   149→  const onPointerMove = (e: React.PointerEvent) => {
   150→    if (!dragRef.current || status !== 'dragging') return;
   151→    const delta = e.clientX - dragRef.current.start;
   152→    const next = Math.max(0, Math.min(trackWidth - handleWidth, dragRef.current.startOffset + delta));
   153→    setOffset(next);
   154→  };
   155→
   156→  const onPointerUp = () => {
   157→    if (status !== 'dragging') return;
   158→    dragRef.current = null;
   159→    if (Math.abs(offset - target) <= tolerance) {
   160→      setStatus('success');
   161→      setTimeout(() => onVerified(), 450);
   162→    } else {
   163→      setStatus('fail');
   164→      setTimeout(() => {
   165→        setStatus('idle');
   166→        setOffset(0);
   167→      }, 700);
   168→    }
   169→  };
   170→
   171→  const pieceColor = status === 'success' ? 'bg-orange-500' : status === 'fail' ? 'bg-rose-500' : 'bg-primary';
   172→  const ringColor = status === 'success' ? 'ring-orange-300' : status === 'fail' ? 'ring-rose-300' : 'ring-primary/40';
   173→
   174→  return (
   175→    <div className="space-y-3">
   176→      <div className="text-xs font-medium text-muted-foreground flex items-center gap-1.5">
   177→        <Puzzle className="w-3.5 h-3.5" /> Security verification — slide to complete
   178→      </div>
   179→      {/* Puzzle canvas */}
   180→      <div className="relative h-12 rounded-xl overflow-hidden border border-border bg-muted/40 select-none">
   181→        {/* pattern background */}
   182→        <div
   183→          className="absolute inset-0 opacity-50"
   184→          style={{
   185→            backgroundImage:
   186→              'repeating-linear-gradient(45deg, hsl(var(--muted-foreground) / 0.08) 0 6px, transparent 6px 12px), repeating-linear-gradient(-45deg, hsl(var(--muted-foreground) / 0.06) 0 6px, transparent 6px 12px)',
   187→          }}
   188→        />
   189→        {/* target cutout (notch) */}
   190→        <div
   191→          className="absolute top-1/2 -translate-y-1/2 h-9 w-9 rounded-lg border-2 border-dashed border-muted-foreground/40 bg-background/60"
   192→          style={{ left: target + 4 }}
   193→        />
   194→        {/* moving puzzle piece */}
   195→        <div
   196→          className={`absolute top-1/2 -translate-y-1/2 h-9 w-9 rounded-lg ${pieceColor} shadow-lg ring-2 ${ringColor} flex items-center justify-center text-white transition-shadow ${status === 'fail' ? 'animate-[shake_0.4s_ease-in-out]' : ''}`}
   197→          style={{ left: 4 + offset, transition: status === 'dragging' ? 'none' : 'left 0.25s ease' }}
   198→        >
   199→          {status === 'success' ? <CheckCircle2 className="w-5 h-5" /> : status === 'fail' ? <XCircle className="w-5 h-5" /> : <Puzzle className="w-4 h-4" />}
   200→        </div>
   201→        {/* hint text */}
   202→        {status === 'idle' && (
   203→          <span className="absolute inset-0 flex items-center justify-center text-xs text-muted-foreground pointer-events-none">
   204→            Drag the slider below to fit the puzzle
   205→          </span>
   206→        )}
   207→        {status === 'success' && (
   208→          <span className="absolute inset-0 flex items-center justify-center text-xs font-medium text-orange-600 bg-orange-50/60 pointer-events-none">
   209→            Verified
   210→          </span>
   211→        )}
   212→      </div>
   213→      {/* Slider track */}
   214→      <div className="relative h-11 rounded-xl bg-muted border border-border overflow-hidden">
   215→        <div
   216→          className={`absolute inset-y-0 left-0 ${status === 'success' ? 'bg-orange-100 dark:bg-orange-950/40' : status === 'fail' ? 'bg-rose-100 dark:bg-rose-950/40' : 'bg-primary/10'} transition-all`}
   217→          style={{ width: offset + handleWidth }}
   218→        />
   219→        <div
   220→          className="absolute top-0 h-full flex items-center justify-center text-xs font-medium text-muted-foreground pointer-events-none"
   221→          style={{ left: handleWidth, right: 0 }}
   222→        >
   223→          {status === 'success' ? 'Verification complete' : status === 'fail' ? 'Verification failed, retrying…' : 'Slide right to verify'}
   224→        </div>
   225→        <div
   226→          onPointerDown={onPointerDown}
   227→          onPointerMove={onPointerMove}
   228→          onPointerUp={onPointerUp}
   229→          onPointerCancel={onPointerUp}
   230→          className={`absolute top-1/2 -translate-y-1/2 h-9 w-9 rounded-lg cursor-grab active:cursor-grabbing flex items-center justify-center text-white shadow-md ${pieceColor} ${status === 'success' ? 'cursor-default' : ''} touch-none`}
   231→          style={{ left: 4 + offset, transition: status === 'dragging' ? 'none' : 'left 0.25s ease' }}
   232→        >
   233→          {status === 'success' ? <CheckCircle2 className="w-5 h-5" /> : <ArrowRight className="w-4 h-4" />}
   234→        </div>
   235→      </div>
   236→    </div>
   237→  );
   238→}

   240→/* ───────────────────────── Step Indicator ───────────────────────── */
   241→function StepIndicator({ currentStep, totalSteps = 5 }: { currentStep: RegStep; totalSteps?: number }) {
   242→  return (
   243→    <div className="flex items-center justify-center gap-1 mb-6">
   244→      {Array.from({ length: totalSteps }, (_, i) => (i + 1) as RegStep).map((step, i) => {
   245→        const meta = REG_STEP_META[step];
   246→        const isCompleted = currentStep > step;
   247→        const isCurrent = currentStep === step;
   248→        const Icon = meta.icon;
   249→        return (
   250→          <React.Fragment key={step}>
   251→            <div className="flex items-center gap-1.5">
   252→              <div
   253→                className={`h-7 w-7 rounded-full flex items-center justify-center text-xs font-bold transition-all ${
   254→                  isCompleted || isCurrent
   255→                    ? 'gradient-orange text-white shadow-md shadow-orange-500/30'
   256→                    : 'bg-muted text-muted-foreground'
   257→                }`}
   258→              >
   259→                {isCompleted ? <CheckCircle2 className="w-4 h-4" /> : <Icon className="w-3.5 h-3.5" />}
   260→              </div>
   261→              <span className={`text-xs font-medium hidden sm:inline ${isCompleted || isCurrent ? 'text-foreground' : 'text-muted-foreground'}`}>
   262→                {meta.label}
   263→              </span>
   264→            </div>
   265→            {i < totalSteps - 1 && (
   266→              <div className={`h-0.5 w-4 sm:w-8 rounded-full ${currentStep > step ? 'bg-orange-500' : 'bg-muted'}`} />
   267→            )}
   268→          </React.Fragment>
   269→        );
   270→      })}
   271→    </div>
   272→  );
   273→}
