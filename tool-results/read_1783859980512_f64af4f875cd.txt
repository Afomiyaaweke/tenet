    45→  const [loading, setLoading] = useState(true);
    46→  const [statusFilter, setStatusFilter] = useState('');
    47→  const [expandedId, setExpandedId] = useState<string | null>(null);
    48→  const [activeTab, setActiveTab] = useState<BidTab>('all');
    49→  const [stampSelectorOpen, setStampSelectorOpen] = useState(false);
    50→  const [selectedBidId, setSelectedBidId] = useState<string | null>(null);
    51→  const [bidSignatures, setBidSignatures] = useState<Record<string, SavedSignature>>({});
    52→  const stampSigHook = useStampSignature();
    53→  const [visibleCount, setVisibleCount] = useState(10);
    54→  const BID_PAGE_SIZE = 10;
    55→
    56→  // Document/OCR/AI state
    57→  const [ocrLoading, setOcrLoading] = useState<Set<string>>(new Set());
    58→  const [reviewLoading, setReviewLoading] = useState<Set<string>>(new Set());
    59→  const [docOcrText, setDocOcrText] = useState<Record<string, string>>({});
    60→  const [docReview, setDocReview] = useState<Record<string, AIReviewData>>({});
    61→  const [expandedDocId, setExpandedDocId] = useState<string | null>(null);
    62→  const [reviewDialogOpen, setReviewDialogOpen] = useState(false);
    63→  const [reviewDialogDocId, setReviewDialogDocId] = useState<string | null>(null);
    64→  const [docUploadBidId, setDocUploadBidId] = useState<string | null>(null);
    65→  const docFileRef = useRef<HTMLInputElement>(null);
    66→  const [docUploadType, setDocUploadType] = useState('bid_attachment');
    67→  const [submitUrlForBid, setSubmitUrlForBid] = useState<Record<string, string>>({});
    68→  const [reviewPromptForBid, setReviewPromptForBid] = useState<Record<string, string>>({});
    69→  const fileInputRefs = useRef<Record<string, HTMLInputElement | null>>({});
    70→
    71→  // Saved tenders state
    72→  const [savedTenders, setSavedTenders] = useState<any[]>([]);
    73→  const [savedLoading, setSavedLoading] = useState(false);
    74→
    75→  const loadSavedTenders = useCallback(async () => {
    76→    setSavedLoading(true);
    77→    const res = await api.get('/tenders/saved');
    78→    if (res.success) setSavedTenders(res.data);
    79→    setSavedLoading(false);
    80→  }, []);
    81→
    82→  const handleRemoveSaved = async (id: string) => {
    83→    const res = await api.delete(`/tenders/saved/${id}`);
    84→    if (res.success) {
    85→      toast.success('Removed from saved tenders');
    86→      loadSavedTenders();
    87→    } else {
    88→      toast.error('Failed to remove');
    89→    }
    90→  };
    91→
    92→  const loadBids = useCallback(async () => {
    93→    setLoading(true);
    94→    const params: Record<string, string> = {};
    95→    if (statusFilter && statusFilter !== 'all') params.status = statusFilter;
    96→    const res = await api.get('/bids', params);
    97→    if (res.success) setBids(res.data);
    98→    setLoading(false);
    99→  }, [statusFilter]);
   100→
   101→  useEffect(() => { loadBids(); loadSavedTenders(); }, [loadBids, loadSavedTenders]);
   102→
   103→  const handleStatusUpdate = async (bidId: string, status: string) => {
   104→    const res = await api.patch(`/bids/${bidId}/status`, { status });
   105→    if (res.success) {
   106→      toast.success(`Bid ${status}`);
   107→      loadBids();
   108→    } else {
   109→      toast.error(res.error || 'Failed to update bid');
   110→    }
   111→  };
   112→
   113→  // ── Document Upload (with auto-OCR + auto-AI Review) ──
   114→  const handleDocUpload = useCallback(async (bidId: string, docType?: string, fileInputRef?: React.RefObject<HTMLInputElement | null>) => {
   115→    const ref = fileInputRef || docFileRef;
   116→    const file = ref.current?.files?.[0];
   117→    if (!file) return;
   118→    const type = docType || docUploadType;
   119→    const formData = new FormData();
   120→    formData.append('file', file);
   121→    formData.append('docType', type);
   122→    formData.append('autoOcr', 'true');
   123→    formData.append('autoReview', 'true');
   124→    formData.append('submitUrl', submitUrlForBid[bidId] || '');
   125→    formData.append('reviewPrompt', reviewPromptForBid[bidId] || '');
   126→    try {
   127→      const res = await api.upload(`/bids/${bidId}/documents`, formData);
   128→      if (res.success) {
   129→        toast.success('Document uploaded — OCR & AI Review started automatically');
   130→        if (ref.current) ref.current.value = '';
   131→        loadBids();
   132→        // Start polling for the newly uploaded document
   133→        const newDocId = res.data?.id;
   134→        if (newDocId) {
   135→          setOcrLoading(prev => new Set(prev).add(newDocId));
   136→          pollOcrThenReview(newDocId);
   137→        }
   138→      } else {
   139→        toast.error(res.error || 'Upload failed');
   140→      }
   141→    } catch {
   142→      toast.error('Upload failed');
   143→    } finally {
   144→      setDocUploadBidId(null);
   145→    }
   146→  }, [docUploadType, loadBids, submitUrlForBid, reviewPromptForBid]);
   147→
   148→  // ── Ref for AI Review handler (avoids circular dep with pollOcrThenReview) ──
   149→  const runReviewRef = useRef<(docId: string) => void>(() => {});
   150→
   151→  // ── AI Review Processing ──
   152→  const handleRunReview = useCallback(async (docId: string) => {
   153→    setReviewLoading(prev => new Set(prev).add(docId));
   154→    try {
   155→      const res = await api.post(`/document-review/${docId}`);
   156→      if (!res.success && res.error?.includes('OCR must be completed')) {
   157→        setReviewLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   158→        toast.error('Run OCR first before AI review');
   159→        return;
   160→      }
   161→      const poll = async (attempts = 0): Promise<void> => {
   162→        if (attempts > 60) {
   163→          setReviewLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   164→          toast.error('AI Review processing timed out');
   165→          return;
   166→        }
   167→        await new Promise(r => setTimeout(r, 2000));
   168→        const pollRes = await api.get(`/document-review/${docId}`);
   169→        if (pollRes.success && pollRes.data?.aiReviewStatus === 'completed') {
   170→          setReviewLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   171→          let reviewData = pollRes.data.aiReview || {};
   172→          if (typeof reviewData === 'string') {
   173→            try { reviewData = JSON.parse(reviewData); } catch { reviewData = { summary: reviewData }; }
   174→          }
   175→          setDocReview(prev => ({ ...prev, [docId]: reviewData as AIReviewData }));
   176→          loadBids();
   177→          toast.success('AI Review completed — Document fully processed');
   178→        } else if (pollRes.success && pollRes.data?.aiReviewStatus === 'failed') {
   179→          setReviewLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   180→          toast.error('AI Review processing failed');
   181→          loadBids();
   182→        } else {
   183→          await poll(attempts + 1);
   184→        }
   185→      };
   186→      poll();
   187→    } catch {
   188→      setReviewLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   189→      toast.error('Failed to start AI Review');
   190→    }
   191→  }, [loadBids]);
   192→
   193→  // Keep ref updated
   194→  runReviewRef.current = handleRunReview;
   195→
   196→  // ── Poll OCR then auto-trigger AI Review ──
   197→  const pollOcrThenReview = useCallback(async (docId: string) => {
   198→    const poll = async (attempts = 0): Promise<void> => {
   199→      if (attempts > 60) {
   200→        setOcrLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   201→        toast.error('OCR processing timed out');
   202→        return;
   203→      }
   204→      await new Promise(r => setTimeout(r, 2000));
   205→      const res = await api.get(`/document-ocr/${docId}`);
   206→      if (res.success && res.data?.ocrStatus === 'completed') {
   207→        setOcrLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   208→        setDocOcrText(prev => ({ ...prev, [docId]: res.data.ocrText || '' }));
   209→        loadBids();
   210→        toast.success('OCR completed — Starting AI Review...');
   211→        // Auto-chain: trigger AI Review after OCR completes
   212→        runReviewRef.current(docId);
   213→      } else if (res.success && res.data?.ocrStatus === 'failed') {
   214→        setOcrLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   215→        toast.error('OCR processing failed');
   216→        loadBids();
   217→      } else {
   218→        await poll(attempts + 1);
   219→      }
   220→    };
   221→    poll();
   222→  }, [loadBids]);
   223→
   224→  // ── Drag & Drop Upload ──
   225→  const handleDocDrop = useCallback(async (bidId: string, files: FileList | null, docType?: string) => {
   226→    if (!files || files.length === 0) return;
   227→    const file = files[0];
   228→    const ext = file.name.split('.').pop()?.toLowerCase() || '';
   229→    const allowed = ['pdf', 'jpg', 'jpeg', 'png', 'docx', 'doc', 'txt'];
   230→    if (!allowed.includes(ext)) {
   231→      toast.error('Invalid file type. Allowed: PDF, JPEG, PNG, DOCX, DOC, TXT');
   232→      return;
   233→    }
   234→    if (file.size > 10 * 1024 * 1024) {
   235→      toast.error('File too large. Maximum 10MB.');
   236→      return;
   237→    }
   238→    const formData = new FormData();
   239→    formData.append('file', file);
   240→    formData.append('docType', docType || docUploadType);
   241→    formData.append('autoOcr', 'true');
   242→    formData.append('autoReview', 'true');
   243→    formData.append('submitUrl', submitUrlForBid[bidId] || '');
   244→    formData.append('reviewPrompt', reviewPromptForBid[bidId] || '');
   245→    try {
   246→      const res = await api.upload(`/bids/${bidId}/documents`, formData);
   247→      if (res.success) {
   248→        toast.success('Document dropped — OCR & AI Review started');
   249→        loadBids();
   250→        const newDocId = res.data?.id;
   251→        if (newDocId) {
   252→          setOcrLoading(prev => new Set(prev).add(newDocId));
   253→          pollOcrThenReview(newDocId);
   254→        }
   255→      } else {
   256→        toast.error(res.error || 'Upload failed');
   257→      }
   258→    } catch {
   259→      toast.error('Upload failed');
   260→    }
   261→  }, [docUploadType, loadBids, pollOcrThenReview, submitUrlForBid, reviewPromptForBid]);
   262→
   263→  // ── OCR Processing (manual trigger) ──
   264→  const handleRunOcr = useCallback(async (docId: string) => {
   265→    setOcrLoading(prev => new Set(prev).add(docId));
   266→    try {
   267→      await api.post(`/document-ocr/${docId}`);
   268→      pollOcrThenReview(docId);
   269→    } catch {
   270→      setOcrLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   271→      toast.error('Failed to start OCR');
   272→    }
   273→  }, [pollOcrThenReview]);
   274→
   275→  // ── View OCR text inline ──
   276→  const handleViewOcr = useCallback(async (docId: string) => {
   277→    if (expandedDocId === docId) {
   278→      setExpandedDocId(null);
   279→      return;
   280→    }
   281→    setExpandedDocId(docId);
   282→    if (!docOcrText[docId]) {
   283→      const res = await api.get(`/document-ocr/${docId}`);
   284→      if (res.success) {
   285→        setDocOcrText(prev => ({ ...prev, [docId]: res.data?.ocrText || '' }));
   286→      }
   287→    }
   288→  }, [expandedDocId, docOcrText]);
   289→
   290→  const stats = useMemo(() => ({
   291→    pending: bids.filter(b => b.status === 'pending_review').length,
   292→    shortlisted: bids.filter(b => b.status === 'shortlisted').length,
   293→    awarded: bids.filter(b => b.status === 'awarded').length,
   294→    rejected: bids.filter(b => b.status === 'rejected').length,
   295→    total: bids.length,
   296→  }), [bids]);
   297→
   298→  const filteredBids = useMemo(() => {
   299→    if (activeTab === 'saved') return [] as Bid[];
   300→    if (activeTab === 'all') return bids;
   301→    return bids.filter(b => b.status === activeTab);
   302→  }, [bids, activeTab]);
   303→
   304→  const statusBadge = (status: string) => {
   305→    switch (status) {
   306→      case 'pending_review': return 'bg-amber-100 text-amber-700 hover:bg-amber-100';
   307→      case 'shortlisted': return 'bg-teal-100 text-teal-700 hover:bg-teal-100';
   308→      case 'awarded': return 'bg-emerald-100 text-emerald-700 hover:bg-emerald-100';
   309→      case 'rejected': return 'bg-rose-100 text-rose-700 hover:bg-rose-100';
   310→      default: return 'bg-muted text-muted-foreground hover:bg-muted';
   311→    }
   312→  };
   313→
   314→  const statusIcon = (status: string) => {
   315→    switch (status) {
   316→      case 'pending_review': return { icon: Clock, bg: 'bg-amber-50', color: 'text-amber-600' };
   317→      case 'shortlisted': return { icon: Award, bg: 'bg-teal-50', color: 'text-teal-600' };
   318→      case 'awarded': return { icon: CheckCircle, bg: 'bg-emerald-50', color: 'text-emerald-600' };
   319→      case 'rejected': return { icon: AlertCircle, bg: 'bg-rose-50', color: 'text-rose-600' };
   320→      default: return { icon: Gavel, bg: 'bg-muted/50', color: 'text-muted-foreground' };
   321→    }
   322→  };
   323→
   324→  const isAdminOrOwner = user?.role === 'team_admin';
   325→
   326→  const getScoreColor = (score: number) => {
   327→    if (score >= 80) return 'text-emerald-600';
   328→    if (score >= 60) return 'text-amber-600';
   329→    return 'text-rose-600';
   330→  };
   331→
   332→  const getScoreBg = (score: number) => {
   333→    if (score >= 80) return 'bg-emerald-50';
   334→    if (score >= 60) return 'bg-amber-50';
   335→    return 'bg-rose-50';
   336→  };
   337→
   338→  const getRiskColor = (risk: string) => {
   339→    switch (risk?.toLowerCase()) {
   340→      case 'low': return 'bg-emerald-100 text-emerald-700';
   341→      case 'medium': return 'bg-amber-100 text-amber-700';
   342→      case 'high': case 'critical': return 'bg-rose-100 text-rose-700';
   343→      default: return 'bg-gray-100 text-gray-700';
   344→    }
   345→  };
   346→
   347→  const tabs: { key: BidTab; label: string; icon: typeof Clock; count: number; color: string }[] = [
   348→    { key: 'all', label: 'All', icon: Gavel, count: stats.total, color: 'emerald' },
   349→    { key: 'pending_review', label: 'Pending', icon: Clock, count: stats.pending, color: 'amber' },
   350→    { key: 'shortlisted', label: 'Shortlisted', icon: Award, count: stats.shortlisted, color: 'teal' },
   351→    { key: 'awarded', label: 'Awarded', icon: CheckCircle, count: stats.awarded, color: 'emerald' },
   352→    { key: 'rejected', label: 'Rejected', icon: AlertCircle, count: stats.rejected, color: 'rose' },
   353→    { key: 'saved', label: 'Saved Tenders', icon: Bookmark, count: savedTenders.length, color: 'violet' },
   354→  ];
   355→
   356→  // Review dialog data
   357→  const reviewDialogData = reviewDialogDocId ? docReview[reviewDialogDocId] : null;
   358→
   359→  return (
   360→    <div className="p-4 md:p-6 space-y-6 max-w-5xl mx-auto view-enter">
   361→      {/* Header */}
   362→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4 animate-[fadeIn_0.3s_ease-out]">
   363→        <div className="flex items-center gap-4">
   364→          <div className="p-3 rounded-2xl gradient-amber shadow-md flex-shrink-0">
   365→            <Gavel className="h-6 w-6 text-white" />
   366→          </div>
   367→          <div>
   368→            <h2 className="text-2xl md:text-3xl font-bold tracking-tight">
   369→              <span className="text-gradient-emerald">{isAdminOrOwner ? 'Review' : 'My'}</span> Bids
   370→            </h2>
   371→            <p className="text-muted-foreground text-sm mt-0.5">
   372→              {isAdminOrOwner ? 'Manage and evaluate submitted bids' : 'Track your bid submissions'}
   373→            </p>
   374→          </div>
   375→        </div>
   376→      </div>
   377→
   378→      {/* Stats Summary */}
   379→      {!loading && (bids.length > 0 || savedTenders.length > 0) && (
   380→        <div className="grid grid-cols-2 sm:grid-cols-4 gap-4 animate-[fadeIn_0.3s_ease-out]">
   381→          {[
   382→            { label: 'Pending', count: stats.pending, icon: Clock, bg: 'bg-amber-50', color: 'text-amber-600' },
   383→            { label: 'Shortlisted', count: stats.shortlisted, icon: Award, bg: 'bg-teal-50', color: 'text-teal-600' },
   384→            { label: 'Awarded', count: stats.awarded, icon: CheckCircle, bg: 'bg-emerald-50', color: 'text-emerald-600' },
   385→            { label: 'Rejected', count: stats.rejected, icon: AlertCircle, bg: 'bg-rose-50', color: 'text-rose-600' },
   386→          ].map(stat => (
   387→            <div key={stat.label}>
   388→              <Card className="premium-shadow rounded-xl border-0 bg-card hover:-translate-y-0.5 transition-all duration-200">
   389→                <CardContent className="p-4 flex items-center gap-3">
   390→                  <div className={`p-2 rounded-lg ${stat.bg} flex-shrink-0`}>
   391→                    <stat.icon className={`h-4 w-4 ${stat.color}`} />
   392→                  </div>
   393→                  <div>
   394→                    <p className="text-xl font-bold">{stat.count}</p>
   395→                    <p className="text-xs text-muted-foreground">{stat.label}</p>
   396→                  </div>
   397→                </CardContent>
   398→              </Card>
   399→            </div>
   400→          ))}
   401→        </div>
   402→      )}
   403→
   404→      {/* Tab Navigation */}
   405→      {!loading && (
   406→        <div className="animate-[fadeIn_0.3s_ease-out]">
   407→          <Card className="premium-shadow rounded-xl border-0 bg-card">
   408→            <CardContent className="p-1.5">
   409→              <div className="flex gap-1 overflow-x-auto">
   410→                {tabs.map(tab => (
   411→                  <button
   412→                    key={tab.key}
   413→                    onClick={() => setActiveTab(tab.key)}
   414→                    className={`flex items-center gap-1.5 px-3 py-2 rounded-lg text-sm font-medium transition-all duration-200 whitespace-nowrap ${
   415→                      activeTab === tab.key
   416→                        ? 'gradient-emerald text-white premium-shadow'
   417→                        : 'text-muted-foreground hover:bg-muted/60 hover:text-foreground'
   418→                    }`}
   419→                  >
   420→                    <tab.icon className="h-3.5 w-3.5" />
   421→                    <span>{tab.label}</span>
   422→                    <span className={`text-[10px] px-1.5 py-0 rounded-full ${
   423→                      activeTab === tab.key
   424→                        ? 'bg-white/20 text-white'
   425→                        : 'bg-muted text-muted-foreground'
   426→                    }`}>
   427→                      {tab.count}
   428→                    </span>
   429→                  </button>
   430→                ))}
   431→              </div>
   432→            </CardContent>
   433→          </Card>
   434→        </div>
   435→      )}
   436→
   437→      {/* Saved Tenders Tab */}
   438→      {activeTab === 'saved' && (
   439→        <div className="space-y-4 animate-[fadeIn_0.3s_ease-out]">
   440→          {savedLoading ? (
   441→            [1, 2, 3].map(i => (
   442→              <Card key={i} className="premium-shadow rounded-xl border-0 bg-card animate-pulse">
   443→                <CardContent className="p-5"><div className="h-20 bg-muted/50 rounded-xl" /></CardContent>
   444→              </Card>
   445→            ))
   446→          ) : savedTenders.length === 0 ? (
   447→            <Card className="premium-shadow rounded-xl border-0 bg-card">
   448→              <CardContent className="p-12 text-center">
   449→                <div className="p-3 rounded-2xl bg-violet-50 dark:bg-violet-950/30 w-fit mx-auto mb-4">
   450→                  <Bookmark className="h-8 w-8 text-violet-500" />
   451→                </div>
   452→                <h3 className="text-lg font-semibold">No saved tenders yet</h3>
   453→                <p className="text-muted-foreground text-sm mt-1 max-w-sm mx-auto">
   454→                  Bookmark tenders from Live Tenders to work on them later
   455→                </p>
