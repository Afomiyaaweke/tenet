    28→function daysUntil(dateStr: string): number {

    32→function parseBidAnalysis(analysis: BidAnalysis): BidAnalysisResult | null {
    33→  try {
    34→    return {
    35→      summary: JSON.parse(analysis.summary || '{}'),
    36→      applicants: JSON.parse(analysis.rankings || '[]'),
    37→      budgetAnalysis: analysis.budgetAnalysis || '',
    38→      riskSummary: analysis.riskSummary || '',
    39→      finalRecommendation: analysis.recommendation || '',
    40→    };
    41→  } catch {
    42→    return null;

    46→function ScoreBar({ score, label }: { score: number; label?: string }) {
    47→  const colorClass = score >= 80 ? 'bg-emerald-500' : score >= 60 ? 'bg-amber-500' : 'bg-red-500';
    48→  const textColor = score >= 80 ? 'text-emerald-700' : score >= 60 ? 'text-amber-700' : 'text-red-700';
    49→  return (
    50→    <div className="flex items-center gap-2 min-w-[120px]">
    51→      {label && <span className="text-[10px] text-muted-foreground w-12 shrink-0">{label}</span>}
    52→      <div className="flex-1 h-2 bg-muted/50 rounded-full overflow-hidden">
    53→        <div
    54→ className={`h-full rounded-full ${colorClass} transition-[width] duration-700`} style={{ width: `${Math.min(score, 100)}%` }}
    55→ />
    56→      </div>
    57→      <span className={`text-xs font-semibold ${textColor} w-8 text-right`}>{score}</span>
    58→    </div>
    59→  );
    60→}

    62→function RiskBadge({ level }: { level: 'low' | 'medium' | 'high' }) {
    63→  const config = {
    64→    low: { className: 'bg-emerald-100 text-emerald-700', icon: ShieldCheck },
    65→    medium: { className: 'bg-amber-100 text-amber-700', icon: ShieldQuestion },
    66→    high: { className: 'bg-red-100 text-red-700', icon: ShieldAlert },
    67→  };
    68→  const c = config[level];
    69→  const Icon = c.icon;
    70→  return (
    71→    <Badge className={`text-[10px] px-2 py-0.5 border-0 rounded-lg ${c.className}`}>
    72→      <Icon className="h-3 w-3 mr-1" />
    73→      {level.charAt(0).toUpperCase() + level.slice(1)}
    74→    </Badge>
    75→  );
    76→}
