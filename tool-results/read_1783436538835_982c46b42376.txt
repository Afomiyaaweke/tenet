    76→function getProfileCompleteness(profile: Record<string, unknown> | undefined): { pct: number; missing: string[] } {
    77→  if (!profile) return { pct: 0, missing: ['Full Name', 'Phone', 'Location', 'Skills', 'Bio'] };
    78→  const checks: [string, boolean][] = [
    79→    ['Full Name', !!(profile.fullName as string)],
    80→    ['Phone', !!(profile.phone as string)],
    81→    ['Location', !!(profile.location as string)],
    82→    ['Skills', !!(profile.skillTags as string)],
    83→    ['Bio', !!(profile.bio as string)],
    84→  ];
    85→  const completed = checks.filter(([, v]) => v).length;
    86→  const missing = checks.filter(([, v]) => !v).map(([label]) => label);
    87→  return { pct: Math.round((completed / checks.length) * 100), missing };
    88→}
