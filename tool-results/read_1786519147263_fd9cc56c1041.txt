    85→    85→function timeAgo(dateStr: string): string {
    86→    86→  const diff = Date.now() - new Date(dateStr).getTime();
    87→    87→  const mins = Math.floor(diff / 60000);
    88→    88→  if (mins < 1) return 'just now';
    89→    89→  if (mins < 60) return `${mins}m ago`;
    90→    90→  const hrs = Math.floor(mins / 60);
    91→    91→  if (hrs < 24) return `${hrs}h ago`;
    92→    92→  const days = Math.floor(hrs / 24);
    93→    93→  return `${days}d ago`;
    94→    94→}

   101→   101→function SparklineBars({ data, color = '#10b981', height = 32 }: { data: number[]; color?: string; height?: number }) {
   102→   102→  if (!data.length) return null;
   103→   103→  const max = Math.max(...data, 1);
   104→   104→  const barW = 4;
   105→   105→  const gap = 3;
   106→   106→  const totalW = data.length * (barW + gap) - gap;
   107→   107→
   108→   108→  return (
   109→   109→    <svg width={totalW} height={height} className="flex-shrink-0">
   110→   110→      {data.map((v, i) => {
   111→   111→        const barH = Math.max((v / max) * (height - 4), 3);
   112→   112→        return (
   113→   113→          <rect
   114→   114→            key={i}
   115→   115→            x={i * (barW + gap)}
   116→   116→            y={height - barH}
   117→   117→            width={barW}
   118→   118→            height={barH}
   119→   119→            rx={2}
   120→   120→            fill={color}
   121→   121→            opacity={0.35 + (i / data.length) * 0.45}
   122→   122→          />
   123→   123→        );
   124→   124→      })}
   125→   125→    </svg>
   126→   126→  );
   127→   127→}
