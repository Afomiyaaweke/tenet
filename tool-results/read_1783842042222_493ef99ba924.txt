   160→function formatCurrency(amount: number): string {
   161→  return new Intl.NumberFormat('en-ET', { style: 'currency', currency: 'ETB', maximumFractionDigits: 0 }).format(amount);
   162→}

   164→function formatDate(dateStr: string): string {
   165→  if (!dateStr) return '—';
   166→  return new Date(dateStr).toLocaleDateString('en-GB', { day: '2-digit', month: 'short', year: 'numeric' });
   167→}
   168→
   169→function daysUntil(dateStr: string): number {
   170→  return Math.ceil((new Date(dateStr).getTime() - Date.now()) / (1000 * 60 * 60 * 24));
   171→}

   190→export function ApplicantsView() {
   191→  const { user } = useAuthStore();
   192→  const { setView } = useNavStore();
   193→  const [rows, setRows] = useState<ApplicantRow[]>([]);
   194→  const [meta, setMeta] = useState<ApplicantMeta | null>(null);
   195→  const [openTenders, setOpenTenders] = useState<TenderInfo[]>([]);
   196→  const [closedTenders, setClosedTenders] = useState<TenderInfo[]>([]);
   197→  const [loading, setLoading] = useState(true);
   198→  const [search, setSearch] = useState('');
   199→  const [statusFilter, setStatusFilter] = useState<string>('all');
   200→  const [tenderFilter, setTenderFilter] = useState<string>('all');
   201→  const [sortField, setSortField] = useState<SortField>('submittedAt');
   202→  const [sortDir, setSortDir] = useState<SortDir>('desc');
   203→  const [viewMode, setViewMode] = useState<ViewMode>('table');
   204→  const [expandedRow, setExpandedRow] = useState<string | null>(null);
   205→  const [visibleColumns, setVisibleColumns] = useState<Set<string>>(() =>
   206→    new Set(COLUMNS.filter(c => c.defaultVisible).map(c => c.key))
   207→  );
   208→  const [columnMenuOpen, setColumnMenuOpen] = useState(false);
   209→  const [page, setPage] = useState(1);
   210→  const PAGE_SIZE = 50;
   211→
   212→  // ─── Document / OCR / AI Review state ───────────────────────────
   213→  const [docUploadBidId, setDocUploadBidId] = useState<string | null>(null);
   214→  const [ocrLoading, setOcrLoading] = useState<Set<string>>(new Set());
   215→  const [reviewLoading, setReviewLoading] = useState<Set<string>>(new Set());
   216→  const [viewingDocId, setViewingDocId] = useState<string | null>(null);
   217→  const [viewingDocType, setViewingDocType] = useState<'ocr' | 'review'>('ocr');
   218→  const [docOcrText, setDocOcrText] = useState<Record<string, string>>({});
   219→  const [docReview, setDocReview] = useState<Record<string, Record<string, unknown>>>({});
   220→
   221→  // Fetch data
   222→  const fetchData = useCallback(async () => {
   223→    setLoading(true);
   224→    try {
   225→      const params: Record<string, string> = {
   226→        page: String(page),
   227→        limit: String(PAGE_SIZE),
   228→      };
   229→      if (search) params.search = search;
   230→      if (statusFilter && statusFilter !== 'all') params.status = statusFilter;
   231→      if (tenderFilter && tenderFilter !== 'all') params.tenderId = tenderFilter;
   232→
   233→      const res = await api.get('/applicants', params);
   234→      if (res.success) {
   235→        setRows(res.data);
   236→        setMeta(res.meta);
   237→        setOpenTenders(res.openTenders || []);
   238→        setClosedTenders(res.closedTenders || []);
   239→      }
   240→    } catch (err) {
   241→      console.error('Fetch applicants error:', err);
   242→      toast.error('Failed to load applicants');
   243→    } finally {
   244→      setLoading(false);
   245→    }
   246→  }, [page, search, statusFilter, tenderFilter]);
   247→
   248→  useEffect(() => {
   249→    fetchData();
   250→  }, [fetchData]);
   251→
   252→  // ─── Document handlers ──────────────────────────────────────────
   253→  const handleDocUpload = useCallback(async (bidId: string, file: File) => {
   254→    const formData = new FormData();
   255→    formData.append('file', file);
   256→    formData.append('docType', 'bid_attachment');
   257→    formData.append('autoOcr', 'true');
   258→    try {
   259→      const res = await api.upload(`/bids/${bidId}/documents`, formData);
   260→      if (res.success) {
   261→        toast.success('Document uploaded successfully');
   262→        fetchData();
   263→      } else {
   264→        toast.error(res.message || 'Upload failed');
   265→      }
   266→    } catch {
   267→      toast.error('Upload failed');
   268→    } finally {
   269→      setDocUploadBidId(null);
   270→    }
   271→  }, [fetchData]);
   272→
   273→  const handleRunOcr = useCallback(async (docId: string) => {
   274→    setOcrLoading(prev => new Set(prev).add(docId));
   275→    try {
   276→      await api.post(`/document-ocr/${docId}`);
   277→      // Poll for completion
   278→      const poll = async (attempts = 0): Promise<void> => {
   279→        if (attempts > 30) {
   280→          toast.error('OCR processing timed out');
   281→          setOcrLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   282→          return;
   283→        }
   284→        await new Promise(r => setTimeout(r, 2000));
   285→        const res = await api.get(`/document-ocr/${docId}`);
   286→        if (res.success && res.data?.ocrStatus === 'completed') {
   287→          setOcrLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   288→          setDocOcrText(prev => ({ ...prev, [docId]: res.data.ocrText || '' }));
   289→          fetchData();
   290→          toast.success('OCR completed');
   291→        } else if (res.success && res.data?.ocrStatus === 'failed') {
   292→          setOcrLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   293→          toast.error('OCR processing failed');
   294→          fetchData();
   295→        } else {
   296→          await poll(attempts + 1);
   297→        }
   298→      };
   299→      poll();
   300→    } catch {
   301→      setOcrLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   302→      toast.error('Failed to start OCR');
   303→    }
   304→  }, [fetchData]);
   305→
   306→  const handleRunReview = useCallback(async (docId: string) => {
   307→    setReviewLoading(prev => new Set(prev).add(docId));
   308→    try {
   309→      await api.post(`/document-review/${docId}`);
   310→      // Poll for completion
   311→      const poll = async (attempts = 0): Promise<void> => {
   312→        if (attempts > 30) {
   313→          toast.error('AI Review processing timed out');
   314→          setReviewLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   315→          return;
   316→        }
   317→        await new Promise(r => setTimeout(r, 2000));
   318→        const res = await api.get(`/document-review/${docId}`);
   319→        if (res.success && res.data?.aiReviewStatus === 'completed') {
   320→          setReviewLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   321→          let reviewData = res.data.aiReview || {};
   322→          if (typeof reviewData === 'string') {
   323→            try { reviewData = JSON.parse(reviewData); } catch { reviewData = { summary: reviewData }; }
   324→          }
   325→          setDocReview(prev => ({ ...prev, [docId]: reviewData }));
   326→          fetchData();
   327→          toast.success('AI Review completed');
   328→        } else if (res.success && res.data?.aiReviewStatus === 'failed') {
   329→          setReviewLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   330→          toast.error('AI Review processing failed');
   331→          fetchData();
   332→        } else {
   333→          await poll(attempts + 1);
   334→        }
   335→      };
   336→      poll();
   337→    } catch {
   338→      setReviewLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   339→      toast.error('Failed to start AI Review');
   340→    }
   341→  }, [fetchData]);
   342→
   343→  const handleViewDocDetail = useCallback(async (docId: string, type: 'ocr' | 'review') => {
   344→    if (viewingDocId === docId && viewingDocType === type) {
   345→      // Toggle off
   346→      setViewingDocId(null);
   347→      return;
   348→    }
   349→    setViewingDocId(docId);
   350→    setViewingDocType(type);
   351→    try {
   352→      if (type === 'ocr' && !docOcrText[docId]) {
   353→        const res = await api.get(`/document-ocr/${docId}`);
   354→        if (res.success) {
   355→          setDocOcrText(prev => ({ ...prev, [docId]: res.data?.ocrText || '' }));
   356→        }
   357→      } else if (type === 'review' && !docReview[docId]) {
   358→        const res = await api.get(`/document-review/${docId}`);
   359→        if (res.success) {
   360→          let reviewData = res.data?.aiReview || {};
   361→          // aiReview may come as a JSON string from the API - parse it
   362→          if (typeof reviewData === 'string') {
   363→            try {
   364→              reviewData = JSON.parse(reviewData);
   365→            } catch {
   366→              reviewData = { summary: reviewData };
   367→            }
   368→          }
   369→          setDocReview(prev => ({ ...prev, [docId]: reviewData }));
   370→        }
   371→      }
   372→    } catch {
   373→      toast.error('Failed to load document details');
   374→    }
   375→  }, [viewingDocId, viewingDocType, docOcrText, docReview]);
   376→
   377→  const closeDocDetail = useCallback(() => {
   378→    setViewingDocId(null);
   379→  }, []);
   380→
   381→  // Get unique tenders for filter dropdown (from closedTenders API data)
   382→  const tenderOptions = useMemo(() => {
   383→    return closedTenders.map(t => [t.id, t.title] as [string, string]);
   384→  }, [closedTenders]);
   385→
   386→  // Sorted rows (client-side sorting on current page)
   387→  const sortedRows = useMemo(() => {
   388→    const copy = [...rows];
   389→    copy.sort((a, b) => {
   390→      const aVal = a[sortField];
   391→      const bVal = b[sortField];
   392→      if (aVal === bVal) return 0;
   393→      if (aVal == null) return 1;
   394→      if (bVal == null) return -1;
   395→      let cmp = 0;
   396→      if (typeof aVal === 'number' && typeof bVal === 'number') {
   397→        cmp = aVal - bVal;
   398→      } else if (typeof aVal === 'boolean' && typeof bVal === 'boolean') {
   399→        cmp = Number(aVal) - Number(bVal);
   400→      } else {
   401→        cmp = String(aVal).localeCompare(String(bVal));
   402→      }
   403→      return sortDir === 'asc' ? cmp : -cmp;
   404→    });
   405→    return copy;
   406→  }, [rows, sortField, sortDir]);
   407→
   408→  // Visible columns
   409→  const activeColumns = useMemo(
   410→    () => COLUMNS.filter(c => visibleColumns.has(c.key)),
   411→    [visibleColumns]
   412→  );
   413→
   414→  // Column groups
   415→  const columnGroups = useMemo(() => {
   416→    const groups: Record<string, ColumnDef[]> = {};
   417→    activeColumns.forEach(c => {
   418→      if (!groups[c.group]) groups[c.group] = [];
   419→      groups[c.group].push(c);
   420→    });
   421→    return groups;
   422→  }, [activeColumns]);
   423→
   424→  // Toggle column visibility
   425→  const toggleColumn = (key: string) => {
   426→    setVisibleColumns(prev => {
   427→      const next = new Set(prev);
   428→      if (next.has(key)) next.delete(key);
   429→      else next.add(key);
   430→      return next;
   431→    });
   432→  };
   433→
   434→  // Toggle sort
   435→  const handleSort = (field: SortField) => {
   436→    if (sortField === field) {
   437→      setSortDir(prev => prev === 'asc' ? 'desc' : 'asc');
   438→    } else {
   439→      setSortField(field);
   440→      setSortDir('asc');
   441→    }
   442→  };
   443→
   444→  // Sort icon
   445→  const SortIcon = ({ field }: { field: SortField }) => {
   446→    if (sortField !== field) return <ArrowUpDown className="h-3 w-3 text-muted-foreground/40" />;
   447→    return sortDir === 'asc'
   448→      ? <ArrowUp className="h-3 w-3 text-primary" />
   449→      : <ArrowDown className="h-3 w-3 text-primary" />;
   450→  };
   451→
   452→  // Export CSV
   453→  const exportCSV = () => {
   454→    if (!rows.length) return;
   455→    const headers = activeColumns.map(c => c.label);
   456→    const csvRows = rows.map(row =>
   457→      activeColumns.map(c => {
   458→        let val = row[c.key];
   459→        if (typeof val === 'boolean') val = val ? 'Yes' : 'No';
   460→        if (typeof val === 'number') val = val.toLocaleString();
   461→        val = String(val || '');
   462→        // Escape CSV
   463→        if (val.includes(',') || val.includes('"') || val.includes('\n')) {
   464→          val = `"${val.replace(/"/g, '""')}"`;
   465→        }
   466→        return val;
   467→      }).join(',')
   468→    );
   469→    const csv = [headers.join(','), ...csvRows].join('\n');
   470→    const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });
