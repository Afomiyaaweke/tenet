   166→function relativeTime(dateStr: string): string {
   167→  const now = Date.now();
   168→  const then = new Date(dateStr).getTime();
   169→  const diff = Math.max(0, now - then);
   170→  const seconds = Math.floor(diff / 1000);
   171→  if (seconds < 60) return 'just now';
   172→  const minutes = Math.floor(seconds / 60);
   173→  if (minutes < 60) return `${minutes}m ago`;
   174→  const hours = Math.floor(minutes / 60);
   175→  if (hours < 24) return `${hours}h ago`;
   176→  const days = Math.floor(hours / 24);
   177→  if (days < 7) return `${days}d ago`;
   178→  return new Date(dateStr).toLocaleDateString();
   179→}

   427→                      <p className="text-[13px] font-semibold truncate text-foreground">{n.title}</p>

   443→              variant="ghost"

   459→export function AppShell() {
   460→  const { user, company, logout } = useAuthStore();
   461→  const { view, viewParams, setView } = useNavStore();
   462→  const { fetchNotifications, notifications } = useDataStore();
   463→  const role = user?.role || 'user';
   464→  const navSections = useMemo(() => getNavItemsForRole(role), [role]);
   465→  const unreadCount = notifications.filter((n) => !n.read).length;
   466→
   467→  // Flat list for title lookup
   468→  const allNavItems = useMemo(
   469→    () => navSections.flatMap((s) => s.items),
   470→    [navSections]
   471→  );
   472→
   473→  useEffect(() => {
   474→    fetchNotifications();
   475→  }, [fetchNotifications]);
   476→
   477→  const handleMarkAsRead = async (id: string) => {
   478→    await api.patch(`/notifications/${id}`, { read: true });
   479→    fetchNotifications();
   480→  };
   481→
