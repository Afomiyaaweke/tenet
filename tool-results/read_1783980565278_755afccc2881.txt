    38→const STORAGE_KEY = 'tenet_signatures';
    39→
    40→export const STAMP_TEMPLATES = [
    41→  { text: 'APPROVED', label: 'Approved', color: '#16a34a' },
    42→  { text: 'VERIFIED', label: 'Verified', color: '#2563eb' },
    43→  { text: 'CONFIDENTIAL', label: 'Confidential', color: '#dc2626' },
    44→  { text: 'REJECTED', label: 'Rejected', color: '#dc2626' },
    45→  { text: 'DRAFT', label: 'Draft', color: '#9333ea' },
    46→] as const;

    48→/* ══════════════════════════════════════════════════════════════
    49→   HELPERS
    50→   ══════════════════════════════════════════════════════════════ */
    51→
    52→function loadItems(): SavedSignature[] {
    53→  if (typeof window === 'undefined') return [];
    54→  try {
    55→    const raw = localStorage.getItem(STORAGE_KEY);
    56→    return raw ? JSON.parse(raw) : [];
    57→  } catch {
    58→    return [];
    59→  }
    60→}
    61→
    62→function saveItems(items: SavedSignature[]) {
    63→  if (typeof window === 'undefined') return;
    64→  localStorage.setItem(STORAGE_KEY, JSON.stringify(items));
    65→}
    66→
    67→function generateSeal(text: string, date: string): string {
    68→  const template = STAMP_TEMPLATES.find(t => t.text === text);
    69→  const color = template?.color || '#dc2626';
    70→  const svg = `<svg xmlns="http://www.w3.org/2000/svg" width="120" height="120" viewBox="0 0 120 120">
    71→    <circle cx="60" cy="60" r="55" fill="none" stroke="${color}" stroke-width="3"/>
    72→    <circle cx="60" cy="60" r="48" fill="none" stroke="${color}" stroke-width="1.5"/>
    73→    <path id="topArc" d="M 15,60 A 45,45 0 0,1 105,60" fill="none"/>
    74→    <text font-size="10" fill="${color}" font-weight="bold" letter-spacing="3">
    75→      <textPath href="#topArc" startOffset="50%" text-anchor="middle">${text}</textPath>
    76→    </text>
    77→    <text x="60" y="68" text-anchor="middle" font-size="9" fill="${color}">${date}</text>
    78→    <text x="60" y="80" text-anchor="middle" font-size="7" fill="${color}">TENET</text>
    79→  </svg>`;
    80→  return 'data:image/svg+xml;base64,' + btoa(svg);
    81→}

    84→   HOOK: useStampSignature
    85→   ══════════════════════════════════════════════════════════════ */
    86→
    87→export function useStampSignature(): UseStampSignatureReturn {
    88→  const [savedItems, setSavedItems] = useState<SavedSignature[]>(() => {
    89→    if (typeof window === 'undefined') return [];
    90→    try {
    91→      const raw = localStorage.getItem(STORAGE_KEY);
    92→      return raw ? JSON.parse(raw) : [];
    93→    } catch {
    94→      return [];
    95→    }
    96→  });
    97→
    98→  const addSignature = useCallback((dataUrl: string, label: string, type: 'signature' | 'stamp') => {
    99→    const newItem: SavedSignature = {
   100→      id: Date.now().toString(),
   101→      dataUrl,
   102→      label,
   103→      type,
   104→    };
   105→    setSavedItems(prev => {
   106→      const updated = [...prev, newItem];
   107→      saveItems(updated);
   108→      return updated;
   109→    });
   110→  }, []);
   111→
   112→  const removeItem = useCallback((id: string) => {
   113→    setSavedItems(prev => {
   114→      const updated = prev.filter(s => s.id !== id);
   115→      saveItems(updated);
   116→      return updated;
   117→    });
   118→    toast.success('Signature/stamp removed');
   119→  }, []);
   120→
   121→  const uploadFromFile = useCallback(async (type: 'signature' | 'stamp') => {
   122→    return new Promise<void>((resolve) => {
   123→      const input = document.createElement('input');
   124→      input.type = 'file';
   125→      input.accept = 'image/*';
   126→      input.onchange = (e) => {
   127→        const file = (e.target as HTMLInputElement).files?.[0];
   128→        if (!file) { resolve(); return; }
   129→        const reader = new FileReader();
   130→        reader.onload = (ev) => {
   131→          const dataUrl = ev.target?.result as string;
   132→          const label = type === 'signature'
   133→            ? `Signature ${loadItems().filter(s => s.type === 'signature').length + 1}`
   134→            : `Stamp ${loadItems().filter(s => s.type === 'stamp').length + 1}`;
   135→          const newItem: SavedSignature = {
   136→            id: Date.now().toString(),
   137→            dataUrl,
   138→            label,
   139→            type,
   140→          };
   141→          setSavedItems(prev => {
   142→            const updated = [...prev, newItem];
   143→            saveItems(updated);
   144→            return updated;
   145→          });
   146→          toast.success(`${type === 'signature' ? 'Signature' : 'Stamp'} uploaded`);
   147→          resolve();
   148→        };
   149→        reader.readAsDataURL(file);
   150→      };
   151→      input.oncancel = () => resolve();
   152→      input.click();
   153→    });
   154→  }, []);
   155→
   156→  const generateStamp = useCallback((text: string): string => {
   157→    const dateStr = new Date().toLocaleDateString();
   158→    return generateSeal(text, dateStr);
   159→  }, []);
   160→
   161→  return {
   162→    savedItems,
   163→    addSignature,
   164→    removeItem,
   165→    uploadFromFile,
   166→    generateStamp,
   167→  };
   168→}

   170→/* ══════════════════════════════════════════════════════════════
   171→   COMPONENT: SignatureDrawingCanvas
   172→   ══════════════════════════════════════════════════════════════ */
   173→
   174→function SignatureDrawingCanvas({
   175→  open,
   176→  onClose,
   177→  onSave,
   178→}: {
   179→  open: boolean;
   180→  onClose: () => void;
   181→  onSave: (dataUrl: string) => void;
   182→}) {
   183→  const canvasRef = useRef<HTMLCanvasElement>(null);
   184→  const [isDrawing, setIsDrawing] = useState(false);
   185→
   186→  const startDraw = (e: React.MouseEvent<HTMLCanvasElement>) => {
   187→    setIsDrawing(true);
   188→    const ctx = canvasRef.current?.getContext('2d');
   189→    if (ctx && canvasRef.current) {
   190→      const rect = canvasRef.current.getBoundingClientRect();
   191→      const scaleX = canvasRef.current.width / rect.width;
   192→      const scaleY = canvasRef.current.height / rect.height;
   193→      ctx.beginPath();
   194→      ctx.moveTo((e.clientX - rect.left) * scaleX, (e.clientY - rect.top) * scaleY);
   195→    }
   196→  };
   197→
   198→  const draw = (e: React.MouseEvent<HTMLCanvasElement>) => {
   199→    if (!isDrawing) return;
   200→    const ctx = canvasRef.current?.getContext('2d');
   201→    if (ctx && canvasRef.current) {
   202→      const rect = canvasRef.current.getBoundingClientRect();
   203→      const scaleX = canvasRef.current.width / rect.width;
   204→      const scaleY = canvasRef.current.height / rect.height;
   205→      ctx.lineWidth = 2;
   206→      ctx.lineCap = 'round';
   207→      ctx.strokeStyle = '#1a1a1a';
   208→      ctx.lineTo((e.clientX - rect.left) * scaleX, (e.clientY - rect.top) * scaleY);
   209→      ctx.stroke();
   210→    }
   211→  };
   212→
   213→  const endDraw = () => {
   214→    setIsDrawing(false);
   215→  };
   216→
   217→  const clearCanvas = () => {
   218→    const ctx = canvasRef.current?.getContext('2d');
   219→    if (ctx && canvasRef.current) {
   220→      ctx.clearRect(0, 0, canvasRef.current.width, canvasRef.current.height);
   221→    }
   222→  };
   223→
   224→  const save = () => {
   225→    const dataUrl = canvasRef.current?.toDataURL('image/png');
   226→    if (dataUrl) {
   227→      onSave(dataUrl);
   228→      clearCanvas();
   229→    }
   230→  };
   231→
   232→  return (
   233→    <Dialog open={open} onOpenChange={(v) => { if (!v) onClose(); }}>
   234→      <DialogContent className="sm:max-w-[500px]">
   235→        <DialogHeader>
   236→          <DialogTitle className="flex items-center gap-2">
   237→            <Pen className="h-4 w-4 text-emerald-600" /> Draw Your Signature
   238→          </DialogTitle>
   239→        </DialogHeader>
   240→        <div className="space-y-3">
   241→          <div className="border-2 border-dashed border-border rounded-lg overflow-hidden bg-white">
   242→            <canvas
   243→              ref={canvasRef}
   244→              width={460}
   245→              height={200}
   246→              onMouseDown={startDraw}
   247→              onMouseMove={draw}
   248→              onMouseUp={endDraw}
   249→              onMouseLeave={endDraw}
   250→              className="w-full cursor-crosshair touch-none"
   251→            />
   252→          </div>
   253→          <div className="flex items-center justify-between">
   254→            <Button variant="ghost" size="sm" onClick={clearCanvas} className="text-xs">
   255→              <Eraser className="h-3.5 w-3.5 mr-1" /> Clear
   256→            </Button>
   257→            <div className="flex gap-2">
   258→              <Button variant="outline" size="sm" onClick={onClose} className="text-xs">Cancel</Button>
   259→              <Button size="sm" onClick={save}
   260→                className="text-xs gradient-emerald text-white border-0 premium-shadow hover:opacity-90">
   261→                <Check className="h-3.5 w-3.5 mr-1" /> Save Signature
   262→              </Button>
   263→            </div>
   264→          </div>
   265→        </div>
   266→      </DialogContent>
   267→    </Dialog>
   268→  );
   269→}

   275→interface StampSignatureManagerProps {
   276→  /** If provided, the hook instance is used instead of creating one internally */
   277→  hook?: UseStampSignatureReturn;
   278→  /** Whether to show the drawing canvas feature */
   279→  showDraw?: boolean;
   280→  /** Whether to show stamp templates */
   281→  showStampTemplates?: boolean;
   282→  /** Whether to show upload buttons */
   283→  showUpload?: boolean;
   284→  /** Whether to show the gallery of saved items */
   285→  showGallery?: boolean;
   286→  /** Compact mode for inline use */
   287→  compact?: boolean;
   288→  /** Callback when a stamp/signature is selected */
   289→  onSelect?: (item: SavedSignature) => void;
   290→  /** Title override */
   291→  title?: string;
   292→}
   293→
   294→export function StampSignatureManager({
   295→  hook: externalHook,
   296→  showDraw = true,
   297→  showStampTemplates = true,
   298→  showUpload = true,
   299→  showGallery = true,
   300→  compact = false,
   301→  onSelect,
   302→  title,
   303→}: StampSignatureManagerProps) {
   304→  const internalHook = useStampSignature();
   305→  const { savedItems, addSignature, removeItem, uploadFromFile, generateStamp } = externalHook || internalHook;
   306→
   307→  const [drawOpen, setDrawOpen] = useState(false);
   308→
   309→  const handleDrawSave = (dataUrl: string) => {
   310→    const sigCount = savedItems.filter(s => s.type === 'signature').length + 1;
   311→    addSignature(dataUrl, `Signature ${sigCount}`, 'signature');
   312→    toast.success('Signature saved');
   313→    setDrawOpen(false);
   314→  };
   315→
   316→  const handleUploadSignature = () => {
   317→    uploadFromFile('signature');
   318→  };
   319→
   320→  const handleUploadStamp = () => {
   321→    uploadFromFile('stamp');
   322→  };
   323→
   324→  const handleAddStamp = (text: string) => {
   325→    const dataUrl = generateStamp(text);
   326→    addSignature(dataUrl, `${text} Stamp`, 'stamp');
   327→    toast.success(`${text} stamp created`);
   328→  };
   329→
   330→  if (compact) {
   331→    return (
   332→      <div className="space-y-3">
   333→        <SignatureDrawingCanvas open={drawOpen} onClose={() => setDrawOpen(false)} onSave={handleDrawSave} />
   334→
   335→        {/* Action buttons */}
   336→        <div className="flex flex-wrap gap-2">
   337→          {showDraw && (
   338→            <Button variant="outline" size="sm" className="h-8 text-xs rounded-lg" onClick={() => setDrawOpen(true)}>
   339→              <Pen className="h-3.5 w-3.5 mr-1.5" /> Draw
   340→            </Button>
   341→          )}
   342→          {showUpload && (
   343→            <>
   344→              <Button variant="outline" size="sm" className="h-8 text-xs rounded-lg" onClick={handleUploadSignature}>
   345→                <Upload className="h-3.5 w-3.5 mr-1.5" /> Upload Sig
   346→              </Button>
   347→              <Button variant="outline" size="sm" className="h-8 text-xs rounded-lg" onClick={handleUploadStamp}>
   348→                <Stamp className="h-3.5 w-3.5 mr-1.5" /> Upload Stamp
   349→              </Button>
   350→            </>
   351→          )}
   352→          {showStampTemplates && (
   353→            <div className="flex flex-wrap gap-1">
   354→              {STAMP_TEMPLATES.map(st => (
   355→                <button
   356→                  key={st.text}
   357→                  onClick={() => handleAddStamp(st.text)}
   358→                  className="h-8 px-2.5 text-[10px] font-semibold rounded-lg border border-border/60 hover:bg-muted transition-colors"
   359→                  style={{ color: st.color }}
   360→                >
   361→                  {st.label}
   362→                </button>
   363→              ))}
   364→            </div>
   365→          )}
   366→        </div>
   367→
   368→        {/* Gallery */}
   369→        {showGallery && savedItems.length > 0 && (
   370→          <div className="flex gap-2 overflow-x-auto pb-1">
   371→            {savedItems.map(item => (
   372→              <div key={item.id} className="flex flex-col items-center gap-1 flex-shrink-0 group relative">
   373→                <button
   374→                  onClick={() => onSelect?.(item)}
   375→                  className={`w-16 h-14 border border-border rounded-lg hover:border-emerald-400 transition-colors overflow-hidden bg-white p-1 ${onSelect ? 'cursor-pointer' : 'cursor-default'}`}
   376→                >
   377→                  <img src={item.dataUrl} alt={item.label} className="max-w-full max-h-full object-contain" />
   378→                </button>
   379→                <span className="text-[9px] text-muted-foreground truncate max-w-[64px]">{item.label}</span>
   380→                <button
   381→                  onClick={(e) => { e.stopPropagation(); removeItem(item.id); }}
   382→                  className="absolute -top-1 -right-1 w-4 h-4 rounded-full bg-rose-500 text-white flex items-center justify-center opacity-0 group-hover:opacity-100 transition-opacity"
   383→                >
   384→                  <X className="h-2.5 w-2.5" />
   385→                </button>
   386→              </div>
   387→            ))}
   388→          </div>
   389→        )}
   390→      </div>
   391→    );
   392→  }
   393→
   394→  return (
   395→    <Card className="premium-shadow rounded-xl border-0 bg-card overflow-hidden">
   396→      <CardHeader className="pb-3">
   397→        <CardTitle className="text-base font-bold flex items-center gap-2">
   398→          <div className="p-1.5 rounded-lg bg-orange-500/10">
   399→            <FileSignature className="h-3.5 w-3.5 text-orange-600" />
   400→          </div>
   401→          {title || 'Signature & Stamp'}
   402→          {savedItems.length > 0 && (
   403→            <Badge className="text-[10px] px-1.5 py-0 bg-orange-50 text-orange-700 border-0 font-medium ml-1">
   404→              {savedItems.length}
   405→            </Badge>
   406→          )}
   407→        </CardTitle>
   408→      </CardHeader>
   409→      <CardContent className="space-y-5">
   410→        <SignatureDrawingCanvas open={drawOpen} onClose={() => setDrawOpen(false)} onSave={handleDrawSave} />
   411→
   412→        {/* Upload Section */}
   413→        {showUpload && (
   414→          <div>
   415→            <p className="text-xs font-semibold text-muted-foreground uppercase tracking-wider mb-2">Upload</p>
   416→            <div className="grid grid-cols-2 gap-3">

   421→                <div className="p-2 rounded-lg bg-emerald-50">
   422→                  <ImageIcon className="h-4 w-4 text-emerald-600" />
   423→                </div>
   424→                <span className="text-xs font-medium text-foreground">Upload Signature</span>
   425→                <span className="text-[10px] text-muted-foreground">PNG, JPG, SVG</span>

   431→                <div className="p-2 rounded-lg bg-orange-50">
   432→                  <Stamp className="h-4 w-4 text-orange-600" />
   433→                </div>
   434→                <span className="text-xs font-medium text-foreground">Upload Stamp</span>
   435→                <span className="text-[10px] text-muted-foreground">PNG, JPG, SVG</span>
   436→              </button>

   452→          </div>
