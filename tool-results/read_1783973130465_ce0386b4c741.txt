    60→    60→function daysUntil(dateStr: string): number {
    61→    61→  return Math.ceil((new Date(dateStr).getTime() - Date.now()) / (1000 * 60 * 60 * 24));
    62→    62→}
    63→    63→
    64→    64→function deadlineColor(days: number) {
    65→    65→  if (days <= 0) return 'rose';
    66→    66→  if (days <= 3) return 'rose';
    67→    67→  if (days <= 7) return 'amber';
    68→    68→  return 'emerald';
    69→    69→}
    70→    70→
    71→    71→function deadlineBg(days: number) {
    72→    72→  const c = deadlineColor(days);
    73→    73→  const map: Record<string, string> = {
    74→    74→    rose: 'bg-rose-50 text-rose-700',
    75→    75→    amber: 'bg-amber-50 text-amber-700',
    76→    76→    emerald: 'bg-emerald-50 text-emerald-700',
    77→    77→  };
    78→    78→  return map[c] || map.emerald;
    79→    79→}

    81→    81→function formatBudget(min: number, max: number) {
    82→    82→  const fmt = (n: number) => n >= 1_000_000 ? `${(n / 1_000_000).toFixed(1)}M` : n >= 1_000 ? `${(n / 1_000).toFixed(0)}K` : n.toLocaleString();
    83→    83→  return `ETB ${fmt(min)} – ${fmt(max)}`;
    84→    84→}
