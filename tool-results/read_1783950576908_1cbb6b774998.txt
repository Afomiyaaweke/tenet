   254→/* ─────────────────────────────────────────────────────────────────────
   255→ * Utility helpers

   257→
   258→function parseAmount(value: number | string | undefined): number {
   259→  if (value === undefined || value === null) return 0;
   260→  if (typeof value === 'number') return value;
   261→  const n = Number(String(value).replace(/[^0-9.-]/g, ''));
   262→  return Number.isFinite(n) ? n : 0;
   263→}
   264→
   265→function truncate(text: string, max = 280): string {
   266→  if (!text) return '';
   267→  return text.length > max ? text.slice(0, max).trimEnd() + '…' : text;
   268→}

   288→}
   289→
   290→interface WorldBankResponse {
   291→  total?: number;
   292→  rows?: WorldBankProcurementRow[];
   293→  procurements?: WorldBankProcurementRow[];
   294→  [k: string]: unknown;
   295→}
   296→
   297→function inferCategory(row: WorldBankProcurementRow): string {
   298→  const t = (row.contract_type || '').toLowerCase();
   299→  if (t.includes('work') || t.includes('civil') || t.includes('construction')) return 'Construction';
   300→  if (t.includes('consult')) return 'Consulting';
   301→  if (t.includes('good') || t.includes('supply') || t.includes('equipment')) return 'Supply';
   302→  if (t.includes('service')) return 'Consulting';
   303→  const p = (row.project_name || '').toLowerCase();
   304→  if (p.includes('road') || p.includes('bridge') || p.includes('infrastructure')) return 'Construction';
   305→  if (p.includes('agric') || p.includes('farm') || p.includes('irrigat')) return 'Agriculture';
   306→  if (p.includes('health') || p.includes('clinic') || p.includes('hospital')) return 'Healthcare';
   307→  if (p.includes('energy') || p.includes('power') || p.includes('electric')) return 'Engineering';
   308→  if (p.includes('edu') || p.includes('school')) return 'Education';
   309→  if (p.includes('water') || p.includes('sanitation')) return 'Engineering';
   310→  return 'Consulting';
   311→}
   312→
   313→function buildDeadline(signingDate?: string): string {
   314→  if (!signingDate) return new Date(Date.now() + 45 * 86400000).toISOString();
   315→  const base = new Date(signingDate);
   316→  if (Number.isNaN(base.getTime())) return new Date(Date.now() + 45 * 86400000).toISOString();
   317→  return new Date(base.getTime() + 45 * 86400000).toISOString();
   318→}

   320→export async function fetchWorldBankTenders(opts: {
   321→  search?: string;
   322→  rows?: number;
   323→  offset?: number;
   324→}): Promise<{ tenders: LiveTender[]; total: number; ok: boolean }> {
   325→  const rows = Math.min(Math.max(opts.rows ?? 20, 1), 500);
   326→  const offset = opts.offset || 0;
   327→  const params = new URLSearchParams({
   328→    format: 'json',
   329→    rows: String(rows),
   330→    fl: [
   331→      'project_id',
   332→      'procurement_group_id',
   333→      'project_name',
   334→      'borrower',
   335→      'country',
   336→      'region',
   337→      'contract_description',
   338→      'contract_type',
   339→      'supplier',
   340→      'supplier_country',
   341→      'contract_signing_date',
   342→      'total_contract_amount',
   343→      'wb_contract_number',
   344→    ].join(','),
   345→  });
   346→  if (opts.search) params.set('q', opts.search);
   347→  if (offset > 0) params.set('osstart', String(offset + 1));
   348→
   349→  const url = `https://search.worldbank.org/api/v2/procurement?${params.toString()}`;
   350→  const ctrl = new AbortController();
   351→  const timer = setTimeout(() => ctrl.abort(), 8000);
   352→
   353→  try {
   354→    const res = await fetch(url, {
   355→      signal: ctrl.signal,
   356→      headers: { Accept: 'application/json', 'User-Agent': 'Tenet-Tender-Ecosystem/1.0' },
   357→      cache: 'no-store',
   358→    });
   359→    clearTimeout(timer);
   360→    if (!res.ok) {
   361→      return { tenders: [], total: 0, ok: false };
   362→    }
   363→    const json = (await res.json()) as WorldBankResponse;
   364→    const rowsArr = json.rows || json.procurements || [];
   365→    const total = typeof json.total === 'number' ? json.total : rowsArr.length;
   366→
   367→    const tenders: LiveTender[] = rowsArr
   368→      .filter((r) => r && (r.contract_description || r.project_name))
   369→      .map((r, idx) => {
   370→        const externalId = r.procurement_group_id || r.project_id || `wb-${idx}`;
   371→        const title = truncate(r.contract_description || r.project_name || 'World Bank Contract Award', 160);
   372→        const amount = parseAmount(r.total_contract_amount);
   373→        const signingDate = r.contract_signing_date;
   374→        return {
   375→          id: `worldbank-${externalId}`,
   376→          title,
   377→          scope: truncate(r.contract_description || r.project_name || '', 400),
   378→          budgetMin: amount,
   379→          budgetMax: amount,
   380→          deadline: buildDeadline(signingDate),
   381→          location: r.country || r.region || 'Multiple',
   382→          categoryTags: inferCategory(r),
   383→          requiredDocs: r.project_id
   384→            ? `https://projects.worldbank.org/en/projects-operations/project-detail/${r.project_id}`
   385→            : 'https://financesone.worldbank.org',
   386→          status: 'awarded',
   387→          createdBy: 'worldbank',
   388→          createdAt: signingDate || new Date().toISOString(),
   389→          updatedAt: signingDate || new Date().toISOString(),
   390→          source: 'worldbank',
   391→          externalId,
   392→          externalUrl: r.project_id
   393→            ? `https://projects.worldbank.org/en/projects-operations/project-detail/${r.project_id}`
   394→            : 'https://financesone.worldbank.org',
   395→          currency: 'USD',
   396→          borrower: r.borrower || undefined,
   397→          supplier: r.supplier || undefined,
   398→          contractType: r.contract_type || undefined,
   399→          signingDate: signingDate || undefined,
   400→          region: r.region || undefined,
   401→          documentUrl: r.project_id ? `https://projects.worldbank.org/en/projects-operations/project-detail/${r.project_id}` : undefined,
   402→        } satisfies LiveTender;
   403→      });
   404→
   405→    return { tenders, total, ok: true };
   406→  } catch {
   407→    clearTimeout(timer);
   408→    return { tenders: [], total: 0, ok: false };
   409→  }
   410→}

   412→/* ─────────────────────────────────────────────────────────────────────
   413→ * EU TED adapter (live, public RSS-style feed)
   414→ * ───────────────────────────────────────────────────────────────────── */
   415→
   416→export async function fetchEuTedTenders(opts: {
   417→  search?: string;
   418→  rows?: number;
   419→  offset?: number;
   420→}): Promise<{ tenders: LiveTender[]; total: number; ok: boolean }> {
   421→  const rows = Math.min(Math.max(opts.rows ?? 20, 1), 500);
   422→  const offset = opts.offset || 0;
   423→  const base = 'https://api.ted.europa.eu/v3/notices/search';
   424→  const body = {
   425→    query: opts.search ? `(${opts.search})` : '*',
   426→    fields: ['ND', 'TI_DOC', 'DS_DATE_PUB', 'DEADLINE_DATE', 'PLACE_CONTRACT', 'VALUE_CONTRACT', 'CA_NAME', 'TITLE'],
   427→    limit: rows,
   428→    offset,
   429→    pagination: { pageNumber: Math.floor(offset / rows) + 1, pageSize: rows },
   430→    sort: { field: 'DS_DATE_PUB', order: 'DESC' },
   431→  };
   432→  const ctrl = new AbortController();
   433→  const timer = setTimeout(() => ctrl.abort(), 8000);
   434→
   435→  try {
   436→    const res = await fetch(base, {
   437→      method: 'POST',
   438→      signal: ctrl.signal,
   439→      headers: { 'Content-Type': 'application/json', Accept: 'application/json' },
   440→      body: JSON.stringify(body),
   441→      cache: 'no-store',
   442→    });
   443→    clearTimeout(timer);
   444→    if (!res.ok) return { tenders: [], total: 0, ok: false };
   445→    const json = (await res.json()) as { notices?: unknown[]; total?: number };
   446→    const notices = Array.isArray(json.notices) ? json.notices : [];
   447→    const total = typeof json.total === 'number' ? json.total : notices.length;
   448→
   449→    const tenders: LiveTender[] = notices.map((n, idx) => {
   450→      const row = (n || {}) as Record<string, unknown>;
   451→      const nd = String(row.ND || row['ND'] || `ted-${idx}`);
   452→      const title = String(row.TI_DOC || row.TITLE || 'EU TED Notice');
   453→      const deadlineRaw = row.DEADLINE_DATE as string | undefined;
   454→      const amount = Number(row.VALUE_CONTRACT) || 0;
   455→      return {
   456→        id: `eu_ted-${nd}`,
   457→        title: truncate(title, 160),
   458→        scope: truncate(title, 400),
   459→        budgetMin: amount,
   460→        budgetMax: amount,
   461→        deadline: deadlineRaw || new Date(Date.now() + 30 * 86400000).toISOString(),
   462→        location: String(row.PLACE_CONTRACT || 'EU'),
   463→        categoryTags: 'Consulting',
   464→        requiredDocs: `https://ted.europa.eu/udl?uri=TED:NOTICE:${nd}:TEXT:EN:HTML`,
   465→        status: 'open',
   466→        createdBy: 'eu_ted',
   467→        createdAt: String(row.DS_DATE_PUB || new Date().toISOString()),
   468→        updatedAt: String(row.DS_DATE_PUB || new Date().toISOString()),
   469→        source: 'eu_ted',
   470→        externalId: nd,
   471→        externalUrl: `https://ted.europa.eu/udl?uri=TED:NOTICE:${nd}:TEXT:EN:HTML`,
   472→        currency: 'EUR',
   473→        borrower: String(row.CA_NAME || '') || undefined,
   474→        documentUrl: `https://ted.europa.eu/udl?uri=TED:NOTICE:${nd}:TEXT:EN:HTML`,
   475→      } satisfies LiveTender;
   476→    });
   477→
   478→    return { tenders, total, ok: true };
   479→  } catch {
   480→    clearTimeout(timer);
   481→    return { tenders: [], total: 0, ok: false };
   482→  }
   483→}

   485→/* ─────────────────────────────────────────────────────────────────────
   486→ * UNGM — UN Global Procurement adapter (live via RSS)
   487→ * ───────────────────────────────────────────────────────────────────── */
   488→
   489→export async function fetchUngmTenders(opts: {
   490→  search?: string;
   491→  rows?: number;
   492→  offset?: number;
   493→}): Promise<{ tenders: LiveTender[]; total: number; ok: boolean }> {
   494→  const ctrl = new AbortController();
   495→  const timer = setTimeout(() => ctrl.abort(), 8000);
   496→
   497→  try {
   498→    const res = await fetch('https://www.ungm.org/Public/Notice/RSS', {
   499→      signal: ctrl.signal,
   500→      headers: { 'User-Agent': 'Tenet-Tender-Ecosystem/1.0', Accept: 'application/rss+xml, text/xml, */*' },
   501→      cache: 'no-store',
   502→    });
   503→    clearTimeout(timer);
   504→
   505→    if (!res.ok) return { tenders: [], total: 0, ok: false };
   506→
   507→    const xml = await res.text();
   508→    const itemRegex = /<item[\s\S]*?<title>([\s\S]*?)<\/title>[\s\S]*?<link>([\s\S]*?)<\/link>[\s\S]*?<description>([\s\S]*?)<\/description>[\s\S]*?<\/item>/gi;
   509→    const matches = [...xml.matchAll(itemRegex)];
   510→
   511→    if (matches.length === 0) return { tenders: [], total: 0, ok: false };
   512→
   513→    const tenders: LiveTender[] = matches.slice(0, opts.rows ?? 10).map((m, idx) => {
   514→      const rawTitle = m[1]?.replace(/<!\[CDATA\[([\s\S]*?)\]\]>/, '$1').trim() || `UNGM Notice ${idx + 1}`;
   515→      const rawLink = m[2]?.trim() || 'https://www.ungm.org';
   516→      const rawDesc = m[3]?.replace(/<!\[CDATA\[([\s\S]*?)\]\]>/, '$1').trim() || '';
   517→      return {
   518→        id: `ungm-rss-${idx}`,
   519→        title: truncate(rawTitle, 160),
   520→        scope: truncate(rawDesc, 400),
   521→        budgetMin: 0,
   522→        budgetMax: 0,
   523→        deadline: new Date(Date.now() + 30 * 86400000).toISOString(),
