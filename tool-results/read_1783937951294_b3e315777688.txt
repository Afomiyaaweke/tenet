   227→   227→
   228→   228→function parseAmount(value: number | string | undefined): number {
   229→   229→  if (value === undefined || value === null) return 0;
   230→   230→  if (typeof value === 'number') return value;
   231→   231→  const n = Number(String(value).replace(/[^0-9.-]/g, ''));
   232→   232→  return Number.isFinite(n) ? n : 0;
   233→   233→}
   234→   234→
   235→   235→function truncate(text: string, max = 280): string {
   236→   236→  if (!text) return '';
   237→   237→  return text.length > max ? text.slice(0, max).trimEnd() + '…' : text;
   238→   238→}

   256→   256→  total_contract_amount?: number | string;

   260→   260→interface WorldBankResponse {
   261→   261→  total?: number;

   266→   266→
   267→   267→function inferCategory(row: WorldBankProcurementRow): string {
   268→   268→  const t = (row.contract_type || '').toLowerCase();
   269→   269→  if (t.includes('work') || t.includes('civil') || t.includes('construction')) return 'Construction';
   270→   270→  if (t.includes('consult')) return 'Consulting';
   271→   271→  if (t.includes('good') || t.includes('supply') || t.includes('equipment')) return 'Supply';
   272→   272→  if (t.includes('service')) return 'Consulting';
   273→   273→  const p = (row.project_name || '').toLowerCase();
   274→   274→  if (p.includes('road') || p.includes('bridge') || p.includes('infrastructure')) return 'Construction';
   275→   275→  if (p.includes('agric') || p.includes('farm') || p.includes('irrigat')) return 'Agriculture';
   276→   276→  if (p.includes('health') || p.includes('clinic') || p.includes('hospital')) return 'Healthcare';
   277→   277→  if (p.includes('energy') || p.includes('power') || p.includes('electric')) return 'Engineering';
   278→   278→  if (p.includes('edu') || p.includes('school')) return 'Education';
   279→   279→  if (p.includes('water') || p.includes('sanitation')) return 'Engineering';
   280→   280→  return 'Consulting';
   281→   281→}
   282→   282→
   283→   283→function buildDeadline(signingDate?: string): string {
   284→   284→  if (!signingDate) return new Date(Date.now() + 45 * 86400000).toISOString();
   285→   285→  const base = new Date(signingDate);
   286→   286→  if (Number.isNaN(base.getTime())) return new Date(Date.now() + 45 * 86400000).toISOString();
   287→   287→  return new Date(base.getTime() + 45 * 86400000).toISOString();
   288→   288→}

   290→   290→export async function fetchWorldBankTenders(opts: {
   291→   291→  search?: string;
   292→   292→  rows?: number;
   293→   293→}): Promise<{ tenders: LiveTender[]; total: number; ok: boolean }> {
   294→   294→  const rows = Math.min(Math.max(opts.rows ?? 20, 1), 500);
   295→   295→  const params = new URLSearchParams({
   296→   296→    format: 'json',
   297→   297→    rows: String(rows),
   298→   298→    fl: [
   299→   299→      'project_id',
   300→   300→      'procurement_group_id',
   301→   301→      'project_name',
   302→   302→      'borrower',
   303→   303→      'country',
   304→   304→      'region',
   305→   305→      'contract_description',
   306→   306→      'contract_type',
   307→   307→      'supplier',
   308→   308→      'supplier_country',
   309→   309→      'contract_signing_date',
   310→   310→      'total_contract_amount',
   311→   311→      'wb_contract_number',
   312→   312→    ].join(','),
   313→   313→  });
   314→   314→  if (opts.search) params.set('q', opts.search);
   315→   315→
   316→   316→  const url = `https://search.worldbank.org/api/v2/procurement?${params.toString()}`;
   317→   317→  const ctrl = new AbortController();
   318→   318→  const timer = setTimeout(() => ctrl.abort(), 8000);
   319→   319→
   320→   320→  try {
   321→   321→    const res = await fetch(url, {
   322→   322→      signal: ctrl.signal,
   323→   323→      headers: { Accept: 'application/json', 'User-Agent': 'Tenet-Tender-Ecosystem/1.0' },
   324→   324→      cache: 'no-store',
   325→   325→    });
   326→   326→    clearTimeout(timer);
   327→   327→    if (!res.ok) {
   328→   328→      return { tenders: [], total: 0, ok: false };
   329→   329→    }
   330→   330→    const json = (await res.json()) as WorldBankResponse;
   331→   331→    const rowsArr = json.rows || json.procurements || [];
   332→   332→    const total = typeof json.total === 'number' ? json.total : rowsArr.length;
   333→   333→
   334→   334→    const tenders: LiveTender[] = rowsArr
   335→   335→      .filter((r) => r && (r.contract_description || r.project_name))
   336→   336→      .map((r, idx) => {
   337→   337→        const externalId = r.procurement_group_id || r.project_id || `wb-${idx}`;
   338→   338→        const title = truncate(r.contract_description || r.project_name || 'World Bank Contract Award', 160);
   339→   339→        const amount = parseAmount(r.total_contract_amount);
   340→   340→        const signingDate = r.contract_signing_date;
   341→   341→        return {
   342→   342→          id: `worldbank-${externalId}`,
   343→   343→          title,
   344→   344→          scope: truncate(r.contract_description || r.project_name || '', 400),
   345→   345→          budgetMin: amount,
   346→   346→          budgetMax: amount,
   347→   347→          deadline: buildDeadline(signingDate),
   348→   348→          location: r.country || r.region || 'Multiple',
   349→   349→          categoryTags: inferCategory(r),
   350→   350→          requiredDocs: r.project_id
   351→   351→            ? `https://projects.worldbank.org/en/projects-operations/project-detail/${r.project_id}`
   352→   352→            : 'https://financesone.worldbank.org',
   353→   353→          status: 'awarded',
   354→   354→          createdBy: 'worldbank',
   355→   355→          createdAt: signingDate || new Date().toISOString(),
   356→   356→          updatedAt: signingDate || new Date().toISOString(),
   357→   357→          source: 'worldbank',
   358→   358→          externalId,
   359→   359→          externalUrl: r.project_id
   360→   360→            ? `https://projects.worldbank.org/en/projects-operations/project-detail/${r.project_id}`
   361→   361→            : 'https://financesone.worldbank.org',
   362→   362→          currency: 'USD',
   363→   363→          borrower: r.borrower || undefined,
   364→   364→          supplier: r.supplier || undefined,
   365→   365→          contractType: r.contract_type || undefined,
   366→   366→          signingDate: signingDate || undefined,
   367→   367→          region: r.region || undefined,
   368→   368→        } satisfies LiveTender;
   369→   369→      });
   370→   370→
   371→   371→    return { tenders, total, ok: true };
   372→   372→  } catch {
   373→   373→    clearTimeout(timer);
   374→   374→    return { tenders: [], total: 0, ok: false };
   375→   375→  }
   376→   376→}

   418→   418→      const amount = Number(row.VALUE_CONTRACT) || 0;
   419→   419→      return {
   420→   420→        id: `eu_ted-${nd}`,
   421→   421→        title: truncate(title, 160),
   422→   422→        scope: truncate(title, 400),
   423→   423→        budgetMin: amount,
   424→   424→        budgetMax: amount,
   425→   425→        deadline: deadlineRaw || new Date(Date.now() + 30 * 86400000).toISOString(),
   426→   426→        location: String(row.PLACE_CONTRACT || 'EU'),
   427→   427→        categoryTags: 'Consulting',
   428→   428→        requiredDocs: `https://ted.europa.eu/udl?uri=TED:NOTICE:${nd}:TEXT:EN:HTML`,
   429→   429→        status: 'open',
   430→   430→        createdBy: 'eu_ted',
   431→   431→        createdAt: String(row.DS_DATE_PUB || new Date().toISOString()),
   432→   432→        updatedAt: String(row.DS_DATE_PUB || new Date().toISOString()),
   433→   433→        source: 'eu_ted',
   434→   434→        externalId: nd,
   435→   435→        externalUrl: `https://ted.europa.eu/udl?uri=TED:NOTICE:${nd}:TEXT:EN:HTML`,
   436→   436→        currency: 'EUR',
   437→   437→        borrower: String(row.CA_NAME || '') || undefined,
   438→   438→      } satisfies LiveTender;
   439→   439→    });
   440→   440→
   441→   441→    return { tenders, total, ok: true };
   442→   442→  } catch {
   443→   443→    clearTimeout(timer);
