    89→    89→/* ───────────────────────── Password Strength Meter ───────────────────────── */
    90→    90→function scorePassword(pw: string): { score: number; label: string; color: string } {
    91→    91→  let score = 0;
    92→    92→  if (pw.length >= 8) score++;
    93→    93→  if (pw.length >= 12) score++;
    94→    94→  if (/[a-z]/.test(pw) && /[A-Z]/.test(pw)) score++;
    95→    95→  if (/[0-9]/.test(pw)) score++;
    96→    96→  if (/[^A-Za-z0-9]/.test(pw)) score++;
    97→    97→  const labels = ['Too weak', 'Weak', 'Fair', 'Good', 'Strong', 'Very strong'];
    98→    98→  const colors = ['bg-rose-500', 'bg-rose-500', 'bg-amber-500', 'bg-yellow-500', 'bg-emerald-500', 'bg-emerald-600'];
    99→    99→  return { score, label: labels[score], color: colors[score] };
   100→   100→}

   103→   103→function SlideCaptcha({ onVerified }: { onVerified: () => void }) {
   104→   104→  const [offset, setOffset] = useState(0);
   105→   105→  const [status, setStatus] = useState<'idle' | 'dragging' | 'success' | 'fail'>('idle');
   106→   106→  const [target] = useState(() => 180 + Math.floor(Math.random() * 80)); // target px 180–260
   107→   107→  const trackWidth = 320;
   108→   108→  const handleWidth = 44;
   109→   109→  const tolerance = 8;
   110→   110→  const dragRef = React.useRef<{ start: number; startOffset: number } | null>(null);
   111→   111→
   112→   112→  const onPointerDown = (e: React.PointerEvent) => {
   113→   113→    if (status === 'success') return;
   114→   114→    setStatus('dragging');
   115→   115→    dragRef.current = { start: e.clientX, startOffset: offset };
   116→   116→    (e.target as HTMLElement).setPointerCapture(e.pointerId);
   117→   117→  };
   118→   118→
   119→   119→  const onPointerMove = (e: React.PointerEvent) => {
   120→   120→    if (!dragRef.current || status !== 'dragging') return;
   121→   121→    const delta = e.clientX - dragRef.current.start;
   122→   122→    const next = Math.max(0, Math.min(trackWidth - handleWidth, dragRef.current.startOffset + delta));
   123→   123→    setOffset(next);
   124→   124→  };
   125→   125→
   126→   126→  const onPointerUp = () => {
   127→   127→    if (status !== 'dragging') return;
   128→   128→    dragRef.current = null;
   129→   129→    if (Math.abs(offset - target) <= tolerance) {
   130→   130→      setStatus('success');
   131→   131→      setTimeout(() => onVerified(), 450);
   132→   132→    } else {
   133→   133→      setStatus('fail');
   134→   134→      setTimeout(() => {
   135→   135→        setStatus('idle');
   136→   136→        setOffset(0);
   137→   137→      }, 700);
   138→   138→    }
   139→   139→  };
   140→   140→
   141→   141→  const pieceColor = status === 'success' ? 'bg-emerald-500' : status === 'fail' ? 'bg-rose-500' : 'bg-primary';
   142→   142→  const ringColor = status === 'success' ? 'ring-emerald-300' : status === 'fail' ? 'ring-rose-300' : 'ring-primary/40';
   143→   143→
   144→   144→  return (
   145→   145→    <div className="space-y-3">
   146→   146→      <div className="text-xs font-medium text-muted-foreground flex items-center gap-1.5">
   147→   147→        <Puzzle className="w-3.5 h-3.5" /> Security verification — slide to complete
   148→   148→      </div>
   149→   149→      {/* Puzzle canvas */}
   150→   150→      <div className="relative h-12 rounded-xl overflow-hidden border border-border bg-muted/40 select-none">
   151→   151→        {/* pattern background */}
   152→   152→        <div
   153→   153→          className="absolute inset-0 opacity-50"
   154→   154→          style={{
   155→   155→            backgroundImage:
   156→   156→              'repeating-linear-gradient(45deg, hsl(var(--muted-foreground) / 0.08) 0 6px, transparent 6px 12px), repeating-linear-gradient(-45deg, hsl(var(--muted-foreground) / 0.06) 0 6px, transparent 6px 12px)',
   157→   157→          }}
   158→   158→        />
   159→   159→        {/* target cutout (notch) */}
   160→   160→        <div
   161→   161→          className="absolute top-1/2 -translate-y-1/2 h-9 w-9 rounded-lg border-2 border-dashed border-muted-foreground/40 bg-background/60"
   162→   162→          style={{ left: target + 4 }}
   163→   163→        />
   164→   164→        {/* moving puzzle piece */}
   165→   165→        <div
   166→   166→          className={`absolute top-1/2 -translate-y-1/2 h-9 w-9 rounded-lg ${pieceColor} shadow-lg ring-2 ${ringColor} flex items-center justify-center text-white transition-shadow ${status === 'fail' ? 'animate-[shake_0.4s_ease-in-out]' : ''}`}
   167→   167→          style={{ left: 4 + offset, transition: status === 'dragging' ? 'none' : 'left 0.25s ease' }}
   168→   168→        >
   169→   169→          {status === 'success' ? <CheckCircle2 className="w-5 h-5" /> : status === 'fail' ? <XCircle className="w-5 h-5" /> : <Puzzle className="w-4 h-4" />}
   170→   170→        </div>
   171→   171→        {/* hint text */}
   172→   172→        {status === 'idle' && (
   173→   173→          <span className="absolute inset-0 flex items-center justify-center text-xs text-muted-foreground pointer-events-none">
   174→   174→            Drag the slider below to fit the puzzle
   175→   175→          </span>
   176→   176→        )}
   177→   177→        {status === 'success' && (
   178→   178→          <span className="absolute inset-0 flex items-center justify-center text-xs font-medium text-emerald-600 bg-emerald-50/60 pointer-events-none">
   179→   179→            Verified
   180→   180→          </span>
   181→   181→        )}
   182→   182→      </div>
   183→   183→      {/* Slider track */}
   184→   184→      <div className="relative h-11 rounded-xl bg-muted border border-border overflow-hidden">
   185→   185→        <div
   186→   186→          className={`absolute inset-y-0 left-0 ${status === 'success' ? 'bg-emerald-100 dark:bg-emerald-950/40' : status === 'fail' ? 'bg-rose-100 dark:bg-rose-950/40' : 'bg-primary/10'} transition-all`}
   187→   187→          style={{ width: offset + handleWidth }}
   188→   188→        />
   189→   189→        <div
   190→   190→          className="absolute top-0 h-full flex items-center justify-center text-xs font-medium text-muted-foreground pointer-events-none"
   191→   191→          style={{ left: handleWidth, right: 0 }}
   192→   192→        >
   193→   193→          {status === 'success' ? 'Verification complete' : status === 'fail' ? 'Verification failed, retrying…' : 'Slide right to verify'}
   194→   194→        </div>
   195→   195→        <div
   196→   196→          onPointerDown={onPointerDown}
   197→   197→          onPointerMove={onPointerMove}
   198→   198→          onPointerUp={onPointerUp}
   199→   199→          onPointerCancel={onPointerUp}
   200→   200→          className={`absolute top-1/2 -translate-y-1/2 h-9 w-9 rounded-lg cursor-grab active:cursor-grabbing flex items-center justify-center text-white shadow-md ${pieceColor} ${status === 'success' ? 'cursor-default' : ''} touch-none`}
   201→   201→          style={{ left: 4 + offset, transition: status === 'dragging' ? 'none' : 'left 0.25s ease' }}
   202→   202→        >

   204→   204→        </div>
   205→   205→      </div>
   206→   206→    </div>
   207→   207→  );
   208→   208→}

   211→   211→function SecurityCodeInput({ value, onChange, disabled }: { value: string; onChange: (v: string) => void; disabled?: boolean }) {
   212→   212→  const inputsRef = React.useRef<(HTMLInputElement | null)[]>([]);
   213→   213→  const chars = value.split('');
   214→   214→
   215→   215→  const setChar = (i: number, c: string) => {
   216→   216→    const sanitized = c.replace(/\D/g, '').slice(-1);
   217→   217→    const arr = value.split('');
   218→   218→    arr[i] = sanitized || ' ';
   219→   219→    const next = arr.join('').replace(/ /g, '');
   220→   220→    // pad to length i+1 so typing works
   221→   221→    const padded = (next + '      ').slice(0, 6).replace(/\s/g, '');
   222→   222→    onChange(padded);
   223→   223→    if (sanitized && i < 5) inputsRef.current[i + 1]?.focus();
   224→   224→  };
   225→   225→
   226→   226→  const onKeyDown = (i: number, e: React.KeyboardEvent<HTMLInputElement>) => {
   227→   227→    if (e.key === 'Backspace' && !chars[i] && i > 0) {
   228→   228→      inputsRef.current[i - 1]?.focus();
   229→   229→    }
   230→   230→  };
   231→   231→
   232→   232→  const onPaste = (e: React.ClipboardEvent) => {
   233→   233→    const text = e.clipboardData.getData('text').replace(/\D/g, '').slice(0, 6);
   234→   234→    if (text) {
   235→   235→      e.preventDefault();
   236→   236→      onChange(text);
   237→   237→      inputsRef.current[Math.min(text.length, 5)]?.focus();
   238→   238→    }
   239→   239→  };
   240→   240→
   241→   241→  return (
   242→   242→    <div className="flex justify-between gap-2" onPaste={onPaste}>
   243→   243→      {Array.from({ length: 6 }).map((_, i) => (
   244→   244→        <input
   245→   245→          key={i}
   246→   246→          ref={(el) => { inputsRef.current[i] = el; }}
   247→   247→          type="text"
   248→   248→          inputMode="numeric"
   249→   249→          maxLength={1}
   250→   250→          disabled={disabled}
   251→   251→          value={chars[i] || ''}
   252→   252→          onChange={(e) => setChar(i, e.target.value)}
   253→   253→          onKeyDown={(e) => onKeyDown(i, e)}
   254→   254→          className="w-11 h-13 sm:w-12 sm:h-14 text-center text-xl font-bold rounded-xl border-2 bg-muted/30 focus:bg-background focus:border-primary focus:ring-2 focus:ring-primary/20 outline-none transition-all"
   255→   255→          style={{ aspectRatio: '1 / 1.15' }}
   256→   256→        />
   257→   257→      ))}
   258→   258→    </div>
   259→   259→  );
   260→   260→}
