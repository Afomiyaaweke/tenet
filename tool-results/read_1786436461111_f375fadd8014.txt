   548→function parseAmount(value: number | string | undefined): number {
   549→  if (value === undefined || value === null) return 0;
   550→  if (typeof value === 'number') return value;
   551→  const n = Number(String(value).replace(/[^0-9.-]/g, ''));
   552→  return Number.isFinite(n) ? n : 0;
   553→}
   554→
   555→function truncate(text: string, max = 280): string {
   556→  if (!text) return '';
   557→  return text.length > max ? text.slice(0, max).trimEnd() + '…' : text;
   558→}

   587→function inferCategory(row: WorldBankProcurementRow): string {
   588→  const t = (row.contract_type || '').toLowerCase();
   589→  if (t.includes('work') || t.includes('civil') || t.includes('construction')) return 'Construction';
   590→  if (t.includes('consult')) return 'Consulting';
   591→  if (t.includes('good') || t.includes('supply') || t.includes('equipment')) return 'Supply';
   592→  if (t.includes('service')) return 'Consulting';
   593→  const p = (row.project_name || '').toLowerCase();
   594→  if (p.includes('road') || p.includes('bridge') || p.includes('infrastructure')) return 'Construction';
   595→  if (p.includes('agric') || p.includes('farm') || p.includes('irrigat')) return 'Agriculture';
   596→  if (p.includes('health') || p.includes('clinic') || p.includes('hospital')) return 'Healthcare';
   597→  if (p.includes('energy') || p.includes('power') || p.includes('electric')) return 'Engineering';
   598→  if (p.includes('edu') || p.includes('school')) return 'Education';
   599→  if (p.includes('water') || p.includes('sanitation')) return 'Engineering';
   600→  return 'Consulting';
   601→}

   603→function buildDeadline(signingDate?: string): string {
   604→  if (!signingDate) return new Date(Date.now() + 45 * 86400000).toISOString();
   605→  const base = new Date(signingDate);
   606→  if (Number.isNaN(base.getTime())) return new Date(Date.now() + 45 * 86400000).toISOString();
   607→  return new Date(base.getTime() + 45 * 86400000).toISOString();
   608→}
