    24→function daysUntil(dateStr: string): number {
    25→  return Math.ceil((new Date(dateStr).getTime() - Date.now()) / (1000 * 60 * 60 * 24));
    26→}
    27→
    28→function deadlineColor(days: number) {
    29→  if (days <= 0) return 'rose';
    30→  if (days <= 3) return 'rose';
    31→  if (days <= 7) return 'amber';
    32→  return 'emerald';
    33→}
    34→
    35→function deadlineBg(days: number) {
    36→  const c = deadlineColor(days);
    37→  return `bg-${c}-50 text-${c}-700`;
    38→}
