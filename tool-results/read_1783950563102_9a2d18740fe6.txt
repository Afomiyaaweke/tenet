   368→function fmtMoney(amount: number, currency: string): string {
   369→  if (!amount) return '—';
   370→  try {
   371→    return new Intl.NumberFormat('en-US', {
   372→      style: 'currency',
   373→      currency: currency || 'USD',
   374→      maximumFractionDigits: 0,
   375→    }).format(amount);
   376→  } catch {
   377→    return `${currency} ${amount.toLocaleString()}`;
   378→  }
   379→}
   380→
   381→function daysUntil(dateStr: string): number {
   382→  if (!dateStr) return 999;
   383→  return Math.ceil((new Date(dateStr).getTime() - Date.now()) / (1000 * 60 * 60 * 24));
   384→}
   385→
   386→function deadlineBadge(days: number): string {
   387→  if (days <= 0) return 'bg-rose-50 text-rose-700 dark:bg-rose-950/40 dark:text-rose-300';
   388→  if (days <= 7) return 'bg-amber-50 text-amber-700 dark:bg-amber-950/40 dark:text-amber-300';
   389→  if (days <= 21) return 'bg-sky-50 text-sky-700 dark:bg-sky-950/40 dark:text-sky-300';
   390→  return 'bg-emerald-50 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300';
   391→}
   392→
   393→function deadlineLabel(days: number): string {
   394→  if (days <= 0) return 'Closed';
   395→  if (days <= 7) return `${days}d left`;
   396→  if (days <= 30) return `${days}d left`;
   397→  return `${days}d left`;
   398→}
   399→
   400→function relativeTime(dateStr: string): string {
   401→  if (!dateStr) return '';
   402→  const diff = Date.now() - new Date(dateStr).getTime();
   403→  const day = 86400000;
   404→  if (diff < day) return 'today';
   405→  if (diff < 2 * day) return 'yesterday';
   406→  if (diff < 30 * day) return `${Math.floor(diff / day)}d ago`;
   407→  return new Date(dateStr).toLocaleDateString();
   408→}
   409→
   410→function riskBadgeColor(level: string): string {
   411→  if (level === 'low') return 'bg-emerald-50 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300';
   412→  if (level === 'medium') return 'bg-amber-50 text-amber-700 dark:bg-amber-950/40 dark:text-amber-300';
   413→  return 'bg-rose-50 text-rose-700 dark:bg-rose-950/40 dark:text-rose-300';
   414→}

   416→function readinessColor(score: number): string {
   417→  if (score >= 8) return 'bg-emerald-500';
   418→  if (score >= 5) return 'bg-amber-500';
   419→  return 'bg-rose-500';

   422→/* ─────────────────────────────────────────────────────────────────────

   425→
   426→function InlineDocumentViewer({ doc, onClose }: { doc: InlineDocument; onClose: () => void }) {
   427→  const copyContent = () => {
   428→    const text = doc.sections
   429→      ? doc.sections.map((s) => `${s.heading}\n${s.content}`).join('\n\n')
   430→      : doc.content;
   431→    navigator.clipboard.writeText(text);
   432→    toast.success('Content copied to clipboard');
   433→  };
   434→
   435→  return (
   436→    <div className="overflow-hidden animate-[fadeIn_0.3s_ease-out]">
   437→      <div className="border-t border-border bg-gradient-to-b from-muted/30 to-background">
   438→        <div className="p-4 md:p-6 space-y-4">
   439→          {/* Document header */}
   440→          <div className="flex items-start justify-between gap-3">
   441→            <div className="min-w-0 flex-1">
   442→              <div className="flex items-center gap-2 text-xs text-muted-foreground mb-1">
   443→                <BookOpen className="h-3.5 w-3.5" />
   444→                <span>Full tender content loaded from {new URL(doc.url).hostname}</span>
   445→                <span>·</span>
   446→                <span>{new Date(doc.fetchedAt).toLocaleTimeString()}</span>
   447→              </div>
   448→              <h3 className="text-lg font-semibold text-foreground leading-snug">
   449→                {doc.title || 'Tender Document'}
   450→              </h3>
   451→              {doc.metaDescription && (
   452→                <p className="text-sm text-muted-foreground mt-1 line-clamp-2">
   453→                  {doc.metaDescription}
   454→                </p>
   455→              )}
   456→            </div>
   457→            <div className="flex items-center gap-2 shrink-0">
   458→              <Button variant="ghost" size="sm" className="gap-1.5 text-xs" onClick={copyContent}>
   459→                <Copy className="h-3.5 w-3.5" />
   460→                Copy
   461→              </Button>
   462→              <a
   463→                href={doc.url}
   464→                target="_blank"
   465→                rel="noopener noreferrer"
   466→                className="inline-flex items-center gap-1.5 text-xs font-medium text-primary hover:underline"
   467→              >
   468→                <ExternalLink className="h-3.5 w-3.5" />
   469→                Open original
   470→              </a>
   471→              <Button variant="ghost" size="sm" onClick={onClose} className="gap-1.5 text-xs text-muted-foreground">
   472→                <ChevronUp className="h-3.5 w-3.5" />
   473→                Collapse
   474→              </Button>
   475→            </div>
   476→          </div>
   477→
   478→          {/* Extracted metadata pills */}
   479→          {(doc.deadlines && doc.deadlines.length > 0) || (doc.budgets && doc.budgets.length > 0) ? (
