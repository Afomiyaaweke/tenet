   145→// ─── Content Formatting (similar to agent.tsx formatAIContent) ───────────────
   146→
   147→function formatInline(text: string): React.ReactNode {
   148→  const parts: React.ReactNode[] = [];
   149→  let remaining = text;
   150→  let partKey = 0;
   151→
   152→  while (remaining.length > 0) {
   153→    const boldMatch = remaining.match(/\*\*(.+?)\*\*/);
   154→    const italicMatch = remaining.match(/\*(.+?)\*/);
   155→
   156→    let firstMatch: { index: number; length: number; content: string; type: 'bold' | 'italic' } | null = null;
   157→
   158→    if (boldMatch && boldMatch.index !== undefined) {
   159→      firstMatch = { index: boldMatch.index, length: boldMatch[0].length, content: boldMatch[1], type: 'bold' };
   160→    }
   161→
   162→    if (italicMatch && italicMatch.index !== undefined) {
   163→      if (!firstMatch || italicMatch.index < firstMatch.index) {
   164→        firstMatch = { index: italicMatch.index, length: italicMatch[0].length, content: italicMatch[1], type: 'italic' };
   165→      }
   166→    }
   167→
   168→    if (firstMatch) {
   169→      if (firstMatch.index > 0) {
   170→        parts.push(<span key={`t-${partKey++}`}>{remaining.slice(0, firstMatch.index)}</span>);
   171→      }
   172→      if (firstMatch.type === 'bold') {
   173→        parts.push(<strong key={`b-${partKey++}`} className="font-semibold text-foreground">{firstMatch.content}</strong>);
   174→      } else {
   175→        parts.push(<em key={`i-${partKey++}`} className="italic text-muted-foreground">{firstMatch.content}</em>);
   176→      }
   177→      remaining = remaining.slice(firstMatch.index + firstMatch.length);
   178→    } else {
   179→      parts.push(<span key={`t-${partKey++}`}>{remaining}</span>);
   180→      break;
   181→    }
   182→  }
   183→
   184→  return <>{parts}</>;
   185→}

   187→function formatDocContent(content: string): React.ReactNode[] {
   188→  const lines = content.split('\n');
   189→  const result: React.ReactNode[] = [];
   190→
   191→  for (let i = 0; i < lines.length; i++) {
   192→    const line = lines[i];
   193→    const key = `doc-line-${i}`;
   194→
   195→    // Heading lines (## Title, ### Subtitle)
   196→    const h2Match = line.match(/^##\s+(.*)/);
   197→    if (h2Match) {
   198→      result.push(
   199→        <h2 key={key} className="text-xl font-bold text-foreground mt-6 mb-3 pb-2 border-b border-border/40">
   200→          {formatInline(h2Match[1])}
   201→        </h2>
   202→      );
   203→      continue;
   204→    }
   205→
   206→    const h3Match = line.match(/^###\s+(.*)/);
   207→    if (h3Match) {
   208→      result.push(
   209→        <h3 key={key} className="text-lg font-semibold text-foreground mt-4 mb-2">
   210→          {formatInline(h3Match[1])}
   211→        </h3>
   212→      );
   213→      continue;
   214→    }
   215→
   216→    const h1Match = line.match(/^#\s+(.*)/);
   217→    if (h1Match) {
   218→      result.push(
   219→        <h1 key={key} className="text-2xl font-bold text-foreground mt-4 mb-3">
   220→          {formatInline(h1Match[1])}
   221→        </h1>
   222→      );
   223→      continue;
   224→    }
   225→
   226→    // Horizontal rule
   227→    if (line.match(/^---+$/)) {
   228→      result.push(<Separator key={key} className="my-4" />);
   229→      continue;
   230→    }
   231→
   232→    // Bullet point lines
   233→    const bulletMatch = line.match(/^[\s]*[-*•]\s+(.*)/);
   234→    if (bulletMatch) {
   235→      const bulletContent = bulletMatch[1];
   236→      result.push(
   237→        <div key={key} className="flex items-start gap-2.5 ml-3 my-1">
   238→          <span className="text-emerald-500 mt-1 flex-shrink-0 text-xs">&#9679;</span>
   239→          <span className="text-sm leading-relaxed">{formatInline(bulletContent)}</span>
   240→        </div>
   241→      );
   242→      continue;
   243→    }
   244→
   245→    // Numbered list lines
   246→    const numberedMatch = line.match(/^[\s]*(\d+)[.)]\s+(.*)/);
   247→    if (numberedMatch) {
   248→      const num = numberedMatch[1];
   249→      const numberedContent = numberedMatch[2];
   250→      result.push(
   251→        <div key={key} className="flex items-start gap-2.5 ml-3 my-1">
   252→          <span className="text-emerald-600 font-semibold flex-shrink-0 text-sm min-w-[1.25rem]">{num}.</span>
   253→          <span className="text-sm leading-relaxed">{formatInline(numberedContent)}</span>
   254→        </div>
   255→      );
   256→      continue;
   257→    }
   258→
   259→    // Empty lines
   260→    if (line.trim() === '') {
   261→      result.push(<div key={key} className="h-2" />);
   262→      continue;
   263→    }
   264→
   265→    // Regular line
   266→    result.push(<div key={key} className="text-sm leading-relaxed">{formatInline(line)}</div>);
   267→  }
   268→
   269→  return result;
   270→}

   296→  const activeTemplate = TEMPLATES.find(t => t.id === selectedTemplate) || null;

   522→            <div key={field.key} className="space-y-2">
   523→              <Label className="text-sm font-medium">{field.label}</Label>
   524→              {field.type === 'textarea' ? (
   525→                <Textarea
   526→                  placeholder={field.placeholder}
   527→                  rows={field.rows || 3}
   528→                  value={inputData[field.key] || ''}
   529→                  onChange={e => handleInputChange(field.key, e.target.value)}
   530→                  className="bg-muted/50 border-border/60 rounded-xl resize-none focus:border-emerald-300 focus:ring-emerald-200/40"
   531→                />
   532→              ) : field.type === 'select-tender' ? (
   533→                <Select
   534→                  value={inputData[field.key] || ''}
   535→                  onValueChange={val => handleInputChange(field.key, val)}
   536→                >
   537→                  <SelectTrigger className="bg-muted/50 border-border/60 rounded-xl w-full">
   538→                    <SelectValue placeholder={field.placeholder} />
   539→                  </SelectTrigger>
   540→                  <SelectContent>
   541→                    {tenders.length === 0 ? (
   542→                      <SelectItem value="_none" disabled>No open tenders available</SelectItem>
   543→                    ) : (
   544→                      tenders.map(tender => (
   545→                        <SelectItem key={tender.id} value={tender.id}>
   546→                          {tender.title}
   547→                        </SelectItem>
   548→                      ))
   549→                    )}
   550→                  </SelectContent>
   551→                </Select>
   552→              ) : field.type === 'select-project' ? (
   553→                <Select
   554→                  value={inputData[field.key] || ''}
   555→                  onValueChange={val => handleInputChange(field.key, val)}
   556→                >
   557→                  <SelectTrigger className="bg-muted/50 border-border/60 rounded-xl w-full">
   558→                    <SelectValue placeholder={field.placeholder} />
   559→                  </SelectTrigger>
   560→                  <SelectContent>
   561→                    {projects.filter(p => p.status === 'active').length === 0 ? (
   562→                      <SelectItem value="_none" disabled>No active projects available</SelectItem>
   563→                    ) : (
   564→                      projects
   565→                        .filter(p => p.status === 'active')
   566→                        .map(project => (
   567→                          <SelectItem key={project.id} value={project.id}>
   568→                            {project.tender?.title || `Project ${project.id.slice(0, 8)}`}
   569→                          </SelectItem>
   570→                        ))
   571→                    )}
   572→                  </SelectContent>
   573→                </Select>
   574→              ) : (
   575→                <Input
   576→                  type={field.type === 'number' ? 'number' : 'text'}
   577→                  placeholder={field.placeholder}
   578→                  value={inputData[field.key] || ''}
   579→                  onChange={e => handleInputChange(field.key, e.target.value)}
   580→                  className="bg-muted/50 border-border/60 rounded-xl focus:border-emerald-300 focus:ring-emerald-200/40"
   581→                />
   582→              )}
   583→            </div>
   584→          ))}
   585→        </div>
   586→
   587→        {/* Generate Button */}
   588→        <Button
   589→          onClick={handleGenerate}
   590→          disabled={generating}
   591→          className="w-full gradient-emerald hover:opacity-90 text-white rounded-xl h-12 premium-shadow transition-all hover:-translate-y-0.5 disabled:opacity-50 disabled:hover:translate-y-0 text-sm font-semibold"
   592→        >
   593→          {generating ? (
   594→            <>
   595→              <Loader2 className="h-4 w-4 mr-2 animate-spin" />
   596→              Generating Document...
   597→            </>
   598→          ) : (
   599→            <>
   600→              <Sparkles className="h-4 w-4 mr-2" />
   601→              Generate Document
   602→            </>
   603→          )}
   604→        </Button>
   605→      </div>
   606→    );
   607→  };
   608→
   609→  // ─── Render: Generating Animation ──────────────────────────────────────────
   610→
   611→  const renderGenerating = () => (
   612→    <div className="flex flex-col items-center justify-center py-16 gap-6 view-enter">
   613→      <div className="relative">
   614→        <div className="w-16 h-16 rounded-2xl bg-emerald-50 flex items-center justify-center animate-pulse">
   615→          <FileText className="h-8 w-8 text-emerald-500" />
   616→        </div>
   617→        <div className="absolute -top-1 -right-1 w-6 h-6 rounded-full gradient-emerald flex items-center justify-center">
   618→          <Sparkles className="h-3 w-3 text-white animate-spin" />
   619→        </div>
   620→      </div>
   621→      <div className="text-center">
   622→        <h4 className="font-semibold text-base mb-1">AI is writing your document...</h4>
   623→        <p className="text-sm text-muted-foreground">This may take a moment. Our AI is crafting your professional document.</p>
   624→      </div>
   625→      <div className="flex gap-1.5">
   626→        <span className="w-2.5 h-2.5 bg-emerald-400 rounded-full animate-bounce" style={{ animationDelay: '0ms' }} />
   627→        <span className="w-2.5 h-2.5 bg-emerald-400 rounded-full animate-bounce" style={{ animationDelay: '150ms' }} />
   628→        <span className="w-2.5 h-2.5 bg-emerald-400 rounded-full animate-bounce" style={{ animationDelay: '300ms' }} />
   629→      </div>
   630→    </div>
   631→  );
   632→
   633→  // ─── Render: Generated Document Preview ────────────────────────────────────
   634→
   635→  const renderGeneratedDoc = () => {
   636→    if (!generatedContent) return null;
   637→    const templateDef = TEMPLATES.find(t => t.id === generatedTemplateType);
   638→    const DocIcon = templateDef?.icon || FileText;
   639→
   640→    return (
   641→      <div className="space-y-4 view-enter">
   642→        {/* Document Actions */}
   643→        <div className="flex items-center justify-between">
   644→          <div className="flex items-center gap-2">
   645→            <div className={`w-8 h-8 rounded-lg ${templateDef?.gradientClass || 'gradient-emerald'} flex items-center justify-center`}>
   646→              <DocIcon className="h-4 w-4 text-white" />
   647→            </div>
   648→            <div>
   649→              <h4 className="font-semibold text-sm">{templateDef?.name || 'Generated Document'}</h4>
   650→              {generatedAt && (
   651→                <p className="text-[11px] text-muted-foreground">
   652→                  Generated {new Date(generatedAt).toLocaleDateString()} at {new Date(generatedAt).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })}
   653→                </p>
   654→              )}
   655→            </div>
   656→          </div>
   657→          <div className="flex items-center gap-1.5">
   658→            <Button variant="ghost" size="sm" onClick={handleCopy}
   659→              className="h-8 gap-1.5 text-xs text-muted-foreground hover:text-emerald-600 hover:bg-emerald-50 rounded-lg">
   660→              <Copy className="h-3.5 w-3.5" />
   661→              Copy
   662→            </Button>
   663→            <Button variant="ghost" size="sm" onClick={handlePrint}
   664→              className="h-8 gap-1.5 text-xs text-muted-foreground hover:text-emerald-600 hover:bg-emerald-50 rounded-lg">
   665→              <Printer className="h-3.5 w-3.5" />
   666→              Print
   667→            </Button>
   668→            <Button variant="ghost" size="sm" onClick={handleNewDocument}
   669→              className="h-8 gap-1.5 text-xs text-muted-foreground hover:text-emerald-600 hover:bg-emerald-50 rounded-lg">
   670→              <RotateCcw className="h-3.5 w-3.5" />
   671→              New
   672→            </Button>
   673→          </div>
   674→        </div>
   675→
   676→        {/* Document Paper */}
   677→        <div className="bg-white rounded-xl border border-border/40 premium-shadow-lg overflow-hidden">
   678→          {/* Document Header Bar */}
   679→          <div className="px-6 py-3 bg-muted/30 border-b border-border/30 flex items-center justify-between">
   680→            <div className="flex items-center gap-2">
   681→              <div className={`w-2 h-2 rounded-full ${templateDef?.bgClass || 'bg-emerald-50'}`} style={{ backgroundColor: 'oklch(0.558 0.155 163)' }} />
   682→              <span className="text-[11px] font-medium text-muted-foreground uppercase tracking-wider">
   683→                {templateDef?.name || 'Document'}
   684→              </span>
   685→            </div>
   686→            <Badge className="text-[9px] px-1.5 py-0 gradient-emerald text-white border-0 font-medium">
   687→              AI Generated
   688→            </Badge>
   689→          </div>
   690→
   691→          {/* Document Content */}
   692→          <ScrollArea className="max-h-[60vh]">
   693→            <div className="p-6 md:p-8">
   694→              {/* Document Meta */}
   695→              <div className="flex items-center justify-between mb-6 pb-4 border-b border-border/30">
   696→                <div>
   697→                  <p className="text-[11px] text-muted-foreground uppercase tracking-wider mb-0.5">Prepared by</p>
   698→                  <p className="text-sm font-medium">
   699→                    {user?.company?.name || user?.profile?.fullName || user?.email || 'Unknown User'}
   700→                  </p>
   701→                </div>
   702→                <div className="text-right">
   703→                  <p className="text-[11px] text-muted-foreground uppercase tracking-wider mb-0.5">Date</p>
   704→                  <p className="text-sm font-medium">
   705→                    {generatedAt ? new Date(generatedAt).toLocaleDateString('en-US', { year: 'numeric', month: 'long', day: 'numeric' }) : new Date().toLocaleDateString('en-US', { year: 'numeric', month: 'long', day: 'numeric' })}
   706→                  </p>
   707→                </div>
   708→              </div>
   709→
   710→              {/* Formatted Content */}
   711→              <div className="prose-sm space-y-1">
   712→                {formatDocContent(generatedContent)}
   713→              </div>
   714→
   715→              {/* Document Footer */}
   716→              <div className="mt-8 pt-4 border-t border-border/30 flex items-center justify-between">
   717→                <p className="text-[10px] text-muted-foreground">
   718→                  Tenets Tender Ecosystem &middot; AI-Powered Document
   719→                </p>
   720→                <p className="text-[10px] text-muted-foreground">
   721→                  {generatedTemplateType?.replace(/-/g, ' ').replace(/\b\w/g, c => c.toUpperCase())}
   722→                </p>
   723→              </div>
   724→            </div>
   725→          </ScrollArea>
   726→        </div>
   727→      </div>
   728→    );
   729→  };
   730→
   731→  // ─── Render: Right Panel ───────────────────────────────────────────────────
   732→
   733→  const renderRightPanel = () => {
   734→    if (!selectedTemplate) return null;
   735→
   736→    return (
   737→      <Card className="premium-shadow rounded-xl border-0 bg-white">
   738→        <CardContent className="p-5 md:p-6">
   739→          {generating ? renderGenerating() : generatedContent ? renderGeneratedDoc() : renderTemplateForm()}
   740→        </CardContent>
   741→      </Card>
   742→    );
   743→  };
   744→
   745→  // ─── Main Render ───────────────────────────────────────────────────────────
   746→
   747→  return (
   748→    <div className="p-4 md:p-6 space-y-6 max-w-6xl mx-auto view-enter">
   749→      {/* Header */}
   750→      <div className="flex items-center gap-4">
   751→        <div className="p-3 rounded-2xl gradient-emerald shadow-md flex-shrink-0">
   752→          <BookOpen className="h-6 w-6 text-white" />
   753→        </div>
   754→        <div className="flex-1">
   755→          <h2 className="text-2xl md:text-3xl font-bold tracking-tight">
   756→            <span className="text-gradient-emerald">Document</span> Builder
   757→          </h2>
   758→          <p className="text-muted-foreground text-sm mt-0.5">AI-powered document preparation for the tender ecosystem</p>
   759→        </div>
   760→        {selectedTemplate && (
   761→          <Button
   762→            variant="outline"
   763→            size="sm"
   764→            onClick={handleNewDocument}
   765→            className="gap-1.5 text-xs rounded-xl border-border/60 hover:bg-emerald-50 hover:text-emerald-700 hover:border-emerald-300 transition-all"
   766→          >
   767→            <RotateCcw className="h-3.5 w-3.5" />
   768→            Start Over
   769→          </Button>
   770→        )}
   771→      </div>
   772→
   773→      {/* Body */}
   774→      {!selectedTemplate ? (
   775→        renderEmptyState()
   776→      ) : (
   777→        <div className="grid grid-cols-1 lg:grid-cols-12 gap-5">
   778→          {/* Left Panel */}
   779→          <div className="lg:col-span-4 xl:col-span-3">
   780→            <Card className="premium-shadow rounded-xl border-0 bg-white">
   781→              <CardContent className="p-4">
   782→                {renderTemplateList()}
   783→              </CardContent>
   784→            </Card>
   785→          </div>
   786→
   787→          {/* Right Panel */}
   788→          <div className="lg:col-span-8 xl:col-span-9">
   789→            {renderRightPanel()}
   790→          </div>
   791→        </div>
   792→      )}
   793→    </div>
   794→  );
