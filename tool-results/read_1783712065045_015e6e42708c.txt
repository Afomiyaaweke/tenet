    31→function daysUntil(dateStr: string): number {

    35→function parseBidAnalysis(analysis: BidAnalysis): BidAnalysisResult | null {
    36→  try {
    37→    return {
    38→      summary: JSON.parse(analysis.summary || '{}'),
    39→      applicants: JSON.parse(analysis.rankings || '[]'),
    40→      budgetAnalysis: analysis.budgetAnalysis || '',

    49→function ScoreBar({ score, label }: { score: number; label?: string }) {
    50→  const colorClass = score >= 80 ? 'bg-emerald-500' : score >= 60 ? 'bg-amber-500' : 'bg-red-500';
    51→  const textColor = score >= 80 ? 'text-emerald-700' : score >= 60 ? 'text-amber-700' : 'text-red-700';
    52→  return (
    53→    <div className="flex items-center gap-2 min-w-[120px]">
    54→      {label && <span className="text-[10px] text-muted-foreground w-12 shrink-0">{label}</span>}
    55→      <div className="flex-1 h-2 bg-muted/50 rounded-full overflow-hidden">
    56→        <div
    57→ className={`h-full rounded-full ${colorClass} transition-[width] duration-700`} style={{ width: `${Math.min(score, 100)}%` }}
    58→ />
    59→      </div>
    60→      <span className={`text-xs font-semibold ${textColor} w-8 text-right`}>{score}</span>
    61→    </div>
    62→  );
    63→}

    65→function RiskBadge({ level }: { level: 'low' | 'medium' | 'high' }) {
    66→  const config = {
    67→    low: { className: 'bg-emerald-100 text-emerald-700', icon: ShieldCheck },
    68→    medium: { className: 'bg-amber-100 text-amber-700', icon: ShieldQuestion },
    69→    high: { className: 'bg-red-100 text-red-700', icon: ShieldAlert },
    70→  };
    71→  const c = config[level];
    72→  const Icon = c.icon;
    73→  return (
    74→    <Badge className={`text-[10px] px-2 py-0.5 border-0 rounded-lg ${c.className}`}>
    75→      <Icon className="h-3 w-3 mr-1" />
    76→      {level.charAt(0).toUpperCase() + level.slice(1)}
    77→    </Badge>
    78→  );
    79→}
