   127→function fmtMoney(amount: number, currency: string): string {
   128→  if (!amount) return '—';
   129→  try {
   130→    return new Intl.NumberFormat('en-US', {
   131→      style: 'currency',
   132→      currency: currency || 'USD',
   133→      maximumFractionDigits: 0,
   134→    }).format(amount);
   135→  } catch {
   136→    return `${currency} ${amount.toLocaleString()}`;
   137→  }
   138→}
   139→
   140→function daysUntil(dateStr: string): number {
   141→  return Math.ceil((new Date(dateStr).getTime() - Date.now()) / (1000 * 60 * 60 * 24));
   142→}
   143→
   144→function deadlineBadge(days: number): string {
   145→  if (days <= 0) return 'bg-rose-50 text-rose-700 dark:bg-rose-950/40 dark:text-rose-300';
   146→  if (days <= 7) return 'bg-amber-50 text-amber-700 dark:bg-amber-950/40 dark:text-amber-300';
   147→  if (days <= 21) return 'bg-sky-50 text-sky-700 dark:bg-sky-950/40 dark:text-sky-300';
   148→  return 'bg-emerald-50 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300';
   149→}
   150→
   151→function relativeTime(dateStr: string): string {
   152→  const diff = Date.now() - new Date(dateStr).getTime();
   153→  const day = 86400000;
   154→  if (diff < day) return 'today';
   155→  if (diff < 2 * day) return 'yesterday';
   156→  if (diff < 30 * day) return `${Math.floor(diff / day)}d ago`;
   157→  return new Date(dateStr).toLocaleDateString();
   158→}

   160→/* ─────────────────────────────────────────────────────────────────────
   161→ * Inline Document Viewer Component
   162→ * ───────────────────────────────────────────────────────────────────── */
   163→
   164→function InlineDocumentViewer({ doc, onClose }: { doc: InlineDocument; onClose: () => void }) {
   165→  const scrollRef = useRef<HTMLDivElement>(null);
   166→
   167→  const copyContent = () => {
   168→    const text = doc.sections
   169→      ? doc.sections.map((s) => `${s.heading}\n${s.content}`).join('\n\n')
   170→      : doc.content;
   171→    navigator.clipboard.writeText(text);
   172→    toast.success('Content copied to clipboard');
   173→  };
   174→
   175→  return (
   176→    <motion.div
   177→      initial={{ opacity: 0, height: 0 }}
   178→      animate={{ opacity: 1, height: 'auto' }}
   179→      exit={{ opacity: 0, height: 0 }}
   180→      transition={{ duration: 0.3, ease: 'easeInOut' }}
   181→      className="overflow-hidden"
   182→    >
   183→      <div className="border-t border-border bg-gradient-to-b from-muted/30 to-background">
   184→        <div className="p-4 md:p-6 space-y-4">
   185→          {/* Document header */}
   186→          <div className="flex items-start justify-between gap-3">
   187→            <div className="min-w-0 flex-1">
   188→              <div className="flex items-center gap-2 text-xs text-muted-foreground mb-1">
   189→                <BookOpen className="h-3.5 w-3.5" />
   190→                <span>Document loaded from {new URL(doc.url).hostname}</span>
   191→                <span>·</span>
   192→                <span>{new Date(doc.fetchedAt).toLocaleTimeString()}</span>
   193→              </div>
   194→              <h3 className="text-lg font-semibold text-foreground leading-snug">
   195→                {doc.title || 'Tender Document'}
   196→              </h3>
   197→              {doc.metaDescription && (
   198→                <p className="text-sm text-muted-foreground mt-1 line-clamp-2">
   199→                  {doc.metaDescription}
   200→                </p>
   201→              )}
   202→            </div>
   203→            <div className="flex items-center gap-2 shrink-0">
   204→              <Button variant="ghost" size="sm" className="gap-1.5 text-xs" onClick={copyContent}>
   205→                <Copy className="h-3.5 w-3.5" />
   206→                Copy
   207→              </Button>
   208→              <a
   209→                href={doc.url}
   210→                target="_blank"
   211→                rel="noopener noreferrer"
   212→                className="inline-flex items-center gap-1.5 text-xs font-medium text-primary hover:underline"
   213→              >
   214→                <ExternalLink className="h-3.5 w-3.5" />
   215→                Open original
   216→              </a>
   217→              <Button variant="ghost" size="sm" onClick={onClose} className="gap-1.5 text-xs text-muted-foreground">
   218→                <ChevronUp className="h-3.5 w-3.5" />
   219→                Collapse
   220→              </Button>
   221→            </div>
   222→          </div>
   223→
   224→          {/* Extracted metadata pills */}
   225→          {(doc.deadlines && doc.deadlines.length > 0) || (doc.budgets && doc.budgets.length > 0) ? (
   226→            <div className="flex flex-wrap gap-2">
   227→              {doc.deadlines?.map((d, i) => (
   228→                <Badge key={`dl-${i}`} variant="outline" className="gap-1.5 text-xs border-amber-300 text-amber-700 dark:border-amber-800 dark:text-amber-300">
   229→                  <Calendar className="h-3 w-3" />
   230→                  {d}
   231→                </Badge>
   232→              ))}
   233→              {doc.budgets?.map((b, i) => (
   234→                <Badge key={`bg-${i}`} variant="outline" className="gap-1.5 text-xs border-emerald-300 text-emerald-700 dark:border-emerald-800 dark:text-emerald-300">
   235→                  <DollarSign className="h-3 w-3" />
   236→                  {b}
   237→                </Badge>
   238→              ))}
   239→            </div>
   240→          ) : null}
   241→
   242→          {/* Sections view */}
   243→          {doc.sections && doc.sections.length > 0 ? (
   244→            <div ref={scrollRef} className="max-h-80 overflow-y-auto rounded-lg border border-border bg-background p-4 space-y-4 scrollbar-thin">
   245→              {doc.sections.map((section, i) => (
   246→                <div key={i}>
   247→                  <h4 className="text-sm font-semibold text-foreground mb-1.5">{section.heading}</h4>
   248→                  <p className="text-sm text-muted-foreground leading-relaxed whitespace-pre-line">
   249→                    {section.content}
   250→                  </p>
   251→                  {i < doc.sections!.length - 1 && <Separator className="mt-4 bg-border" />}
   252→                </div>
   253→              ))}
   254→            </div>
   255→          ) : (
   256→            <div ref={scrollRef} className="max-h-80 overflow-y-auto rounded-lg border border-border bg-background p-4 scrollbar-thin">
   257→              <p className="text-sm text-muted-foreground leading-relaxed whitespace-pre-line">
   258→                {doc.content || 'No content could be extracted from this page.'}
   259→              </p>
   260→            </div>
   261→          )}
   262→        </div>
   263→      </div>
   264→    </motion.div>
   265→  );
   266→}

   268→/* ─────────────────────────────────────────────────────────────────────
   269→ * Main LiveTendersView component — Notion-style
   270→ * ───────────────────────────────────────────────────────────────────── */
   271→
   272→export function LiveTendersView() {
   273→  const [tenders, setTenders] = useState<LiveTender[]>([]);
   274→  const [dataSources, setDataSources] = useState<DataSource[]>([]);
   275→  const [sourceMeta, setSourceMeta] = useState<
   276→    { id: string; name: string; live: boolean; ok: boolean; count: number }[]
   277→  >([]);
   278→  const [loading, setLoading] = useState(true);
   279→  const [refreshing, setRefreshing] = useState(false);
   280→  const [fallback, setFallback] = useState(false);
   281→  const [search, setSearch] = useState('');
   282→  const [sourceFilter, setSourceFilter] = useState('all');
   283→
   284→  // Inline document state
   285→  const [expandedId, setExpandedId] = useState<string | null>(null);
   286→  const [docLoading, setDocLoading] = useState<string | null>(null);
   287→  const [docData, setDocData] = useState<Record<string, InlineDocument>>({});
   288→  const [docError, setDocError] = useState<Record<string, string>>({});
   289→
   290→  // Import state
   291→  const [importing, setImporting] = useState<string | null>(null);
   292→
   293→  const load = useCallback(
   294→    async (isRefresh = false) => {
   295→      if (isRefresh) setRefreshing(true);
   296→      else setLoading(true);
   297→      const params: Record<string, string> = { rows: '20' };
   298→      if (search) params.search = search;
   299→      if (sourceFilter && sourceFilter !== 'all') params.source = sourceFilter;
   300→      const res = await api.get('/tenders/live', params);
   301→      if (res.success) {
   302→        setTenders(res.data as LiveTender[]);
   303→        setSourceMeta(res.meta?.sources || []);
   304→        setFallback(Boolean(res.meta?.fallback));
   305→        if (Array.isArray(res.meta?.dataSources)) setDataSources(res.meta.dataSources);
   306→      } else {
   307→        toast.error(res.error || 'Failed to load live tenders');
   308→      }
   309→      setLoading(false);
   310→      setRefreshing(false);
   311→    },
   312→    [search, sourceFilter],
   313→  );
   314→
   315→  useEffect(() => {
   316→    const t = setTimeout(() => load(), 250);
   317→    return () => clearTimeout(t);
   318→  }, [load]);
   319→
   320→  const liveSourcesCount = useMemo(
   321→    () => sourceMeta.filter((s) => s.live && s.ok).length,
   322→    [sourceMeta],
   323→  );
   324→
   325→  const totalTenderValue = useMemo(() => {
   326→    return tenders.reduce((sum, t) => sum + (t.budgetMax || 0), 0);
   327→  }, [tenders]);
   328→
   329→  /* ────── Inline document loading ────── */
   330→  const loadDocument = useCallback(async (tender: LiveTender) => {
   331→    const id = tender.id;
   332→    if (expandedId === id) {
   333→      setExpandedId(null);
   334→      return;
   335→    }
   336→
   337→    // Already fetched?
   338→    if (docData[id]) {
   339→      setExpandedId(id);
   340→      return;
   341→    }
   342→
   343→    setExpandedId(id);
   344→    setDocLoading(id);
   345→    setDocError((prev) => { const n = { ...prev }; delete n[id]; return n; });
   346→
   347→    try {
   348→      const res = await api.get(`/tenders/${encodeURIComponent(id)}/documents`, { url: tender.externalUrl });
   349→      if (res.success && res.data) {
   350→        setDocData((prev) => ({ ...prev, [id]: res.data as InlineDocument }));
   351→      } else {
   352→        setDocError((prev) => ({ ...prev, [id]: res.error || 'Failed to load document' }));
   353→      }
   354→    } catch {
   355→      setDocError((prev) => ({ ...prev, [id]: 'Network error' }));
   356→    }
   357→    setDocLoading(null);
   358→  }, [expandedId, docData]);
   359→
   360→  /* ────── Import to local tenders ────── */
   361→  const importTender = useCallback(async (tender: LiveTender) => {
   362→    setImporting(tender.id);
   363→    try {
   364→      const res = await api.post('/tenders', {
   365→        title: tender.title,
   366→        scope: tender.scope,
   367→        budgetMin: tender.budgetMin,
   368→        budgetMax: tender.budgetMax,
   369→        deadline: tender.deadline,
   370→        location: tender.location,
   371→        categoryTags: tender.categoryTags,
   372→        requiredDocs: `Source: ${SOURCE_LABELS[tender.source] || tender.source} | External ID: ${tender.externalId} | URL: ${tender.externalUrl}`,
   373→        status: 'open',
   374→        currency: tender.currency,
   375→      });
   376→      if (res.success) {
   377→        toast.success('Tender imported successfully', {
   378→          description: `"${tender.title}" is now in your tenders list.`,
   379→        });
   380→      } else {
   381→        toast.error(res.error || 'Failed to import tender');
   382→      }
   383→    } catch {
   384→      toast.error('Failed to import tender');
   385→    }
   386→    setImporting(null);
   387→  }, []);
   388→
   389→  /* ────── Render ────── */
   390→  return (
   391→    <div className="view-enter">
   392→      {/* ───────────────── Notion-style Cover ───────────────── */}
   393→      <div className="relative h-32 md:h-40 bg-gradient-to-br from-emerald-600 via-teal-700 to-cyan-800 overflow-hidden">
   394→        <div className="absolute inset-0 bg-[url('data:image/svg+xml;base64,PHN2ZyB3aWR0aD0iNDAiIGhlaWdodD0iNDAiIHZpZXdCb3g9IjAgMCA0MCA0MCIgeG1sbnM9Imh0dHA6Ly93d3cudzMub3JnLzIwMDAvc3ZnIj48Y2lyY2xlIGN4PSIyMCIgY3k9IjIwIiByPSIxIiBmaWxsPSJyZ2JhKDI1NSwyNTUsMjU1LDAuMDUpIi8+PC9zdmc+')] opacity-60" />
   395→        <div className="absolute bottom-0 left-0 right-0 h-16 bg-gradient-to-t from-background to-transparent" />
   396→        <div className="absolute top-4 right-4 md:right-6">
   397→          <Button
   398→            variant="secondary"
   399→            size="sm"
   400→            onClick={() => load(true)}
   401→            disabled={refreshing}
   402→            className="gap-1.5 shadow-lg bg-white/90 dark:bg-gray-800/90 backdrop-blur-sm"
   403→          >
   404→            <RefreshCw className={`h-3.5 w-3.5 ${refreshing ? 'animate-spin' : ''}`} />
   405→            {refreshing ? 'Refreshing…' : 'Refresh'}
   406→          </Button>
   407→        </div>
   408→      </div>
   409→
   410→      <div className="px-4 md:px-6 max-w-5xl mx-auto -mt-8 relative z-10 space-y-6">
   411→        {/* ───────────────── Notion-style Icon + Title ───────────────── */}
   412→        <div className="flex items-end gap-4">
   413→          <div className="h-14 w-14 md:h-16 md:w-16 rounded-xl bg-gradient-to-br from-emerald-500 to-teal-600 flex items-center justify-center shadow-lg shadow-emerald-500/20 border-4 border-background shrink-0">
   414→            <Globe2 className="h-7 w-7 md:h-8 md:w-8 text-white" />
   415→          </div>
   416→          <div className="flex-1 min-w-0 pb-1">
   417→            <div className="flex items-center gap-2 flex-wrap">
