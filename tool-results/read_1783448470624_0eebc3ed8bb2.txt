     1→'use client';
     2→
     3→import { useEffect, useState, useCallback, useMemo } from 'react';
     4→import { useAuthStore, useNavStore } from '@/store';
     5→import { api, Tender, Bid } from '@/lib/api';
     6→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
     7→import { Badge } from '@/components/ui/badge';
     8→import { Button } from '@/components/ui/button';
     9→import { ScrollArea } from '@/components/ui/scroll-area';
    10→import { toast } from 'sonner';
    11→import {
    12→  ArrowLeft, MapPin, Calendar, DollarSign, Tag, FileText, Gavel, Clock, Users,
    13→  ChevronRight, TrendingUp, Timer, Target, Award, CheckCircle, X,
    14→  Briefcase, GitCompareArrows, Zap, Eye, Building2, CircleCheck,
    15→  BarChart3, Scale, FileStack,
    16→} from 'lucide-react';
    17→
    18→function daysUntil(dateStr: string): number {
    19→  return Math.ceil((new Date(dateStr).getTime() - Date.now()) / (1000 * 60 * 60 * 24));
    20→}
    21→
    22→function formatETB(amount: number): string {
    23→  return `ETB ${amount.toLocaleString()}`;
    24→}
    25→
    26→/* ────────────── Tender Comparison View ────────────── */
    27→
    28→export function TenderCompareView({ tenderIds }: { tenderIds?: string }) {
    29→  const { user } = useAuthStore();
    30→  const { setView } = useNavStore();
    31→  const [tenders, setTenders] = useState<Tender[]>([]);
    32→  const [loading, setLoading] = useState(true);
    33→
    34→  const ids = useMemo(() => {
    35→    if (!tenderIds) return [];
    36→    return tenderIds.split(',').filter(Boolean);
    37→  }, [tenderIds]);
    38→
    39→  const loadTenders = useCallback(async () => {
    40→    if (ids.length === 0) { setLoading(false); return; }
    41→    setLoading(true);
    42→    const loaded: Tender[] = [];
    43→    for (const id of ids) {
    44→      const res = await api.get(`/tenders/${id}`);
    45→      if (res.success) loaded.push(res.data);
    46→    }
    47→    setTenders(loaded);
    48→    setLoading(false);
    49→  }, [ids]);
    50→
    51→  /* eslint-disable react-hooks/set-state-in-effect */
    52→  useEffect(() => { loadTenders(); }, [loadTenders]);
    53→  /* eslint-enable react-hooks/set-state-in-effect */
    54→
    55→  if (loading) {
    56→    return (
    57→      <div className="p-4 md:p-6 space-y-4 max-w-7xl mx-auto view-enter">
    58→        <div className="h-8 bg-muted/50 rounded-xl w-1/3 animate-pulse" />
    59→        <div className="grid grid-cols-2 gap-4">
    60→          {[1, 2].map(i => (
    61→            <div key={i} className="h-96 bg-muted/50 rounded-xl animate-pulse premium-shadow" />
    62→          ))}
    63→        </div>
    64→      </div>
    65→    );
    66→  }
    67→
    68→  if (tenders.length === 0) {
    69→    return (
    70→      <div className="p-6 text-center view-enter">
    71→        <div className="relative w-20 h-20 mx-auto mb-6">
    72→          <div className="absolute inset-0 rounded-2xl gradient-emerald opacity-20" />
    73→          <div className="absolute inset-2 rounded-xl gradient-emerald flex items-center justify-center">
    74→            <GitCompareArrows className="h-8 w-8 text-white" />
    75→          </div>
    76→        </div>
    77→        <h3 className="text-lg font-semibold">No tenders to compare</h3>
    78→        <p className="text-muted-foreground text-sm mt-2">Go back to tenders and select at least 2 to compare</p>
    79→        <Button variant="outline" onClick={() => setView('tenders')} className="mt-4 rounded-xl">
    80→          <ArrowLeft className="h-4 w-4 mr-2" /> Back to Tenders
    81→        </Button>
    82→      </div>
    83→    );

    86→  // Comparison rows
    87→  const comparisonFields = [
    88→    {
    89→      label: 'Status',

    95→          awarded: 'bg-teal-100 text-teal-700',
    96→          cancelled: 'bg-muted text-muted-foreground',
    97→          draft: 'bg-muted text-muted-foreground',
    98→        };
    99→        return (
   100→          <Badge className={`text-xs px-2.5 py-1 border-0 rounded-lg ${colors[t.status] || 'bg-muted text-muted-foreground'}`}>

   126→            <p className={`text-xs font-medium ${days <= 0 ? 'text-rose-600' : days <= 3 ? 'text-rose-600' : days <= 7 ? 'text-amber-600' : 'text-emerald-600'}`}>
   127→              {days <= 0 ? 'Expired' : days === 1 ? '1 day left' : `${days} days left`}
   128→            </p>
   129→          </div>
   130→        );

   132→    },
   133→    {

   139→      label: 'Categories',

   142→        <div className="flex flex-wrap gap-1">
   143→          {t.categoryTags.split(',').filter(Boolean).map(tag => (
   144→            <Badge key={tag} className="text-[10px] px-1.5 py-0 border-0 rounded-lg bg-emerald-50 text-emerald-700 hover:bg-primary/10">
   145→              {tag.trim()}
   146→            </Badge>
   147→          ))}
   148→        </div>
   149→      ),
   150→    },
   151→    {
   152→      label: 'Total Bids',
   153→      icon: Users,
   154→      render: (t: Tender) => <p className="text-sm font-bold">{t._count?.bids || 0}</p>,
   155→    },
   156→    {
   157→      label: 'Match Score',
   158→      icon: Target,

   160→        if (t.matchScore === undefined) return <p className="text-xs text-muted-foreground">N/A</p>;
   161→        const barColor = t.matchScore >= 70 ? 'from-emerald-400 to-emerald-600' : t.matchScore >= 40 ? 'from-amber-400 to-amber-600' : 'from-muted to-muted-foreground/50';
   162→        const textColor = t.matchScore >= 70 ? 'text-emerald-700' : t.matchScore >= 40 ? 'text-amber-700' : 'text-muted-foreground';
   163→        return (
   164→          <div className="space-y-1">
   165→            <p className={`text-sm font-bold ${textColor}`}>{t.matchScore}%</p>
   166→            <div className="h-1.5 bg-muted/50 rounded-full overflow-hidden">
   167→              <div className={`h-full rounded-full bg-gradient-to-r ${barColor}`} style={{ width: `${t.matchScore}%` }} />
   168→            </div>
   169→          </div>
   170→        );
   171→      },
   172→    },
   173→    {
   174→      label: 'Required Documents',
   175→      icon: FileStack,
   176→      render: (t: Tender) => {
   177→        const docs = t.requiredDocs?.split(',').filter(Boolean) || [];
   178→        if (docs.length === 0) return <p className="text-xs text-muted-foreground">None specified</p>;
   179→        return (

   181→            {docs.map(doc => (

   184→          </div>
   185→        );
   186→      },
   187→    },
   188→  ];
   189→
   190→  // Find the "best" values for highlighting
   191→  const bestBudget = tenders.reduce((best, t) => t.budgetMax > (best?.budgetMax || 0) ? t : best, tenders[0]);
   192→  const mostDays = tenders.reduce((best, t) => daysUntil(t.deadline) > daysUntil(best.deadline) ? t : best, tenders[0]);
   193→  const mostBids = tenders.reduce((best, t) => (t._count?.bids || 0) > (best._count?.bids || 0) ? t : best, tenders[0]);
   194→  const bestMatch = tenders.reduce((best, t) => (t.matchScore || 0) > (best.matchScore || 0) ? t : best, tenders[0]);
   195→
   196→  return (
   197→    <div className="p-4 md:p-6 space-y-6 max-w-7xl mx-auto view-enter">
   198→      {/* Header */}
   199→      <div
   200→ className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4 animate-[fadeIn_0.3s_ease-out]"
   201→ >
   202→        <div className="flex items-center gap-4">
   203→          <div className="p-3 rounded-2xl gradient-emerald shadow-md flex-shrink-0">
   204→            <GitCompareArrows className="h-6 w-6 text-white" />
   205→          </div>
   206→          <div>
   207→            <h2 className="text-2xl md:text-3xl font-bold tracking-tight">
   208→              <span className="text-gradient-emerald">Compare</span> Tenders
   209→            </h2>
   210→            <p className="text-muted-foreground text-sm mt-0.5">Side-by-side comparison of {tenders.length} tenders</p>
   211→          </div>
   212→        </div>
   213→        <Button variant="outline" onClick={() => setView('tenders')}
   214→          className="rounded-xl hover:text-emerald-700 hover:bg-primary/10 transition-colors">
   215→          <ArrowLeft className="h-4 w-4 mr-2" /> Back to Tenders
   216→        </Button>
   217→      </div>
   218→
   219→      {/* Quick Insights */}
   220→      <div className="animate-[fadeIn_0.3s_ease-out]"
   221→ >
   222→        <Card className="premium-shadow rounded-xl border-0 bg-card overflow-hidden">
   223→          <div className="h-1.5 bg-gradient-to-r from-emerald-400 via-teal-400 to-amber-400" />
   224→          <CardContent className="p-4">
   225→            <div className="flex items-center gap-2 mb-3">
   226→              <div className="p-1.5 rounded-lg gradient-emerald">
   227→                <Zap className="h-3.5 w-3.5 text-white" />
   228→              </div>
   229→              <h3 className="text-sm font-bold">Quick Insights</h3>
   230→            </div>
   231→            <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   232→              <div className="bg-emerald-50/60 rounded-xl p-3">
   233→                <div className="flex items-center gap-1.5 mb-1">
   234→                  <DollarSign className="h-3 w-3 text-emerald-600" />
   235→                  <span className="text-[10px] text-emerald-600 font-medium">Highest Budget</span>
   236→                </div>
   237→                <p className="text-xs font-bold text-emerald-700 truncate">{bestBudget?.title}</p>
   238→                <p className="text-[10px] text-emerald-600">{formatETB(bestBudget?.budgetMax || 0)}</p>
   239→              </div>
   240→              <div className="bg-amber-50/60 rounded-xl p-3">
   241→                <div className="flex items-center gap-1.5 mb-1">
   242→                  <Timer className="h-3 w-3 text-amber-600" />
   243→                  <span className="text-[10px] text-amber-600 font-medium">Most Time</span>
   244→                </div>
   245→                <p className="text-xs font-bold text-amber-700 truncate">{mostDays?.title}</p>
   246→                <p className="text-[10px] text-amber-600">{daysUntil(mostDays?.deadline || '')} days left</p>
   247→              </div>
   248→              <div className="bg-teal-50/60 rounded-xl p-3">
   249→                <div className="flex items-center gap-1.5 mb-1">
   250→                  <Users className="h-3 w-3 text-teal-600" />
   251→                  <span className="text-[10px] text-teal-600 font-medium">Most Bids</span>
   252→                </div>
   253→                <p className="text-xs font-bold text-teal-700 truncate">{mostBids?.title}</p>
   254→                <p className="text-[10px] text-teal-600">{mostBids?._count?.bids || 0} bids</p>
   255→              </div>
   256→              <div className="bg-purple-50/60 rounded-xl p-3">
   257→                <div className="flex items-center gap-1.5 mb-1">
   258→                  <Target className="h-3 w-3 text-purple-600" />
   259→                  <span className="text-[10px] text-purple-600 font-medium">Best Match</span>
   260→                </div>
   261→                <p className="text-xs font-bold text-purple-700 truncate">{bestMatch?.title}</p>
   262→                <p className="text-[10px] text-purple-600">{bestMatch?.matchScore || 0}% match</p>
   263→              </div>
   264→            </div>
   265→          </CardContent>
   266→        </Card>
   267→      </div>
   268→
   269→      {/* Comparison Table */}
   270→      <div className="animate-[fadeIn_0.3s_ease-out]"
   271→ >
   272→        <Card className="premium-shadow rounded-xl border-0 bg-card overflow-hidden">
   273→          <div className="h-1.5 bg-gradient-to-r from-emerald-400 to-teal-400" />
   274→          <ScrollArea className="w-full">
   275→            <div className="min-w-[600px]">
   276→              {/* Header Row */}
   277→              <div className="grid border-b border-border/50" style={{ gridTemplateColumns: `200px repeat(${tenders.length}, 1fr)` }}>
   278→                <div className="p-4 bg-muted/30 flex items-center gap-2 sticky left-0 z-10 bg-card">
   279→                  <Scale className="h-4 w-4 text-emerald-600" />
   280→                  <span className="text-sm font-bold">Attributes</span>
   281→                </div>
   282→                {tenders.map((tender, i) => (
   283→                  <div key={tender.id} className="p-4 border-l border-border/30">
   284→                    <div className="flex items-start gap-3">
   285→                      <div className="flex-1 min-w-0">
   286→                        <h4 className="text-sm font-bold truncate">{tender.title}</h4>
   287→                        <p className="text-xs text-muted-foreground mt-0.5">ID: {tender.id.slice(0, 8)}...</p>
   288→                      </div>
   289→                      <Button size="sm" variant="ghost"
   290→                        className="h-7 w-7 p-0 rounded-lg hover:bg-primary/10 hover:text-emerald-700 flex-shrink-0"
   291→                        onClick={() => setView('tender-detail', { id: tender.id })}>
   292→                        <Eye className="h-3.5 w-3.5" />
   293→                      </Button>
   294→                    </div>
   295→                  </div>
   296→                ))}
   297→              </div>
   298→
   299→              {/* Comparison Rows */}
   300→              {comparisonFields.map((field, idx) => (

   303→ className="grid border-b border-border/30 hover:bg-muted/20 transition-colors"
   304→ style={{ gridTemplateColumns: `200px repeat(${tenders.length}, 1fr)` }}
   305→ >
   306→                  <div className="p-3 flex items-center gap-2 sticky left-0 z-10 bg-card">
   307→                    <div className="p-1 rounded bg-muted/50">
   308→                      <field.icon className="h-3.5 w-3.5 text-muted-foreground" />
   309→                    </div>

   311→                  </div>
   312→                  {tenders.map(tender => (
   313→                    <div key={tender.id} className="p-3 border-l border-border/20">
   314→                      {field.render(tender)}
   315→                    </div>
   316→                  ))}
   317→                </div>
   318→              ))}
   319→
   320→              {/* Scope Row */}
   321→              <div
   322→ className="grid"
   323→ style={{ gridTemplateColumns: `200px repeat(${tenders.length}, 1fr)` }}

   325→                <div className="p-3 flex items-center gap-2 sticky left-0 z-10 bg-card">

   327→                    <Briefcase className="h-3.5 w-3.5 text-muted-foreground" />

   329→                  <span className="text-xs font-semibold text-muted-foreground">Scope</span>

   331→                {tenders.map(tender => (
   332→                  <div key={tender.id} className="p-3 border-l border-border/20">

   338→          </ScrollArea>

   342→      {/* Visual Budget Comparison Chart */}

   347→          <CardHeader className="pb-3">
