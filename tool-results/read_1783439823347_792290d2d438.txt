   154→function relativeTime(dateStr: string): string {
   155→  const now = Date.now();
   156→  const then = new Date(dateStr).getTime();
   157→  const diff = Math.max(0, now - then);
   158→  const seconds = Math.floor(diff / 1000);
   159→  if (seconds < 60) return 'just now';
   160→  const minutes = Math.floor(seconds / 60);
   161→  if (minutes < 60) return `${minutes}m ago`;
   162→  const hours = Math.floor(minutes / 60);
   163→  if (hours < 24) return `${hours}h ago`;
   164→  const days = Math.floor(hours / 24);
   165→  if (days < 7) return `${days}d ago`;
   166→  return new Date(dateStr).toLocaleDateString();
   167→}

   356→  onMarkAsRead,

   423→                  <button
   424→                    key={n.id}
   425→                    onClick={() => {
   426→                      onMarkAsRead(n.id);
   427→                      setOpen(false);
   428→                    }}
   429→                    className="w-full flex items-start gap-3 px-4 py-3 hover:bg-muted/50 transition-colors text-left group"

   435→                      <Icon className={`h-4 w-4 ${colorClass}`} />

   439→                      <p className="text-[11px] text-muted-foreground truncate mt-0.5 leading-relaxed">{n.message}</p>
   440→                      <p className="text-[10px] text-muted-foreground/50 mt-1 font-medium">{relativeTime(n.createdAt)}</p>
   441→                    </div>
   442→                    <div className="w-2 h-2 rounded-full bg-orange-500 flex-shrink-0 mt-2 ring-2 ring-orange-100 dark:ring-orange-900/40" />
   443→                  </button>
   444→                );

   447→          )}
   448→        </ScrollArea>
   449→
   450→        {/* Footer */}
   451→        {notifications.length > 6 && (
   452→          <div className="px-3 py-2 border-t border-border/40">

   454→              variant="ghost"
   455→              size="sm"
   456→              className="w-full text-[11px] text-muted-foreground hover:text-foreground font-medium"
   457→              onClick={() => setOpen(false)}
   458→            >
   459→              View all notifications
   460→            </Button>
   461→          </div>
   462→        )}
   463→      </PopoverContent>
   464→    </Popover>
   465→  );
   466→}

   468→/* ──────────────────────────── AppShell ──────────────────────────── */
   469→
   470→export function AppShell() {
   471→  const { user, company, logout } = useAuthStore();
   472→  const { view, viewParams, setView } = useNavStore();
   473→  const { fetchNotifications, notifications } = useDataStore();
   474→  const role = user?.role || 'user';
   475→  const navSections = useMemo(() => getNavItemsForRole(role), [role]);
   476→  const unreadCount = notifications.filter((n) => !n.read).length;
   477→
   478→  // Flat list for title lookup
   479→  const allNavItems = useMemo(
   480→    () => navSections.flatMap((s) => s.items),
   481→    [navSections]
   482→  );
   483→
   484→  useEffect(() => {
   485→    fetchNotifications();
   486→  }, [fetchNotifications]);
   487→
   488→  const handleMarkAsRead = async (id: string) => {
   489→    await api.patch(`/notifications/${id}`, { read: true });
   490→    fetchNotifications();
   491→  };
   492→
   493→  const handleMarkAllRead = async () => {
   494→    const unread = notifications.filter((n) => !n.read);
   495→    await Promise.all(unread.map((n) => api.patch(`/notifications/${n.id}`, { read: true })));
   496→    fetchNotifications();
   497→  };
   498→
   499→  const renderView = () => {
   500→    switch (view) {
   501→      case 'dashboard':
   502→        return <DashboardView />;
   503→      case 'tenders':
   504→        return <TendersView />;
   505→      case 'live-tenders':
   506→        return <LiveTendersView />;
   507→      case 'tender-detail':
   508→        return <TenderDetailView tenderId={viewParams.id} />;
   509→      case 'tender-compare':
   510→        return <TenderCompareView tenderIds={viewParams.ids} />;
