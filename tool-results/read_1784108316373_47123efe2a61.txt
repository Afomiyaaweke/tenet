    95→    95→/* ───────────────────────── Password Strength Meter ───────────────────────── */
    96→    96→function getPasswordStrength(password: string): { score: number; label: string; color: string } {
    97→    97→  let score = 0;
    98→    98→  if (password.length >= 12) score += 20;
    99→    99→  if (password.length >= 16) score += 10;
   100→   100→  if (/[A-Z]/.test(password)) score += 15;
   101→   101→  if (/[a-z]/.test(password)) score += 15;
   102→   102→  if (/[0-9]/.test(password)) score += 15;
   103→   103→  if (/[^A-Za-z0-9]/.test(password)) score += 15;
   104→   104→  if (new Set(password.toLowerCase()).size >= 8) score += 10;
   105→   105→
   106→   106→  if (score >= 80) return { score, label: 'Strong', color: 'bg-green-500' };
   107→   107→  if (score >= 60) return { score, label: 'Good', color: 'bg-blue-500' };
   108→   108→  if (score >= 40) return { score, label: 'Fair', color: 'bg-yellow-500' };
   109→   109→  return { score, label: 'Weak', color: 'bg-red-500' };
   110→   110→}
   111→   111→
   112→   112→function PasswordStrengthMeter({ password }: { password: string }) {
   113→   113→  if (!password) return null;
   114→   114→  const { score, label, color } = getPasswordStrength(password);
   115→   115→  return (
   116→   116→    <div className="space-y-1.5 mt-2">
   117→   117→      <div className="flex gap-1">
   118→   118→        {[25, 50, 75, 100].map(threshold => (
   119→   119→          <div key={threshold} className="h-1.5 flex-1 rounded-full bg-muted">
   120→   120→            <div
   121→   121→              className={`h-full rounded-full transition-all duration-300 ${score >= threshold ? color : 'bg-transparent'}`}
   122→   122→              style={{ width: score >= threshold ? '100%' : '0%' }}
   123→   123→            />
   124→   124→          </div>
   125→   125→        ))}
   126→   126→      </div>
   127→   127→      <div className="flex justify-between items-center">
   128→   128→        <span className="text-xs text-muted-foreground">Strength:</span>
   129→   129→        <span className={`text-xs font-medium ${score >= 80 ? 'text-green-600' : score >= 60 ? 'text-blue-600' : score >= 40 ? 'text-yellow-600' : 'text-red-600'}`}>{label}</span>
   130→   130→      </div>
   131→   131→    </div>
   132→   132→  );
   133→   133→}

   135→   135→function PasswordRequirements({ password }: { password: string }) {

   143→   143→  return (

   160→   160→function StepIndicator({ currentStep, totalSteps = 4 }: { currentStep: RegStep; totalSteps?: number }) {

   162→   162→    <div className="flex items-center justify-center gap-1 mb-6">
   163→   163→      {Array.from({ length: totalSteps }, (_, i) => (i + 1) as RegStep).map((step, i) => {
   164→   164→        const meta = REG_STEP_META[step];
   165→   165→        const isCompleted = currentStep > step;
   166→   166→        const isCurrent = currentStep === step;
   167→   167→        const Icon = meta.icon;
   168→   168→        return (
   169→   169→          <React.Fragment key={step}>
   170→   170→            <div className="flex items-center gap-1.5">
   171→   171→              <div
   172→   172→                className={`h-7 w-7 rounded-full flex items-center justify-center text-xs font-bold transition-all ${
   173→   173→                  isCompleted || isCurrent
   174→   174→                    ? 'gradient-orange text-white shadow-md shadow-orange-500/30'
   175→   175→                    : 'bg-muted text-muted-foreground'
   176→   176→                }`}
   177→   177→              >
   178→   178→                {isCompleted ? <CheckCircle2 className="w-4 h-4" /> : <Icon className="w-3.5 h-3.5" />}
   179→   179→              </div>
   180→   180→              <span className={`text-xs font-medium hidden sm:inline ${isCompleted || isCurrent ? 'text-foreground' : 'text-muted-foreground'}`}>
   181→   181→                {meta.label}

   183→   183→            </div>
   184→   184→            {i < totalSteps - 1 && (
   185→   185→              <div className={`h-0.5 w-4 sm:w-8 rounded-full ${currentStep > step ? 'bg-orange-500' : 'bg-muted'}`} />
   186→   186→            )}
   187→   187→          </React.Fragment>
   188→   188→        );
   189→   189→      })}
