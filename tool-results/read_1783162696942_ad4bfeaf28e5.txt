   127→   127→function fmtMoney(amount: number, currency: string): string {
   128→   128→  if (!amount) return '—';
   129→   129→  try {
   130→   130→    return new Intl.NumberFormat('en-US', {
   131→   131→      style: 'currency',
   132→   132→      currency: currency || 'USD',
   133→   133→      maximumFractionDigits: 0,
   134→   134→    }).format(amount);
   135→   135→  } catch {
   136→   136→    return `${currency} ${amount.toLocaleString()}`;
   137→   137→  }
   138→   138→}

   140→   140→function daysUntil(dateStr: string): number {
   141→   141→  return Math.ceil((new Date(dateStr).getTime() - Date.now()) / (1000 * 60 * 60 * 24));
   142→   142→}
   143→   143→
   144→   144→function deadlineBadge(days: number): string {
   145→   145→  if (days <= 0) return 'bg-rose-50 text-rose-700 dark:bg-rose-950/40 dark:text-rose-300';
   146→   146→  if (days <= 7) return 'bg-amber-50 text-amber-700 dark:bg-amber-950/40 dark:text-amber-300';
   147→   147→  if (days <= 21) return 'bg-sky-50 text-sky-700 dark:bg-sky-950/40 dark:text-sky-300';
   148→   148→  return 'bg-emerald-50 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300';
   149→   149→}
   150→   150→
   151→   151→function relativeTime(dateStr: string): string {
   152→   152→  const diff = Date.now() - new Date(dateStr).getTime();
   153→   153→  const day = 86400000;
   154→   154→  if (diff < day) return 'today';
   155→   155→  if (diff < 2 * day) return 'yesterday';
   156→   156→  if (diff < 30 * day) return `${Math.floor(diff / day)}d ago`;
   157→   157→  return new Date(dateStr).toLocaleDateString();
   158→   158→}
