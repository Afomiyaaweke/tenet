   187→function relativeTime(dateStr: string): string {
   188→  const now = Date.now();
   189→  const then = new Date(dateStr).getTime();
   190→  const diff = Math.max(0, now - then);
   191→  const seconds = Math.floor(diff / 1000);
   192→  if (seconds < 60) return 'just now';
   193→  const minutes = Math.floor(seconds / 60);
   194→  if (minutes < 60) return `${minutes}m ago`;
   195→  const hours = Math.floor(minutes / 60);
   196→  if (hours < 24) return `${hours}h ago`;
   197→  const days = Math.floor(hours / 24);
   198→  if (days < 7) return `${days}d ago`;
   199→  return new Date(dateStr).toLocaleDateString();
   200→}

   362→/* ──────────────────── NotificationDropdown ──────────────────── */
   363→
   364→function NotificationDropdown({
   365→  notifications,
   366→  onMarkAsRead,
   367→  onMarkAllRead,
   368→}: {
   369→  notifications: { id: string; title: string; message: string; type: string; read: boolean; createdAt: string }[];
   370→  onMarkAsRead: (id: string) => void;
   371→  onMarkAllRead: () => void;

   373→  const [open, setOpen] = useState(false);
   374→  const unread = useMemo(() => notifications.filter((n) => !n.read), [notifications]);
   375→  const displayItems = useMemo(() => unread.slice(0, 6), [unread]);
   376→
   377→  return (
   378→    <Popover open={open} onOpenChange={setOpen}>
   379→      <PopoverTrigger asChild>
   380→        <Button variant="ghost" size="icon" className="relative h-9 w-9 hover:bg-muted/80 transition-colors">
   381→          <Bell className="h-[18px] w-[18px] text-muted-foreground" />

   383→            <span className="absolute -top-0.5 -right-0.5 min-w-[18px] h-[18px] px-1 bg-red-500 text-white text-[10px] font-bold rounded-full flex items-center justify-center shadow-sm shadow-red-200 animate-in zoom-in-50 duration-200">
   384→              {unread.length > 99 ? '99+' : unread.length}
   385→            </span>
   386→          )}
   387→        </Button>
   388→      </PopoverTrigger>
   389→      <PopoverContent

   392→        sideOffset={8}

   396→          <div className="flex items-center gap-2">
   397→            <h4 className="text-sm font-bold text-foreground">Notifications</h4>
   398→            {unread.length > 0 && (
   399→              <Badge className="text-[10px] px-1.5 py-0 bg-red-500 text-white font-bold border-0">

   406→              variant="ghost"
   407→              size="sm"

   409→              onClick={onMarkAllRead}

   411→              <Check className="h-3 w-3 mr-1" /> Mark all read

   413→          )}
   414→        </div>

   416→        {/* Items */}
   417→        <ScrollArea className="max-h-[340px]">
   418→          {displayItems.length === 0 ? (
   419→            <div className="py-8 text-center">
   420→              <div className="w-12 h-12 mx-auto mb-3 rounded-full bg-orange-50 dark:bg-orange-950/30 flex items-center justify-center">
   421→                <CheckCircle className="h-6 w-6 text-orange-400" />
   422→              </div>
   423→              <p className="text-sm font-medium text-muted-foreground">All caught up!</p>
   424→              <p className="text-xs text-muted-foreground/60 mt-0.5">No new notifications</p>
   425→            </div>
   426→          ) : (
   427→            <div className="divide-y divide-border/40">
   428→              {displayItems.map((n, i) => {
   429→                const Icon = NOTIFICATION_ICONS[n.type] || Info;
   430→                const colorClass = NOTIFICATION_COLORS[n.type] || 'text-gray-500';
   431→                const bgClass = NOTIFICATION_BG[n.type] || 'bg-gray-50';
   432→                return (
   433→                  <button
   434→                    key={n.id}
   435→                    onClick={() => {
   436→                      onMarkAsRead(n.id);
   437→                      setOpen(false);
   438→                    }}
   439→                    className="w-full flex items-start gap-3 px-4 py-3 hover:bg-muted/50 transition-colors text-left group"
   440→                    style={{ animationDelay: `${i * 40}ms` }}
   441→                  >
   442→                    <div
   443→                      className={`w-8 h-8 rounded-lg ${bgClass} flex items-center justify-center flex-shrink-0 mt-0.5 transition-transform group-hover:scale-110`}
   444→                    >
   445→                      <Icon className={`h-4 w-4 ${colorClass}`} />
   446→                    </div>
   447→                    <div className="min-w-0 flex-1">
   448→                      <p className="text-[13px] font-semibold truncate text-foreground">{n.title}</p>
   449→                      <p className="text-[11px] text-muted-foreground truncate mt-0.5 leading-relaxed">{n.message}</p>
   450→                      <p className="text-[10px] text-muted-foreground/50 mt-1 font-medium">{relativeTime(n.createdAt)}</p>
   451→                    </div>
   452→                    <div className="w-2 h-2 rounded-full bg-orange-500 flex-shrink-0 mt-2 ring-2 ring-orange-100 dark:ring-orange-900/40" />
   453→                  </button>
   454→                );
   455→              })}
   456→            </div>
   457→          )}
   458→        </ScrollArea>
   459→
   460→        {/* Footer */}
   461→        {notifications.length > 6 && (
   462→          <div className="px-3 py-2 border-t border-border/40">
   463→            <Button
   464→              variant="ghost"
   465→              size="sm"
   466→              className="w-full text-[11px] text-muted-foreground hover:text-foreground font-medium"
   467→              onClick={() => setOpen(false)}
   468→            >
   469→              View all notifications
   470→            </Button>
   471→          </div>
   472→        )}
   473→      </PopoverContent>
   474→    </Popover>
   475→  );
   476→}
