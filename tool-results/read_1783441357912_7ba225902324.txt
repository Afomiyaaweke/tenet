   111→function generateSeal(text: string, date: string): string {
   112→  const svg = `<svg xmlns="http://www.w3.org/2000/svg" width="120" height="120" viewBox="0 0 120 120">
   113→    <circle cx="60" cy="60" r="55" fill="none" stroke="#dc2626" stroke-width="3"/>
   114→    <circle cx="60" cy="60" r="48" fill="none" stroke="#dc2626" stroke-width="1.5"/>
   115→    <path id="topArc" d="M 15,60 A 45,45 0 0,1 105,60" fill="none"/>
   116→    <text font-size="10" fill="#dc2626" font-weight="bold" letter-spacing="3">
   117→      <textPath href="#topArc" startOffset="50%" text-anchor="middle">${text}</textPath>
   118→    </text>
   119→    <text x="60" y="68" text-anchor="middle" font-size="9" fill="#dc2626">${date}</text>
   120→    <text x="60" y="80" text-anchor="middle" font-size="7" fill="#dc2626">TENETS</text>
   121→  </svg>`;
   122→  return 'data:image/svg+xml;base64,' + btoa(svg);
   123→}

   125→function formatAIResultToHTML(data: Record<string, unknown>): string {
   126→  let html = '';
   127→  for (const [key, value] of Object.entries(data)) {
   128→    const title = key.replace(/([A-Z])/g, ' $1').replace(/^./, s => s.toUpperCase());
   129→    const content = typeof value === 'string' ? value : JSON.stringify(value, null, 2);
   130→    const formattedContent = content
   131→      .replace(/&/g, '&amp;')
   132→      .replace(/</g, '&lt;')
   133→      .replace(/>/g, '&gt;')
   134→      .replace(/\*\*(.+?)\*\*/g, '<strong>$1</strong>')
   135→      .replace(/^[-*•]\s+(.*)/gm, '<li>$1</li>')
   136→      .replace(/^(\d+)[.)]\s+(.*)/gm, '<li>$2</li>')
   137→      .replace(/^#{1,3}\s+(.*)/gm, '<strong>$1</strong>')
   138→      .replace(/\n\n/g, '</p><p>')
   139→      .replace(/\n/g, '<br/>');
   140→    html += `<h2 style="font-size:16px;font-weight:700;margin:16px 0 8px;color:#059669;border-bottom:1px solid #d1fae5;padding-bottom:4px;">${title}</h2>`;
   141→    html += `<p>${formattedContent}</p>`;
   142→  }
   143→  return html;
   144→}

   146→function loadSignatures(): SavedSignature[] {
   147→  if (typeof window === 'undefined') return [];
   148→  try {
   149→    const raw = localStorage.getItem('tenets_signatures');
   150→    return raw ? JSON.parse(raw) : [];
   151→  } catch { return []; }
   152→}
   153→
   154→function saveSignaturesToStorage(sigs: SavedSignature[]) {
   155→  if (typeof window === 'undefined') return;
   156→  localStorage.setItem('tenets_signatures', JSON.stringify(sigs));
   157→}

   162→
   163→function SkillTagSelector({ selected, onChange }: { selected: string[]; onChange: (s: string[]) => void }) {
   164→  const toggle = (skill: string) => {
   165→    if (selected.includes(skill)) onChange(selected.filter(s => s !== skill));
   166→    else onChange([...selected, skill]);
   167→  };
   168→  return (
   169→    <div className="flex flex-wrap gap-1">
   170→      {COMMON_SKILLS.map(skill => (
   171→        <button key={skill} type="button" onClick={() => toggle(skill)}
   172→          className={`px-2 py-0.5 rounded-full text-[10px] font-medium transition-all border ${
   173→            selected.includes(skill)
   174→              ? 'bg-emerald-50 text-emerald-700 border-emerald-300'
   175→              : 'bg-muted/50 text-muted-foreground border-border hover:border-emerald-300'
   176→          }`}>
   177→          {skill}
   178→        </button>
   179→      ))}
   180→    </div>
   181→  );
   182→}
