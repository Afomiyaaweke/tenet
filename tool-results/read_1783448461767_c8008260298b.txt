   129→/* ───────────────────────── Password Strength Meter ───────────────────────── */
   130→function scorePassword(pw: string): { score: number; label: string; color: string } {
   131→  let score = 0;
   132→  if (pw.length >= 8) score++;
   133→  if (pw.length >= 12) score++;
   134→  if (/[a-z]/.test(pw) && /[A-Z]/.test(pw)) score++;
   135→  if (/[0-9]/.test(pw)) score++;
   136→  if (/[^A-Za-z0-9]/.test(pw)) score++;
   137→  const labels = ['Too weak', 'Weak', 'Fair', 'Good', 'Strong', 'Very strong'];
   138→  const colors = ['bg-rose-500', 'bg-rose-500', 'bg-amber-500', 'bg-yellow-500', 'bg-orange-500', 'bg-orange-600'];
   139→  return { score, label: labels[score], color: colors[score] };
   140→}

   142→/* ───────────────────────── Slide-to-Verify Captcha ───────────────────────── */
   143→function SlideCaptcha({ onVerified }: { onVerified: () => void }) {
   144→  const [offset, setOffset] = useState(0);
   145→  const [status, setStatus] = useState<'idle' | 'dragging' | 'success' | 'fail'>('idle');
   146→  const [target] = useState(() => 180 + Math.floor(Math.random() * 80));
   147→  const trackWidth = 320;
   148→  const handleWidth = 44;
   149→  const tolerance = 8;
   150→  const dragRef = React.useRef<{ start: number; startOffset: number } | null>(null);
   151→
   152→  const onPointerDown = (e: React.PointerEvent) => {
   153→    if (status === 'success') return;
   154→    setStatus('dragging');
   155→    dragRef.current = { start: e.clientX, startOffset: offset };
   156→    (e.target as HTMLElement).setPointerCapture(e.pointerId);
   157→  };
   158→
   159→  const onPointerMove = (e: React.PointerEvent) => {
   160→    if (!dragRef.current || status !== 'dragging') return;
   161→    const delta = e.clientX - dragRef.current.start;
   162→    const next = Math.max(0, Math.min(trackWidth - handleWidth, dragRef.current.startOffset + delta));
   163→    setOffset(next);
   164→  };
   165→
   166→  const onPointerUp = () => {
   167→    if (status !== 'dragging') return;
   168→    dragRef.current = null;
   169→    if (Math.abs(offset - target) <= tolerance) {
   170→      setStatus('success');
   171→      setTimeout(() => onVerified(), 450);
   172→    } else {
   173→      setStatus('fail');
   174→      setTimeout(() => {
   175→        setStatus('idle');
   176→        setOffset(0);
   177→      }, 700);
   178→    }
   179→  };
   180→
   181→  const pieceColor = status === 'success' ? 'bg-orange-500' : status === 'fail' ? 'bg-rose-500' : 'bg-primary';
   182→  const ringColor = status === 'success' ? 'ring-orange-300' : status === 'fail' ? 'ring-rose-300' : 'ring-primary/40';
   183→
   184→  return (
   185→    <div className="space-y-3">
   186→      <div className="text-xs font-medium text-muted-foreground flex items-center gap-1.5">
   187→        <Puzzle className="w-3.5 h-3.5" /> Security verification — slide to complete
   188→      </div>
   189→      {/* Puzzle canvas */}
   190→      <div className="relative h-12 rounded-xl overflow-hidden border border-border bg-muted/40 select-none">
   191→        {/* pattern background */}
   192→        <div
   193→          className="absolute inset-0 opacity-50"
   194→          style={{
   195→            backgroundImage:
   196→              'repeating-linear-gradient(45deg, hsl(var(--muted-foreground) / 0.08) 0 6px, transparent 6px 12px), repeating-linear-gradient(-45deg, hsl(var(--muted-foreground) / 0.06) 0 6px, transparent 6px 12px)',
   197→          }}
   198→        />
   199→        {/* target cutout (notch) */}
   200→        <div
   201→          className="absolute top-1/2 -translate-y-1/2 h-9 w-9 rounded-lg border-2 border-dashed border-muted-foreground/40 bg-background/60"
   202→          style={{ left: target + 4 }}
   203→        />
   204→        {/* moving puzzle piece */}
   205→        <div
   206→          className={`absolute top-1/2 -translate-y-1/2 h-9 w-9 rounded-lg ${pieceColor} shadow-lg ring-2 ${ringColor} flex items-center justify-center text-white transition-shadow ${status === 'fail' ? 'animate-[shake_0.4s_ease-in-out]' : ''}`}
   207→          style={{ left: 4 + offset, transition: status === 'dragging' ? 'none' : 'left 0.25s ease' }}
   208→        >
   209→          {status === 'success' ? <CheckCircle2 className="w-5 h-5" /> : status === 'fail' ? <XCircle className="w-5 h-5" /> : <Puzzle className="w-4 h-4" />}
   210→        </div>
   211→        {/* hint text */}
   212→        {status === 'idle' && (
   213→          <span className="absolute inset-0 flex items-center justify-center text-xs text-muted-foreground pointer-events-none">
   214→            Drag the slider below to fit the puzzle
   215→          </span>
   216→        )}
   217→        {status === 'success' && (
   218→          <span className="absolute inset-0 flex items-center justify-center text-xs font-medium text-orange-600 bg-orange-50/60 pointer-events-none">
   219→            Verified
   220→          </span>
   221→        )}
   222→      </div>
   223→      {/* Slider track */}
   224→      <div className="relative h-11 rounded-xl bg-muted border border-border overflow-hidden">
   225→        <div
   226→          className={`absolute inset-y-0 left-0 ${status === 'success' ? 'bg-orange-100 dark:bg-orange-950/40' : status === 'fail' ? 'bg-rose-100 dark:bg-rose-950/40' : 'bg-primary/10'} transition-all`}
   227→          style={{ width: offset + handleWidth }}
   228→        />
   229→        <div
   230→          className="absolute top-0 h-full flex items-center justify-center text-xs font-medium text-muted-foreground pointer-events-none"
   231→          style={{ left: handleWidth, right: 0 }}
   232→        >
   233→          {status === 'success' ? 'Verification complete' : status === 'fail' ? 'Verification failed, retrying…' : 'Slide right to verify'}
   234→        </div>
   235→        <div
   236→          onPointerDown={onPointerDown}
   237→          onPointerMove={onPointerMove}
   238→          onPointerUp={onPointerUp}
   239→          onPointerCancel={onPointerUp}
   240→          className={`absolute top-1/2 -translate-y-1/2 h-9 w-9 rounded-lg cursor-grab active:cursor-grabbing flex items-center justify-center text-white shadow-md ${pieceColor} ${status === 'success' ? 'cursor-default' : ''} touch-none`}
   241→          style={{ left: 4 + offset, transition: status === 'dragging' ? 'none' : 'left 0.25s ease' }}
   242→        >
   243→          {status === 'success' ? <CheckCircle2 className="w-5 h-5" /> : <ArrowRight className="w-4 h-4" />}
   244→        </div>
   245→      </div>
   246→    </div>
   247→  );
   248→}

   250→/* ───────────────────────── 6-digit Security Code Input ───────────────────────── */
   251→function SecurityCodeInput({ value, onChange, disabled }: { value: string; onChange: (v: string) => void; disabled?: boolean }) {
   252→  const inputsRef = React.useRef<(HTMLInputElement | null)[]>([]);
   253→  const chars = value.split('');
   254→
   255→  const setChar = (i: number, c: string) => {
   256→    const sanitized = c.replace(/\D/g, '').slice(-1);
   257→    const arr = value.split('');
   258→    arr[i] = sanitized || ' ';
   259→    const next = arr.join('').replace(/ /g, '');
   260→    const padded = (next + '      ').slice(0, 6).replace(/\s/g, '');
   261→    onChange(padded);
   262→    if (sanitized && i < 5) inputsRef.current[i + 1]?.focus();
   263→  };
   264→
   265→  const onKeyDown = (i: number, e: React.KeyboardEvent<HTMLInputElement>) => {
   266→    if (e.key === 'Backspace' && !chars[i] && i > 0) {
   267→      inputsRef.current[i - 1]?.focus();
   268→    }
   269→  };
   270→
   271→  const onPaste = (e: React.ClipboardEvent) => {
   272→    const text = e.clipboardData.getData('text').replace(/\D/g, '').slice(0, 6);
   273→    if (text) {
   274→      e.preventDefault();
   275→      onChange(text);
   276→      inputsRef.current[Math.min(text.length, 5)]?.focus();
   277→    }
   278→  };
   279→
   280→  return (
   281→    <div className="flex justify-between gap-2" onPaste={onPaste}>
   282→      {Array.from({ length: 6 }).map((_, i) => (
   283→        <input
   284→          key={i}
   285→          ref={(el) => { inputsRef.current[i] = el; }}
   286→          type="text"
   287→          inputMode="numeric"
   288→          maxLength={1}
   289→          disabled={disabled}
   290→          value={chars[i] || ''}
   291→          onChange={(e) => setChar(i, e.target.value)}
   292→          onKeyDown={(e) => onKeyDown(i, e)}
   293→          className="w-11 h-13 sm:w-12 sm:h-14 text-center text-xl font-bold rounded-xl border-2 bg-muted/30 focus:bg-background focus:border-primary focus:ring-2 focus:ring-primary/20 outline-none transition-all"
   294→          style={{ aspectRatio: '1 / 1.15' }}
   295→        />
   296→      ))}
   297→    </div>
   298→  );
   299→}
