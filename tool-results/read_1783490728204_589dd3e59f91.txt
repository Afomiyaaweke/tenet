    41→function timeAgo(dateStr: string): string {
    42→  const diff = Date.now() - new Date(dateStr).getTime();
    43→  const mins = Math.floor(diff / 60000);
    44→  if (mins < 1) return 'just now';
    45→  if (mins < 60) return `${mins}m ago`;
    46→  const hrs = Math.floor(mins / 60);
    47→  if (hrs < 24) return `${hrs}h ago`;
    48→  const days = Math.floor(hrs / 24);
    49→  return `${days}d ago`;
    50→}
