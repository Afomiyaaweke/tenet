    92→    92→function timeAgo(dateStr: string): string {
    93→    93→  const diff = Date.now() - new Date(dateStr).getTime();
    94→    94→  const mins = Math.floor(diff / 60000);
    95→    95→  if (mins < 1) return 'just now';
    96→    96→  if (mins < 60) return `${mins}m ago`;
    97→    97→  const hrs = Math.floor(mins / 60);
    98→    98→  if (hrs < 24) return `${hrs}h ago`;
    99→    99→  const days = Math.floor(hrs / 24);
   100→   100→  return `${days}d ago`;
   101→   101→}

   107→   107→// ─── Sparkline SVG Component ────────────────────────────────────────
   108→   108→function SparklineBars({ data, color = '#10b981', height = 32 }: { data: number[]; color?: string; height?: number }) {
   109→   109→  if (!data.length) return null;
   110→   110→  const max = Math.max(...data, 1);
   111→   111→  const barW = 4;
   112→   112→  const gap = 3;
   113→   113→  const totalW = data.length * (barW + gap) - gap;
   114→   114→
   115→   115→  return (
   116→   116→    <svg width={totalW} height={height} className="flex-shrink-0">
   117→   117→      {data.map((v, i) => {
   118→   118→        const barH = Math.max((v / max) * (height - 4), 3);
   119→   119→        return (
   120→   120→          <rect
   121→   121→            key={i}
   122→   122→            x={i * (barW + gap)}
   123→   123→            y={height - barH}
   124→   124→            width={barW}
   125→   125→            height={barH}
   126→   126→            rx={2}
   127→   127→            fill={color}
   128→   128→            opacity={0.35 + (i / data.length) * 0.45}
   129→   129→          />
   130→   130→        );
   131→   131→      })}
   132→   132→    </svg>
   133→   133→  );
   134→   134→}
