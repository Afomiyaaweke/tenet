   117→/* ───────────────────────── Password Strength Meter ───────────────────────── */
   118→function scorePassword(pw: string): { score: number; label: string; color: string } {
   119→  let score = 0;
   120→  if (pw.length >= 8) score++;
   121→  if (pw.length >= 12) score++;
   122→  if (/[a-z]/.test(pw) && /[A-Z]/.test(pw)) score++;
   123→  if (/[0-9]/.test(pw)) score++;
   124→  if (/[^A-Za-z0-9]/.test(pw)) score++;
   125→  const labels = ['Too weak', 'Weak', 'Fair', 'Good', 'Strong', 'Very strong'];
   126→  const colors = ['bg-rose-500', 'bg-rose-500', 'bg-amber-500', 'bg-yellow-500', 'bg-orange-500', 'bg-orange-600'];
   127→  return { score, label: labels[score], color: colors[score] };
   128→}

   130→/* ───────────────────────── Step Indicator ───────────────────────── */
   131→function StepIndicator({ currentStep, totalSteps = 5 }: { currentStep: RegStep; totalSteps?: number }) {
   132→  return (
   133→    <div className="flex items-center justify-center gap-1 mb-6">
   134→      {Array.from({ length: totalSteps }, (_, i) => (i + 1) as RegStep).map((step, i) => {
   135→        const meta = REG_STEP_META[step];
   136→        const isCompleted = currentStep > step;
   137→        const isCurrent = currentStep === step;
   138→        const Icon = meta.icon;
   139→        return (
   140→          <React.Fragment key={step}>
   141→            <div className="flex items-center gap-1.5">
   142→              <div
   143→                className={`h-7 w-7 rounded-full flex items-center justify-center text-xs font-bold transition-all ${
   144→                  isCompleted || isCurrent
   145→                    ? 'gradient-orange text-white shadow-md shadow-orange-500/30'
   146→                    : 'bg-muted text-muted-foreground'
   147→                }`}
   148→              >
   149→                {isCompleted ? <CheckCircle2 className="w-4 h-4" /> : <Icon className="w-3.5 h-3.5" />}
   150→              </div>
   151→              <span className={`text-xs font-medium hidden sm:inline ${isCompleted || isCurrent ? 'text-foreground' : 'text-muted-foreground'}`}>
   152→                {meta.label}
   153→              </span>
   154→            </div>
   155→            {i < totalSteps - 1 && (
   156→              <div className={`h-0.5 w-4 sm:w-8 rounded-full ${currentStep > step ? 'bg-orange-500' : 'bg-muted'}`} />
   157→            )}
   158→          </React.Fragment>
   159→        );
   160→      })}
   161→    </div>
   162→  );
   163→}
