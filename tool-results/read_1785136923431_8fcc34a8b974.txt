   177→// ── Main Component ─────────────────────────────────────────────────────
   178→
   179→export function InfraDashboardView() {
   180→  const { user } = useAuthStore();
   181→  const isAdmin = user?.role === 'team_admin';
   182→
   183→  const [loading, setLoading] = useState(true);
   184→  const [healthData, setHealthData] = useState<Record<string, unknown> | null>(null);
   185→  const [metricsData, setMetricsData] = useState<Record<string, unknown> | null>(null);
   186→  const [alerts, setAlerts] = useState<Array<Record<string, unknown>>>([]);
   187→  const [webhooks, setWebhooks] = useState<Array<Record<string, unknown>>>([]);
   188→  const [rateLimits, setRateLimits] = useState<Array<Record<string, unknown>>>([]);
   189→  const [secrets, setSecrets] = useState<Array<Record<string, unknown>>>([]);
   190→  const [cacheEntries, setCacheEntries] = useState<Array<Record<string, unknown>>>([]);
   191→
   192→  // Dialog states
   193→  const [dlgOpen, setDlgOpen] = useState(false);
   194→  const [dlgType, setDlgType] = useState<string>('');
   195→  const [dlgData, setDlgData] = useState<Record<string, unknown>>({});
   196→
   197→  const concerns = buildConcerns(healthData, metricsData);
   198→
   199→  const fetchData = useCallback(async () => {
   200→    setLoading(true);
   201→    try {
   202→      const [healthRes, metricsRes, alertsRes, webhooksRes, rateLimitsRes, secretsRes, cacheRes] = await Promise.allSettled([
   203→        api.get('/infra/health'),
   204→        api.get('/infra/metrics'),
   205→        api.get('/infra/alerts'),
   206→        api.get('/infra/webhooks'),
   207→        api.get('/infra/rate-limits'),
   208→        isAdmin ? api.get('/infra/secrets') : Promise.resolve({ success: false, data: [] }),
   209→        api.get('/infra/cache'),
   210→      ]);
   211→
   212→      if (healthRes.status === 'fulfilled' && healthRes.value.success) setHealthData(healthRes.value.data);
   213→      if (metricsRes.status === 'fulfilled' && metricsRes.value.success) setMetricsData(metricsRes.value.data);
   214→      if (alertsRes.status === 'fulfilled' && alertsRes.value.success) setAlerts(alertsRes.value.data || []);
   215→      if (webhooksRes.status === 'fulfilled' && webhooksRes.value.success) setWebhooks(webhooksRes.value.data || []);
   216→      if (rateLimitsRes.status === 'fulfilled' && rateLimitsRes.value.success) setRateLimits(rateLimitsRes.value.data || []);
   217→      if (secretsRes.status === 'fulfilled' && secretsRes.value.success) setSecrets(secretsRes.value.data || []);
   218→      if (cacheRes.status === 'fulfilled' && cacheRes.value.success) setCacheEntries(cacheRes.value.data?.entries || []);
   219→    } catch {
   220→      toast.error('Failed to load infrastructure data');
   221→    }
   222→    setLoading(false);
   223→  }, [isAdmin]);
   224→
   225→  /* eslint-disable react-hooks/set-state-in-effect */
   226→  useEffect(() => { fetchData(); }, [fetchData]);
   227→  /* eslint-enable react-hooks/set-state-in-effect */
   228→
   229→  const score = (healthData?.score as number) ?? 85;
   230→  const activeAlerts = alerts.filter((a) => a.status === 'active').length;
   231→  const healthyCount = concerns.filter((c) => c.status === 'healthy').length;
   232→  const warningCount = concerns.filter((c) => c.status === 'warning').length;
   233→  const criticalCount = concerns.filter((c) => c.status === 'critical').length;
   234→  const uptime = healthData?.uptime ? `${(Number(healthData.uptime) / 3600).toFixed(1)}h` : '—';
   235→
   236→  const scoreColor = score >= 80 ? 'text-emerald-500' : score >= 60 ? 'text-amber-500' : score >= 40 ? 'text-orange-500' : 'text-red-500';
   237→  const scoreStroke = score >= 80 ? '#10b981' : score >= 60 ? '#f59e0b' : score >= 40 ? '#f97316' : '#ef4444';
   238→
   239→  // ── Dialog handlers ──────────────────────────────────────────────────
   240→
   241→  const openConfig = (concernId: string) => {
   242→    setDlgType(concernId);
   243→    if (concernId === 'rate_limiting') {
   244→      setDlgData({ endpoint: '', windowMs: 60000, maxRequests: 100, strategy: 'sliding_window' });
   245→    } else if (concernId === 'webhooks') {
   246→      setDlgData({ name: '', url: '', events: 'tender.created,bid.submitted', active: true });
   247→    } else if (concernId === 'secret_management') {
   248→      setDlgData({ key: '', value: '', category: 'general', description: '' });
   249→    } else {
   250→      setDlgData({ enabled: true, notes: '' });
   251→    }
   252→    setDlgOpen(true);
   253→  };
   254→
   255→  const saveConfig = async () => {
   256→    try {
   257→      if (dlgType === 'rate_limiting') {
   258→        const res = await api.post('/infra/rate-limits', dlgData);
   259→        if (res.success) { toast.success('Rate limit rule created'); fetchData(); }
   260→        else toast.error(res.error || 'Failed');
   261→      } else if (dlgType === 'webhooks') {
   262→        const res = await api.post('/infra/webhooks', dlgData);
   263→        if (res.success) { toast.success('Webhook created'); fetchData(); }
   264→        else toast.error(res.error || 'Failed');
   265→      } else if (dlgType === 'secret_management') {
   266→        const res = await api.post('/infra/secrets', dlgData);
   267→        if (res.success) { toast.success('Secret created'); fetchData(); }
   268→        else toast.error(res.error || 'Failed');
   269→      } else {
   270→        toast.success(`${dlgType} configuration saved`);
   271→      }
   272→      setDlgOpen(false);
   273→    } catch { toast.error('Failed to save configuration'); }
   274→  };
   275→
   276→  const testWebhook = async (id: string) => {
   277→    try {
   278→      const res = await api.post('/infra/webhooks', { action: 'test', id });
   279→      if (res.success) toast.success('Webhook test sent successfully');
   280→      else toast.error(res.error || 'Webhook test failed');
   281→    } catch { toast.error('Webhook test failed'); }
   282→  };
   283→
   284→  const rotateSecret = async (id: string) => {
   285→    try {
   286→      const res = await api.post('/infra/secrets', { action: 'rotate', id });
   287→      if (res.success) { toast.success('Secret rotated'); fetchData(); }
   288→      else toast.error(res.error || 'Rotation failed');
   289→    } catch { toast.error('Secret rotation failed'); }
   290→  };
   291→
   292→  const acknowledgeAlert = async (id: string) => {
   293→    try {
   294→      const res = await api.put('/infra/alerts', { id, status: 'acknowledged', acknowledgedBy: user?.id });
   295→      if (res.success) { toast.success('Alert acknowledged'); fetchData(); }
   296→    } catch { toast.error('Failed to acknowledge'); }
   297→  };
   298→
   299→  const resolveAlert = async (id: string) => {
   300→    try {
   301→      const res = await api.put('/infra/alerts', { id, status: 'resolved' });
   302→      if (res.success) { toast.success('Alert resolved'); fetchData(); }
   303→    } catch { toast.error('Failed to resolve'); }
   304→  };
   305→
   306→  // ── Render ───────────────────────────────────────────────────────────
   307→
   308→  if (loading && !healthData) {
   309→    return (
   310→      <div className="p-6 space-y-4">
   311→        <Skeleton className="h-10 w-64" />
   312→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-5 gap-4">
   313→          {Array.from({ length: 5 }).map((_, i) => <Skeleton key={i} className="h-28" />)}
   314→        </div>
   315→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-4">
   316→          {Array.from({ length: 8 }).map((_, i) => <Skeleton key={i} className="h-44" />)}
   317→        </div>
   318→      </div>
   319→    );
   320→  }
   321→
   322→  return (
   323→    <div className="p-4 md:p-6 space-y-6">
   324→      {/* ── Header ── */}
   325→      <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-4">
   326→        <div>
   327→          <h1 className="text-2xl font-bold text-foreground">Infrastructure & DevOps</h1>
   328→          <p className="text-sm text-muted-foreground mt-1">Monitor and manage all 23 infrastructure concerns</p>
   329→        </div>
   330→        <div className="flex items-center gap-2">
   331→          {activeAlerts > 0 && (
   332→            <Badge variant="destructive" className="gap-1">
   333→              <AlertTriangle className="w-3 h-3" /> {activeAlerts} Alert{activeAlerts > 1 ? 's' : ''}
   334→            </Badge>
   335→          )}
   336→          <Button variant="outline" size="sm" onClick={fetchData} disabled={loading}>
   337→            <RefreshCw className={`w-4 h-4 mr-1.5 ${loading ? 'animate-spin' : ''}`} /> Refresh
   338→          </Button>
   339→        </div>
   340→      </div>
   341→
   342→      {/* ── Health Score + Quick Stats ── */}
   343→      <div className="grid grid-cols-1 md:grid-cols-6 gap-4">
   344→        {/* Health Score */}
   345→        <Card className="md:col-span-2">
   346→          <CardContent className="pt-6 flex items-center gap-6">
   347→            <div className="relative w-28 h-28 flex-shrink-0">
   348→              <svg className="w-28 h-28 -rotate-90" viewBox="0 0 120 120">
   349→                <circle cx="60" cy="60" r="50" fill="none" stroke="currentColor" strokeWidth="8" className="text-muted/30" />
   350→                <circle cx="60" cy="60" r="50" fill="none" stroke={scoreStroke} strokeWidth="8"
   351→                  strokeDasharray={`${(score / 100) * 314} 314`} strokeLinecap="round" />
   352→              </svg>
   353→              <div className="absolute inset-0 flex items-center justify-center">
   354→                <span className={`text-3xl font-bold ${scoreColor}`}>{score}</span>
   355→              </div>
   356→            </div>
   357→            <div>
   358→              <p className="text-sm font-medium text-muted-foreground">Overall Health</p>
   359→              <p className={`text-lg font-bold ${scoreColor}`}>
   360→                {score >= 80 ? 'Healthy' : score >= 60 ? 'Degraded' : 'Unhealthy'}
   361→              </p>
   362→              <p className="text-xs text-muted-foreground mt-1">Uptime: {uptime}</p>
   363→            </div>
   364→          </CardContent>
   365→        </Card>
   366→
   367→        {/* Quick Stats */}
   368→        {[
   369→          { label: 'Healthy', value: healthyCount, color: 'text-emerald-600', bg: 'bg-emerald-50 dark:bg-emerald-950/20', icon: CheckCircle2 },
   370→          { label: 'Warnings', value: warningCount, color: 'text-amber-600', bg: 'bg-amber-50 dark:bg-amber-950/20', icon: AlertTriangle },
   371→          { label: 'Critical', value: criticalCount, color: 'text-red-600', bg: 'bg-red-50 dark:bg-red-950/20', icon: XCircle },
   372→          { label: 'Uptime', value: uptime, color: 'text-slate-600', bg: 'bg-slate-50 dark:bg-slate-900/30', icon: Clock },
   373→        ].map((stat) => (
   374→          <Card key={stat.label}>
   375→            <CardContent className="pt-6">
   376→              <div className={`w-10 h-10 rounded-lg ${stat.bg} flex items-center justify-center mb-2`}>
   377→                <stat.icon className={`w-5 h-5 ${stat.color}`} />
   378→              </div>
   379→              <p className="text-2xl font-bold">{stat.value}</p>
   380→              <p className="text-xs text-muted-foreground">{stat.label}</p>
   381→            </CardContent>
   382→          </Card>
   383→        ))}
   384→      </div>
   385→
   386→      {/* ── Active Alerts ── */}
   387→      {activeAlerts > 0 && (
   388→        <Card className="border-amber-200 dark:border-amber-800">
   389→          <CardHeader className="pb-2">
   390→            <CardTitle className="text-base flex items-center gap-2">
   391→              <AlertTriangle className="w-4 h-4 text-amber-500" /> Active Alerts
   392→            </CardTitle>
   393→          </CardHeader>
   394→          <CardContent>
   395→            <ScrollArea className="max-h-40">
