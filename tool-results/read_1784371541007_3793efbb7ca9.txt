    57→function getCategoryMeta(cat: string) {

    61→function daysUntil(dateStr: string): number {
    62→  return Math.ceil((new Date(dateStr).getTime() - Date.now()) / (1000 * 60 * 60 * 24));
    63→}
    64→
    65→function deadlineColor(days: number) {
    66→  if (days <= 0) return 'rose';
    67→  if (days <= 3) return 'rose';
    68→  if (days <= 7) return 'amber';
    69→  return 'emerald';
    70→}
    71→
    72→function deadlineBg(days: number) {
    73→  const c = deadlineColor(days);
    74→  const map: Record<string, string> = {
    75→    rose: 'bg-rose-50 text-rose-700',
    76→    amber: 'bg-amber-50 text-amber-700',
    77→    emerald: 'bg-emerald-50 text-emerald-700',
    78→  };
    79→  return map[c] || map.emerald;
    80→}
    81→
    82→function formatBudget(min: number, max: number) {
    83→  const fmt = (n: number) => n >= 1_000_000 ? `${(n / 1_000_000).toFixed(1)}M` : n >= 1_000 ? `${(n / 1_000).toFixed(0)}K` : n.toLocaleString();
    84→  return `ETB ${fmt(min)} – ${fmt(max)}`;
    85→}
