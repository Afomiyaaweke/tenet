   441→                {/* Rate Limits Toggle */}
   442→                <button
   443→                  onClick={() => setShowRateLimits(prev => ({ ...prev, [plan.id]: !prev[plan.id] }))}

   450→                  <div className="grid grid-cols-2 gap-1.5 mt-2 mb-4">

   467→                      <Check className="h-4 w-4 mr-2" />
   468→                      Current Plan

   471→                    <Button
   472→                      variant="outline"
   473→                      className="w-full h-11"

   482→                      className={`w-full h-11 text-white font-semibold ${theme.buttonBg}`}
   483→                      onClick={() => setConfirmDialog({ open: true, plan })}

   485→                    >
   486→                      {upgrading === plan.id ? <RefreshCw className="h-4 w-4 mr-2 animate-spin" /> : <Sparkles className="h-4 w-4 mr-2" />}
   487→                      Upgrade to {plan.name}
   488→                    </Button>

   490→                </div>
   491→              </CardContent>
   492→            </Card>
   493→          );
   494→        })}
   495→      </div>

   497→      {/* ── Feature Comparison ────────────────────────────────────────── */}
   498→      <Card className="overflow-hidden border-0 shadow-md">
   499→        <CardHeader className="bg-muted/30 pb-4">
   500→          <CardTitle className="text-lg flex items-center gap-2">
   501→            <BarChart3 className="h-5 w-5 text-primary" />
   502→            Detailed Feature Comparison
   503→          </CardTitle>
   504→          <CardDescription>See exactly what each plan includes at a glance</CardDescription>
   505→        </CardHeader>
   506→        <CardContent className="p-0">
   507→          <ScrollArea className="max-h-[500px]">
   508→            <table className="w-full text-sm">
   509→              <thead>
   510→                <tr className="border-b bg-muted/20">
   511→                  <th className="py-3 px-5 text-left font-semibold text-muted-foreground w-[40%]">Feature</th>
   512→                  {allPlans.map(p => (
   513→                    <th key={p.id} className={`py-3 px-4 text-center font-bold ${p.id === currentPlanName ? PLAN_THEMES[p.id]?.accent : ''}`}>

   526→                      return (

   529→                            <Check className="h-4 w-4 text-emerald-500 mx-auto" />

   531→                            <X className="h-4 w-4 text-red-300 mx-auto" />

   533→                            <span>{val}</span>
   534→                          )}
   535→                        </td>

   537→                    })}
   538→                  </tr>
   539→                ))}
   540→              </tbody>
   541→            </table>
   542→          </ScrollArea>
   543→        </CardContent>

   549→          <CardTitle className="text-lg flex items-center gap-2">
   550→            <Star className="h-5 w-5 text-primary" />
   551→            Frequently Asked Questions
   552→          </CardTitle>
   553→        </CardHeader>
   554→        <CardContent className="p-0 divide-y divide-muted/50">

   557→              <button
   558→                onClick={() => setExpandedFaq(expandedFaq === idx ? null : idx)}

   567→              </button>

   569→                <div className="px-6 pb-4 -mt-1">
   570→                  <p className="text-sm text-muted-foreground leading-relaxed">{faq.a}</p>

   578→      {/* ── Confirmation Dialog ──────────────────────────────────────── */}
   579→      <Dialog open={confirmDialog.open} onOpenChange={(open) => setConfirmDialog({ open, plan: open ? confirmDialog.plan : null })}>
   580→        <DialogContent className="sm:max-w-md">

   584→                <span className={PLAN_THEMES[confirmDialog.plan.id]?.iconText}>

   590→                : `Upgrade to ${confirmDialog.plan?.name}`}
   591→            </DialogTitle>
   592→            <DialogDescription className="text-sm mt-2">

   594→                ? 'You will lose access to premium features and your limits will be reduced to the Free tier.'

   596→            </DialogDescription>
   597→          </DialogHeader>
   598→
   599→          {confirmDialog.plan && confirmDialog.plan.price > 0 && (
   600→            <div className="space-y-3 py-3">
   601→              <h4 className="text-sm font-semibold">You&apos;ll unlock:</h4>
   602→              <ul className="space-y-1.5">
   603→                {confirmDialog.plan.features.slice(0, 5).map((f, i) => (

   613→          <DialogFooter className="gap-2 sm:gap-2">
   614→            <Button variant="outline" onClick={() => setConfirmDialog({ open: false, plan: null })}>
   615→              Cancel

   627→            >
   628→              {upgrading ? <RefreshCw className="h-4 w-4 mr-2 animate-spin" /> : null}
   629→              {confirmDialog.plan?.price === 0 ? 'Switch to Free' : `Upgrade to ${confirmDialog.plan?.name}`}

   632→        </DialogContent>
