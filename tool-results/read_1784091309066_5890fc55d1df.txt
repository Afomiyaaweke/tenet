   170→   170→function relativeTime(dateStr: string): string {
   171→   171→  const now = Date.now();
   172→   172→  const then = new Date(dateStr).getTime();
   173→   173→  const diff = Math.max(0, now - then);
   174→   174→  const seconds = Math.floor(diff / 1000);
   175→   175→  if (seconds < 60) return 'just now';
   176→   176→  const minutes = Math.floor(seconds / 60);
   177→   177→  if (minutes < 60) return `${minutes}m ago`;
   178→   178→  const hours = Math.floor(minutes / 60);
   179→   179→  if (hours < 24) return `${hours}h ago`;
   180→   180→  const days = Math.floor(hours / 24);
   181→   181→  if (days < 7) return `${days}d ago`;
   182→   182→  return new Date(dateStr).toLocaleDateString();
   183→   183→}
