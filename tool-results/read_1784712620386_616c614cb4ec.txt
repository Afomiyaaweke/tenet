    96→/* ───────────────────────── Password Strength Meter ───────────────────────── */
    97→function getPasswordStrength(password: string): { score: number; label: string; color: string } {
    98→  let score = 0;
    99→  if (password.length >= 12) score += 20;
   100→  if (password.length >= 16) score += 10;
   101→  if (/[A-Z]/.test(password)) score += 15;
   102→  if (/[a-z]/.test(password)) score += 15;
   103→  if (/[0-9]/.test(password)) score += 15;
   104→  if (/[^A-Za-z0-9]/.test(password)) score += 15;
   105→  if (new Set(password.toLowerCase()).size >= 8) score += 10;
   106→
   107→  if (score >= 80) return { score, label: 'Strong', color: 'bg-green-500' };
   108→  if (score >= 60) return { score, label: 'Good', color: 'bg-blue-500' };
   109→  if (score >= 40) return { score, label: 'Fair', color: 'bg-yellow-500' };
   110→  return { score, label: 'Weak', color: 'bg-red-500' };
   111→}
   112→
   113→function PasswordStrengthMeter({ password }: { password: string }) {
   114→  if (!password) return null;
   115→  const { score, label, color } = getPasswordStrength(password);
   116→  return (
   117→    <div className="space-y-1.5 mt-2">
   118→      <div className="flex gap-1">
   119→        {[25, 50, 75, 100].map(threshold => (
   120→          <div key={threshold} className="h-1.5 flex-1 rounded-full bg-muted">
   121→            <div
   122→              className={`h-full rounded-full transition-all duration-300 ${score >= threshold ? color : 'bg-transparent'}`}
   123→              style={{ width: score >= threshold ? '100%' : '0%' }}
   124→            />
   125→          </div>
   126→        ))}
   127→      </div>
   128→      <div className="flex justify-between items-center">
   129→        <span className="text-xs text-muted-foreground">Strength:</span>
   130→        <span className={`text-xs font-medium ${score >= 80 ? 'text-green-600' : score >= 60 ? 'text-blue-600' : score >= 40 ? 'text-yellow-600' : 'text-red-600'}`}>{label}</span>
   131→      </div>
   132→    </div>
   133→  );
   134→}

   136→function PasswordRequirements({ password }: { password: string }) {

   160→/* ───────────────────────── Step Indicator ───────────────────────── */
   161→function StepIndicator({ currentStep, totalSteps = 4 }: { currentStep: RegStep; totalSteps?: number }) {
   162→  return (
   163→    <div className="flex items-center justify-center gap-1 mb-6">
   164→      {Array.from({ length: totalSteps }, (_, i) => (i + 1) as RegStep).map((step, i) => {
   165→        const meta = REG_STEP_META[step];
   166→        const isCompleted = currentStep > step;
   167→        const isCurrent = currentStep === step;
   168→        const Icon = meta.icon;
   169→        return (
   170→          <React.Fragment key={step}>
   171→            <div className="flex items-center gap-1.5">
   172→              <div
   173→                className={`h-7 w-7 rounded-full flex items-center justify-center text-xs font-bold transition-all ${
   174→                  isCompleted || isCurrent
   175→                    ? 'gradient-orange text-white shadow-md shadow-orange-500/30'
   176→                    : 'bg-muted text-muted-foreground'
   177→                }`}
   178→              >
   179→                {isCompleted ? <CheckCircle2 className="w-4 h-4" /> : <Icon className="w-3.5 h-3.5" />}
   180→              </div>
   181→              <span className={`text-xs font-medium hidden sm:inline ${isCompleted || isCurrent ? 'text-foreground' : 'text-muted-foreground'}`}>
   182→                {meta.label}
   183→              </span>
   184→            </div>
   185→            {i < totalSteps - 1 && (
   186→              <div className={`h-0.5 w-4 sm:w-8 rounded-full ${currentStep > step ? 'bg-orange-500' : 'bg-muted'}`} />
   187→            )}
   188→          </React.Fragment>
   189→        );
   190→      })}
   191→    </div>
   192→  );
   193→}

   207→    companyCountry: '', companyEmail: '', companyWebsite: '',

   210→  });

   212→
   213→  const [activeTab, setActiveTab] = useState<'login' | 'register'>('login');
   214→  const [showPassword, setShowPassword] = useState(false);
   215→  const [regStep, setRegStep] = useState<RegStep>(1);
   216→
   217→  // Forgot / Reset password state
   218→  const [authMode, setAuthMode] = useState<'login' | 'register' | 'forgot-password' | 'forgot-sent' | 'reset-password'>('login');
   219→  const [forgotEmail, setForgotEmail] = useState('');
   220→  const [resetToken, setResetToken] = useState('');
   221→  const [tokenValidation, setTokenValidation] = useState<'idle' | 'checking' | 'valid' | 'invalid'>('idle');
   222→  const [newPassword, setNewPassword] = useState('');
   223→  const [confirmPassword, setConfirmPassword] = useState('');
   224→  const [regConfirmPassword, setRegConfirmPassword] = useState('');
   225→  const [resetSuccess, setResetSuccess] = useState(false);
   226→
   227→  // Handle ?token=xxx in URL for email reset links
   228→  useEffect(() => {
   229→    const params = new URLSearchParams(window.location.search);
   230→    const tokenFromUrl = params.get('token');
   231→    if (tokenFromUrl) {
   232→      setResetToken(tokenFromUrl);
   233→      setAuthMode('reset-password');
   234→      // Clean up URL without reloading
   235→      window.history.replaceState({}, '', window.location.pathname);
   236→    }
   237→  }, []);
   238→
   239→  // Debounced token validation — checks the code against the server as user types
   240→  useEffect(() => {
   241→    if (!resetToken || resetToken.length < 10) {
   242→      setTokenValidation('idle');
   243→      return;
   244→    }
   245→
   246→    const timer = setTimeout(async () => {
   247→      setTokenValidation('checking');
   248→      try {
   249→        const { api } = await import('@/lib/api');
   250→        const res = await api.get(`/auth/validate-reset-token?token=${encodeURIComponent(resetToken)}`);
   251→        setTokenValidation(res.valid ? 'valid' : 'invalid');
   252→      } catch {
   253→        setTokenValidation('idle'); // Network error — don't block
   254→      }
   255→    }, 600); // 600ms debounce
   256→
   257→    return () => clearTimeout(timer);
   258→  }, [resetToken]);
   259→
   260→  const handleLoginSubmit = async (e: React.FormEvent) => {
   261→    e.preventDefault();
   262→    if (!loginData.email || !loginData.password) return;
   263→    setLoading(true);
   264→    try {
   265→      const ok = await login(loginData.email, loginData.password);
   266→      if (!ok) {
   267→        toast.error('Invalid email or password. Please try again.');
   268→      }
   269→    } catch {
   270→      toast.error('Login failed. Please check your connection and try again.');
   271→    }
   272→    setLoading(false);
   273→  };
   274→
   275→  const handleForgotPassword = async (e: React.FormEvent) => {
   276→    e.preventDefault();
   277→    if (!forgotEmail) return;
   278→    setLoading(true);
   279→    try {
   280→      const { api } = await import('@/lib/api');
   281→      const res = await api.post('/auth/forgot-password', { email: forgotEmail });
   282→      if (res.success) {
   283→        // Always show the confirmation screen to prevent email enumeration
   284→        // The reset token is sent ONLY to the user's email - never in the API response
   285→        setResetToken(''); // Clear any previous token
   286→        setTokenValidation('idle');
   287→        setAuthMode('forgot-sent');
   288→      } else {
   289→        toast.error(res.error || 'Failed to process request');
   290→      }
   291→    } catch {
   292→      toast.error('Network error. Please try again.');
   293→    }
   294→    setLoading(false);
   295→  };
   296→
   297→  const handleResetPassword = async (e: React.FormEvent) => {
   298→    e.preventDefault();
   299→    if (!resetToken || !newPassword || !confirmPassword) return;
   300→    if (newPassword !== confirmPassword) {
   301→      toast.error('Passwords do not match');
   302→      return;
   303→    }
   304→    if (newPassword.length < 12) {
   305→      toast.error('Password must be at least 12 characters');
   306→      return;
   307→    }
   308→    if (!/[A-Z]/.test(newPassword) || !/[a-z]/.test(newPassword) || !/[0-9]/.test(newPassword) || !/[^A-Za-z0-9]/.test(newPassword)) {
   309→      toast.error('Password must include uppercase, lowercase, number, and special character');
   310→      return;
   311→    }
   312→    setLoading(true);
   313→    try {
   314→      const { api } = await import('@/lib/api');
   315→      const res = await api.post('/auth/reset-password', {
   316→        token: resetToken,
   317→        newPassword,
   318→        confirmPassword,
   319→      });
   320→      if (res.success) {
   321→        setResetSuccess(true);
   322→        toast.success('Password reset successfully! You can now sign in.');
   323→      } else {
   324→        toast.error(res.error || 'Failed to reset password');
   325→      }
   326→    } catch {
   327→      toast.error('Network error. Please try again.');
   328→    }
   329→    setLoading(false);
   330→  };
   331→
   332→  /* ─── Registration step navigation ─── */
   333→  const canGoNext = (): boolean => {
   334→    switch (regStep) {
   335→      case 1:
   336→        return !!regData.email &&
   337→          !!regData.password &&
   338→          regData.password.length >= 12 &&
   339→          /[A-Z]/.test(regData.password) &&
   340→          /[a-z]/.test(regData.password) &&
   341→          /[0-9]/.test(regData.password) &&
   342→          /[^A-Za-z0-9]/.test(regData.password) &&
   343→          !!regConfirmPassword &&
   344→          regData.password === regConfirmPassword;
   345→      case 2:
   346→        return !!regData.companyName;
   347→      case 3:
   348→        return !!regData.fullName;
   349→      case 4:
   350→        return true;
   351→      default:
   352→        return false;
   353→    }
   354→  };

   356→  const goNext = () => {
   357→    if (!canGoNext()) return;
   358→    if (regStep < 4) setRegStep((regStep + 1) as RegStep);
   359→  };
   360→
   361→  const goBack = () => {
   362→    if (regStep > 1) setRegStep((regStep - 1) as RegStep);

   383→
   384→        {/* ═══════════ LEFT PANEL (Desktop) ═══════════ */}
   385→        <div className="hidden lg:flex lg:w-[52%] relative overflow-hidden gradient-slate flex-col">
