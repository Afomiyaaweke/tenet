   103→function formatAIResultToHTML(data: Record<string, unknown>): string {
   104→  let html = '';
   105→  for (const [key, value] of Object.entries(data)) {
   106→    const title = key.replace(/([A-Z])/g, ' $1').replace(/^./, s => s.toUpperCase());
   107→    const content = typeof value === 'string' ? value : JSON.stringify(value, null, 2);
   108→    const formattedContent = content
   109→      .replace(/&/g, '&amp;')
   110→      .replace(/</g, '&lt;')
   111→      .replace(/>/g, '&gt;')
   112→      .replace(/\*\*(.+?)\*\*/g, '<strong>$1</strong>')
   113→      .replace(/^[-*•]\s+(.*)/gm, '<li>$1</li>')
   114→      .replace(/^(\d+)[.)]\s+(.*)/gm, '<li>$2</li>')
   115→      .replace(/^#{1,3}\s+(.*)/gm, '<strong>$1</strong>')
   116→      .replace(/\n\n/g, '</p><p>')
   117→      .replace(/\n/g, '<br/>');
   118→    html += `<h2 style="font-size:16px;font-weight:700;margin:16px 0 8px;color:#059669;border-bottom:1px solid #d1fae5;padding-bottom:4px;">${title}</h2>`;
   119→    html += `<p>${formattedContent}</p>`;
   120→  }
   121→  return html;
   122→}

   129→
   130→function SkillTagSelector({ selected, onChange }: { selected: string[]; onChange: (s: string[]) => void }) {
   131→  const toggle = (skill: string) => {
   132→    if (selected.includes(skill)) onChange(selected.filter(s => s !== skill));
   133→    else onChange([...selected, skill]);
   134→  };
   135→  return (
   136→    <div className="flex flex-wrap gap-1">
   137→      {COMMON_SKILLS.map(skill => (
   138→        <button key={skill} type="button" onClick={() => toggle(skill)}
   139→          className={`px-2 py-0.5 rounded-full text-[10px] font-medium transition-all border ${
   140→            selected.includes(skill)
   141→              ? 'bg-emerald-50 text-emerald-700 border-emerald-300'
   142→              : 'bg-muted/50 text-muted-foreground border-border hover:border-emerald-300'
   143→          }`}>
   144→          {skill}
   145→        </button>
   146→      ))}
   147→    </div>
   148→  );
   149→}
