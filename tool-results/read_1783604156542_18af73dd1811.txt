   127→   127→/* ───────────────────────── Password Strength Meter ───────────────────────── */
   128→   128→function scorePassword(pw: string): { score: number; label: string; color: string } {
   129→   129→  let score = 0;
   130→   130→  if (pw.length >= 8) score++;
   131→   131→  if (pw.length >= 12) score++;
   132→   132→  if (/[a-z]/.test(pw) && /[A-Z]/.test(pw)) score++;
   133→   133→  if (/[0-9]/.test(pw)) score++;
   134→   134→  if (/[^A-Za-z0-9]/.test(pw)) score++;
   135→   135→  const labels = ['Too weak', 'Weak', 'Fair', 'Good', 'Strong', 'Very strong'];
   136→   136→  const colors = ['bg-rose-500', 'bg-rose-500', 'bg-amber-500', 'bg-yellow-500', 'bg-orange-500', 'bg-orange-600'];
   137→   137→  return { score, label: labels[score], color: colors[score] };
   138→   138→}

   140→   140→/* ───────────────────────── Slide-to-Verify Captcha ───────────────────────── */
   141→   141→function SlideCaptcha({ onVerified }: { onVerified: () => void }) {
   142→   142→  const [offset, setOffset] = useState(0);
   143→   143→  const [status, setStatus] = useState<'idle' | 'dragging' | 'success' | 'fail'>('idle');
   144→   144→  const [target] = useState(() => 180 + Math.floor(Math.random() * 80));
   145→   145→  const trackWidth = 320;
   146→   146→  const handleWidth = 44;
   147→   147→  const tolerance = 8;
   148→   148→  const dragRef = React.useRef<{ start: number; startOffset: number } | null>(null);
   149→   149→
   150→   150→  const onPointerDown = (e: React.PointerEvent) => {
   151→   151→    if (status === 'success') return;
   152→   152→    setStatus('dragging');
   153→   153→    dragRef.current = { start: e.clientX, startOffset: offset };
   154→   154→    (e.target as HTMLElement).setPointerCapture(e.pointerId);
   155→   155→  };
   156→   156→
   157→   157→  const onPointerMove = (e: React.PointerEvent) => {
   158→   158→    if (!dragRef.current || status !== 'dragging') return;
   159→   159→    const delta = e.clientX - dragRef.current.start;
   160→   160→    const next = Math.max(0, Math.min(trackWidth - handleWidth, dragRef.current.startOffset + delta));
   161→   161→    setOffset(next);
   162→   162→  };
   163→   163→
   164→   164→  const onPointerUp = () => {
   165→   165→    if (status !== 'dragging') return;
   166→   166→    dragRef.current = null;
   167→   167→    if (Math.abs(offset - target) <= tolerance) {
   168→   168→      setStatus('success');
   169→   169→      setTimeout(() => onVerified(), 450);
   170→   170→    } else {
   171→   171→      setStatus('fail');
   172→   172→      setTimeout(() => {
   173→   173→        setStatus('idle');
   174→   174→        setOffset(0);
   175→   175→      }, 700);
   176→   176→    }
   177→   177→  };
   178→   178→
   179→   179→  const pieceColor = status === 'success' ? 'bg-orange-500' : status === 'fail' ? 'bg-rose-500' : 'bg-primary';
   180→   180→  const ringColor = status === 'success' ? 'ring-orange-300' : status === 'fail' ? 'ring-rose-300' : 'ring-primary/40';
   181→   181→
   182→   182→  return (
   183→   183→    <div className="space-y-3">
   184→   184→      <div className="text-xs font-medium text-muted-foreground flex items-center gap-1.5">
   185→   185→        <Puzzle className="w-3.5 h-3.5" /> Security verification — slide to complete
   186→   186→      </div>
   187→   187→      {/* Puzzle canvas */}
   188→   188→      <div className="relative h-12 rounded-xl overflow-hidden border border-border bg-muted/40 select-none">
   189→   189→        {/* pattern background */}
   190→   190→        <div
   191→   191→          className="absolute inset-0 opacity-50"
   192→   192→          style={{
   193→   193→            backgroundImage:
   194→   194→              'repeating-linear-gradient(45deg, hsl(var(--muted-foreground) / 0.08) 0 6px, transparent 6px 12px), repeating-linear-gradient(-45deg, hsl(var(--muted-foreground) / 0.06) 0 6px, transparent 6px 12px)',
   195→   195→          }}
   196→   196→        />
   197→   197→        {/* target cutout (notch) */}
   198→   198→        <div
   199→   199→          className="absolute top-1/2 -translate-y-1/2 h-9 w-9 rounded-lg border-2 border-dashed border-muted-foreground/40 bg-background/60"
   200→   200→          style={{ left: target + 4 }}
   201→   201→        />
   202→   202→        {/* moving puzzle piece */}
   203→   203→        <div
   204→   204→          className={`absolute top-1/2 -translate-y-1/2 h-9 w-9 rounded-lg ${pieceColor} shadow-lg ring-2 ${ringColor} flex items-center justify-center text-white transition-shadow ${status === 'fail' ? 'animate-[shake_0.4s_ease-in-out]' : ''}`}
   205→   205→          style={{ left: 4 + offset, transition: status === 'dragging' ? 'none' : 'left 0.25s ease' }}
   206→   206→        >
   207→   207→          {status === 'success' ? <CheckCircle2 className="w-5 h-5" /> : status === 'fail' ? <XCircle className="w-5 h-5" /> : <Puzzle className="w-4 h-4" />}
   208→   208→        </div>
   209→   209→        {/* hint text */}
   210→   210→        {status === 'idle' && (
   211→   211→          <span className="absolute inset-0 flex items-center justify-center text-xs text-muted-foreground pointer-events-none">
   212→   212→            Drag the slider below to fit the puzzle
   213→   213→          </span>
   214→   214→        )}
   215→   215→        {status === 'success' && (
   216→   216→          <span className="absolute inset-0 flex items-center justify-center text-xs font-medium text-orange-600 bg-orange-50/60 pointer-events-none">
   217→   217→            Verified
   218→   218→          </span>
   219→   219→        )}
   220→   220→      </div>
   221→   221→      {/* Slider track */}
   222→   222→      <div className="relative h-11 rounded-xl bg-muted border border-border overflow-hidden">
   223→   223→        <div
   224→   224→          className={`absolute inset-y-0 left-0 ${status === 'success' ? 'bg-orange-100 dark:bg-orange-950/40' : status === 'fail' ? 'bg-rose-100 dark:bg-rose-950/40' : 'bg-primary/10'} transition-all`}
   225→   225→          style={{ width: offset + handleWidth }}
   226→   226→        />
   227→   227→        <div
   228→   228→          className="absolute top-0 h-full flex items-center justify-center text-xs font-medium text-muted-foreground pointer-events-none"
   229→   229→          style={{ left: handleWidth, right: 0 }}
   230→   230→        >
   231→   231→          {status === 'success' ? 'Verification complete' : status === 'fail' ? 'Verification failed, retrying…' : 'Slide right to verify'}
   232→   232→        </div>
   233→   233→        <div
   234→   234→          onPointerDown={onPointerDown}
   235→   235→          onPointerMove={onPointerMove}
   236→   236→          onPointerUp={onPointerUp}
   237→   237→          onPointerCancel={onPointerUp}
   238→   238→          className={`absolute top-1/2 -translate-y-1/2 h-9 w-9 rounded-lg cursor-grab active:cursor-grabbing flex items-center justify-center text-white shadow-md ${pieceColor} ${status === 'success' ? 'cursor-default' : ''} touch-none`}
   239→   239→          style={{ left: 4 + offset, transition: status === 'dragging' ? 'none' : 'left 0.25s ease' }}
   240→   240→        >
   241→   241→          {status === 'success' ? <CheckCircle2 className="w-5 h-5" /> : <ArrowRight className="w-4 h-4" />}
   242→   242→        </div>
   243→   243→      </div>
   244→   244→    </div>
   245→   245→  );
   246→   246→}

   369→   369→      return;
   370→   370→    }
   371→   371→    if (newPassword.length < 8) {
   372→   372→      toast.error('Password must be at least 8 characters');
   373→   373→      return;
   374→   374→    }
   375→   375→    setLoading(true);
   376→   376→    try {
   377→   377→      const { api } = await import('@/lib/api');
   378→   378→      const res = await api.post('/auth/reset-password', {
   379→   379→        token: resetToken,
   380→   380→        newPassword,
   381→   381→        confirmPassword,
   382→   382→      });
   383→   383→      if (res.success) {
   384→   384→        setResetSuccess(true);
   385→   385→        toast.success('Password reset successfully! You can now sign in.');
   386→   386→      } else {
   387→   387→        toast.error(res.error || 'Failed to reset password');
   388→   388→      }
   389→   389→    } catch {
   390→   390→      toast.error('Network error. Please try again.');
   391→   391→    }
   392→   392→    setLoading(false);
   393→   393→  };
   394→   394→
   395→   395→  /* ─── Registration step navigation ─── */
   396→   396→  const canGoNext = (): boolean => {
   397→   397→    switch (regStep) {
   398→   398→      case 1:

   400→   400→      case 2:
   401→   401→        return !!regData.companyName;
   402→   402→      case 3:
   403→   403→        return !!regData.fullName;
   404→   404→      case 4:
   405→   405→        return !!regData.role;
   406→   406→      case 5:
   407→   407→        return true;
   408→   408→      default:
   409→   409→        return false;
   410→   410→    }
   411→   411→  };
   412→   412→
   413→   413→  const goNext = () => {
   414→   414→    if (!canGoNext()) return;
   415→   415→    if (regStep < 5) setRegStep((regStep + 1) as RegStep);
   416→   416→  };
   417→   417→
   418→   418→  const goBack = () => {
   419→   419→    if (regStep > 1) setRegStep((regStep - 1) as RegStep);
   420→   420→  };
   421→   421→
   422→   422→  const handleRegister = async () => {
   423→   423→    setLoading(true);
   424→   424→    const ok = await register(regData);
   425→   425→    if (!ok) toast.error('Registration failed. Email may already exist or Super Admin authorization required.');
   426→   426→    else toast.success('Welcome to Tenet!');
   427→   427→    setLoading(false);
   428→   428→  };
