   134→function formatCurrency(amount: number): string {
   135→  return new Intl.NumberFormat('en-ET', { style: 'currency', currency: 'ETB', maximumFractionDigits: 0 }).format(amount);
   136→}
   137→
   138→function formatDate(dateStr: string): string {
   139→  if (!dateStr) return '—';
   140→  return new Date(dateStr).toLocaleDateString('en-GB', { day: '2-digit', month: 'short', year: 'numeric' });
   141→}
   142→
   143→function daysUntil(dateStr: string): number {
   144→  return Math.ceil((new Date(dateStr).getTime() - Date.now()) / (1000 * 60 * 60 * 24));
   145→}

   164→export function ApplicantsView() {
   165→  const { user } = useAuthStore();
   166→  const { setView } = useNavStore();
   167→  const [rows, setRows] = useState<ApplicantRow[]>([]);
   168→  const [meta, setMeta] = useState<ApplicantMeta | null>(null);
   169→  const [loading, setLoading] = useState(true);
   170→  const [search, setSearch] = useState('');
   171→  const [statusFilter, setStatusFilter] = useState<string>('all');
   172→  const [tenderFilter, setTenderFilter] = useState<string>('all');
   173→  const [sortField, setSortField] = useState<SortField>('submittedAt');
   174→  const [sortDir, setSortDir] = useState<SortDir>('desc');
   175→  const [viewMode, setViewMode] = useState<ViewMode>('table');
   176→  const [expandedRow, setExpandedRow] = useState<string | null>(null);
   177→  const [visibleColumns, setVisibleColumns] = useState<Set<string>>(() =>
   178→    new Set(COLUMNS.filter(c => c.defaultVisible).map(c => c.key))
   179→  );
   180→  const [columnMenuOpen, setColumnMenuOpen] = useState(false);
   181→  const [page, setPage] = useState(1);
   182→  const PAGE_SIZE = 50;
   183→
   184→  // Fetch data
   185→  const fetchData = useCallback(async () => {
   186→    setLoading(true);
   187→    try {
   188→      const params: Record<string, string> = {
   189→        page: String(page),
   190→        limit: String(PAGE_SIZE),
   191→      };
   192→      if (search) params.search = search;
   193→      if (statusFilter && statusFilter !== 'all') params.status = statusFilter;
   194→      if (tenderFilter && tenderFilter !== 'all') params.tenderId = tenderFilter;
   195→
   196→      const res = await api.get('/applicants', params);
   197→      if (res.success) {
   198→        setRows(res.data);
   199→        setMeta(res.meta);
   200→      }
   201→    } catch (err) {
   202→      console.error('Fetch applicants error:', err);
   203→      toast.error('Failed to load applicants');
   204→    } finally {
   205→      setLoading(false);
   206→    }
   207→  }, [page, search, statusFilter, tenderFilter]);
   208→
   209→  useEffect(() => {
   210→    fetchData();
   211→  }, [fetchData]);
   212→
   213→  // Get unique tenders for filter dropdown
   214→  const tenderOptions = useMemo(() => {
   215→    const seen = new Map<string, string>();
   216→    rows.forEach(r => { if (!seen.has(r.tenderId)) seen.set(r.tenderId, r.tenderTitle); });
   217→    return Array.from(seen.entries());
   218→  }, [rows]);
   219→
   220→  // Sorted rows (client-side sorting on current page)
   221→  const sortedRows = useMemo(() => {
   222→    const copy = [...rows];
   223→    copy.sort((a, b) => {
   224→      const aVal = a[sortField];
   225→      const bVal = b[sortField];
   226→      if (aVal === bVal) return 0;
   227→      if (aVal == null) return 1;
   228→      if (bVal == null) return -1;
   229→      let cmp = 0;
   230→      if (typeof aVal === 'number' && typeof bVal === 'number') {
   231→        cmp = aVal - bVal;
   232→      } else if (typeof aVal === 'boolean' && typeof bVal === 'boolean') {
   233→        cmp = Number(aVal) - Number(bVal);
   234→      } else {
   235→        cmp = String(aVal).localeCompare(String(bVal));
   236→      }
   237→      return sortDir === 'asc' ? cmp : -cmp;
   238→    });
   239→    return copy;
   240→  }, [rows, sortField, sortDir]);
   241→
   242→  // Visible columns
   243→  const activeColumns = useMemo(
   244→    () => COLUMNS.filter(c => visibleColumns.has(c.key)),
   245→    [visibleColumns]
   246→  );
   247→
   248→  // Column groups
   249→  const columnGroups = useMemo(() => {
   250→    const groups: Record<string, ColumnDef[]> = {};
   251→    activeColumns.forEach(c => {
   252→      if (!groups[c.group]) groups[c.group] = [];
   253→      groups[c.group].push(c);
   254→    });
   255→    return groups;
   256→  }, [activeColumns]);
   257→
   258→  // Toggle column visibility
   259→  const toggleColumn = (key: string) => {
   260→    setVisibleColumns(prev => {
   261→      const next = new Set(prev);
   262→      if (next.has(key)) next.delete(key);
   263→      else next.add(key);
   264→      return next;
   265→    });
   266→  };
   267→
   268→  // Toggle sort
   269→  const handleSort = (field: SortField) => {
   270→    if (sortField === field) {
   271→      setSortDir(prev => prev === 'asc' ? 'desc' : 'asc');
   272→    } else {
   273→      setSortField(field);
   274→      setSortDir('asc');
   275→    }
   276→  };
   277→
   278→  // Sort icon
   279→  const SortIcon = ({ field }: { field: SortField }) => {
   280→    if (sortField !== field) return <ArrowUpDown className="h-3 w-3 text-muted-foreground/40" />;
   281→    return sortDir === 'asc'
   282→      ? <ArrowUp className="h-3 w-3 text-primary" />
   283→      : <ArrowDown className="h-3 w-3 text-primary" />;
   284→  };
   285→
   286→  // Export CSV
   287→  const exportCSV = () => {
   288→    if (!rows.length) return;
   289→    const headers = activeColumns.map(c => c.label);
   290→    const csvRows = rows.map(row =>
   291→      activeColumns.map(c => {
   292→        let val = row[c.key];
   293→        if (typeof val === 'boolean') val = val ? 'Yes' : 'No';
   294→        if (typeof val === 'number') val = val.toLocaleString();
   295→        val = String(val || '');
   296→        // Escape CSV
   297→        if (val.includes(',') || val.includes('"') || val.includes('\n')) {
   298→          val = `"${val.replace(/"/g, '""')}"`;
   299→        }
   300→        return val;
   301→      }).join(',')
   302→    );
   303→    const csv = [headers.join(','), ...csvRows].join('\n');
   304→    const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });
   305→    const url = URL.createObjectURL(blob);
   306→    const link = document.createElement('a');
   307→    link.href = url;
   308→    link.download = `applicants-export-${new Date().toISOString().slice(0, 10)}.csv`;
   309→    link.click();
   310→    URL.revokeObjectURL(url);
   311→    toast.success('Spreadsheet exported successfully');
   312→  };
   313→
   314→  // Stats
   315→  const statusCounts = meta?.statusCounts || {};
   316→  const totalApplicants = meta?.total || 0;
   317→  const totalBudget = meta?.totalBudgetSum || 0;
   318→  const uniqueTenders = meta?.uniqueTenders || 0;
   319→  const uniqueCompanies = meta?.uniqueCompanies || 0;
   320→
   321→  // ─── Render ───────────────────────────────────────────────────────
   322→
   323→  return (
   324→    <div className="flex flex-col gap-4 p-4 md:p-6">
   325→      {/* Header */}
   326→      <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-3">
   327→        <div className="flex items-center gap-3">
   328→          <div className="p-2 rounded-xl bg-orange-100 dark:bg-orange-900/30">
   329→            <FileSpreadsheet className="h-6 w-6 text-orange-600 dark:text-orange-400" />
   330→          </div>
   331→          <div>
   332→            <h1 className="text-xl font-bold">Applicants Spreadsheet</h1>
   333→            <p className="text-sm text-muted-foreground">
   334→              {totalApplicants} applicant{totalApplicants !== 1 ? 's' : ''} across {uniqueTenders} tender{uniqueTenders !== 1 ? 's' : ''}
   335→            </p>
   336→          </div>
   337→        </div>
   338→        <div className="flex items-center gap-2">
   339→          <Button variant="outline" size="sm" onClick={fetchData}>
   340→            <RefreshCw className={`h-4 w-4 mr-1 ${loading ? 'animate-spin' : ''}`} />
   341→            Refresh
   342→          </Button>
   343→          <Button variant="outline" size="sm" onClick={exportCSV} disabled={!rows.length}>
   344→            <Download className="h-4 w-4 mr-1" />
   345→            Export CSV
   346→          </Button>
   347→        </div>
   348→      </div>
   349→
   350→      {/* Stats bar */}
   351→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   352→        <Card className="bg-gradient-to-br from-orange-50 to-white dark:from-orange-950/30 dark:to-background border-orange-200/50 dark:border-orange-800/30">
   353→          <CardContent className="p-3 flex items-center gap-3">
   354→            <div className="p-2 rounded-lg bg-orange-100 dark:bg-orange-900/30">
   355→              <Users className="h-5 w-5 text-orange-600 dark:text-orange-400" />
   356→            </div>
   357→            <div>
   358→              <p className="text-xs text-muted-foreground">Total Applicants</p>
   359→              <p className="text-lg font-bold">{totalApplicants}</p>
   360→            </div>
   361→          </CardContent>
   362→        </Card>
   363→        <Card className="bg-gradient-to-br from-emerald-50 to-white dark:from-emerald-950/30 dark:to-background border-emerald-200/50 dark:border-emerald-800/30">
   364→          <CardContent className="p-3 flex items-center gap-3">
   365→            <div className="p-2 rounded-lg bg-emerald-100 dark:bg-emerald-900/30">
   366→              <DollarSign className="h-5 w-5 text-emerald-600 dark:text-emerald-400" />
   367→            </div>
   368→            <div>
   369→              <p className="text-xs text-muted-foreground">Total Bid Value</p>
   370→              <p className="text-lg font-bold">{formatCurrency(totalBudget)}</p>
   371→            </div>
   372→          </CardContent>
   373→        </Card>
   374→        <Card className="bg-gradient-to-br from-blue-50 to-white dark:from-blue-950/30 dark:to-background border-blue-200/50 dark:border-blue-800/30">
   375→          <CardContent className="p-3 flex items-center gap-3">
   376→            <div className="p-2 rounded-lg bg-blue-100 dark:bg-blue-900/30">
   377→              <FileSearch className="h-5 w-5 text-blue-600 dark:text-blue-400" />
   378→            </div>
   379→            <div>
   380→              <p className="text-xs text-muted-foreground">Tenders</p>
   381→              <p className="text-lg font-bold">{uniqueTenders}</p>
   382→            </div>
   383→          </CardContent>
   384→        </Card>
   385→        <Card className="bg-gradient-to-br from-violet-50 to-white dark:from-violet-950/30 dark:to-background border-violet-200/50 dark:border-violet-800/30">
   386→          <CardContent className="p-3 flex items-center gap-3">
   387→            <div className="p-2 rounded-lg bg-violet-100 dark:bg-violet-900/30">
   388→              <Building2 className="h-5 w-5 text-violet-600 dark:text-violet-400" />
   389→            </div>
   390→            <div>
   391→              <p className="text-xs text-muted-foreground">Companies</p>
   392→              <p className="text-lg font-bold">{uniqueCompanies}</p>
   393→            </div>
   394→          </CardContent>
   395→        </Card>
   396→      </div>
   397→
   398→      {/* Status filter pills */}
   399→      <div className="flex items-center gap-2 flex-wrap">
   400→        <span className="text-xs font-medium text-muted-foreground mr-1">Status:</span>
   401→        {[
   402→          { key: 'all', label: 'All', count: totalApplicants },
   403→          { key: 'pending_review', label: 'Pending', count: statusCounts.pending_review || 0 },
   404→          { key: 'shortlisted', label: 'Shortlisted', count: statusCounts.shortlisted || 0 },
   405→          { key: 'awarded', label: 'Awarded', count: statusCounts.awarded || 0 },
   406→          { key: 'rejected', label: 'Rejected', count: statusCounts.rejected || 0 },
   407→        ].map(s => (
   408→          <Button
   409→            key={s.key}
   410→            variant={statusFilter === s.key ? 'default' : 'outline'}
   411→            size="sm"
   412→            className={`h-7 text-xs ${statusFilter === s.key ? 'bg-primary text-primary-foreground' : ''}`}
   413→            onClick={() => { setStatusFilter(s.key); setPage(1); }}
   414→          >
   415→            {s.label}
   416→            <Badge variant="secondary" className="ml-1.5 h-4 px-1.5 text-[10px]">
   417→              {s.count}
   418→            </Badge>
   419→          </Button>
   420→        ))}
   421→      </div>
   422→
   423→      {/* Toolbar: search, view mode, column picker */}
   424→      <div className="flex flex-col sm:flex-row items-stretch sm:items-center gap-2">
   425→        <div className="relative flex-1">
   426→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   427→          <Input
   428→            placeholder="Search applicants, companies, tenders..."
   429→            className="pl-9 h-9"
   430→            value={search}
   431→            onChange={(e) => { setSearch(e.target.value); setPage(1); }}
   432→          />
   433→        </div>
   434→        <div className="flex items-center gap-2">
   435→          {/* View mode toggle */}
   436→          <div className="flex items-center border rounded-lg overflow-hidden">
   437→            <Button
   438→              variant={viewMode === 'table' ? 'secondary' : 'ghost'}
   439→              size="sm"

   446→              variant={viewMode === 'cards' ? 'secondary' : 'ghost'}
   447→              size="sm"

   449→              onClick={() => setViewMode('cards')}
   450→            >
   451→              <LayoutGrid className="h-4 w-4" />
   452→            </Button>
   453→          </div>
   454→
   455→          {/* Column picker */}
   456→          <DropdownMenu open={columnMenuOpen} onOpenChange={setColumnMenuOpen}>

   655→                variant="outline"
   656→                size="sm"
   657→                className="h-7 w-7 p-0"
   658→                disabled={!meta || page >= meta.totalPages}
   659→                onClick={() => setPage(p => p + 1)}
   660→              >
   661→                <ChevronRight className="h-4 w-4" />
   662→              </Button>
   663→            </div>
   664→          </div>
   665→        </Card>
   666→      ) : (
   667→        /* ─── Card View ─── */
   668→        <div className="grid grid-cols-1 md:grid-cols-2 xl:grid-cols-3 gap-3">
   669→          {sortedRows.map((row) => {
   670→            const bidCfg = STATUS_CONFIG[row.bidStatus] || STATUS_CONFIG.pending_review;
   671→            const BidStatusIcon = bidCfg.icon;
   672→            return (
   673→              <Card key={row.id} className={`overflow-hidden hover:shadow-md transition-shadow ${row.bidStatus === 'awarded' ? 'border-emerald-300 dark:border-emerald-700' : ''}`}>
   674→                <CardContent className="p-4">
   675→                  {/* Header: applicant + status */}
   676→                  <div className="flex items-start justify-between mb-3">
   677→                    <div className="flex items-center gap-2.5 min-w-0">
   678→                      <div className="h-9 w-9 rounded-full bg-gradient-to-br from-orange-200 to-orange-400 dark:from-orange-800 dark:to-orange-600 flex items-center justify-center text-white font-bold text-sm shrink-0">
   679→                        {row.applicantName.charAt(0).toUpperCase()}
   680→                      </div>
   681→                      <div className="min-w-0">
   682→                        <div className="flex items-center gap-1.5">
   683→                          <p className="font-semibold text-sm truncate">{row.applicantName}</p>
   684→                          {row.applicantVerified && <ShieldCheck className="h-3.5 w-3.5 text-emerald-500 shrink-0" />}

   686→                        <p className="text-xs text-muted-foreground truncate">{row.companyName}</p>

   690→                      <BidStatusIcon className="h-3 w-3 mr-1" />
   691→                      {bidCfg.label}
   692→                    </Badge>
   693→                  </div>
   694→
   695→                  {/* Tender title */}
   696→                  <p className="text-xs font-medium text-primary mb-3 line-clamp-2 cursor-pointer hover:underline"
   697→                    onClick={() => setView('tender-detail', { id: row.tenderId })}
   698→                  >
   699→                    {row.tenderTitle}
   700→                  </p>
   701→
   702→                  {/* Detail grid */}
   703→                  <div className="grid grid-cols-2 gap-x-4 gap-y-1.5 text-xs mb-3">
   704→                    <div>
   705→                      <span className="text-muted-foreground">Bid Amount</span>
   706→                      <p className="font-semibold">{formatCurrency(row.financialProposal)}</p>
   707→                    </div>
   708→                    <div>
   709→                      <span className="text-muted-foreground">Timeline</span>
   710→                      <p className="font-semibold">{row.timeline}</p>
   711→                    </div>
   712→                    <div>
   713→                      <span className="text-muted-foreground">Submitted</span>
   714→                      <p>{formatDate(row.submittedAt)}</p>
   715→                    </div>
   716→                    <div>
   717→                      <span className="text-muted-foreground">Location</span>
   718→                      <p className="truncate">{row.applicantLocation || row.tenderLocation}</p>
   719→                    </div>
   720→                  </div>
   721→
   722→                  {/* Skills */}
   723→                  {row.applicantSkills && (

   725→                      {row.applicantSkills.split(',').slice(0, 4).map((skill, i) => (

   732→                          +{row.applicantSkills.split(',').length - 4}

   734→                      )}
   735→                    </div>
   736→                  )}
   737→
   738→                  {/* Footer */}
   739→                  <div className="flex items-center justify-between pt-2 border-t">
   740→                    <div className="flex items-center gap-1.5">
   741→                      {row.companyVerified && (
   742→                        <Badge className="text-[9px] px-1.5 py-0 bg-emerald-100 text-emerald-700 border-0">
   743→                          <ShieldCheck className="h-2.5 w-2.5 mr-0.5" />Verified Co.
   744→                        </Badge>
   745→                      )}
   746→                    </div>
   747→                    <Button
   748→                      variant="ghost"
   749→                      size="sm"
   750→                      className="h-7 text-xs"
   751→                      onClick={() => setView('tender-detail', { id: row.tenderId })}

   753→                      View Tender <ExternalLink className="h-3 w-3 ml-1" />

   755→                  </div>
   756→                </CardContent>
   757→              </Card>
   758→            );
   759→          })}
   760→        </div>
   761→      )}
   762→    </div>
   763→  );

   767→
   768→function CellRenderer({ row, col, setView }: { row: ApplicantRow; col: ColumnDef; setView: (v: string, p?: Record<string, string>) => void }) {
   769→  const val = row[col.key];
   770→
   771→  switch (col.key) {
   772→    case 'applicantName':
   773→      return (
   774→        <div className="flex items-center gap-2 min-w-0">
   775→          <div className="h-7 w-7 rounded-full bg-gradient-to-br from-orange-200 to-orange-400 dark:from-orange-800 dark:to-orange-600 flex items-center justify-center text-white font-bold text-[10px] shrink-0">
   776→            {String(val).charAt(0).toUpperCase()}
   777→          </div>
   778→          <div className="min-w-0">
   779→            <div className="flex items-center gap-1">
   780→              <span className="font-medium truncate">{String(val)}</span>
   781→              {row.applicantVerified && <ShieldCheck className="h-3 w-3 text-emerald-500 shrink-0" />}
   782→            </div>
   783→            <span className="text-[10px] text-muted-foreground truncate block">{row.applicantJobTitle}</span>
   784→          </div>
   785→        </div>
   786→      );
   787→
   788→    case 'applicantEmail':
   789→      return (
   790→        <div className="flex items-center gap-1.5 min-w-0">
   791→          <Mail className="h-3 w-3 text-muted-foreground shrink-0" />
   792→          <span className="truncate text-muted-foreground">{String(val)}</span>
   793→        </div>
   794→      );
   795→
   796→    case 'applicantPhone':
   797→      return (
   798→        <div className="flex items-center gap-1.5 min-w-0">
   799→          <Phone className="h-3 w-3 text-muted-foreground shrink-0" />
   800→          <span className="truncate">{String(val) || '—'}</span>
   801→        </div>
   802→      );
   803→
   804→    case 'applicantVerified':
   805→    case 'companyVerified':
   806→      return val
   807→        ? <Badge className="text-[9px] px-1.5 py-0 bg-emerald-100 text-emerald-700 border-0 rounded-md"><ShieldCheck className="h-3 w-3 mr-0.5" />Yes</Badge>
   808→        : <Badge className="text-[9px] px-1.5 py-0 bg-gray-100 text-gray-500 border-0 rounded-md">No</Badge>;
   809→
   810→    case 'companyName':
   811→      return (
   812→        <div className="flex items-center gap-1.5 min-w-0">
   813→          <Building2 className="h-3 w-3 text-violet-500 shrink-0" />
   814→          <span className="font-medium truncate">{String(val) || 'Individual'}</span>
   815→        </div>
   816→      );
   817→
   818→    case 'companyIndustry':
   819→      return (
   820→        <Badge variant="secondary" className="text-[10px] px-1.5 py-0">
   821→          <Briefcase className="h-2.5 w-2.5 mr-0.5" />
   822→          {String(val) || '—'}
   823→        </Badge>
   824→      );
   825→
   826→    case 'tenderTitle':
   827→      return (
   828→        <span
   829→          className="text-primary font-medium truncate cursor-pointer hover:underline block"
   830→          onClick={() => setView('tender-detail', { id: row.tenderId })}
   831→        >
   832→          {String(val)}
   833→        </span>
   834→      );
   835→
   836→    case 'tenderStatus': {
   837→      const cfg = TENDER_STATUS_CONFIG[String(val)] || TENDER_STATUS_CONFIG.open;
   838→      return <Badge className={`text-[9px] px-1.5 py-0 border-0 rounded-md ${cfg.className}`}>{cfg.label}</Badge>;
   839→    }
   840→
   841→    case 'tenderBudgetMin':
   842→    case 'tenderBudgetMax':
   843→      return <span className="font-mono text-xs">{typeof val === 'number' ? formatCurrency(val) : '—'}</span>;

   855→        </div>
   856→      );
   857→    }
   858→
   859→    case 'applicantLocation':
   860→    case 'tenderLocation':
   861→      return (
   862→        <div className="flex items-center gap-1 min-w-0">
   863→          <MapPin className="h-3 w-3 text-muted-foreground shrink-0" />

   865→        </div>
   866→      );

   873→              {tag.trim()}
   874→            </Badge>
   875→          ))}

   882→          {typeof val === 'number' ? formatCurrency(val) : '—'}

   887→      return <Badge variant="outline" className="text-[10px] px-1.5 py-0"><Clock className="h-2.5 w-2.5 mr-0.5" />{String(val)}</Badge>;
   888→
   889→    case 'bidStatus': {
   890→      const cfg = STATUS_CONFIG[String(val)] || STATUS_CONFIG.pending_review;
   891→      const Icon = cfg.icon;
   892→      return (
   893→        <Badge className={`text-[9px] px-2 py-0.5 border-0 rounded-lg ${cfg.className}`}>
   894→          <Icon className="h-3 w-3 mr-1" />
   895→          {cfg.label}
   896→        </Badge>
   897→      );
   898→    }
   899→
   900→    case 'submittedAt':
   901→      return <span className="text-muted-foreground">{formatDate(String(val))}</span>;
   902→
   903→    case 'technicalProposal':
   904→      return (
   905→        <span className="truncate text-muted-foreground block" style={{ maxWidth: '200px' }}>
   906→          {String(val) ? String(val).substring(0, 80) + (String(val).length > 80 ? '...' : '') : '—'}
   907→        </span>
   908→      );
   909→
   910→    case 'applicantSkills':
   911→      return (
   912→        <div className="flex flex-wrap gap-0.5">
   913→          {String(val).split(',').filter(Boolean).slice(0, 3).map((skill, i) => (
   914→            <Badge key={i} variant="secondary" className="text-[9px] px-1 py-0">
   915→              {skill.trim()}
   916→            </Badge>
   917→          ))}
   918→          {String(val).split(',').filter(Boolean).length > 3 && (
   919→            <Badge variant="secondary" className="text-[9px] px-1 py-0">+{String(val).split(',').filter(Boolean).length - 3}</Badge>
   920→          )}
   921→        </div>
   922→      );
   923→
   924→    case 'rejectionNote':
   925→      return val
   926→        ? <span className="text-red-600 dark:text-red-400 truncate block" style={{ maxWidth: '180px' }}>{String(val)}</span>
   927→        : <span className="text-muted-foreground">—</span>;
   928→
   929→    case 'applicantLicense':
   930→    case 'applicantTin':
   931→    case 'companyRegistration':
   932→    case 'companyTin':
   933→      return <span className="font-mono text-xs text-muted-foreground">{String(val) || '—'}</span>;
   934→
