     1→'use client';
     2→
     3→import { useEffect, useState, useCallback, useMemo } from 'react';
     4→import { useAuthStore, useNavStore } from '@/store';
     5→import { api, Bid } from '@/lib/api';
     6→import { Card, CardContent } from '@/components/ui/card';
     7→import { Badge } from '@/components/ui/badge';
     8→import { Button } from '@/components/ui/button';
     9→import { toast } from 'sonner';
    10→import {
    11→  Gavel, Clock, DollarSign, FileSearch, Award, AlertCircle,
    12→  CheckCircle, ChevronDown, ChevronUp, ArrowRight, TrendingUp,
    13→  Briefcase, X, Eye, RotateCcw, Filter, Target,
    14→  CircleDot, Building2, FileSignature, Stamp,
    15→} from 'lucide-react';
    16→import { useStampSignature, StampSignatureSelector, type SavedSignature } from '@/components/stamp-signature';
    17→
    18→type BidTab = 'all' | 'pending_review' | 'shortlisted' | 'awarded' | 'rejected';
    19→
    20→export function BidsView() {
    21→  const { user } = useAuthStore();
    22→  const { setView } = useNavStore();
    23→  const [bids, setBids] = useState<Bid[]>([]);
    24→  const [loading, setLoading] = useState(true);
    25→  const [statusFilter, setStatusFilter] = useState('');
    26→  const [expandedId, setExpandedId] = useState<string | null>(null);
    27→  const [activeTab, setActiveTab] = useState<BidTab>('all');
    28→  const [stampSelectorOpen, setStampSelectorOpen] = useState(false);
    29→  const [selectedBidId, setSelectedBidId] = useState<string | null>(null);
    30→  const [bidSignatures, setBidSignatures] = useState<Record<string, SavedSignature>>({});
    31→  const stampSigHook = useStampSignature();
    32→
    33→  const loadBids = useCallback(async () => {
    34→    setLoading(true);
    35→    const params: Record<string, string> = {};
    36→    if (statusFilter && statusFilter !== 'all') params.status = statusFilter;
    37→    const res = await api.get('/bids', params);
    38→    if (res.success) setBids(res.data);
    39→    setLoading(false);
    40→  }, [statusFilter]);
    41→
    42→  // eslint-disable-next-line react-hooks/set-state-in-effect
    43→  useEffect(() => { loadBids(); }, [loadBids]);
    44→
    45→  const handleStatusUpdate = async (bidId: string, status: string) => {
    46→    const res = await api.patch(`/bids/${bidId}/status`, { status });
    47→    if (res.success) {
    48→      toast.success(`Bid ${status}`);
    49→      loadBids();
    50→    } else {
    51→      toast.error(res.error || 'Failed to update bid');
    52→    }
    53→  };
    54→
    55→  const stats = useMemo(() => ({
    56→    pending: bids.filter(b => b.status === 'pending_review').length,
    57→    shortlisted: bids.filter(b => b.status === 'shortlisted').length,
    58→    awarded: bids.filter(b => b.status === 'awarded').length,
    59→    rejected: bids.filter(b => b.status === 'rejected').length,
    60→    total: bids.length,
    61→  }), [bids]);
    62→
    63→  const filteredBids = useMemo(() => {
    64→    if (activeTab === 'all') return bids;
    65→    return bids.filter(b => b.status === activeTab);
    66→  }, [bids, activeTab]);
    67→
    68→  const statusBadge = (status: string) => {
    69→    switch (status) {
    70→      case 'pending_review': return 'bg-amber-100 text-amber-700 hover:bg-amber-100';
    71→      case 'shortlisted': return 'bg-teal-100 text-teal-700 hover:bg-teal-100';
    72→      case 'awarded': return 'bg-emerald-100 text-emerald-700 hover:bg-emerald-100';
    73→      case 'rejected': return 'bg-rose-100 text-rose-700 hover:bg-rose-100';
    74→      default: return 'bg-muted text-muted-foreground hover:bg-muted';
    75→    }
    76→  };
    77→
    78→  const statusIcon = (status: string) => {
    79→    switch (status) {
    80→      case 'pending_review': return { icon: Clock, bg: 'bg-amber-50', color: 'text-amber-600' };
    81→      case 'shortlisted': return { icon: Award, bg: 'bg-teal-50', color: 'text-teal-600' };
    82→      case 'awarded': return { icon: CheckCircle, bg: 'bg-emerald-50', color: 'text-emerald-600' };
    83→      case 'rejected': return { icon: AlertCircle, bg: 'bg-rose-50', color: 'text-rose-600' };
    84→      default: return { icon: Gavel, bg: 'bg-muted/50', color: 'text-muted-foreground' };
    85→    }
    86→  };
    87→
    88→  const isAdminOrOwner = user?.role === 'super_admin' || user?.role === 'team_admin';
    89→
    90→  const tabs: { key: BidTab; label: string; icon: typeof Clock; count: number; color: string }[] = [
    91→    { key: 'all', label: 'All', icon: Gavel, count: stats.total, color: 'emerald' },
    92→    { key: 'pending_review', label: 'Pending', icon: Clock, count: stats.pending, color: 'amber' },
    93→    { key: 'shortlisted', label: 'Shortlisted', icon: Award, count: stats.shortlisted, color: 'teal' },
    94→    { key: 'awarded', label: 'Awarded', icon: CheckCircle, count: stats.awarded, color: 'emerald' },
    95→    { key: 'rejected', label: 'Rejected', icon: AlertCircle, count: stats.rejected, color: 'rose' },
    96→  ];
    97→
    98→  return (
    99→    <div className="p-4 md:p-6 space-y-6 max-w-5xl mx-auto view-enter">
   100→      {/* Header */}
   101→      <div
   102→ className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4 animate-[fadeIn_0.3s_ease-out]"
   103→ >
   104→        <div className="flex items-center gap-4">
   105→          <div className="p-3 rounded-2xl gradient-amber shadow-md flex-shrink-0">
   106→            <Gavel className="h-6 w-6 text-white" />
   107→          </div>
   108→          <div>
   109→            <h2 className="text-2xl md:text-3xl font-bold tracking-tight">
   110→              <span className="text-gradient-emerald">{isAdminOrOwner ? 'Review' : 'My'}</span> Bids
   111→            </h2>
   112→            <p className="text-muted-foreground text-sm mt-0.5">
   113→              {isAdminOrOwner ? 'Manage and evaluate submitted bids' : 'Track your bid submissions'}
   114→            </p>
   115→          </div>
   116→        </div>
   117→      </div>
   118→
   119→      {/* Stats Summary */}
   120→      {!loading && bids.length > 0 && (
   121→        <div
   122→ className="grid grid-cols-2 sm:grid-cols-4 gap-4 animate-[fadeIn_0.3s_ease-out]"
   123→ >
   124→          {[
   125→            { label: 'Pending', count: stats.pending, icon: Clock, bg: 'bg-amber-50', color: 'text-amber-600' },
   126→            { label: 'Shortlisted', count: stats.shortlisted, icon: Award, bg: 'bg-teal-50', color: 'text-teal-600' },
   127→            { label: 'Awarded', count: stats.awarded, icon: CheckCircle, bg: 'bg-emerald-50', color: 'text-emerald-600' },
   128→            { label: 'Rejected', count: stats.rejected, icon: AlertCircle, bg: 'bg-rose-50', color: 'text-rose-600' },
   129→          ].map(stat => (
   130→            <div key={stat.label}>
   131→              <Card className="premium-shadow rounded-xl border-0 bg-card hover:-translate-y-0.5 transition-all duration-200">
   132→                <CardContent className="p-4 flex items-center gap-3">
   133→                  <div className={`p-2 rounded-lg ${stat.bg} flex-shrink-0`}>
   134→                    <stat.icon className={`h-4 w-4 ${stat.color}`} />
   135→                  </div>
   136→                  <div>
   137→                    <p className="text-xl font-bold">{stat.count}</p>
   138→                    <p className="text-xs text-muted-foreground">{stat.label}</p>
   139→                  </div>
   140→                </CardContent>
   141→              </Card>
   142→            </div>
   143→          ))}
   144→        </div>
   145→      )}
   146→
   147→      {/* Tab Navigation */}
   148→      {!loading && bids.length > 0 && (
   149→        <div className="animate-[fadeIn_0.3s_ease-out]"
   150→ >
   151→          <Card className="premium-shadow rounded-xl border-0 bg-card">
   152→            <CardContent className="p-1.5">
   153→              <div className="flex gap-1 overflow-x-auto">
   154→                {tabs.map(tab => (
   155→                  <button
   156→                    key={tab.key}
   157→                    onClick={() => setActiveTab(tab.key)}
   158→                    className={`flex items-center gap-1.5 px-3 py-2 rounded-lg text-sm font-medium transition-all duration-200 whitespace-nowrap ${
   159→                      activeTab === tab.key
   160→                        ? 'gradient-emerald text-white premium-shadow'
   161→                        : 'text-muted-foreground hover:bg-muted/60 hover:text-foreground'
   162→                    }`}
   163→                  >
   164→                    <tab.icon className="h-3.5 w-3.5" />
   165→                    <span>{tab.label}</span>
   166→                    <span className={`text-[10px] px-1.5 py-0 rounded-full ${
   167→                      activeTab === tab.key
   168→                        ? 'bg-white/20 text-white'
   169→                        : 'bg-muted text-muted-foreground'
   170→                    }`}>
   171→                      {tab.count}
   172→                    </span>
   173→                  </button>
   174→                ))}
   175→              </div>
   176→            </CardContent>
   177→          </Card>
   178→        </div>
   179→      )}
   180→
   181→      {/* Bids List */}
   182→      {loading ? (
   183→        <div className="space-y-4">
   184→          {[1, 2, 3].map(i => (
   185→            <Card key={i} className="premium-shadow rounded-xl border-0 bg-card animate-pulse overflow-hidden">
   186→              <div className="h-1 bg-muted/30" />
   187→              <CardContent className="p-5"><div className="h-20 bg-muted/50 rounded-xl" /></CardContent>
   188→            </Card>
   189→          ))}
   190→        </div>
   191→      ) : bids.length === 0 ? (
   192→        <div className="animate-[fadeIn_0.3s_ease-out]">
   193→          <Card className="premium-shadow rounded-xl border-0 bg-card">
   194→            <CardContent className="p-16 text-center">
   195→              <div className="relative w-20 h-20 mx-auto mb-6">
   196→                <div className="absolute inset-0 rounded-2xl gradient-amber opacity-20" />
   197→                <div className="absolute inset-2 rounded-xl gradient-amber flex items-center justify-center">
   198→                  <Gavel className="h-8 w-8 text-white" />
   199→                </div>
   200→              </div>
   201→              <h3 className="text-lg font-semibold">No bids found</h3>
   202→              <p className="text-muted-foreground text-sm mt-2 max-w-sm mx-auto">
   203→                {user?.role === 'user'
   204→                  ? 'Start by browsing open tenders and submitting your proposals'
   205→                  : 'No bids match your current filters'}
   206→              </p>
   207→              {user?.role === 'user' && (
   208→                <Button
   209→                  className="mt-4 gradient-emerald hover:opacity-90 text-white rounded-xl premium-shadow transition-all hover:-translate-y-0.5"
   210→                  onClick={() => setView('tenders')}>
   211→                  <FileSearch className="h-4 w-4 mr-2" /> Browse Tenders
   212→                </Button>
   213→              )}
   214→            </CardContent>
   215→          </Card>
   216→        </div>
   217→      ) : filteredBids.length === 0 ? (
   218→        <div className="animate-[fadeIn_0.3s_ease-out]">
   219→          <Card className="premium-shadow rounded-xl border-0 bg-card">
   220→            <CardContent className="p-12 text-center">
   221→              <div className="p-3 rounded-2xl bg-muted/50 w-fit mx-auto mb-4">
   222→                <Filter className="h-8 w-8 text-muted-foreground" />
   223→              </div>
   224→              <h3 className="text-lg font-semibold">No {activeTab === 'all' ? '' : activeTab.replace('_', ' ')} bids</h3>
   225→              <p className="text-muted-foreground text-sm mt-1">Try selecting a different tab</p>
   226→            </CardContent>
   227→          </Card>
   228→        </div>
   229→      ) : (
   230→        <div
   231→ className="space-y-4 animate-[fadeIn_0.3s_ease-out]"
   232→ >
   233→          {filteredBids.map(bid => {
   234→              const sInfo = statusIcon(bid.status);
   235→              const SIcon = sInfo.icon;
   236→              const isExpanded = expandedId === bid.id;
   237→              const companyName = bid.user?.company?.name;
   238→              const jobTitle = bid.user?.profile?.jobTitle;
   239→
   240→              return (
   241→                <div className="hover:-translate-y-[2px] transition-all duration-200"
   242→ key={bid.id}
   243→ >
   244→                  <Card className="premium-shadow rounded-xl border-0 bg-card overflow-hidden">
   245→                    {/* Status accent strip */}
   246→                    <div className={`h-1 ${
   247→                      bid.status === 'pending_review' ? 'bg-gradient-to-r from-amber-400 to-amber-500' :
   248→                      bid.status === 'shortlisted' ? 'bg-gradient-to-r from-teal-400 to-teal-600' :
   249→                      bid.status === 'awarded' ? 'bg-gradient-to-r from-emerald-400 to-emerald-600' :
   250→                      bid.status === 'rejected' ? 'bg-gradient-to-r from-rose-400 to-rose-500' :
   251→                      'bg-gradient-to-r from-muted to-muted-foreground/50'
   252→                    }`} />
   253→
   254→                    <CardContent className="p-0">
   255→                      {/* Bid Header Row */}
   256→                      <div className="p-4 cursor-pointer hover:bg-muted/20 transition-colors"
   257→                        onClick={() => setExpandedId(isExpanded ? null : bid.id)}>
   258→                        <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
   259→                          <div className="flex items-center gap-3 min-w-0">
   260→                            {/* Status Icon */}
   261→                            <div className={`p-2.5 rounded-xl flex-shrink-0 ${sInfo.bg}`}>
   262→                              <SIcon className={`h-5 w-5 ${sInfo.color}`} />
   263→                            </div>
   264→                            <div className="min-w-0">
   265→                              <p className="font-semibold text-sm truncate hover:text-emerald-700 transition-colors">
   266→                                {bid.tender?.title || 'Tender'}
   267→                              </p>
   268→                              <p className="text-xs text-muted-foreground mt-0.5">
   269→                                {isAdminOrOwner
   270→                                  ? (
   271→                                    <span className="flex items-center gap-1 flex-wrap">
   272→                                      <span>{bid.user?.profile?.fullName || bid.user?.email || 'Contractor'}</span>
   273→                                      {jobTitle && <span className="text-muted-foreground/60">&middot; {jobTitle}</span>}
   274→                                      {companyName && (
   275→                                        <span className="inline-flex items-center gap-0.5">
   276→                                          <span className="text-muted-foreground/60">&middot;</span>
   277→                                          <Building2 className="h-3 w-3" /> {companyName}
   278→                                        </span>
   279→                                      )}
   280→                                      <span className="text-muted-foreground/60">&middot; {new Date(bid.createdAt).toLocaleDateString()}</span>
   281→                                    </span>
   282→                                  )
   283→                                  : (
   284→                                    <span className="flex items-center gap-1.5 flex-wrap">
   285→                                      <span>{bid.user?.profile?.fullName || 'Contractor'}</span>
   286→                                      {companyName && (
   287→                                        <span className="inline-flex items-center gap-0.5">
   288→                                          <span className="text-muted-foreground/60">&middot;</span>
   289→                                          <Building2 className="h-3 w-3" /> {companyName}
   290→                                        </span>
   291→                                      )}
   292→                                      <span className="text-muted-foreground/60">&middot; {new Date(bid.createdAt).toLocaleDateString()}</span>
   293→                                    </span>
   294→                                  )
   295→                                }
   296→                              </p>
   297→                            </div>
   298→                          </div>
   299→                          <div className="flex items-center gap-2 flex-shrink-0 flex-wrap">
   300→                            {/* Financial badge */}
   301→                            <Badge className="text-xs border-0 rounded-lg bg-emerald-50 text-emerald-700 hover:bg-primary/10">
   302→                              <DollarSign className="h-3 w-3 mr-1" /> ETB {bid.financialProposal.toLocaleString()}
   303→                            </Badge>
   304→                            {/* Timeline badge */}
   305→                            <Badge className="text-xs border-0 rounded-lg bg-teal-50 text-teal-700 hover:bg-teal-50">
   306→                              <Clock className="h-3 w-3 mr-1" /> {bid.timeline}
   307→                            </Badge>
   308→                            {/* Status badge */}
   309→                            <div className="flex items-center gap-1">
   310→                              <CircleDot className={`h-2 w-2 ${
   311→                                bid.status === 'awarded' ? 'text-emerald-500' :
   312→                                bid.status === 'rejected' ? 'text-rose-500' :
   313→                                bid.status === 'shortlisted' ? 'text-teal-500' :
   314→                                'text-amber-500'
   315→                              }`} />
   316→                              <Badge className={`text-[10px] px-2 py-0.5 border-0 rounded-lg ${statusBadge(bid.status)}`}>
   317→                                {bid.status.replace('_', ' ')}
   318→                              </Badge>
   319→                            </div>
   320→                            <div className="text-muted-foreground ml-1">
   321→                              {isExpanded ? <ChevronUp className="h-4 w-4" /> : <ChevronDown className="h-4 w-4" />}
   322→                            </div>
   323→                          </div>
   324→                        </div>
   325→                      </div>
   326→
   327→                      {/* Expanded Content */}
   328→                      {isExpanded && (
   329→                          <div
   330→ className="overflow-hidden animate-[fadeIn_0.3s_ease-out]"
   331→ >
   332→                            <div className="px-5 pb-5 pt-3 border-t border-border/40 space-y-4">
   333→                              {/* Technical Proposal */}
   334→                              <div>
   335→                                <div className="flex items-center gap-2 mb-2">
   336→                                  <div className="p-1.5 rounded-lg gradient-emerald">
   337→                                    <Briefcase className="h-3.5 w-3.5 text-white" />
   338→                                  </div>
   339→                                  <p className="text-xs font-semibold text-foreground uppercase tracking-wide">Technical Proposal</p>
   340→                                </div>
   341→                                <p className="text-sm whitespace-pre-wrap bg-muted/50 rounded-xl p-4 leading-relaxed">{bid.technicalProposal}</p>
   342→                              </div>
   343→
   344→                              {/* Rejection Note */}
   345→                              {bid.rejectionNote && (
   346→                                <div>
   347→                                  <div className="flex items-center gap-2 mb-2">
   348→                                    <div className="p-1.5 rounded-lg bg-rose-50">
   349→                                      <AlertCircle className="h-3.5 w-3.5 text-rose-600" />
   350→                                    </div>
   351→                                    <p className="text-xs font-semibold text-rose-700 uppercase tracking-wide">Rejection Note</p>
   352→                                  </div>
   353→                                  <p className="text-sm text-rose-600 bg-rose-50 rounded-xl p-4">{bid.rejectionNote}</p>
   354→                                </div>
   355→                              )}
   356→
   357→                              {/* Quick Actions */}
   358→                              <div className="flex flex-wrap gap-2 pt-2 border-t border-border/30">
   359→                                {/* Admin/Tender Owner actions */}
   360→                                {isAdminOrOwner && bid.status === 'pending_review' && (
   361→                                  <>
   362→                                    <Button
   363→                                      size="sm"
   364→                                      className="gradient-teal text-white rounded-xl hover:opacity-90 premium-shadow transition-all hover:-translate-y-0.5"
   365→                                      onClick={(e) => { e.stopPropagation(); handleStatusUpdate(bid.id, 'shortlisted'); }}>
   366→                                      <Award className="h-3.5 w-3.5 mr-1.5" /> Shortlist
   367→                                    </Button>
   368→                                    <Button
   369→                                      size="sm"
   370→                                      variant="outline"
   371→                                      className="rounded-xl border-rose-200 text-rose-600 hover:bg-rose-50 hover:text-rose-700 transition-all"
   372→                                      onClick={(e) => { e.stopPropagation(); handleStatusUpdate(bid.id, 'rejected'); }}>
   373→                                      <X className="h-3.5 w-3.5 mr-1.5" /> Reject
   374→                                    </Button>
   375→                                  </>
   376→                                )}
   377→                                {isAdminOrOwner && bid.status === 'shortlisted' && (
   378→                                  <>
   379→                                    <Button
   380→                                      size="sm"
   381→                                      className="gradient-emerald text-white rounded-xl hover:opacity-90 premium-shadow transition-all hover:-translate-y-0.5"
   382→                                      onClick={(e) => { e.stopPropagation(); handleStatusUpdate(bid.id, 'awarded'); }}>
   383→                                      <CheckCircle className="h-3.5 w-3.5 mr-1.5" /> Award Bid
   384→                                    </Button>
   385→                                    <Button
   386→                                      size="sm"
   387→                                      variant="outline"
   388→                                      className="rounded-xl border-rose-200 text-rose-600 hover:bg-rose-50 hover:text-rose-700 transition-all"
   389→                                      onClick={(e) => { e.stopPropagation(); handleStatusUpdate(bid.id, 'rejected'); }}>
   390→                                      <X className="h-3.5 w-3.5 mr-1.5" /> Reject
   391→                                    </Button>
   392→                                  </>
   393→                                )}
   394→
   395→                                {/* Sign Bid - available to all */}
   396→                                <Button
   397→                                  size="sm"
   398→                                  variant="outline"
   399→                                  className="rounded-xl border-orange-200 text-orange-600 hover:bg-orange-50 hover:text-orange-700 transition-all"
   400→                                  onClick={(e) => { e.stopPropagation(); setSelectedBidId(bid.id); setStampSelectorOpen(true); }}>
   401→                                  <FileSignature className="h-3.5 w-3.5 mr-1.5" /> {bidSignatures[bid.id] ? 'Signed' : 'Sign Bid'}
   402→                                </Button>
   403→
   404→                                {/* View Tender link */}
   405→                                {bid.tender && (
   406→                                  <Button
   407→                                    size="sm"
   408→                                    variant="ghost"
   409→                                    className="rounded-xl text-emerald-600 hover:bg-primary/10 hover:text-emerald-700 transition-all"
   410→                                    onClick={(e) => { e.stopPropagation(); setView('tender-detail', { id: bid.tender!.id }); }}>
   411→                                    <Eye className="h-3.5 w-3.5 mr-1.5" /> View Tender
   412→                                  </Button>
   413→                                )}
   414→
   415→                                {/* Contractor: Withdraw action for pending bids */}
   416→                                {user?.role === 'user' && bid.status === 'pending_review' && (
   417→                                  <Button
   418→                                    size="sm"
   419→                                    variant="outline"
   420→                                    className="rounded-xl text-muted-foreground hover:text-rose-600 hover:border-rose-200 hover:bg-rose-50 transition-all"
   421→                                    onClick={(e) => { e.stopPropagation(); handleStatusUpdate(bid.id, 'rejected'); }}>
   422→                                    <RotateCcw className="h-3.5 w-3.5 mr-1.5" /> Withdraw
   423→                                  </Button>
   424→                                )}
   425→
   426→                              {/* Applied Signature Preview */}
   427→                              {bidSignatures[bid.id] && (
   428→                                <div className="flex items-center gap-2 mt-2 p-2 bg-orange-50/50 border border-orange-100 rounded-xl">
   429→                                  <div className="w-10 h-8 bg-white rounded overflow-hidden p-0.5 flex-shrink-0">
   430→                                    <img src={bidSignatures[bid.id].dataUrl} alt="signature" className="max-w-full max-h-full object-contain" />
