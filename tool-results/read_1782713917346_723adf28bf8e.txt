   107→   107→function formatETB(amount: number): string {
   108→   108→  if (amount >= 1_000_000) return `ETB ${(amount / 1_000_000).toFixed(1)}M`;
   109→   109→  if (amount >= 1_000) return `ETB ${(amount / 1_000).toFixed(0)}K`;
   110→   110→  return `ETB ${amount.toLocaleString()}`;

   113→   113→function timeAgo(dateStr: string): string {
   114→   114→  const diff = Date.now() - new Date(dateStr).getTime();
   115→   115→  const mins = Math.floor(diff / 60000);
   116→   116→  if (mins < 1) return 'just now';
   117→   117→  if (mins < 60) return `${mins}m ago`;
   118→   118→  const hrs = Math.floor(mins / 60);
   119→   119→  if (hrs < 24) return `${hrs}h ago`;
   120→   120→  const days = Math.floor(hrs / 24);
   121→   121→  return `${days}d ago`;
   122→   122→}
   123→   123→
   124→   124→function daysUntil(dateStr: string): number {

   128→   128→// ─── Sparkline SVG Component ────────────────────────────────────────
   129→   129→function SparklineBars({ data, color = '#10b981', height = 32 }: { data: number[]; color?: string; height?: number }) {
   130→   130→  if (!data.length) return null;
   131→   131→  const max = Math.max(...data, 1);
   132→   132→  const barW = 4;
   133→   133→  const gap = 3;
   134→   134→  const totalW = data.length * (barW + gap) - gap;
   135→   135→
   136→   136→  return (
   137→   137→    <svg width={totalW} height={height} className="flex-shrink-0">
   138→   138→      {data.map((v, i) => {
   139→   139→        const barH = Math.max((v / max) * (height - 4), 3);
   140→   140→        return (
   141→   141→          <rect
   142→   142→            key={i}
   143→   143→            x={i * (barW + gap)}
   144→   144→            y={height - barH}
   145→   145→            width={barW}
   146→   146→            height={barH}
   147→   147→            rx={2}
   148→   148→            fill={color}
   149→   149→            opacity={0.35 + (i / data.length) * 0.45}
   150→   150→          />
   151→   151→        );
   152→   152→      })}
   153→   153→    </svg>
   154→   154→  );
   155→   155→}
