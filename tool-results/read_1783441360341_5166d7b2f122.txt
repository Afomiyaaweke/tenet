    46→function formatETB(amount: number): string {
    47→  if (amount >= 1_000_000) return `ETB ${(amount / 1_000_000).toFixed(1)}M`;
    48→  if (amount >= 1_000) return `ETB ${(amount / 1_000).toFixed(0)}K`;
    49→  return `ETB ${amount.toLocaleString()}`;
    50→}

    62→function statusColorMap(status: string) {
    63→  switch (status) {
    64→    case 'active': return { bg: 'bg-emerald-100 dark:bg-emerald-900/30', text: 'text-emerald-700 dark:text-emerald-300', dot: 'bg-emerald-500', bar: '#10b981', barBg: 'bg-emerald-100 dark:bg-emerald-900/30' };
    65→    case 'completed': return { bg: 'bg-teal-100 dark:bg-teal-900/30', text: 'text-teal-700 dark:text-teal-300', dot: 'bg-teal-500', bar: '#14b8a6', barBg: 'bg-teal-100 dark:bg-teal-900/30' };
    66→    case 'on_hold': return { bg: 'bg-amber-100 dark:bg-amber-900/30', text: 'text-amber-700 dark:text-amber-300', dot: 'bg-amber-500', bar: '#f59e0b', barBg: 'bg-amber-100 dark:bg-amber-900/30' };
    67→    case 'cancelled': return { bg: 'bg-rose-100 dark:bg-rose-900/30', text: 'text-rose-700 dark:text-rose-300', dot: 'bg-rose-500', bar: '#f43f5e', barBg: 'bg-rose-100 dark:bg-rose-900/30' };
    68→    default: return { bg: 'bg-muted', text: 'text-muted-foreground', dot: 'bg-muted-foreground/50', bar: '#94a3b8', barBg: 'bg-muted' };
    69→  }

    72→function computeProgress(project: Project) {
    73→  const tasks = project.tasks || [];
    74→  if (tasks.length === 0) return 0;
    75→  const done = tasks.filter(t => t.status === 'done').length;
    76→  return Math.round((done / tasks.length) * 100);
    77→}
    78→
    79→function getNextMilestone(project: Project) {
    80→  return (project.milestones || [])
    81→    .filter(m => !m.completed)
    82→    .sort((a, b) => new Date(a.dueDate).getTime() - new Date(b.dueDate).getTime())[0];
    83→}

    85→// ─── Timeline (Gantt) Helpers ────────────────────────────────────────
    86→function getTimelineRange(projects: Project[]) {
    87→  const now = new Date();
    88→  const start = new Date(now.getFullYear(), now.getMonth() - 2, 1);
    89→  const end = new Date(now.getFullYear(), now.getMonth() + 6, 0);
    90→  return { start, end };
    91→}
    92→
    93→function monthRange(start: Date, end: Date): Date[] {
    94→  const months: Date[] = [];
    95→  const d = new Date(start.getFullYear(), start.getMonth(), 1);
    96→  while (d <= end) {
    97→    months.push(new Date(d));
    98→    d.setMonth(d.getMonth() + 1);
    99→  }
   100→  return months;
   101→}
   102→
   103→function dateToX(date: Date, start: Date, end: Date, width: number): number {
   104→  const total = end.getTime() - start.getTime();
   105→  if (total === 0) return 0;
   106→  return ((date.getTime() - start.getTime()) / total) * width;
   107→}

   109→// ─── Board View ──────────────────────────────────────────────────────
   110→function BoardView({ projects, onCardClick }: { projects: Project[]; onCardClick: (id: string) => void }) {
   111→  const grouped = useMemo(() => {
   112→    const map: Record<string, Project[]> = { active: [], on_hold: [], completed: [], cancelled: [] };
   113→    projects.forEach(p => {
   114→      const key = p.status === 'on_hold' ? 'on_hold' : p.status;
   115→      if (map[key]) map[key].push(p);
   116→      else map.active.push(p);
   117→    });
   118→    return map;
   119→  }, [projects]);
   120→
   121→  return (
   122→    <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   123→      {STATUS_COLUMNS.map(col => {

   126→          <div key={col.key} className="flex flex-col">
   127→            {/* Column header */}

   129→              <span className="text-sm">{col.emoji}</span>

   222→// ─── Sort Icon (top-level to avoid render-time component creation) ──
   223→function SortIcon({ col, sortKey, sortDir }: { col: SortKey; sortKey: SortKey; sortDir: SortDir }) {
   224→  if (sortKey !== col) return <ArrowUpDown className="h-3 w-3 ml-1 text-muted-foreground/40" />;
   225→  return sortDir === 'asc'
   226→    ? <ArrowUp className="h-3 w-3 ml-1 text-emerald-500" />
   227→    : <ArrowDown className="h-3 w-3 ml-1 text-emerald-500" />;
   228→}

   230→// ─── List View ───────────────────────────────────────────────────────
   231→function ListView({
   232→  projects,
   233→  onRowClick,
   234→  sortKey,
   235→  sortDir,
   236→  onSort,
   237→}: {
   238→  projects: Project[];
   239→  onRowClick: (id: string) => void;
   240→  sortKey: SortKey;
   241→  sortDir: SortDir;
   242→  onSort: (key: SortKey) => void;
   243→}) {
   244→  const sorted = useMemo(() => {
   245→    const copy = [...projects];
   246→    copy.sort((a, b) => {
   247→      let cmp = 0;
   248→      switch (sortKey) {
   249→        case 'title':
   250→          cmp = (a.tender?.title || '').localeCompare(b.tender?.title || '');
   251→          break;
   252→        case 'contractValue':
   253→          cmp = a.contractValue - b.contractValue;
   254→          break;
   255→        case 'status':
   256→          cmp = a.status.localeCompare(b.status);
   257→          break;
   258→        case 'progress':
   259→          cmp = computeProgress(a) - computeProgress(b);
   260→          break;
   261→      }
   262→      return sortDir === 'asc' ? cmp : -cmp;
   263→    });
   264→    return copy;
   265→  }, [projects, sortKey, sortDir]);
   266→
   267→  return (
   268→    <Card className="rounded-xl border border-border/60 bg-card shadow-sm overflow-hidden">
   269→      <Table>
   270→        <TableHeader>
   271→          <TableRow className="hover:bg-transparent">
   272→            <TableHead className="cursor-pointer select-none" onClick={() => onSort('title')}>
   273→              <span className="flex items-center">Project <SortIcon col="title" sortKey={sortKey} sortDir={sortDir} /></span>
   274→            </TableHead>
   275→            <TableHead className="hidden md:table-cell">Tender</TableHead>
   276→            <TableHead className="hidden sm:table-cell">Contractor</TableHead>
   277→            <TableHead className="cursor-pointer select-none" onClick={() => onSort('contractValue')}>
   278→              <span className="flex items-center">Value <SortIcon col="contractValue" sortKey={sortKey} sortDir={sortDir} /></span>
   279→            </TableHead>
   280→            <TableHead className="cursor-pointer select-none" onClick={() => onSort('status')}>
   281→              <span className="flex items-center">Status <SortIcon col="status" sortKey={sortKey} sortDir={sortDir} /></span>
   282→            </TableHead>
   283→            <TableHead className="cursor-pointer select-none" onClick={() => onSort('progress')}>
   284→              <span className="flex items-center">Progress <SortIcon col="progress" sortKey={sortKey} sortDir={sortDir} /></span>
   285→            </TableHead>
   286→            <TableHead className="hidden lg:table-cell">Next Milestone</TableHead>
   287→          </TableRow>
   288→        </TableHeader>
   289→        <TableBody>
   290→          {sorted.map(project => {
   291→            const sc = statusColorMap(project.status);
   292→            const progress = computeProgress(project);
   293→            const totalTasks = project.tasks?.length || 0;
   294→            const doneTasks = project.tasks?.filter(t => t.status === 'done').length || 0;
   295→            const nextMilestone = getNextMilestone(project);
   296→
   297→            return (
   298→              <TableRow
   299→                key={project.id}
   300→                className="cursor-pointer hover:bg-emerald-50/50 dark:hover:bg-emerald-900/10 transition-colors"
   301→                onClick={() => onRowClick(project.id)}
   302→              >
   303→                <TableCell>
   304→                  <div className="flex items-center gap-2">
   305→                    <FolderKanban className="h-4 w-4 text-emerald-500 flex-shrink-0" />
   306→                    <span className="font-medium text-sm truncate max-w-[180px]">
   307→                      {project.tender?.title || 'Project'}
   308→                    </span>
   309→                  </div>
   310→                </TableCell>
   311→                <TableCell className="hidden md:table-cell text-xs text-muted-foreground truncate max-w-[150px]">
   312→                  {project.tender?.title || '-'}
   313→                </TableCell>
   314→                <TableCell className="hidden sm:table-cell text-xs text-muted-foreground truncate max-w-[120px]">
   315→                  {project.bid?.user?.profile?.fullName || '-'}
   316→                </TableCell>
   317→                <TableCell className="text-sm font-semibold">{formatETB(project.contractValue)}</TableCell>
   318→                <TableCell>
   319→                  <Badge className={`text-[10px] px-2 py-0 border-0 font-semibold ${sc.bg} ${sc.text}`}>
   320→                    <span className={`w-1.5 h-1.5 rounded-full ${sc.dot} mr-1 inline-block`} />
   321→                    {project.status.replace('_', ' ')}
   322→                  </Badge>
   323→                </TableCell>
   324→                <TableCell>
   325→                  <div className="flex items-center gap-2 min-w-[100px]">
   326→                    <div className={`h-1.5 flex-1 ${sc.barBg} rounded-full overflow-hidden`}>
   327→                      <div className="h-full rounded-full" style={{ width: `${progress}%`, backgroundColor: sc.bar }} />
   328→                    </div>
   329→                    <span className="text-[10px] font-medium text-muted-foreground whitespace-nowrap">
   330→                      {doneTasks}/{totalTasks}
   331→                    </span>
   332→                  </div>
   333→                </TableCell>
   334→                <TableCell className="hidden lg:table-cell">
   335→                  {nextMilestone ? (
   336→                    <div className="flex items-center gap-1.5 text-xs text-muted-foreground">
   337→                      <Calendar className="h-3 w-3 text-amber-500 flex-shrink-0" />
   338→                      <span className="truncate max-w-[120px]">{nextMilestone.title}</span>
   339→                    </div>
   340→                  ) : (
   341→                    <span className="text-xs text-muted-foreground/40">-</span>
   342→                  )}
   343→                </TableCell>
   344→              </TableRow>
   345→            );
   346→          })}
   347→        </TableBody>
   348→      </Table>
   349→    </Card>
   350→  );
   351→}

   353→// ─── Timeline View (SVG Gantt) ───────────────────────────────────────
   354→function TimelineView({ projects, onBarClick }: { projects: Project[]; onBarClick: (id: string) => void }) {
   355→  const timelineWidth = 900;
   356→  const rowHeight = 44;
   357→  const headerHeight = 36;
   358→  const { start, end } = useMemo(() => getTimelineRange(projects), [projects]);
   359→  const months = useMemo(() => monthRange(start, end), [start, end]);
   360→
   361→  const barData = useMemo(() => {
   362→    return projects.map(p => {
   363→      const createdAt = new Date(p.createdAt);
   364→      // Estimate end: use last milestone or +3 months
   365→      const lastMilestone = (p.milestones || []).sort((a, b) => new Date(b.dueDate).getTime() - new Date(a.dueDate).getTime())[0];
   366→      const endDate = lastMilestone
   367→        ? new Date(lastMilestone.dueDate)
   368→        : new Date(createdAt.getTime() + 90 * 86400000);
   369→      const sc = statusColorMap(p.status);
   370→      return { project: p, start: createdAt, end: endDate, color: sc.bar };
   371→    });
   372→  }, [projects]);
   373→
   374→  const milestoneMarkers = useMemo(() => {
   375→    const markers: { x: number; y: number; color: string; completed: boolean; title: string; projectId: string }[] = [];
   376→    barData.forEach((bd, rowIdx) => {
   377→      (bd.project.milestones || []).forEach(m => {
   378→        const mx = dateToX(new Date(m.dueDate), start, end, timelineWidth);
   379→        markers.push({
   380→          x: mx, y: headerHeight + rowIdx * rowHeight + rowHeight / 2,
   381→          color: bd.color, completed: m.completed, title: m.title, projectId: bd.project.id,
   382→        });
   383→      });
   384→    });
   385→    return markers;
   386→  }, [barData, start, end]);
   387→
   388→  const nowX = dateToX(new Date(), start, end, timelineWidth);
   389→  const svgHeight = headerHeight + barData.length * rowHeight + 20;
   390→
   391→  return (
   392→    <Card className="rounded-xl border border-border/60 bg-card shadow-sm overflow-hidden">
   393→      <div className="overflow-x-auto">
   394→        <svg width={timelineWidth + 200} height={svgHeight} className="min-w-full">
   395→          {/* Month columns */}
   396→          {months.map((m, i) => {
   397→            const x = dateToX(m, start, end, timelineWidth) + 200;
   398→            const nextM = new Date(m);
   399→            nextM.setMonth(nextM.getMonth() + 1);
   400→            const x2 = dateToX(nextM, start, end, timelineWidth) + 200;
   401→            return (
   402→              <g key={i}>
   403→                <rect x={x} y={0} width={x2 - x} height={svgHeight} fill={i % 2 === 0 ? 'transparent' : 'rgba(128,128,128,0.03)'} />
   404→                <text x={x + (x2 - x) / 2} y={22} textAnchor="middle" className="text-[10px] fill-muted-foreground font-medium">
   405→                  {m.toLocaleDateString('en-US', { month: 'short', year: '2-digit' })}
   406→                </text>
   407→                <line x1={x} y1={headerHeight} x2={x} y2={svgHeight} stroke="currentColor" strokeOpacity={0.06} />
   408→              </g>
   409→            );
   410→          })}
   411→
   412→          {/* Today line */}
   413→          <line x1={nowX + 200} y1={0} x2={nowX + 200} y2={svgHeight} stroke="#f43f5e" strokeWidth={1.5} strokeDasharray="4 3" />
   414→          <text x={nowX + 200} y={12} textAnchor="middle" className="text-[9px] fill-rose-500 font-semibold">TODAY</text>
   415→
   416→          {/* Project label column */}
   417→          {barData.map((bd, i) => {
   418→            const y = headerHeight + i * rowHeight + rowHeight / 2;
   419→            return (
   420→              <g key={bd.project.id}>
   421→                <text x={8} y={y + 4} className="text-[10px] fill-foreground font-medium cursor-pointer" onClick={() => onBarClick(bd.project.id)}>
   422→                  {(bd.project.tender?.title || 'Project').slice(0, 22)}
   423→                </text>
   424→              </g>
   425→            );
   426→          })}
   427→
   428→          {/* Project bars */}
   429→          {barData.map((bd, i) => {
   430→            const x1 = dateToX(bd.start, start, end, timelineWidth) + 200;
   431→            const x2 = dateToX(bd.end, start, end, timelineWidth) + 200;
   432→            const y = headerHeight + i * rowHeight + 12;
   433→            const barH = 20;
   434→            const barW = Math.max(x2 - x1, 8);
   435→            return (
   436→              <g key={bd.project.id} className="cursor-pointer" onClick={() => onBarClick(bd.project.id)}>
   437→                <rect x={x1} y={y} width={barW} height={barH} rx={4} fill={bd.color} fillOpacity={0.2} />
   438→                <rect x={x1} y={y} width={barW} height={barH} rx={4} fill={bd.color} fillOpacity={0.7} />
   439→                <rect x={x1} y={y} width={barW * (computeProgress(bd.project) / 100)} height={barH} rx={4} fill={bd.color} />
   440→              </g>
   441→            );
   442→          })}
   443→
   444→          {/* Milestone diamonds */}
   445→          {milestoneMarkers.map((m, i) => (
   446→            <g key={i}>
   447→              <polygon
   448→                points={`${m.x + 200},${m.y - 5} ${m.x + 205},${m.y} ${m.x + 200},${m.y + 5} ${m.x + 195},${m.y}`}
   449→                fill={m.completed ? '#10b981' : m.color}
   450→                stroke="white"
   451→                strokeWidth={1}
   452→              />
   453→            </g>
   454→          ))}
   455→        </svg>
   456→      </div>
   457→    </Card>
   458→  );
   459→}
