    56→function ScoreBar({ score, label }: { score: number; label?: string }) {
    57→  const colorClass = score >= 80 ? 'bg-emerald-500' : score >= 60 ? 'bg-amber-500' : 'bg-red-500';
    58→  const textColor = score >= 80 ? 'text-emerald-700' : score >= 60 ? 'text-amber-700' : 'text-red-700';
    59→  return (
    60→    <div className="flex items-center gap-2 min-w-[120px]">
    61→      {label && <span className="text-[10px] text-muted-foreground w-12 shrink-0">{label}</span>}
    62→      <div className="flex-1 h-2 bg-muted/50 rounded-full overflow-hidden">
    63→        <motion.div
    64→          className={`h-full rounded-full ${colorClass}`}
    65→          initial={{ width: 0 }}
    66→          animate={{ width: `${Math.min(score, 100)}%` }}
    67→          transition={{ duration: 0.6, ease: 'easeOut' }}
    68→        />
    69→      </div>
    70→      <span className={`text-xs font-semibold ${textColor} w-8 text-right`}>{score}</span>
    71→    </div>
    72→  );
    73→}

    75→function RiskBadge({ level }: { level: 'low' | 'medium' | 'high' }) {
    76→  const config = {
    77→    low: { className: 'bg-emerald-100 text-emerald-700', icon: ShieldCheck },
    78→    medium: { className: 'bg-amber-100 text-amber-700', icon: ShieldQuestion },
    79→    high: { className: 'bg-red-100 text-red-700', icon: ShieldAlert },
    80→  };
    81→  const c = config[level];
    82→  const Icon = c.icon;
    83→  return (
    84→    <Badge className={`text-[10px] px-2 py-0.5 border-0 rounded-lg ${c.className}`}>
    85→      <Icon className="h-3 w-3 mr-1" />
    86→      {level.charAt(0).toUpperCase() + level.slice(1)}
    87→    </Badge>
    88→  );
    89→}
