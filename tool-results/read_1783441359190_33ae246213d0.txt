    49→function timeAgo(dateStr: string): string {
    50→  const diff = Date.now() - new Date(dateStr).getTime();
    51→  const mins = Math.floor(diff / 60000);
    52→  if (mins < 1) return 'just now';
    53→  if (mins < 60) return `${mins}m ago`;
    54→  const hrs = Math.floor(mins / 60);
    55→  if (hrs < 24) return `${hrs}h ago`;
    56→  const days = Math.floor(hrs / 24);
    57→  return `${days}d ago`;
    58→}
