   164→   164→function formatDate(dateStr: string): string {
   165→   165→  if (!dateStr) return '—';
   166→   166→  return new Date(dateStr).toLocaleDateString('en-GB', { day: '2-digit', month: 'short', year: 'numeric' });
   167→   167→}

   206→   206→    new Set(COLUMNS.filter(c => c.defaultVisible).map(c => c.key))

   211→   211→
   212→   212→  // ─── Document / OCR / AI Review state ───────────────────────────

   214→   214→  const [ocrLoading, setOcrLoading] = useState<Set<string>>(new Set());

   244→   244→      setLoading(false);

   246→   246→  }, [page, search, statusFilter, tenderFilter]);
   247→   247→
   248→   248→  useEffect(() => {

   254→   254→    const formData = new FormData();
   255→   255→    formData.append('file', file);
   256→   256→    formData.append('docType', 'bid_attachment');
   257→   257→    formData.append('autoOcr', 'true');

   259→   259→      const res = await api.upload(`/bids/${bidId}/documents`, formData);
   260→   260→      if (res.success) {
   261→   261→        toast.success('Document uploaded successfully');
   262→   262→        fetchData();

   264→   264→        toast.error(res.message || 'Upload failed');
   265→   265→      }

   286→   286→        if (res.success && res.data?.ocrStatus === 'completed') {
   287→   287→          setOcrLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });

   290→   290→          toast.success('OCR completed');

   292→   292→          setOcrLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   293→   293→          toast.error('OCR processing failed');
   294→   294→          fetchData();
   295→   295→        } else {
   296→   296→          await poll(attempts + 1);
   297→   297→        }
   298→   298→      };
   299→   299→      poll();
   300→   300→    } catch {
   301→   301→      setOcrLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   302→   302→      toast.error('Failed to start OCR');
   303→   303→    }

   309→   309→      await api.post(`/document-review/${docId}`);

   312→   312→        if (attempts > 30) {
   313→   313→          toast.error('AI Review processing timed out');
   314→   314→          setReviewLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   315→   315→          return;

   324→   324→          }

   326→   326→          fetchData();
   327→   327→          toast.success('AI Review completed');
   328→   328→        } else if (res.success && res.data?.aiReviewStatus === 'failed') {
   329→   329→          setReviewLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   330→   330→          toast.error('AI Review processing failed');
   331→   331→          fetchData();
   332→   332→        } else {
   333→   333→          await poll(attempts + 1);
   334→   334→        }
   335→   335→      };
   336→   336→      poll();
   337→   337→    } catch {
   338→   338→      setReviewLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   339→   339→      toast.error('Failed to start AI Review');
   340→   340→    }
   341→   341→  }, [fetchData]);
   342→   342→

   360→   360→          let reviewData = res.data?.aiReview || {};

   363→   363→            try {
   364→   364→              reviewData = JSON.parse(reviewData);
   365→   365→            } catch {
   366→   366→              reviewData = { summary: reviewData };
   367→   367→            }
   368→   368→          }
   369→   369→          setDocReview(prev => ({ ...prev, [docId]: reviewData }));
   370→   370→        }
   371→   371→      }
   372→   372→    } catch {
   373→   373→      toast.error('Failed to load document details');
   374→   374→    }
   375→   375→  }, [viewingDocId, viewingDocType, docOcrText, docReview]);
