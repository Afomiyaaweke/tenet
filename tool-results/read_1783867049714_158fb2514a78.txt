   167→function formatCurrency(amount: number): string {
   168→  return new Intl.NumberFormat('en-ET', { style: 'currency', currency: 'ETB', maximumFractionDigits: 0 }).format(amount);
   169→}

   171→function formatDate(dateStr: string): string {
   172→  if (!dateStr) return '—';
   173→  return new Date(dateStr).toLocaleDateString('en-GB', { day: '2-digit', month: 'short', year: 'numeric' });
   174→}
   175→
   176→function daysUntil(dateStr: string): number {
   177→  return Math.ceil((new Date(dateStr).getTime() - Date.now()) / (1000 * 60 * 60 * 24));
   178→}

   197→export function ApplicantsView() {
   198→  const { user } = useAuthStore();
   199→  const { setView } = useNavStore();
   200→  const [rows, setRows] = useState<ApplicantRow[]>([]);
   201→  const [meta, setMeta] = useState<ApplicantMeta | null>(null);
   202→  const [publishedTenders, setPublishedTenders] = useState<PublishedTenderInfo[]>([]);
   203→  const [openTenders, setOpenTenders] = useState<TenderInfo[]>([]);
   204→  const [closedTenders, setClosedTenders] = useState<TenderInfo[]>([]);
   205→  const [loading, setLoading] = useState(true);
   206→  const [search, setSearch] = useState('');
   207→  const [statusFilter, setStatusFilter] = useState<string>('all');
   208→  const [tenderFilter, setTenderFilter] = useState<string>('all');
   209→  const [sortField, setSortField] = useState<SortField>('submittedAt');
   210→  const [sortDir, setSortDir] = useState<SortDir>('desc');
   211→  const [viewMode, setViewMode] = useState<ViewMode>('table');
   212→  const [expandedRow, setExpandedRow] = useState<string | null>(null);
   213→  const [visibleColumns, setVisibleColumns] = useState<Set<string>>(() =>
   214→    new Set(COLUMNS.filter(c => c.defaultVisible).map(c => c.key))
   215→  );
   216→  const [columnMenuOpen, setColumnMenuOpen] = useState(false);
   217→  const [page, setPage] = useState(1);
   218→  const PAGE_SIZE = 50;
   219→
   220→  // ─── Selected tender for detail view ────────────────────────────
   221→  const [selectedTenderId, setSelectedTenderId] = useState<string | null>(null);
   222→  const [selectedTenderInfo, setSelectedTenderInfo] = useState<PublishedTenderInfo | null>(null);
   223→
   224→  // ─── Document / OCR / AI Review state ───────────────────────────
   225→  const [docUploadBidId, setDocUploadBidId] = useState<string | null>(null);
   226→  const [ocrLoading, setOcrLoading] = useState<Set<string>>(new Set());
   227→  const [reviewLoading, setReviewLoading] = useState<Set<string>>(new Set());
   228→  const [viewingDocId, setViewingDocId] = useState<string | null>(null);
   229→  const [viewingDocType, setViewingDocType] = useState<'ocr' | 'review'>('ocr');
   230→  const [docOcrText, setDocOcrText] = useState<Record<string, string>>({});
   231→  const [docReview, setDocReview] = useState<Record<string, Record<string, unknown>>>({});
   232→
   233→  // Fetch data
   234→  const fetchData = useCallback(async () => {
   235→    setLoading(true);
   236→    try {
   237→      const params: Record<string, string> = {
   238→        page: String(page),
   239→        limit: String(PAGE_SIZE),
   240→      };
   241→      if (search) params.search = search;
   242→      if (statusFilter && statusFilter !== 'all') params.status = statusFilter;
   243→      if (tenderFilter && tenderFilter !== 'all') params.tenderId = tenderFilter;
   244→      // If a specific tender is selected, filter by it
   245→      if (selectedTenderId) params.tenderId = selectedTenderId;
   246→
   247→      const res = await api.get('/applicants', params);
   248→      if (res.success) {
   249→        setRows(res.data);
   250→        setMeta(res.meta);
   251→        setOpenTenders(res.openTenders || []);
   252→        setClosedTenders(res.closedTenders || []);
   253→        setPublishedTenders(res.publishedTenders || []);
   254→      }
   255→    } catch (err) {
   256→      console.error('Fetch applicants error:', err);
   257→      toast.error('Failed to load applicants');
   258→    } finally {
   259→      setLoading(false);
   260→    }
   261→  }, [page, search, statusFilter, tenderFilter, selectedTenderId]);
   262→
   263→  useEffect(() => {
   264→    fetchData();
   265→  }, [fetchData]);
   266→
   267→  // When selecting a tender, set the info
   268→  useEffect(() => {
   269→    if (selectedTenderId && publishedTenders.length > 0) {
   270→      const info = publishedTenders.find(t => t.id === selectedTenderId);
   271→      if (info) setSelectedTenderInfo(info);
   272→    } else {
   273→      setSelectedTenderInfo(null);
   274→    }
   275→  }, [selectedTenderId, publishedTenders]);
   276→
   277→  // Handle tender click → show applicants
   278→  const handleTenderClick = (tenderId: string) => {
   279→    setSelectedTenderId(tenderId);
   280→    setTenderFilter(tenderId);
   281→    setStatusFilter('all');
   282→    setSearch('');
   283→    setPage(1);
   284→    setExpandedRow(null);
   285→  };
   286→
   287→  // Back to tenders list
   288→  const handleBackToTenders = () => {
   289→    setSelectedTenderId(null);
   290→    setSelectedTenderInfo(null);
   291→    setTenderFilter('all');
   292→    setStatusFilter('all');
   293→    setSearch('');
   294→    setPage(1);
   295→    setExpandedRow(null);
   296→  };
   297→
   298→  // ─── Document handlers ──────────────────────────────────────────
   299→  const handleDocUpload = useCallback(async (bidId: string, file: File) => {
   300→    const formData = new FormData();
   301→    formData.append('file', file);
   302→    formData.append('docType', 'bid_attachment');
   303→    formData.append('autoOcr', 'true');
   304→    try {
   305→      const res = await api.upload(`/bids/${bidId}/documents`, formData);
   306→      if (res.success) {
   307→        toast.success('Document uploaded successfully');
   308→        fetchData();
   309→      } else {
   310→        toast.error(res.message || 'Upload failed');
   311→      }
   312→    } catch {
   313→      toast.error('Upload failed');
   314→    } finally {
   315→      setDocUploadBidId(null);
   316→    }
   317→  }, [fetchData]);
   318→
   319→  const handleRunOcr = useCallback(async (docId: string) => {
   320→    setOcrLoading(prev => new Set(prev).add(docId));
   321→    try {
   322→      await api.post(`/document-ocr/${docId}`);
   323→      const poll = async (attempts = 0): Promise<void> => {
   324→        if (attempts > 30) {
   325→          toast.error('OCR processing timed out');
   326→          setOcrLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   327→          return;
   328→        }
   329→        await new Promise(r => setTimeout(r, 2000));
   330→        const res = await api.get(`/document-ocr/${docId}`);
   331→        if (res.success && res.data?.ocrStatus === 'completed') {
   332→          setOcrLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   333→          setDocOcrText(prev => ({ ...prev, [docId]: res.data.ocrText || '' }));
   334→          fetchData();
   335→          toast.success('OCR completed');
   336→        } else if (res.success && res.data?.ocrStatus === 'failed') {
   337→          setOcrLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   338→          toast.error('OCR processing failed');
   339→          fetchData();
   340→        } else {
   341→          await poll(attempts + 1);
   342→        }
   343→      };
   344→      poll();
   345→    } catch {
   346→      setOcrLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   347→      toast.error('Failed to start OCR');
   348→    }
   349→  }, [fetchData]);
   350→
   351→  const handleRunReview = useCallback(async (docId: string) => {
   352→    setReviewLoading(prev => new Set(prev).add(docId));
   353→    try {
   354→      await api.post(`/document-review/${docId}`);
   355→      const poll = async (attempts = 0): Promise<void> => {
   356→        if (attempts > 30) {
   357→          toast.error('AI Review processing timed out');
   358→          setReviewLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   359→          return;
   360→        }
   361→        await new Promise(r => setTimeout(r, 2000));
   362→        const res = await api.get(`/document-review/${docId}`);
   363→        if (res.success && res.data?.aiReviewStatus === 'completed') {
   364→          setReviewLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   365→          let reviewData = res.data.aiReview || {};
   366→          if (typeof reviewData === 'string') {
   367→            try { reviewData = JSON.parse(reviewData); } catch { reviewData = { summary: reviewData }; }
   368→          }
   369→          setDocReview(prev => ({ ...prev, [docId]: reviewData }));
   370→          fetchData();
   371→          toast.success('AI Review completed');
   372→        } else if (res.success && res.data?.aiReviewStatus === 'failed') {
   373→          setReviewLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   374→          toast.error('AI Review processing failed');
   375→          fetchData();
   376→        } else {
   377→          await poll(attempts + 1);
   378→        }
   379→      };
   380→      poll();
   381→    } catch {
   382→      setReviewLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   383→      toast.error('Failed to start AI Review');
   384→    }
   385→  }, [fetchData]);
   386→
   387→  const handleViewDocDetail = useCallback(async (docId: string, type: 'ocr' | 'review') => {
   388→    if (viewingDocId === docId && viewingDocType === type) {
   389→      setViewingDocId(null);
   390→      return;
   391→    }
   392→    setViewingDocId(docId);
   393→    setViewingDocType(type);
   394→    try {
   395→      if (type === 'ocr' && !docOcrText[docId]) {
   396→        const res = await api.get(`/document-ocr/${docId}`);
   397→        if (res.success) {
   398→          setDocOcrText(prev => ({ ...prev, [docId]: res.data?.ocrText || '' }));
   399→        }
   400→      } else if (type === 'review' && !docReview[docId]) {
   401→        const res = await api.get(`/document-review/${docId}`);
   402→        if (res.success) {
   403→          let reviewData = res.data?.aiReview || {};
   404→          if (typeof reviewData === 'string') {
   405→            try {
   406→              reviewData = JSON.parse(reviewData);
   407→            } catch {
   408→              reviewData = { summary: reviewData };
   409→            }
   410→          }
   411→          setDocReview(prev => ({ ...prev, [docId]: reviewData }));
   412→        }
   413→      }
   414→    } catch {
   415→      toast.error('Failed to load document details');
   416→    }
   417→  }, [viewingDocId, viewingDocType, docOcrText, docReview]);
   418→
   419→  const closeDocDetail = useCallback(() => {
   420→    setViewingDocId(null);
   421→  }, []);
   422→
   423→  // Get unique tenders for filter dropdown (from closedTenders API data)
   424→  const tenderOptions = useMemo(() => {
   425→    return closedTenders.map(t => [t.id, t.title] as [string, string]);
   426→  }, [closedTenders]);
   427→
   428→  // Sorted rows (client-side sorting on current page)
   429→  const sortedRows = useMemo(() => {
   430→    const copy = [...rows];
   431→    copy.sort((a, b) => {
   432→      const aVal = a[sortField];
   433→      const bVal = b[sortField];
   434→      if (aVal === bVal) return 0;
   435→      if (aVal == null) return 1;
   436→      if (bVal == null) return -1;
   437→      let cmp = 0;
   438→      if (typeof aVal === 'number' && typeof bVal === 'number') {
   439→        cmp = aVal - bVal;
   440→      } else if (typeof aVal === 'boolean' && typeof bVal === 'boolean') {
   441→        cmp = Number(aVal) - Number(bVal);
   442→      } else {
   443→        cmp = String(aVal).localeCompare(String(bVal));
   444→      }
   445→      return sortDir === 'asc' ? cmp : -cmp;
   446→    });
   447→    return copy;
   448→  }, [rows, sortField, sortDir]);
   449→
   450→  // Visible columns
   451→  const activeColumns = useMemo(
   452→    () => COLUMNS.filter(c => visibleColumns.has(c.key)),
   453→    [visibleColumns]
   454→  );
   455→
   456→  // Column groups
   457→  const columnGroups = useMemo(() => {
   458→    const groups: Record<string, ColumnDef[]> = {};
   459→    activeColumns.forEach(c => {
   460→      if (!groups[c.group]) groups[c.group] = [];
   461→      groups[c.group].push(c);
   462→    });
   463→    return groups;
   464→  }, [activeColumns]);
   465→
   466→  // Toggle column visibility
   467→  const toggleColumn = (key: string) => {
   468→    setVisibleColumns(prev => {
   469→      const next = new Set(prev);
   470→      if (next.has(key)) next.delete(key);
   471→      else next.add(key);
   472→      return next;
   473→    });
   474→  };
   475→
   476→  // Toggle sort
   477→  const handleSort = (field: SortField) => {
   478→    if (sortField === field) {
   479→      setSortDir(prev => prev === 'asc' ? 'desc' : 'asc');
   480→    } else {
   481→      setSortField(field);
   482→      setSortDir('asc');
   483→    }
   484→  };
   485→
