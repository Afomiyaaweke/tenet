   120→function getAvatarGradient(seed: string): string {
   121→  let hash = 0;
   122→  for (let i = 0; i < seed.length; i++) {
   123→    hash = seed.charCodeAt(i) + ((hash << 5) - hash);
   124→  }
   125→  return GRADIENT_OPTIONS[Math.abs(hash) % GRADIENT_OPTIONS.length];
   126→}
   127→
   128→function getSenderColor(seed: string): string {
   129→  return SENDER_COLOR_MAP[getAvatarGradient(seed)] || 'text-primary';
   130→}

   132→function getUserInitial(name?: string): string {
   133→  if (!name) return '?';

   136→  return initials.toUpperCase() || name.charAt(0).toUpperCase() || '?';

   139→function formatRelativeTime(dateStr: string): string {
   140→  const now = new Date();
   141→  const date = new Date(dateStr);
   142→  const diffMs = now.getTime() - date.getTime();
   143→  const diffMin = Math.floor(diffMs / 60000);
   144→  const diffHour = Math.floor(diffMs / 3600000);
   145→  const diffDay = Math.floor(diffMs / 86400000);
   146→  if (diffMin < 1) return 'now';
   147→  if (diffMin < 60) return `${diffMin}m`;
   148→  if (diffHour < 24) return `${diffHour}h`;
   149→  if (diffDay === 1) return 'Yesterday';
   150→  if (diffDay < 7) return `${diffDay}d`;
   151→  return date.toLocaleDateString([], { month: 'short', day: 'numeric' });
   152→}

   161→function formatDateSeparator(dateStr: string): string {
   162→  const now = new Date();
   163→  const date = new Date(dateStr);
   164→  const diffMs = now.getTime() - date.getTime();
   165→  const diffDay = Math.floor(diffMs / 86400000);
   166→  if (diffDay === 0) return 'Today';
   167→  if (diffDay === 1) return 'Yesterday';
   168→  if (diffDay < 7) return date.toLocaleDateString([], { weekday: 'long' });
   169→  return date.toLocaleDateString([], {
   170→    month: 'long',
   171→    day: 'numeric',
   172→    year: 'numeric',
   173→  });
   174→}

   176→function shouldShowDateSeparator(
   177→  messages: ChatMessageItem[],
   178→  index: number
   179→): boolean {
   180→  if (index === 0) return true;
   181→  const prev = new Date(messages[index - 1].createdAt).toDateString();
   182→  const curr = new Date(messages[index].createdAt).toDateString();
   183→  return prev !== curr;
   184→}
   185→
   186→function isGroupedWithPrev(
   187→  messages: ChatMessageItem[],
   188→  index: number
   189→): boolean {
   190→  if (index === 0) return false;
   191→  const prev = messages[index - 1];
   192→  const curr = messages[index];
   193→  if (prev.userId !== curr.userId) return false;
   194→  if (shouldShowDateSeparator(messages, index)) return false;
   195→  const diff =
   196→    new Date(curr.createdAt).getTime() - new Date(prev.createdAt).getTime();
   197→  return diff < 5 * 60 * 1000; // 5 minutes
   198→}

   347→function MemberPicker({
   348→  excludedIds = [],
   349→  onSelect,
   350→  placeholder = 'Search users by name or email…',
   351→}: {
   352→  excludedIds?: string[];
   353→  onSelect: (user: UserSearchResult) => void;
   354→  placeholder?: string;
   355→}) {
   356→  const [query, setQuery] = useState('');
   357→  const [results, setResults] = useState<UserSearchResult[]>([]);
   358→  const [loading, setLoading] = useState(false);
   359→  const debounceRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   360→  const excludedKey = excludedIds.join(',');
   361→
   362→  useEffect(() => {
   363→    if (debounceRef.current) clearTimeout(debounceRef.current);
   364→    if (!query.trim()) {
   365→      setResults([]);
   366→      setLoading(false);
   367→      return;
   368→    }
   369→    setLoading(true);
   370→    debounceRef.current = setTimeout(async () => {
   371→      try {
   372→        const res = await api.get('/users/search', { q: query.trim() });
   373→        if (res.success) {
   374→          const excluded = new Set(excludedIds);
   375→          setResults(
   376→            (res.data || []).filter(
   377→              (u: UserSearchResult) => !excluded.has(u.id)
   378→            )
   379→          );
   380→        } else {
   381→          setResults([]);
   382→        }
   383→      } catch {
   384→        setResults([]);
   385→      } finally {
   386→        setLoading(false);
   387→      }
   388→    }, 250);
   389→    return () => {
   390→      if (debounceRef.current) clearTimeout(debounceRef.current);
   391→    };
   392→  }, [query, excludedKey, excludedIds]);
   393→
   394→  return (
   395→    <div className="space-y-2">
   396→      <div className="relative">
   397→        <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground/60" />
   398→        <Input
   399→          placeholder={placeholder}
   400→          value={query}
   401→          onChange={(e) => setQuery(e.target.value)}
   402→          className="h-9 text-sm pl-8 bg-muted/50 border-border/40 rounded-lg"
   403→        />
   404→      </div>
   405→      {loading && <p className="text-xs text-muted-foreground px-1">Searching…</p>}
   406→      {!loading && query.trim() && results.length === 0 && (
   407→        <p className="text-xs text-muted-foreground px-1">No users found.</p>
   408→      )}
   409→      {!loading && results.length > 0 && (
   410→        <div className="border border-border/40 rounded-lg max-h-56 overflow-y-auto divide-y divide-border/20">
   411→          {results.map((u) => (
   412→            <button
   413→              key={u.id}
   414→              onClick={() => onSelect(u)}
   415→              className="w-full flex items-center gap-3 p-2 hover:bg-muted/50 transition-colors text-left"
   416→            >
   417→              <UserAvatar
   418→                userId={u.id}
   419→                name={u.profile?.fullName}
   420→                photoUrl={u.profile?.profilePhoto}
   421→                size={32}
   422→                rounded="rounded-full"
   423→              />
   424→              <div className="min-w-0 flex-1">
   425→                <p className="text-sm font-medium truncate">
   426→                  {u.profile?.fullName || u.email}
   427→                </p>
   428→                <p className="text-xs text-muted-foreground truncate">{u.email}</p>
   429→              </div>
   430→              <Plus className="h-4 w-4 text-primary flex-shrink-0" />
   431→            </button>
   432→          ))}
   433→        </div>
   434→      )}
   435→    </div>
   436→  );
   437→}

   443→function NewGroupDialog({
   444→  open,
   445→  onOpenChange,
   446→  onCreated,
   447→  currentUserId,
   448→}: {
   449→  open: boolean;
   450→  onOpenChange: (o: boolean) => void;
   451→  onCreated: (conv: Conversation) => void;
   452→  currentUserId: string;
   453→}) {
   454→  const [title, setTitle] = useState('');
   455→  const [description, setDescription] = useState('');
   456→  const [selected, setSelected] = useState<UserSearchResult[]>([]);
   457→  const [creating, setCreating] = useState(false);
   458→
   459→  const reset = () => {
   460→    setTitle('');
   461→    setDescription('');
   462→    setSelected([]);
   463→  };
   464→
   465→  const handleCreate = async () => {
   466→    if (!title.trim()) {
   467→      toast.error('Group name is required');
   468→      return;
   469→    }
   470→    if (selected.length === 0) {
   471→      toast.error('Add at least one member');
   472→      return;
   473→    }
   474→    setCreating(true);
   475→    try {
   476→      const res = await api.post('/conversations', {
   477→        type: 'group',
   478→        title: title.trim(),
   479→        description: description.trim() || undefined,
   480→        memberIds: selected.map((u) => u.id),
   481→      });
   482→      if (res.success) {
   483→        toast.success('Group created');
   484→        onCreated(res.data as Conversation);
   485→        reset();
   486→        onOpenChange(false);
   487→      } else {
   488→        toast.error(res.error || 'Failed to create group');
   489→      }
   490→    } finally {
   491→      setCreating(false);
   492→    }
   493→  };
   494→
   495→  return (
   496→    <Dialog
   497→      open={open}
   498→      onOpenChange={(o) => {
   499→        if (!o) reset();
   500→        onOpenChange(o);
   501→      }}
   502→    >
   503→      <DialogContent className="sm:max-w-md">
   504→        <DialogHeader>
   505→          <DialogTitle>New Group</DialogTitle>
   506→          <DialogDescription>
   507→            Create a group conversation with other users.
   508→          </DialogDescription>
   509→        </DialogHeader>
   510→        <div className="space-y-3 py-2">
   511→          <div className="space-y-1.5">
   512→            <label className="text-xs font-medium text-muted-foreground">
   513→              Group name *
   514→            </label>
   515→            <Input
   516→              placeholder="e.g. Project Alpha Team"
   517→              value={title}
   518→              onChange={(e) => setTitle(e.target.value)}
   519→              className="bg-muted/50"
   520→            />
   521→          </div>
   522→          <div className="space-y-1.5">
   523→            <label className="text-xs font-medium text-muted-foreground">
   524→              Description (optional)
   525→            </label>
   526→            <Textarea
   527→              placeholder="What's this group about?"
   528→              value={description}
   529→              onChange={(e) => setDescription(e.target.value)}
   530→              rows={2}
   531→              className="bg-muted/50 resize-none"
   532→            />
   533→          </div>
   534→          <div className="space-y-1.5">
   535→            <label className="text-xs font-medium text-muted-foreground">
   536→              Add members *
   537→            </label>
   538→            <MemberPicker
   539→              excludedIds={[currentUserId, ...selected.map((u) => u.id)]}
   540→              onSelect={(u) => setSelected((prev) => [...prev, u])}
   541→            />
   542→            {selected.length > 0 && (
   543→              <div className="flex flex-wrap gap-1.5 pt-1">
   544→                {selected.map((u) => (
   545→                  <span
   546→                    key={u.id}
   547→                    className="inline-flex items-center gap-1.5 bg-primary/10 text-primary text-xs px-2 py-1 rounded-full"
   548→                  >
   549→                    {u.profile?.fullName || u.email}
   550→                    <button
   551→                      onClick={() =>
   552→                        setSelected((prev) =>
   553→                          prev.filter((x) => x.id !== u.id)
   554→                        )
   555→                      }
   556→                    >
   557→                      <X className="h-3 w-3" />
   558→                    </button>
   559→                  </span>
   560→                ))}
   561→              </div>
   562→            )}
   563→          </div>
   564→        </div>
   565→        <DialogFooter>
   566→          <Button variant="ghost" onClick={() => onOpenChange(false)}>
   567→            Cancel
   568→          </Button>
   569→          <Button
   570→            className="gradient-emerald"
   571→            onClick={handleCreate}
   572→            disabled={creating}
   573→          >
   574→            {creating ? 'Creating…' : 'Create Group'}
   575→          </Button>
   576→        </DialogFooter>
   577→      </DialogContent>
   578→    </Dialog>
   579→  );
   580→}

   761→function ConversationListItem({
   762→  conv,
   763→  active,
   764→  index,
   765→  onlineUsers,
   766→  typingUsers,
   767→  currentUserId,
   768→  onSelect,
   769→}: {
   770→  conv: Conversation;
   771→  active: boolean;
   772→  index: number;
   773→  onlineUsers: Set<string>;
   774→  typingUsers: string[];
   775→  currentUserId: string;
   776→  onSelect: (conv: Conversation) => void;
   777→}) {
   778→  const myMembership = conv.members?.find((m) => m.userId === currentUserId);
   779→  const pinned = !!(myMembership?.pinned || (conv as Conversation & { pinned?: boolean }).pinned);
   780→  const muted = !!myMembership?.muted;
   781→  const unread = conv.unreadCount || 0;
   782→
   783→  // Title and avatar resolution
   784→  let title = conv.title || '';
   785→  let photoUrl: string | undefined;
   786→  let otherUserId: string | undefined;
   787→  if (conv.type === 'direct') {
   788→    const other = conv.members?.find((m) => m.userId !== currentUserId);
   789→    title = other?.user.profile?.fullName || other?.user.email || 'Direct Message';
   790→    photoUrl = other?.user.profile?.profilePhoto;
   791→    otherUserId = other?.userId;
   792→  } else if (!title) {
   793→    title = conv.type === 'channel' ? 'Channel' : 'Untitled Group';
   794→  }
   795→
   796→  // Last message preview
   797→  const lastMsg = conv.messages?.[0];
   798→  const typingOthers = typingUsers.filter((u) => u !== currentUserId);
   799→  let previewText = '';
   800→  let previewTime = conv.updatedAt;
   801→  if (typingOthers.length > 0) {
   802→    previewText = 'typing…';
   803→  } else if (lastMsg) {
   804→    const senderName =
   805→      lastMsg.userId === currentUserId
   806→        ? 'You'
   807→        : lastMsg.user?.profile?.fullName?.split(' ')[0] || 'User';
   808→    previewText =
   809→      conv.type === 'direct' ? lastMsg.content : `${senderName}: ${lastMsg.content}`;
   810→    previewTime = lastMsg.createdAt;
   811→  } else {
   812→    previewText = 'No messages yet';
   813→  }
   814→
   815→  const isOnline = otherUserId ? onlineUsers.has(otherUserId) : false;
   816→
   817→  return (
   818→    <motion.button
   819→      custom={index}
   820→      variants={chatItemVariants}
   821→      initial="hidden"
   822→      animate="visible"
   823→      exit="hidden"
   824→      layout
   825→      onClick={() => onSelect(conv)}
   826→      className={`w-full p-3 text-left transition-colors duration-150 border-b border-border/20 group relative ${
   827→        active ? 'bg-primary/10' : 'hover:bg-muted/50'
   828→      }`}
   829→    >
   830→      {active && (
   831→        <motion.div
   832→          layoutId="activeConvBar"
   833→          className="absolute left-0 top-2 bottom-2 w-[3px] rounded-r-full gradient-emerald"
   834→          transition={{ type: 'spring', stiffness: 500, damping: 35 }}
   835→        />
   836→      )}
   837→      {pinned && !active && (
   838→        <Pin className="absolute top-2 right-2 h-3 w-3 text-muted-foreground/60" />
   839→      )}
   840→      <div className="flex items-start gap-3">
   841→        <div className="relative flex-shrink-0">
   842→          {conv.type === 'direct' ? (
   843→            <UserAvatar
   844→              userId={otherUserId || conv.id}
   845→              name={title}
   846→              photoUrl={photoUrl}
   847→              size={44}
   848→              rounded="rounded-full"
   849→            />
   850→          ) : (
   851→            <GroupAvatar
   852→              seed={conv.id}
   853→              title={title}
   854→              avatarUrl={conv.avatarUrl}
   855→              size={44}
   856→              rounded="rounded-full"
   857→            />
   858→          )}
   859→          {isOnline && (
   860→            <span className="absolute -bottom-0.5 -right-0.5 w-3 h-3 bg-emerald-500 rounded-full border-2 border-card" />
   861→          )}
   862→        </div>
   863→        <div className="min-w-0 flex-1">
   864→          <div className="flex items-center justify-between gap-2">
   865→            <p
   866→              className={`text-sm truncate flex-1 flex items-center gap-1 ${
   867→                active
   868→                  ? 'font-semibold'
   869→                  : unread > 0
   870→                    ? 'font-semibold'
   871→                    : 'font-medium'
   872→              }`}
   873→            >
   874→              <span className="truncate">{title}</span>
   875→              {muted && (
   876→                <BellOff className="h-3 w-3 text-muted-foreground/60 flex-shrink-0" />
   877→              )}
   878→            </p>
   879→            <div className="flex items-center gap-1.5 flex-shrink-0">
   880→              {unread > 0 && (
   881→                <motion.span
   882→                  initial={{ scale: 0 }}
   883→                  animate={{ scale: 1 }}
   884→                  className="min-w-[18px] h-[18px] px-1 rounded-full gradient-emerald text-white text-[10px] font-bold flex items-center justify-center premium-shadow"
   885→                >
   886→                  {unread > 99 ? '99+' : unread}
   887→                </motion.span>
   888→              )}
   889→              <span className="text-[10px] text-muted-foreground/60 flex-shrink-0">
   890→                {previewTime ? formatRelativeTime(previewTime) : ''}
   891→              </span>
   892→            </div>
   893→          </div>
   894→          <p
   895→            className={`text-[11px] truncate mt-0.5 ${
   896→              typingOthers.length > 0
   897→                ? 'text-primary font-medium'
   898→                : 'text-muted-foreground'
   899→            }`}
   900→          >
   901→            {truncate(previewText, 48)}
   902→          </p>
   903→        </div>
   904→      </div>
   905→    </motion.button>
   906→  );
   907→}

   913→function QuickReactButton({ onReact }: { onReact: (emoji: string) => void }) {
   914→  const [open, setOpen] = useState(false);
   915→  const ref = useRef<HTMLDivElement>(null);
   916→
   917→  useEffect(() => {
   918→    if (!open) return;
   919→    const handler = (e: MouseEvent) => {
   920→      if (ref.current && !ref.current.contains(e.target as Node)) {
   921→        setOpen(false);
   922→      }
   923→    };
   924→    document.addEventListener('mousedown', handler);
   925→    return () => document.removeEventListener('mousedown', handler);
   926→  }, [open]);
   927→
   928→  return (
   929→    <div ref={ref} className="relative">
   930→      <button
   931→        onClick={() => setOpen((o) => !o)}
   932→        className="inline-flex items-center justify-center w-6 h-6 rounded-full bg-muted/60 hover:bg-muted text-muted-foreground transition-colors"
   933→        title="Add reaction"
   934→      >
   935→        <Plus className="h-3 w-3" />
   936→      </button>
   937→      <AnimatePresence>
   938→        {open && (
   939→          <motion.div
   940→            initial={{ opacity: 0, scale: 0.85, y: 5 }}
   941→            animate={{ opacity: 1, scale: 1, y: 0 }}
   942→            exit={{ opacity: 0, scale: 0.85, y: 5 }}
   943→            transition={{ duration: 0.15 }}
   944→            className="absolute z-50 bottom-8 right-0 bg-card border border-border rounded-full shadow-lg p-1 flex items-center gap-0.5"
   945→          >
   946→            {QUICK_REACT_EMOJIS.map((e) => (
   947→              <button
   948→                key={e}
   949→                onClick={() => {
   950→                  onReact(e);
   951→                  setOpen(false);
   952→                }}
   953→                className="w-7 h-7 rounded-full hover:bg-muted flex items-center justify-center text-sm transition-transform hover:scale-125"
   954→              >
   955→                {e}
   956→              </button>
   957→            ))}
   958→          </motion.div>
   959→        )}
   960→      </AnimatePresence>
   961→    </div>
   962→  );
   963→}

   965→function MessageReactions({
   966→  reactions,
   967→  currentUserId,
   968→  onReact,
   969→}: {
   970→  reactions: MessageReactionItem[];
   971→  currentUserId: string;
   972→  onReact: (emoji: string) => void;
   973→}) {
   974→  const grouped = useMemo(() => {
   975→    const map = new Map<
   976→      string,
   977→      { count: number; mine: boolean; users: string[] }
   978→    >();
   979→    for (const r of reactions) {
   980→      const entry = map.get(r.emoji) || { count: 0, mine: false, users: [] };
   981→      entry.count += 1;
   982→      if (r.userId === currentUserId) entry.mine = true;
   983→      entry.users.push(r.user?.profile?.fullName || 'User');
   984→      map.set(r.emoji, entry);
   985→    }
   986→    return Array.from(map.entries());
   987→  }, [reactions, currentUserId]);
   988→
   989→  return (
   990→    <div className="flex flex-wrap items-center gap-1 mt-1">
   991→      {grouped.map(([emoji, info]) => (
   992→        <button
   993→          key={emoji}
   994→          onClick={() => onReact(emoji)}
   995→          className={`inline-flex items-center gap-1 px-1.5 h-6 rounded-full text-[11px] font-medium transition-all hover:scale-105 ${
   996→            info.mine
   997→              ? 'gradient-emerald text-white premium-shadow'
   998→              : 'bg-muted text-foreground hover:bg-muted/70'
   999→          }`}
  1000→          title={info.users.join(', ')}
  1001→        >
  1002→          <span>{emoji}</span>
  1003→          <span>{info.count}</span>
  1004→        </button>
  1005→      ))}
  1006→      <QuickReactButton onReact={onReact} />
  1007→    </div>
  1008→  );
  1009→}
