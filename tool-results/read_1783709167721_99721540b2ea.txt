   179→   179→function fmtMoney(amount: number, currency: string): string {
   180→   180→  if (!amount) return '—';
   181→   181→  try {
   182→   182→    return new Intl.NumberFormat('en-US', {
   183→   183→      style: 'currency',
   184→   184→      currency: currency || 'USD',
   185→   185→      maximumFractionDigits: 0,
   186→   186→    }).format(amount);
   187→   187→  } catch {
   188→   188→    return `${currency} ${amount.toLocaleString()}`;
   189→   189→  }

   196→   196→function deadlineBadge(days: number): string {
   197→   197→  if (days <= 0) return 'bg-rose-50 text-rose-700 dark:bg-rose-950/40 dark:text-rose-300';
   198→   198→  if (days <= 7) return 'bg-amber-50 text-amber-700 dark:bg-amber-950/40 dark:text-amber-300';
   199→   199→  if (days <= 21) return 'bg-sky-50 text-sky-700 dark:bg-sky-950/40 dark:text-sky-300';
   200→   200→  return 'bg-emerald-50 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300';

   202→   202→
   203→   203→function relativeTime(dateStr: string): string {
   204→   204→  const diff = Date.now() - new Date(dateStr).getTime();
   205→   205→  const day = 86400000;
   206→   206→  if (diff < day) return 'today';
   207→   207→  if (diff < 2 * day) return 'yesterday';
   208→   208→  if (diff < 30 * day) return `${Math.floor(diff / day)}d ago`;
   209→   209→  return new Date(dateStr).toLocaleDateString();
   210→   210→}

   352→   352→      if (search) params.search = search;

   358→   358→        setSourceMeta(res.meta?.sources || []);
   359→   359→        setFallback(Boolean(res.meta?.fallback));

   362→   362→      } else {
   363→   363→        toast.error(res.error || 'Failed to load live tenders');
   364→   364→      }
   365→   365→      setLoading(false);
   366→   366→      setRefreshing(false);
   367→   367→    },
   368→   368→    [search, sourceFilter, sectorFilter],
   369→   369→  );
   370→   370→
   371→   371→  useEffect(() => {
   372→   372→    const t = setTimeout(() => load(), 250);
   373→   373→    return () => clearTimeout(t);
   374→   374→  }, [load]);
   375→   375→
   376→   376→  const liveSourcesCount = useMemo(
   377→   377→    () => sourceMeta.filter((s) => s.live && s.ok).length,
   378→   378→    [sourceMeta],

   381→   381→  const totalTenderValue = useMemo(() => {
   382→   382→    return tenders.reduce((sum, t) => sum + (t.budgetMax || 0), 0);
   383→   383→  }, [tenders]);
   384→   384→
   385→   385→  /* ────── Inline document loading ────── */
   386→   386→  const loadDocument = useCallback(async (tender: LiveTender) => {
   387→   387→    const id = tender.id;
   388→   388→    if (expandedId === id) {
   389→   389→      setExpandedId(null);
   390→   390→      return;
   391→   391→    }
   392→   392→
   393→   393→    // Already fetched?
   394→   394→    if (docData[id]) {
   395→   395→      setExpandedId(id);
   396→   396→      return;
   397→   397→    }
   398→   398→
   399→   399→    setExpandedId(id);
   400→   400→    setDocLoading(id);
   401→   401→    setDocError((prev) => { const n = { ...prev }; delete n[id]; return n; });
   402→   402→
   403→   403→    try {
   404→   404→      const res = await api.get(`/tenders/${encodeURIComponent(id)}/documents`, { url: tender.externalUrl });
   405→   405→      if (res.success && res.data) {
   406→   406→        setDocData((prev) => ({ ...prev, [id]: res.data as InlineDocument }));
   407→   407→      } else {
   408→   408→        setDocError((prev) => ({ ...prev, [id]: res.error || 'Failed to load document' }));
   409→   409→      }
   410→   410→    } catch {
   411→   411→      setDocError((prev) => ({ ...prev, [id]: 'Network error' }));
   412→   412→    }
   413→   413→    setDocLoading(null);
   414→   414→  }, [expandedId, docData]);
   415→   415→
