   167→   167→
   168→   168→function parseAmount(value: number | string | undefined): number {
   169→   169→  if (value === undefined || value === null) return 0;
   170→   170→  if (typeof value === 'number') return value;
   171→   171→  const n = Number(String(value).replace(/[^0-9.-]/g, ''));
   172→   172→  return Number.isFinite(n) ? n : 0;
   173→   173→}
   174→   174→
   175→   175→function truncate(text: string, max = 280): string {
   176→   176→  if (!text) return '';
   177→   177→  return text.length > max ? text.slice(0, max).trimEnd() + '…' : text;
   178→   178→}

   207→   207→function inferCategory(row: WorldBankProcurementRow): string {
   208→   208→  const t = (row.contract_type || '').toLowerCase();
   209→   209→  if (t.includes('work') || t.includes('civil') || t.includes('construction')) return 'Construction';
   210→   210→  if (t.includes('consult')) return 'Consulting';
   211→   211→  if (t.includes('good') || t.includes('supply') || t.includes('equipment')) return 'Supply';
   212→   212→  if (t.includes('service')) return 'Consulting';
   213→   213→  const p = (row.project_name || '').toLowerCase();
   214→   214→  if (p.includes('road') || p.includes('bridge') || p.includes('infrastructure')) return 'Construction';
   215→   215→  if (p.includes('agric') || p.includes('farm') || p.includes('irrigat')) return 'Agriculture';
   216→   216→  if (p.includes('health') || p.includes('clinic') || p.includes('hospital')) return 'Healthcare';
   217→   217→  if (p.includes('energy') || p.includes('power') || p.includes('electric')) return 'Engineering';
   218→   218→  if (p.includes('edu') || p.includes('school')) return 'Education';
   219→   219→  if (p.includes('water') || p.includes('sanitation')) return 'Engineering';
   220→   220→  return 'Consulting';
   221→   221→}
   222→   222→
   223→   223→function buildDeadline(signingDate?: string): string {
   224→   224→  if (!signingDate) return new Date(Date.now() + 45 * 86400000).toISOString();
   225→   225→  const base = new Date(signingDate);
   226→   226→  if (Number.isNaN(base.getTime())) return new Date(Date.now() + 45 * 86400000).toISOString();
   227→   227→  return new Date(base.getTime() + 45 * 86400000).toISOString();
   228→   228→}

   230→   230→export async function fetchWorldBankTenders(opts: {
   231→   231→  search?: string;
   232→   232→  rows?: number;
   233→   233→}): Promise<{ tenders: LiveTender[]; total: number; ok: boolean }> {
   234→   234→  const rows = Math.min(Math.max(opts.rows ?? 20, 1), 50);
   235→   235→  const params = new URLSearchParams({
   236→   236→    format: 'json',
   237→   237→    rows: String(rows),
   238→   238→    fl: [
   239→   239→      'project_id',
   240→   240→      'procurement_group_id',
   241→   241→      'project_name',
   242→   242→      'borrower',
   243→   243→      'country',
   244→   244→      'region',
   245→   245→      'contract_description',
   246→   246→      'contract_type',
   247→   247→      'supplier',
   248→   248→      'supplier_country',
   249→   249→      'contract_signing_date',
   250→   250→      'total_contract_amount',
   251→   251→      'wb_contract_number',
   252→   252→    ].join(','),
   253→   253→  });
   254→   254→  if (opts.search) params.set('q', opts.search);
   255→   255→
   256→   256→  const url = `https://search.worldbank.org/api/v2/procurement?${params.toString()}`;
   257→   257→  const ctrl = new AbortController();
   258→   258→  const timer = setTimeout(() => ctrl.abort(), 8000);
   259→   259→
   260→   260→  try {
   261→   261→    const res = await fetch(url, {
   262→   262→      signal: ctrl.signal,
   263→   263→      headers: { Accept: 'application/json', 'User-Agent': 'Tenets-Tender-Ecosystem/1.0' },
   264→   264→      cache: 'no-store',
   265→   265→    });
   266→   266→    clearTimeout(timer);
   267→   267→    if (!res.ok) {
   268→   268→      return { tenders: [], total: 0, ok: false };
   269→   269→    }
   270→   270→    const json = (await res.json()) as WorldBankResponse;
   271→   271→    const rowsArr = json.rows || json.procurements || [];
   272→   272→    const total = typeof json.total === 'number' ? json.total : rowsArr.length;
   273→   273→
   274→   274→    const tenders: LiveTender[] = rowsArr
   275→   275→      .filter((r) => r && (r.contract_description || r.project_name))
   276→   276→      .map((r, idx) => {
   277→   277→        const externalId = r.procurement_group_id || r.project_id || `wb-${idx}`;
   278→   278→        const title = truncate(r.contract_description || r.project_name || 'World Bank Contract Award', 160);
   279→   279→        const amount = parseAmount(r.total_contract_amount);
   280→   280→        const signingDate = r.contract_signing_date;
   281→   281→        return {
   282→   282→          id: `worldbank-${externalId}`,
   283→   283→          title,
   284→   284→          scope: truncate(r.contract_description || r.project_name || '', 400),
   285→   285→          budgetMin: amount,
   286→   286→          budgetMax: amount,
   287→   287→          deadline: buildDeadline(signingDate),
   288→   288→          location: r.country || r.region || 'Multiple',
   289→   289→          categoryTags: inferCategory(r),
   290→   290→          requiredDocs: '',
   291→   291→          status: 'awarded',
   292→   292→          createdBy: 'worldbank',
   293→   293→          createdAt: signingDate || new Date().toISOString(),
   294→   294→          updatedAt: signingDate || new Date().toISOString(),
   295→   295→          source: 'worldbank',
   296→   296→          externalId,
   297→   297→          externalUrl: r.project_id
   298→   298→            ? `https://projects.worldbank.org/en/projects-operations/project-detail/${r.project_id}`
   299→   299→            : 'https://financesone.worldbank.org',
   300→   300→          currency: 'USD',
   301→   301→          borrower: r.borrower || undefined,
   302→   302→          supplier: r.supplier || undefined,
   303→   303→          contractType: r.contract_type || undefined,
   304→   304→          signingDate: signingDate || undefined,
   305→   305→          region: r.region || undefined,
   306→   306→        } satisfies LiveTender;
   307→   307→      });
   308→   308→
   309→   309→    return { tenders, total, ok: true };
   310→   310→  } catch {
   311→   311→    clearTimeout(timer);
   312→   312→    return { tenders: [], total: 0, ok: false };
   313→   313→  }

   365→   365→        categoryTags: 'Consulting',
   366→   366→        requiredDocs: '',

   368→   368→        createdBy: 'eu_ted',
   369→   369→        createdAt: String(row.DS_DATE_PUB || new Date().toISOString()),

   374→   374→        currency: 'EUR',
   375→   375→        borrower: String(row.CA_NAME || '') || undefined,
   376→   376→      } satisfies LiveTender;
   377→   377→    });
   378→   378→

   423→   423→        deadline: new Date(Date.now() + 30 * 86400000).toISOString(),

   426→   426→        requiredDocs: '',
   427→   427→        status: 'open',
   428→   428→        createdBy: 'ungm',
   429→   429→        createdAt: new Date().toISOString(),
   430→   430→        updatedAt: new Date().toISOString(),

   432→   432→        externalId: `ungm-rss-${idx}`,
   433→   433→        externalUrl: rawLink,
   434→   434→        currency: 'USD',
   435→   435→        region: 'Global',
   436→   436→      } satisfies LiveTender;

   438→   438→
   439→   439→    return { tenders, total: matches.length, ok: true };

   449→   449→
   450→   450→export async function fetchSamGovTenders(opts: {
   451→   451→  search?: string;
   452→   452→  rows?: number;
   453→   453→}): Promise<{ tenders: LiveTender[]; total: number; ok: boolean }> {
   454→   454→  const rows = Math.min(Math.max(opts.rows ?? 10, 1), 50);
   455→   455→  const params = new URLSearchParams({
   456→   456→    limit: String(rows),
   457→   457→    offset: '0',
   458→   458→    postedFrom: new Date(Date.now() - 30 * 86400000).toISOString().split('T')[0],
   459→   459→    postedTo: new Date().toISOString().split('T')[0],
   460→   460→  });
   461→   461→  if (opts.search) params.set('q', opts.search);
   462→   462→
   463→   463→  const url = `https://api.sam.gov/opportunities/v2/search?${params.toString()}`;
   464→   464→  const ctrl = new AbortController();
   465→   465→  const timer = setTimeout(() => ctrl.abort(), 8000);
   466→   466→
   467→   467→  try {
   468→   468→    const res = await fetch(url, {
   469→   469→      signal: ctrl.signal,
   470→   470→      headers: { Accept: 'application/json', 'User-Agent': 'Tenets-Tender-Ecosystem/1.0' },
   471→   471→      cache: 'no-store',
   472→   472→    });
   473→   473→    clearTimeout(timer);
   474→   474→    if (!res.ok) return { tenders: [], total: 0, ok: false };
   475→   475→
   476→   476→    const json = (await res.json()) as {
   477→   477→      totalRecords?: number;
   478→   478→      opportunitiesData?: Array<Record<string, unknown>>;
   479→   479→    };
   480→   480→    const opps = json.opportunitiesData || [];
   481→   481→    const total = json.totalRecords || opps.length;
   482→   482→
   483→   483→    const tenders: LiveTender[] = opps.map((opp, idx) => {
   484→   484→      const title = String(opp.title || opp.sol || `SAM.gov Opportunity ${idx + 1}`);
   485→   485→      const amount = Number(opp.awardAmount || opp.estimatedValue || 0) || 0;
   486→   486→      return {
   487→   487→        id: `sam_gov-${opp.noticeId || idx}`,
   488→   488→        title: truncate(title, 160),
   489→   489→        scope: truncate(String(opp.description || title), 400),
   490→   490→        budgetMin: amount,
   491→   491→        budgetMax: amount,
   492→   492→        deadline: String(opp.responseDeadLine || new Date(Date.now() + 30 * 86400000).toISOString()),
   493→   493→        location: String(opp.placeOfPerformance || 'United States'),
   494→   494→        categoryTags: String(opp.naicsCode || 'Consulting'),
   495→   495→        requiredDocs: '',
   496→   496→        status: 'open',
   497→   497→        createdBy: 'sam_gov',
   498→   498→        createdAt: String(opp.postedDate || new Date().toISOString()),
   499→   499→        updatedAt: String(opp.postedDate || new Date().toISOString()),
   500→   500→        source: 'sam_gov',
   501→   501→        externalId: String(opp.noticeId || idx),
   502→   502→        externalUrl: opp.uiLink ? String(opp.uiLink) : 'https://sam.gov',
   503→   503→        currency: 'USD',
   504→   504→        region: 'North America',
   505→   505→      } satisfies LiveTender;
   506→   506→    });
   507→   507→
   508→   508→    return { tenders, total, ok: true };
   509→   509→  } catch {
   510→   510→    clearTimeout(timer);
   511→   511→    return { tenders: [], total: 0, ok: false };
   512→   512→  }
   513→   513→}

   625→   625→/* ─────────────────────────────────────────────────────────────────────
   626→   626→ * Aggregator — called by the API route
   627→   627→ * ───────────────────────────────────────────────────────────────────── */
   628→   628→
   629→   629→export interface FetchLiveTendersResult {
   630→   630→  tenders: LiveTender[];
   631→   631→  meta: {
   632→   632→    total: number;
   633→   633→    sources: { id: string; name: string; live: boolean; ok: boolean; count: number }[];
   634→   634→    fallback: boolean;
   635→   635→    cachedAt: number;
   636→   636→  };
   637→   637→}
   638→   638→
   639→   639→const CACHE_TTL_MS = 5 * 60 * 1000; // 5 minutes
   640→   640→interface CacheEntry {
   641→   641→  result: FetchLiveTendersResult;
   642→   642→  expiresAt: number;
   643→   643→}
   644→   644→const cache = new Map<string, CacheEntry>();
   645→   645→
   646→   646→export async function fetchLiveTenders(opts: {
   647→   647→  source?: string;
   648→   648→  search?: string;
   649→   649→  rows?: number;
   650→   650→}): Promise<FetchLiveTendersResult> {
   651→   651→  const cacheKey = `${opts.source || 'all'}::${opts.search || ''}::${opts.rows || 20}`;
   652→   652→  const hit = cache.get(cacheKey);
   653→   653→  if (hit && hit.expiresAt > Date.now()) {
   654→   654→    return { ...hit.result, meta: { ...hit.result.meta, cachedAt: Date.now() } };
   655→   655→  }
   656→   656→
   657→   657→  const rows = opts.rows ?? 20;
   658→   658→  const wantSource = opts.source || 'all';
   659→   659→  const tasks: { id: string; name: string; live: boolean; p: Promise<{ tenders: LiveTender[]; total: number; ok: boolean }> }[] = [];
   660→   660→
   661→   661→  if (wantSource === 'all' || wantSource === 'worldbank') {
   662→   662→    tasks.push({
   663→   663→      id: 'worldbank',
   664→   664→      name: 'World Bank',
   665→   665→      live: true,
   666→   666→      p: fetchWorldBankTenders({ search: opts.search, rows }),
   667→   667→    });
   668→   668→  }
   669→   669→  if (wantSource === 'all' || wantSource === 'eu_ted') {
   670→   670→    tasks.push({
   671→   671→      id: 'eu_ted',
   672→   672→      name: 'EU TED',
   673→   673→      live: true,
   674→   674→      p: fetchEuTedTenders({ search: opts.search, rows }),
   675→   675→    });
   676→   676→  }
   677→   677→  if (wantSource === 'all' || wantSource === 'ungm') {
   678→   678→    tasks.push({
   679→   679→      id: 'ungm',
   680→   680→      name: 'UNGM',
   681→   681→      live: true,
   682→   682→      p: fetchUngmTenders({ search: opts.search, rows: Math.min(rows, 5) }),
   683→   683→    });
   684→   684→  }
   685→   685→  if (wantSource === 'all' || wantSource === 'sam_gov') {
   686→   686→    tasks.push({
   687→   687→      id: 'sam_gov',
   688→   688→      name: 'SAM.gov',
   689→   689→      live: true,
   690→   690→      p: fetchSamGovTenders({ search: opts.search, rows: Math.min(rows, 5) }),
   691→   691→    });
   692→   692→  }
   693→   693→  if (wantSource === 'all' || wantSource === 'afdb') {
   694→   694→    tasks.push({
   695→   695→      id: 'afdb',
   696→   696→      name: 'AfDB',
   697→   697→      live: true,
   698→   698→      p: fetchAfdbTenders({ search: opts.search, rows: Math.min(rows, 5) }),
   699→   699→    });
   700→   700→  }
   701→   701→  if (wantSource === 'all' || wantSource === 'eu_opentenders') {
   702→   702→    tasks.push({
   703→   703→      id: 'eu_opentenders',
   704→   704→      name: 'OpenTenders EU',
   705→   705→      live: true,
   706→   706→      p: fetchEuOpenTenders({ search: opts.search, rows: Math.min(rows, 5) }),
   707→   707→    });
   708→   708→  }
   709→   709→  if (wantSource === 'all' || wantSource === 'jica') {
   710→   710→    tasks.push({
   711→   711→      id: 'jica',
   712→   712→      name: 'JICA',
   713→   713→      live: true,
   714→   714→      p: fetchJicaTenders({ search: opts.search, rows: Math.min(rows, 5) }),
   715→   715→    });
   716→   716→  }
   717→   717→  if (wantSource === 'all' || wantSource === 'adb') {
   718→   718→    tasks.push({
   719→   719→      id: 'adb',
   720→   720→      name: 'ADB',
   721→   721→      live: true,
   722→   722→      p: fetchAdbTenders({ search: opts.search, rows: Math.min(rows, 5) }),
   723→   723→    });
   724→   724→  }
   725→   725→  if (wantSource === 'all' || wantSource === 'uk_contracts') {
   726→   726→    tasks.push({
   727→   727→      id: 'uk_contracts',
   728→   728→      name: 'UK Contracts Finder',
   729→   729→      live: true,
   730→   730→      p: fetchUkContractsTenders({ search: opts.search, rows: Math.min(rows, 5) }),
   731→   731→    });
   732→   732→  }
   733→   733→  if (wantSource === 'all' || wantSource === 'dgmarket') {
   734→   734→    tasks.push({
   735→   735→      id: 'dgmarket',
   736→   736→      name: 'DgMarket',
   737→   737→      live: true,
   738→   738→      p: fetchDgMarketTenders({ search: opts.search, rows: Math.min(rows, 5) }),
   739→   739→    });
   740→   740→  }
   741→   741→
   742→   742→  const settled = await Promise.all(tasks.map(async (t) => ({ ...t, res: await t.p })));
   743→   743→
   744→   744→  const sourcesMeta = settled.map((t) => ({
   745→   745→    id: t.id,
   746→   746→    name: t.name,
   747→   747→    live: t.live,
   748→   748→    ok: t.res.ok,
   749→   749→    count: t.res.tenders.length,
   750→   750→  }));
   751→   751→
   752→   752→  const tenders = settled.flatMap((t) => t.res.tenders);
   753→   753→  const totalLiveOk = settled.some((t) => t.res.ok && t.res.tenders.length > 0);
   754→   754→  const fallback = !totalLiveOk;
   755→   755→
   756→   756→  // Client-side search filter on top of upstream results
   757→   757→  let filteredTenders = tenders;
   758→   758→  if (opts.search) {
   759→   759→    const q = opts.search.toLowerCase();
   760→   760→    filteredTenders = tenders.filter(
   761→   761→      (t) =>
   762→   762→        t.title.toLowerCase().includes(q) ||
   763→   763→        t.scope.toLowerCase().includes(q) ||
   764→   764→        t.location.toLowerCase().includes(q) ||
   765→   765→        t.categoryTags.toLowerCase().includes(q),
   766→   766→    );
   767→   767→  }
   768→   768→
   769→   769→  const result: FetchLiveTendersResult = {
   770→   770→    tenders: filteredTenders,
   771→   771→    meta: {
   772→   772→      total: filteredTenders.length,
   773→   773→      sources: sourcesMeta,
   774→   774→      fallback,
   775→   775→      cachedAt: Date.now(),
   776→   776→    },
   777→   777→  };
   778→   778→
   779→   779→  cache.set(cacheKey, { result, expiresAt: Date.now() + CACHE_TTL_MS });
   780→   780→  return result;
   781→   781→}
