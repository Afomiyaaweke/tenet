     1→'use client';
     2→

     4→import { useAuthStore, useNavStore } from '@/store';
     5→import { api, Bid, BidDocument } from '@/lib/api';
     6→import { Card, CardContent } from '@/components/ui/card';
     7→import { Badge } from '@/components/ui/badge';
     8→import { Button } from '@/components/ui/button';
     9→import { Dialog, DialogContent, DialogHeader, DialogTitle } from '@/components/ui/dialog';
    10→import { toast } from 'sonner';
    11→import {
    12→  Gavel, Clock, DollarSign, FileSearch, Award, AlertCircle,
    13→  CheckCircle, ChevronDown, ChevronUp, ArrowRight, TrendingUp,
    14→  Briefcase, X, Eye, RotateCcw, Filter, Target,
    15→  CircleDot, Building2, FileSignature, Stamp, Sparkles, Languages,
    16→  Bookmark, MapPin, Calendar, ExternalLink, Trash2, PenLine,
    17→  ScanSearch, Brain, Loader2, AlertTriangle, Upload, FileText,
    18→  CloudUpload, FileUp, ThumbsUp, ThumbsDown, AlertOctagon,

    20→} from 'lucide-react';

    23→
    24→type BidTab = 'all' | 'pending_review' | 'shortlisted' | 'awarded' | 'rejected' | 'saved';
    25→
    26→interface AIReviewData {
    27→  complianceScore?: number;
    28→  completenessScore?: number;
    29→  riskLevel?: string;
    30→  findings?: Array<{ type: string; title?: string; description: string; category?: string; severity?: string }>;
    31→  strengths?: string[];
    32→  weaknesses?: string[];
    33→  missingElements?: string[];
    34→  recommendations?: string[];
    35→  summary?: string;
    36→  overallAssessment?: string;
    37→}
    38→
    39→export function BidsView() {
    40→  const { user } = useAuthStore();
    41→  const { setView } = useNavStore();
    42→  const [bids, setBids] = useState<Bid[]>([]);
    43→  const [loading, setLoading] = useState(true);
    44→  const [statusFilter, setStatusFilter] = useState('');
    45→  const [expandedId, setExpandedId] = useState<string | null>(null);
    46→  const [activeTab, setActiveTab] = useState<BidTab>('all');
    47→  const [stampSelectorOpen, setStampSelectorOpen] = useState(false);
    48→  const [selectedBidId, setSelectedBidId] = useState<string | null>(null);
    49→  const [bidSignatures, setBidSignatures] = useState<Record<string, SavedSignature>>({});
    50→  const stampSigHook = useStampSignature();
    51→  const [visibleCount, setVisibleCount] = useState(10);
    52→  const BID_PAGE_SIZE = 10;
    53→
    54→  // Document/OCR/AI state
    55→  const [ocrLoading, setOcrLoading] = useState<Set<string>>(new Set());
    56→  const [reviewLoading, setReviewLoading] = useState<Set<string>>(new Set());
    57→  const [docOcrText, setDocOcrText] = useState<Record<string, string>>({});
    58→  const [docReview, setDocReview] = useState<Record<string, AIReviewData>>({});
    59→  const [expandedDocId, setExpandedDocId] = useState<string | null>(null);
    60→  const [reviewDialogOpen, setReviewDialogOpen] = useState(false);
    61→  const [reviewDialogDocId, setReviewDialogDocId] = useState<string | null>(null);
    62→  const [docUploadBidId, setDocUploadBidId] = useState<string | null>(null);
    63→  const docFileRef = useRef<HTMLInputElement>(null);
    64→  const [docUploadType, setDocUploadType] = useState('bid_attachment');
    65→
    66→  // Saved tenders state
    67→  const [savedTenders, setSavedTenders] = useState<any[]>([]);
    68→  const [savedLoading, setSavedLoading] = useState(false);
    69→
    70→  const loadSavedTenders = useCallback(async () => {
    71→    setSavedLoading(true);
    72→    const res = await api.get('/tenders/saved');
    73→    if (res.success) setSavedTenders(res.data);
    74→    setSavedLoading(false);
    75→  }, []);
    76→
    77→  const handleRemoveSaved = async (id: string) => {
    78→    const res = await api.delete(`/tenders/saved/${id}`);
    79→    if (res.success) {
    80→      toast.success('Removed from saved tenders');
    81→      loadSavedTenders();
    82→    } else {
    83→      toast.error('Failed to remove');
    84→    }
    85→  };
    86→
    87→  const loadBids = useCallback(async () => {
    88→    setLoading(true);
    89→    const params: Record<string, string> = {};
    90→    if (statusFilter && statusFilter !== 'all') params.status = statusFilter;
    91→    const res = await api.get('/bids', params);
    92→    if (res.success) setBids(res.data);
    93→    setLoading(false);
    94→  }, [statusFilter]);
    95→
    96→  useEffect(() => { loadBids(); loadSavedTenders(); }, [loadBids, loadSavedTenders]);
    97→
    98→  const handleStatusUpdate = async (bidId: string, status: string) => {
    99→    const res = await api.patch(`/bids/${bidId}/status`, { status });
   100→    if (res.success) {
   101→      toast.success(`Bid ${status}`);
   102→      loadBids();
   103→    } else {
   104→      toast.error(res.error || 'Failed to update bid');
   105→    }
   106→  };
   107→
   108→  // ── Document Upload ──
   109→  const handleDocUpload = useCallback(async (bidId: string) => {
   110→    const file = docFileRef.current?.files?.[0];
   111→    if (!file) return;
   112→    const formData = new FormData();
   113→    formData.append('file', file);
   114→    formData.append('docType', docUploadType);
   115→    try {
   116→      const res = await api.upload(`/bids/${bidId}/documents`, formData);
   117→      if (res.success) {
   118→        toast.success('Document uploaded');
   119→        if (docFileRef.current) docFileRef.current.value = '';
   120→        loadBids();
   121→      } else {
   122→        toast.error(res.error || 'Upload failed');
   123→      }
   124→    } catch {
   125→      toast.error('Upload failed');
   126→    } finally {
   127→      setDocUploadBidId(null);
   128→    }
   129→  }, [docUploadType, loadBids]);
   130→
   131→  // ── OCR Processing ──
   132→  const handleRunOcr = useCallback(async (docId: string) => {
   133→    setOcrLoading(prev => new Set(prev).add(docId));
   134→    try {
   135→      await api.post(`/document-ocr/${docId}`);
   136→      const poll = async (attempts = 0): Promise<void> => {
   137→        if (attempts > 30) {
   138→          toast.error('OCR processing timed out');
   139→          setOcrLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   140→          return;
   141→        }
   142→        await new Promise(r => setTimeout(r, 2000));
   143→        const res = await api.get(`/document-ocr/${docId}`);
   144→        if (res.success && res.data?.ocrStatus === 'completed') {
   145→          setOcrLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   146→          setDocOcrText(prev => ({ ...prev, [docId]: res.data.ocrText || '' }));
   147→          loadBids();
   148→          toast.success('OCR completed');
   149→        } else if (res.success && res.data?.ocrStatus === 'failed') {
   150→          setOcrLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   151→          toast.error('OCR processing failed');
   152→          loadBids();
   153→        } else {
   154→          await poll(attempts + 1);
   155→        }
   156→      };
   157→      poll();
   158→    } catch {
   159→      setOcrLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   160→      toast.error('Failed to start OCR');
   161→    }
   162→  }, [loadBids]);
   163→
   164→  // ── AI Review Processing ──
   165→  const handleRunReview = useCallback(async (docId: string) => {
   166→    setReviewLoading(prev => new Set(prev).add(docId));
   167→    try {
   168→      const res = await api.post(`/document-review/${docId}`);
   169→      if (!res.success && res.error?.includes('OCR must be completed')) {
   170→        setReviewLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   171→        toast.error('Run OCR first before AI review');
   172→        return;
   173→      }
   174→      const poll = async (attempts = 0): Promise<void> => {
   175→        if (attempts > 30) {
   176→          toast.error('AI Review processing timed out');
   177→          setReviewLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   178→          return;
   179→        }
   180→        await new Promise(r => setTimeout(r, 2000));
   181→        const pollRes = await api.get(`/document-review/${docId}`);
   182→        if (pollRes.success && pollRes.data?.aiReviewStatus === 'completed') {
   183→          setReviewLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   184→          let reviewData = pollRes.data.aiReview || {};
   185→          if (typeof reviewData === 'string') {
   186→            try { reviewData = JSON.parse(reviewData); } catch { reviewData = { summary: reviewData }; }
   187→          }
   188→          setDocReview(prev => ({ ...prev, [docId]: reviewData as AIReviewData }));
   189→          loadBids();
   190→          toast.success('AI Review completed');
   191→        } else if (pollRes.success && pollRes.data?.aiReviewStatus === 'failed') {
   192→          setReviewLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   193→          toast.error('AI Review processing failed');
   194→          loadBids();
   195→        } else {
   196→          await poll(attempts + 1);
   197→        }
   198→      };
   199→      poll();
   200→    } catch {
   201→      setReviewLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   202→      toast.error('Failed to start AI Review');
   203→    }
   204→  }, [loadBids]);
   205→
   206→  // ── View OCR text inline ──
   207→  const handleViewOcr = useCallback(async (docId: string) => {
   208→    if (expandedDocId === docId) {
   209→      setExpandedDocId(null);
   210→      return;
   211→    }
   212→    setExpandedDocId(docId);
   213→    if (!docOcrText[docId]) {
   214→      const res = await api.get(`/document-ocr/${docId}`);
   215→      if (res.success) {
   216→        setDocOcrText(prev => ({ ...prev, [docId]: res.data?.ocrText || '' }));
   217→      }
   218→    }
   219→  }, [expandedDocId, docOcrText]);
   220→
   221→  const stats = useMemo(() => ({
   222→    pending: bids.filter(b => b.status === 'pending_review').length,
   223→    shortlisted: bids.filter(b => b.status === 'shortlisted').length,
   224→    awarded: bids.filter(b => b.status === 'awarded').length,
   225→    rejected: bids.filter(b => b.status === 'rejected').length,
   226→    total: bids.length,
   227→  }), [bids]);
   228→
   229→  const filteredBids = useMemo(() => {
   230→    if (activeTab === 'saved') return [] as Bid[];
   231→    if (activeTab === 'all') return bids;
   232→    return bids.filter(b => b.status === activeTab);
   233→  }, [bids, activeTab]);
   234→
   235→  const statusBadge = (status: string) => {
   236→    switch (status) {
   237→      case 'pending_review': return 'bg-amber-100 text-amber-700 hover:bg-amber-100';
   238→      case 'shortlisted': return 'bg-teal-100 text-teal-700 hover:bg-teal-100';
   239→      case 'awarded': return 'bg-emerald-100 text-emerald-700 hover:bg-emerald-100';
   240→      case 'rejected': return 'bg-rose-100 text-rose-700 hover:bg-rose-100';
   241→      default: return 'bg-muted text-muted-foreground hover:bg-muted';
   242→    }
   243→  };
   244→
   245→  const statusIcon = (status: string) => {
   246→    switch (status) {
   247→      case 'pending_review': return { icon: Clock, bg: 'bg-amber-50', color: 'text-amber-600' };
   248→      case 'shortlisted': return { icon: Award, bg: 'bg-teal-50', color: 'text-teal-600' };
   249→      case 'awarded': return { icon: CheckCircle, bg: 'bg-emerald-50', color: 'text-emerald-600' };
   250→      case 'rejected': return { icon: AlertCircle, bg: 'bg-rose-50', color: 'text-rose-600' };
   251→      default: return { icon: Gavel, bg: 'bg-muted/50', color: 'text-muted-foreground' };
   252→    }
   253→  };
   254→
   255→  const isAdminOrOwner = user?.role === 'team_admin';
   256→
   257→  const getScoreColor = (score: number) => {
   258→    if (score >= 80) return 'text-emerald-600';
   259→    if (score >= 60) return 'text-amber-600';
   260→    return 'text-rose-600';
   261→  };
   262→
   263→  const getScoreBg = (score: number) => {
   264→    if (score >= 80) return 'bg-emerald-50';
   265→    if (score >= 60) return 'bg-amber-50';
   266→    return 'bg-rose-50';
   267→  };
   268→
   269→  const getRiskColor = (risk: string) => {
   270→    switch (risk?.toLowerCase()) {
   271→      case 'low': return 'bg-emerald-100 text-emerald-700';
   272→      case 'medium': return 'bg-amber-100 text-amber-700';
   273→      case 'high': case 'critical': return 'bg-rose-100 text-rose-700';
   274→      default: return 'bg-gray-100 text-gray-700';
   275→    }
   276→  };
   277→
   278→  const tabs: { key: BidTab; label: string; icon: typeof Clock; count: number; color: string }[] = [
   279→    { key: 'all', label: 'All', icon: Gavel, count: stats.total, color: 'emerald' },
   280→    { key: 'pending_review', label: 'Pending', icon: Clock, count: stats.pending, color: 'amber' },
   281→    { key: 'shortlisted', label: 'Shortlisted', icon: Award, count: stats.shortlisted, color: 'teal' },
   282→    { key: 'awarded', label: 'Awarded', icon: CheckCircle, count: stats.awarded, color: 'emerald' },
   283→    { key: 'rejected', label: 'Rejected', icon: AlertCircle, count: stats.rejected, color: 'rose' },
   284→    { key: 'saved', label: 'Saved Tenders', icon: Bookmark, count: savedTenders.length, color: 'violet' },
   285→  ];
   286→
   287→  // Review dialog data
   288→  const reviewDialogData = reviewDialogDocId ? docReview[reviewDialogDocId] : null;
   289→
   290→  return (
   291→    <div className="p-4 md:p-6 space-y-6 max-w-5xl mx-auto view-enter">
   292→      {/* Header */}
   293→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4 animate-[fadeIn_0.3s_ease-out]">
   294→        <div className="flex items-center gap-4">
   295→          <div className="p-3 rounded-2xl gradient-amber shadow-md flex-shrink-0">
   296→            <Gavel className="h-6 w-6 text-white" />
   297→          </div>
   298→          <div>
   299→            <h2 className="text-2xl md:text-3xl font-bold tracking-tight">
   300→              <span className="text-gradient-emerald">{isAdminOrOwner ? 'Review' : 'My'}</span> Bids
   301→            </h2>
   302→            <p className="text-muted-foreground text-sm mt-0.5">
   303→              {isAdminOrOwner ? 'Manage and evaluate submitted bids' : 'Track your bid submissions'}
   304→            </p>
   305→          </div>
   306→        </div>
   307→      </div>
   308→
   309→      {/* Stats Summary */}
   310→      {!loading && (bids.length > 0 || savedTenders.length > 0) && (
   311→        <div className="grid grid-cols-2 sm:grid-cols-4 gap-4 animate-[fadeIn_0.3s_ease-out]">
   312→          {[
   313→            { label: 'Pending', count: stats.pending, icon: Clock, bg: 'bg-amber-50', color: 'text-amber-600' },
   314→            { label: 'Shortlisted', count: stats.shortlisted, icon: Award, bg: 'bg-teal-50', color: 'text-teal-600' },
   315→            { label: 'Awarded', count: stats.awarded, icon: CheckCircle, bg: 'bg-emerald-50', color: 'text-emerald-600' },
   316→            { label: 'Rejected', count: stats.rejected, icon: AlertCircle, bg: 'bg-rose-50', color: 'text-rose-600' },
   317→          ].map(stat => (
   318→            <div key={stat.label}>
   319→              <Card className="premium-shadow rounded-xl border-0 bg-card hover:-translate-y-0.5 transition-all duration-200">
   320→                <CardContent className="p-4 flex items-center gap-3">
   321→                  <div className={`p-2 rounded-lg ${stat.bg} flex-shrink-0`}>
   322→                    <stat.icon className={`h-4 w-4 ${stat.color}`} />
   323→                  </div>
   324→                  <div>
   325→                    <p className="text-xl font-bold">{stat.count}</p>
   326→                    <p className="text-xs text-muted-foreground">{stat.label}</p>
   327→                  </div>
   328→                </CardContent>
   329→              </Card>
   330→            </div>
   331→          ))}
   332→        </div>
   333→      )}
   334→
   335→      {/* Tab Navigation */}
   336→      {!loading && (
   337→        <div className="animate-[fadeIn_0.3s_ease-out]">
   338→          <Card className="premium-shadow rounded-xl border-0 bg-card">
   339→            <CardContent className="p-1.5">
   340→              <div className="flex gap-1 overflow-x-auto">
   341→                {tabs.map(tab => (
   342→                  <button
   343→                    key={tab.key}
   344→                    onClick={() => setActiveTab(tab.key)}
   345→                    className={`flex items-center gap-1.5 px-3 py-2 rounded-lg text-sm font-medium transition-all duration-200 whitespace-nowrap ${
   346→                      activeTab === tab.key
   347→                        ? 'gradient-emerald text-white premium-shadow'
   348→                        : 'text-muted-foreground hover:bg-muted/60 hover:text-foreground'
   349→                    }`}
   350→                  >
   351→                    <tab.icon className="h-3.5 w-3.5" />
   352→                    <span>{tab.label}</span>
   353→                    <span className={`text-[10px] px-1.5 py-0 rounded-full ${
   354→                      activeTab === tab.key
   355→                        ? 'bg-white/20 text-white'
   356→                        : 'bg-muted text-muted-foreground'
   357→                    }`}>
   358→                      {tab.count}
   359→                    </span>
   360→                  </button>
   361→                ))}
   362→              </div>
   363→            </CardContent>
   364→          </Card>
   365→        </div>
   366→      )}
   367→
   368→      {/* Saved Tenders Tab */}
   369→      {activeTab === 'saved' && (
   370→        <div className="space-y-4 animate-[fadeIn_0.3s_ease-out]">
   371→          {savedLoading ? (
   372→            [1, 2, 3].map(i => (
   373→              <Card key={i} className="premium-shadow rounded-xl border-0 bg-card animate-pulse">
   374→                <CardContent className="p-5"><div className="h-20 bg-muted/50 rounded-xl" /></CardContent>
   375→              </Card>
   376→            ))
   377→          ) : savedTenders.length === 0 ? (
   378→            <Card className="premium-shadow rounded-xl border-0 bg-card">
   379→              <CardContent className="p-12 text-center">
   380→                <div className="p-3 rounded-2xl bg-violet-50 dark:bg-violet-950/30 w-fit mx-auto mb-4">
   381→                  <Bookmark className="h-8 w-8 text-violet-500" />
   382→                </div>
   383→                <h3 className="text-lg font-semibold">No saved tenders yet</h3>
   384→                <p className="text-muted-foreground text-sm mt-1 max-w-sm mx-auto">
   385→                  Bookmark tenders from Live Tenders to work on them later
   386→                </p>
   387→                <Button
   388→                  className="mt-4 gradient-emerald hover:opacity-90 text-white rounded-xl premium-shadow"
   389→                  onClick={() => setView('live-tenders')}
   390→                >
   391→                  <FileSearch className="h-4 w-4 mr-2" /> Browse Live Tenders
   392→                </Button>
   393→              </CardContent>
   394→            </Card>
   395→          ) : (
   396→            savedTenders.map((st: any) => (
   397→              <Card key={st.id} className="premium-shadow rounded-xl border-0 bg-card overflow-hidden hover:-translate-y-[2px] transition-all duration-200">
   398→                <div className="h-1 bg-gradient-to-r from-violet-400 to-purple-500" />
   399→                <CardContent className="p-5">
   400→                  <div className="flex items-start justify-between gap-3">
   401→                    <div className="flex-1 min-w-0">
   402→                      <div className="flex items-center gap-2 mb-1">
   403→                        <Badge variant="secondary" className="text-[10px] bg-violet-50 text-violet-700 dark:bg-violet-950/40 dark:text-violet-300">
   404→                          {st.source}
   405→                        </Badge>
   406→                        <Badge variant="secondary" className={`text-[10px] ${
   407→                          st.status === 'saved' ? 'bg-amber-50 text-amber-700' :
   408→                          st.status === 'bidding' ? 'bg-teal-50 text-teal-700' :
   409→                          st.status === 'applied' ? 'bg-emerald-50 text-emerald-700' :
   410→                          'bg-muted text-muted-foreground'
   411→                        }`}>
   412→                          {st.status}
   413→                        </Badge>
   414→                      </div>
   415→                      <h4 className="font-semibold text-sm line-clamp-2">{st.title}</h4>
   416→                      {st.scope && <p className="text-xs text-muted-foreground mt-1 line-clamp-2">{st.scope}</p>}
   417→                      <div className="flex flex-wrap gap-3 mt-2 text-xs text-muted-foreground">
   418→                        {(st.budgetMin > 0 || st.budgetMax > 0) && (
   419→                          <span className="flex items-center gap-1">
   420→                            <DollarSign className="h-3 w-3" />
   421→                            {st.currency || 'USD'} {st.budgetMin?.toLocaleString() || '—'} – {st.budgetMax?.toLocaleString() || '—'}
   422→                          </span>
   423→                        )}
   424→                        {st.location && (
   425→                          <span className="flex items-center gap-1">
   426→                            <MapPin className="h-3 w-3" /> {st.location}
   427→                          </span>
   428→                        )}
   429→                        {st.deadline && (
   430→                          <span className="flex items-center gap-1">
   431→                            <Calendar className="h-3 w-3" /> {new Date(st.deadline).toLocaleDateString()}
   432→                          </span>
   433→                        )}
   434→                      </div>
   435→                      {st.categoryTags && (
   436→                        <div className="flex flex-wrap gap-1 mt-2">
   437→                          {st.categoryTags.split(',').map((tag: string) => (
   438→                            <Badge key={tag} variant="secondary" className="text-[9px] px-1.5 py-0 bg-muted/50">
   439→                              {tag.trim()}
   440→                            </Badge>
   441→                          ))}
   442→                        </div>
   443→                      )}
   444→                      {st.notes && (
   445→                        <div className="mt-2 p-2 rounded-lg bg-muted/30 text-xs text-muted-foreground">
   446→                          <PenLine className="h-3 w-3 inline mr-1" /> {st.notes}
   447→                        </div>
   448→                      )}
   449→                    </div>
   450→                    <div className="flex flex-col gap-1 flex-shrink-0">
   451→                      {st.externalUrl && (
   452→                        <Button
   453→                          variant="ghost"
