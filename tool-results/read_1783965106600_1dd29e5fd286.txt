   307→
   308→function parseAmount(value: number | string | undefined): number {
   309→  if (value === undefined || value === null) return 0;
   310→  if (typeof value === 'number') return value;
   311→  const n = Number(String(value).replace(/[^0-9.-]/g, ''));
   312→  return Number.isFinite(n) ? n : 0;
   313→}
   314→
   315→function truncate(text: string, max = 280): string {
   316→  if (!text) return '';
   317→  return text.length > max ? text.slice(0, max).trimEnd() + '…' : text;
   318→}

   320→/* ─────────────────────────────────────────────────────────────────────
   321→ * World Bank adapter (live, public, no auth)
   322→ * ───────────────────────────────────────────────────────────────────── */
   323→
   324→interface WorldBankProcurementRow {
   325→  project_id?: string;
   326→  procurement_group_id?: string;
   327→  project_name?: string;
   328→  borrower?: string;
   329→  country?: string;
   330→  region?: string;
   331→  contract_description?: string;
   332→  contract_type?: string;
   333→  supplier?: string;
   334→  supplier_country?: string | string[];
   335→  contract_signing_date?: string;
   336→  total_contract_amount?: number | string;
   337→  wb_contract_number?: string;
   338→}
   339→
   340→interface WorldBankResponse {
   341→  total?: number;
   342→  rows?: WorldBankProcurementRow[];
   343→  procurements?: WorldBankProcurementRow[];
   344→  [k: string]: unknown;
   345→}
   346→
   347→function inferCategory(row: WorldBankProcurementRow): string {
   348→  const t = (row.contract_type || '').toLowerCase();
   349→  if (t.includes('work') || t.includes('civil') || t.includes('construction')) return 'Construction';
   350→  if (t.includes('consult')) return 'Consulting';
   351→  if (t.includes('good') || t.includes('supply') || t.includes('equipment')) return 'Supply';
   352→  if (t.includes('service')) return 'Consulting';
   353→  const p = (row.project_name || '').toLowerCase();
   354→  if (p.includes('road') || p.includes('bridge') || p.includes('infrastructure')) return 'Construction';
   355→  if (p.includes('agric') || p.includes('farm') || p.includes('irrigat')) return 'Agriculture';
   356→  if (p.includes('health') || p.includes('clinic') || p.includes('hospital')) return 'Healthcare';
   357→  if (p.includes('energy') || p.includes('power') || p.includes('electric')) return 'Engineering';
   358→  if (p.includes('edu') || p.includes('school')) return 'Education';
   359→  if (p.includes('water') || p.includes('sanitation')) return 'Engineering';
   360→  return 'Consulting';
   361→}
   362→
   363→function buildDeadline(signingDate?: string): string {
   364→  if (!signingDate) return new Date(Date.now() + 45 * 86400000).toISOString();
   365→  const base = new Date(signingDate);
   366→  if (Number.isNaN(base.getTime())) return new Date(Date.now() + 45 * 86400000).toISOString();
   367→  return new Date(base.getTime() + 45 * 86400000).toISOString();
   368→}

   370→export async function fetchWorldBankTenders(opts: {
   371→  search?: string;
   372→  rows?: number;
   373→  offset?: number;
   374→}): Promise<{ tenders: LiveTender[]; total: number; ok: boolean }> {
   375→  const rows = Math.min(Math.max(opts.rows ?? 20, 1), 500);
   376→  const offset = opts.offset || 0;
   377→  const params = new URLSearchParams({
   378→    format: 'json',
   379→    rows: String(rows),
   380→    fl: [
   381→      'project_id',
   382→      'procurement_group_id',
   383→      'project_name',
   384→      'borrower',
   385→      'country',
   386→      'region',
   387→      'contract_description',
   388→      'contract_type',
   389→      'supplier',
   390→      'supplier_country',
   391→      'contract_signing_date',
   392→      'total_contract_amount',
   393→      'wb_contract_number',
   394→    ].join(','),
   395→  });
   396→  if (opts.search) params.set('q', opts.search);
   397→  if (offset > 0) params.set('osstart', String(offset + 1));
   398→
   399→  const url = `https://search.worldbank.org/api/v2/procurement?${params.toString()}`;
   400→  const ctrl = new AbortController();
   401→  const timer = setTimeout(() => ctrl.abort(), 8000);
   402→
   403→  try {
   404→    const res = await fetch(url, {
   405→      signal: ctrl.signal,
   406→      headers: { Accept: 'application/json', 'User-Agent': 'Tenet-Tender-Ecosystem/1.0' },
   407→      cache: 'no-store',
   408→    });
   409→    clearTimeout(timer);
   410→    if (!res.ok) {
   411→      return { tenders: [], total: 0, ok: false };
   412→    }
   413→    const json = (await res.json()) as WorldBankResponse;
   414→    const rowsArr = json.rows || json.procurements || [];
   415→    const total = typeof json.total === 'number' ? json.total : rowsArr.length;
   416→
   417→    const tenders: LiveTender[] = rowsArr
   418→      .filter((r) => r && (r.contract_description || r.project_name))
   419→      .map((r, idx) => {
   420→        const externalId = r.procurement_group_id || r.project_id || `wb-${idx}`;
   421→        const title = truncate(r.contract_description || r.project_name || 'World Bank Contract Award', 160);
   422→        const amount = parseAmount(r.total_contract_amount);
   423→        const signingDate = r.contract_signing_date;
   424→        return {
   425→          id: `worldbank-${externalId}`,
   426→          title,
   427→          scope: truncate(r.contract_description || r.project_name || '', 400),
   428→          budgetMin: amount,
   429→          budgetMax: amount,
   430→          deadline: buildDeadline(signingDate),
   431→          location: r.country || r.region || 'Multiple',
   432→          categoryTags: inferCategory(r),
   433→          requiredDocs: r.project_id
   434→            ? `https://projects.worldbank.org/en/projects-operations/project-detail/${r.project_id}`
   435→            : 'https://financesone.worldbank.org',
   436→          status: 'awarded',
   437→          createdBy: 'worldbank',
   438→          createdAt: signingDate || new Date().toISOString(),
   439→          updatedAt: signingDate || new Date().toISOString(),
   440→          source: 'worldbank',
   441→          externalId,
   442→          externalUrl: r.project_id
   443→            ? `https://projects.worldbank.org/en/projects-operations/project-detail/${r.project_id}`
   444→            : 'https://financesone.worldbank.org',
   445→          currency: 'USD',
   446→          borrower: r.borrower || undefined,
   447→          supplier: r.supplier || undefined,
   448→          contractType: r.contract_type || undefined,
   449→          signingDate: signingDate || undefined,
   450→          region: r.region || undefined,
   451→          documentUrl: r.project_id ? `https://projects.worldbank.org/en/projects-operations/project-detail/${r.project_id}` : undefined,
   452→        } satisfies LiveTender;
   453→      });
   454→
   455→    return { tenders, total, ok: true };
   456→  } catch {
   457→    clearTimeout(timer);
   458→    return { tenders: [], total: 0, ok: false };
   459→  }
   460→}

   462→/* ─────────────────────────────────────────────────────────────────────
   463→ * EU TED adapter (live, public RSS-style feed)
   464→ * ───────────────────────────────────────────────────────────────────── */
   465→
   466→export async function fetchEuTedTenders(opts: {
   467→  search?: string;
   468→  rows?: number;
   469→  offset?: number;
   470→}): Promise<{ tenders: LiveTender[]; total: number; ok: boolean }> {
   471→  const rows = Math.min(Math.max(opts.rows ?? 20, 1), 500);
   472→  const offset = opts.offset || 0;
   473→  const base = 'https://api.ted.europa.eu/v3/notices/search';
   474→  const body = {
   475→    query: opts.search ? `(${opts.search})` : '*',
   476→    fields: ['ND', 'TI_DOC', 'DS_DATE_PUB', 'DEADLINE_DATE', 'PLACE_CONTRACT', 'VALUE_CONTRACT', 'CA_NAME', 'TITLE'],
   477→    limit: rows,
   478→    offset,
   479→    pagination: { pageNumber: Math.floor(offset / rows) + 1, pageSize: rows },
   480→    sort: { field: 'DS_DATE_PUB', order: 'DESC' },
   481→  };
   482→  const ctrl = new AbortController();
   483→  const timer = setTimeout(() => ctrl.abort(), 8000);
   484→
   485→  try {
   486→    const res = await fetch(base, {
   487→      method: 'POST',
   488→      signal: ctrl.signal,
   489→      headers: { 'Content-Type': 'application/json', Accept: 'application/json' },
   490→      body: JSON.stringify(body),
   491→      cache: 'no-store',
   492→    });
   493→    clearTimeout(timer);
   494→    if (!res.ok) return { tenders: [], total: 0, ok: false };
   495→    const json = (await res.json()) as { notices?: unknown[]; total?: number };
   496→    const notices = Array.isArray(json.notices) ? json.notices : [];
   497→    const total = typeof json.total === 'number' ? json.total : notices.length;
   498→
   499→    const tenders: LiveTender[] = notices.map((n, idx) => {
   500→      const row = (n || {}) as Record<string, unknown>;
   501→      const nd = String(row.ND || row['ND'] || `ted-${idx}`);
   502→      const title = String(row.TI_DOC || row.TITLE || 'EU TED Notice');
   503→      const deadlineRaw = row.DEADLINE_DATE as string | undefined;
   504→      const amount = Number(row.VALUE_CONTRACT) || 0;
   505→      return {
   506→        id: `eu_ted-${nd}`,
   507→        title: truncate(title, 160),
   508→        scope: truncate(title, 400),
   509→        budgetMin: amount,
   510→        budgetMax: amount,
   511→        deadline: deadlineRaw || new Date(Date.now() + 30 * 86400000).toISOString(),
   512→        location: String(row.PLACE_CONTRACT || 'EU'),
   513→        categoryTags: 'Consulting',
   514→        requiredDocs: `https://ted.europa.eu/udl?uri=TED:NOTICE:${nd}:TEXT:EN:HTML`,
   515→        status: 'open',
   516→        createdBy: 'eu_ted',
   517→        createdAt: String(row.DS_DATE_PUB || new Date().toISOString()),
   518→        updatedAt: String(row.DS_DATE_PUB || new Date().toISOString()),
   519→        source: 'eu_ted',
   520→        externalId: nd,
   521→        externalUrl: `https://ted.europa.eu/udl?uri=TED:NOTICE:${nd}:TEXT:EN:HTML`,
   522→        currency: 'EUR',
   523→        borrower: String(row.CA_NAME || '') || undefined,
   524→        documentUrl: `https://ted.europa.eu/udl?uri=TED:NOTICE:${nd}:TEXT:EN:HTML`,
   525→      } satisfies LiveTender;
   526→    });
   527→
   528→    return { tenders, total, ok: true };
   529→  } catch {
   530→    clearTimeout(timer);
   531→    return { tenders: [], total: 0, ok: false };
