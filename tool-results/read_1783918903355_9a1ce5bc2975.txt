   130→function getProfileCompleteness(profile: Record<string, unknown> | undefined | null): { pct: number; missing: string[] } {
   131→  if (!profile) return { pct: 0, missing: ['Full Name', 'Phone', 'Location', 'Skills', 'Bio'] };
   132→  const checks: [string, boolean][] = [
   133→    ['Full Name', !!(profile.fullName as string)],
   134→    ['Phone', !!(profile.phone as string)],
   135→    ['Location', !!(profile.location as string)],
   136→    ['Skills', !!(profile.skillTags as string)],
   137→    ['Bio', !!(profile.bio as string)],
   138→  ];
   139→  const completed = checks.filter(([, v]) => v).length;
   140→  const missing = checks.filter(([, v]) => !v).map(([label]) => label);
   141→  return { pct: Math.round((completed / checks.length) * 100), missing };
   142→}
