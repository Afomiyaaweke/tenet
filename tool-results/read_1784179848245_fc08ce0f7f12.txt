   200→function formatAIResultToHTML(data: Record<string, unknown>): string {
   201→  let html = '';
   202→  for (const [key, value] of Object.entries(data)) {
   203→    const title = key.replace(/([A-Z])/g, ' $1').replace(/^./, s => s.toUpperCase());
   204→    const content = typeof value === 'string' ? value : JSON.stringify(value, null, 2);
   205→    const formattedContent = content
   206→      .replace(/&/g, '&amp;')
   207→      .replace(/</g, '&lt;')
   208→      .replace(/>/g, '&gt;')
   209→      .replace(/\*\*(.+?)\*\*/g, '<strong>$1</strong>')
   210→      .replace(/^[-*•]\s+(.*)/gm, '<li>$1</li>')
   211→      .replace(/^(\d+)[.)]\s+(.*)/gm, '<li>$2</li>')
   212→      .replace(/^#{1,3}\s+(.*)/gm, '<strong>$1</strong>')
   213→      .replace(/\n\n/g, '</p><p>')
   214→      .replace(/\n/g, '<br/>');
   215→    html += `<h2 style="font-size:16px;font-weight:700;margin:16px 0 8px;color:#059669;border-bottom:1px solid #d1fae5;padding-bottom:4px;">${title}</h2>`;
   216→    html += `<p>${formattedContent}</p>`;
   217→  }
   218→  return html;
   219→}

   221→function getFileExtBadge(fileName: string): { color: string; label: string } {
   222→  const ext = fileName.toLowerCase().split('.').pop() || '';
   223→  const map: Record<string, { color: string; label: string }> = {
   224→    pdf: { color: 'bg-rose-100 text-rose-700', label: 'PDF' },
   225→    docx: { color: 'bg-blue-100 text-blue-700', label: 'DOCX' },
   226→    doc: { color: 'bg-blue-100 text-blue-700', label: 'DOC' },
   227→    txt: { color: 'bg-gray-100 text-gray-700', label: 'TXT' },
   228→    jpg: { color: 'bg-amber-100 text-amber-700', label: 'JPG' },
   229→    jpeg: { color: 'bg-amber-100 text-amber-700', label: 'JPEG' },
   230→    png: { color: 'bg-emerald-100 text-emerald-700', label: 'PNG' },
   231→  };
   232→  return map[ext] || { color: 'bg-gray-100 text-gray-600', label: ext.toUpperCase() };
   233→}

   241→function SkillTagSelector({ selected, onChange }: { selected: string[]; onChange: (s: string[]) => void }) {
   242→  const toggle = (skill: string) => {
   243→    if (selected.includes(skill)) onChange(selected.filter(s => s !== skill));
   244→    else onChange([...selected, skill]);
   245→  };
   246→  return (
   247→    <div className="flex flex-wrap gap-1">
   248→      {COMMON_SKILLS.map(skill => (
   249→        <button key={skill} type="button" onClick={() => toggle(skill)}
   250→          className={`px-2 py-0.5 rounded-full text-[10px] font-medium transition-all border ${
   251→            selected.includes(skill)
   252→              ? 'bg-emerald-50 text-emerald-700 border-emerald-300'
   253→              : 'bg-muted/50 text-muted-foreground border-border hover:border-emerald-300'
   254→          }`}>
   255→          {skill}
   256→        </button>
   257→      ))}
   258→    </div>
   259→  );
   260→}

   623→          }
