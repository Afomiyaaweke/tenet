   187→   187→function relativeTime(dateStr: string): string {
   188→   188→  const now = Date.now();
   189→   189→  const then = new Date(dateStr).getTime();
   190→   190→  const diff = Math.max(0, now - then);
   191→   191→  const seconds = Math.floor(diff / 1000);
   192→   192→  if (seconds < 60) return 'just now';
   193→   193→  const minutes = Math.floor(seconds / 60);
   194→   194→  if (minutes < 60) return `${minutes}m ago`;
   195→   195→  const hours = Math.floor(minutes / 60);
   196→   196→  if (hours < 24) return `${hours}h ago`;
   197→   197→  const days = Math.floor(hours / 24);
   198→   198→  if (days < 7) return `${days}d ago`;
   199→   199→  return new Date(dateStr).toLocaleDateString();
   200→   200→}
