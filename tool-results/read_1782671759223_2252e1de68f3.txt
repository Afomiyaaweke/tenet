    73→function formatSectionContent(content: string): React.ReactNode {
    74→  const lines = content.split('\n');
    75→  return lines.map((line, i) => {
    76→    const bulletMatch = line.match(/^[\s]*[-*•]\s+(.*)/);
    77→    if (bulletMatch) {
    78→      return (
    79→        <div key={i} className="flex items-start gap-2 ml-2">
    80→          <span className="text-emerald-500 mt-0.5 flex-shrink-0">•</span>
    81→          <span>{formatInline(bulletMatch[1])}</span>
    82→        </div>
    83→      );
    84→    }
    85→    const numberedMatch = line.match(/^[\s]*(\d+)[.)]\s+(.*)/);
    86→    if (numberedMatch) {
    87→      return (
    88→        <div key={i} className="flex items-start gap-2 ml-2">
    89→          <span className="text-emerald-600 font-semibold flex-shrink-0">{numberedMatch[1]}.</span>
    90→          <span>{formatInline(numberedMatch[2])}</span>
    91→        </div>
    92→      );
    93→    }
    94→    if (line.trim() === '') return <div key={i} className="h-2" />;
    95→    const headingMatch = line.match(/^#{1,3}\s+(.*)/);
    96→    if (headingMatch) {
    97→      return <div key={i} className="font-semibold text-foreground mt-2">{headingMatch[1]}</div>;
    98→    }
    99→    return <div key={i}>{formatInline(line)}</div>;
   100→  });
   101→}
   102→
   103→function formatInline(text: string): React.ReactNode {
   104→  const parts: React.ReactNode[] = [];
   105→  let remaining = text;
   106→  let key = 0;
   107→  while (remaining.length > 0) {
   108→    const boldMatch = remaining.match(/\*\*(.+?)\*\*/);
   109→    if (boldMatch && boldMatch.index !== undefined) {
   110→      if (boldMatch.index > 0) parts.push(<span key={`t-${key++}`}>{remaining.slice(0, boldMatch.index)}</span>);
   111→      parts.push(<strong key={`b-${key++}`} className="font-semibold text-foreground">{boldMatch[1]}</strong>);
   112→      remaining = remaining.slice(boldMatch.index + boldMatch[0].length);
   113→    } else {
   114→      parts.push(<span key={`t-${key++}`}>{remaining}</span>);
   115→      break;
   116→    }
   117→  }
   118→  return <>{parts}</>;
   119→}

   121→function scoreColor(score: number): string {
   122→  if (score >= 75) return 'text-emerald-600';
   123→  if (score >= 50) return 'text-amber-600';
   124→  return 'text-rose-600';
   125→}

   127→function scoreBg(score: number): string {
   128→  if (score >= 75) return 'bg-emerald-500';
   129→  if (score >= 50) return 'bg-amber-500';
   130→  return 'bg-rose-500';

   163→function SkillTagSelector({ selected, onChange }: { selected: string[]; onChange: (s: string[]) => void }) {
   164→  const toggle = (skill: string) => {
   165→    if (selected.includes(skill)) {
   166→      onChange(selected.filter(s => s !== skill));
   167→    } else {
   168→      onChange([...selected, skill]);
   169→    }
   170→  };
   171→  return (
   172→    <div className="flex flex-wrap gap-1.5">
   173→      {COMMON_SKILLS.map(skill => (
   174→        <button key={skill} type="button" onClick={() => toggle(skill)}
   175→          className={`px-2.5 py-1 rounded-full text-xs font-medium transition-all duration-150 border ${
   176→            selected.includes(skill)
   177→              ? 'bg-emerald-50 text-emerald-700 border-emerald-300 premium-shadow'
   178→              : 'bg-muted/50 text-muted-foreground border-border hover:border-emerald-300 hover:text-emerald-600'
   179→          }`}>
   180→          {skill}
   181→        </button>
   182→      ))}
   183→    </div>
   184→  );
