   296→   296→function fmtMoney(amount: number, currency: string): string {
   297→   297→  if (!amount) return '—';
   298→   298→  try {
   299→   299→    return new Intl.NumberFormat('en-US', {
   300→   300→      style: 'currency',
   301→   301→      currency: currency || 'USD',
   302→   302→      maximumFractionDigits: 0,
   303→   303→    }).format(amount);
   304→   304→  } catch {
   305→   305→    return `${currency} ${amount.toLocaleString()}`;
   306→   306→  }
   307→   307→}

   309→   309→function daysUntil(dateStr: string): number {
   310→   310→  if (!dateStr) return 999;
   311→   311→  return Math.ceil((new Date(dateStr).getTime() - Date.now()) / (1000 * 60 * 60 * 24));
   312→   312→}
   313→   313→
   314→   314→function deadlineBadge(days: number): string {
   315→   315→  if (days <= 0) return 'bg-rose-50 text-rose-700 dark:bg-rose-950/40 dark:text-rose-300';
   316→   316→  if (days <= 7) return 'bg-amber-50 text-amber-700 dark:bg-amber-950/40 dark:text-amber-300';
   317→   317→  if (days <= 21) return 'bg-sky-50 text-sky-700 dark:bg-sky-950/40 dark:text-sky-300';
   318→   318→  return 'bg-emerald-50 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300';
   319→   319→}

   321→   321→function deadlineLabel(days: number): string {
   322→   322→  if (days <= 0) return 'Closed';
   323→   323→  if (days <= 7) return `${days}d left`;
   324→   324→  if (days <= 30) return `${days}d left`;
   325→   325→  return `${days}d left`;
   326→   326→}
   327→   327→
   328→   328→function relativeTime(dateStr: string): string {
   329→   329→  if (!dateStr) return '';
   330→   330→  const diff = Date.now() - new Date(dateStr).getTime();
   331→   331→  const day = 86400000;
   332→   332→  if (diff < day) return 'today';
   333→   333→  if (diff < 2 * day) return 'yesterday';
   334→   334→  if (diff < 30 * day) return `${Math.floor(diff / day)}d ago`;
   335→   335→  return new Date(dateStr).toLocaleDateString();
   336→   336→}
