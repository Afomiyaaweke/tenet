   170→function relativeTime(dateStr: string): string {
   171→  const now = Date.now();
   172→  const then = new Date(dateStr).getTime();
   173→  const diff = Math.max(0, now - then);
   174→  const seconds = Math.floor(diff / 1000);
   175→  if (seconds < 60) return 'just now';
   176→  const minutes = Math.floor(seconds / 60);
   177→  if (minutes < 60) return `${minutes}m ago`;
   178→  const hours = Math.floor(minutes / 60);
   179→  if (hours < 24) return `${hours}h ago`;
   180→  const days = Math.floor(hours / 24);
   181→  if (days < 7) return `${days}d ago`;
   182→  return new Date(dateStr).toLocaleDateString();
   183→}

   425→                      <Icon className={`h-4 w-4 ${colorClass}`} />

   428→                      <p className="text-[13px] font-semibold truncate text-foreground">{n.title}</p>
   429→                      <p className="text-[11px] text-muted-foreground truncate mt-0.5 leading-relaxed">{n.message}</p>

   439→
   440→        {/* Footer */}

   442→          <div className="px-3 py-2 border-t border-border/40">
   443→            <Button
   444→              variant="ghost"
   445→              size="sm"
   446→              className="w-full text-[11px] text-muted-foreground hover:text-foreground font-medium"
   447→              onClick={() => setOpen(false)}

   449→              View all notifications
   450→            </Button>
   451→          </div>
   452→        )}
   453→      </PopoverContent>
   454→    </Popover>
   455→  );
   456→}
