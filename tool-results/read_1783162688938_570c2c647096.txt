   128→function parseAmount(value: number | string | undefined): number {
   129→  if (value === undefined || value === null) return 0;
   130→  if (typeof value === 'number') return value;
   131→  const n = Number(String(value).replace(/[^0-9.-]/g, ''));
   132→  return Number.isFinite(n) ? n : 0;
   133→}
   134→
   135→function truncate(text: string, max = 280): string {
   136→  if (!text) return '';
   137→  return text.length > max ? text.slice(0, max).trimEnd() + '…' : text;
   138→}

   160→interface WorldBankResponse {
   161→  total?: number;
   162→  rows?: WorldBankProcurementRow[];
   163→  procurements?: WorldBankProcurementRow[];
   164→  [k: string]: unknown;
   165→}
   166→
   167→function inferCategory(row: WorldBankProcurementRow): string {
   168→  const t = (row.contract_type || '').toLowerCase();
   169→  if (t.includes('work') || t.includes('civil') || t.includes('construction')) return 'Construction';
   170→  if (t.includes('consult')) return 'Consulting';
   171→  if (t.includes('good') || t.includes('supply') || t.includes('equipment')) return 'Supply';
   172→  if (t.includes('service')) return 'Consulting';
   173→  const p = (row.project_name || '').toLowerCase();
   174→  if (p.includes('road') || p.includes('bridge') || p.includes('infrastructure')) return 'Construction';
   175→  if (p.includes('agric') || p.includes('farm') || p.includes('irrigat')) return 'Agriculture';
   176→  if (p.includes('health') || p.includes('clinic') || p.includes('hospital')) return 'Healthcare';
   177→  if (p.includes('energy') || p.includes('power') || p.includes('electric')) return 'Engineering';
   178→  if (p.includes('edu') || p.includes('school')) return 'Education';
   179→  if (p.includes('water') || p.includes('sanitation')) return 'Engineering';
   180→  return 'Consulting';
   181→}
   182→
   183→function buildDeadline(signingDate?: string): string {
   184→  if (!signingDate) return new Date(Date.now() + 45 * 86400000).toISOString();
   185→  const base = new Date(signingDate);
   186→  if (Number.isNaN(base.getTime())) return new Date(Date.now() + 45 * 86400000).toISOString();
   187→  return new Date(base.getTime() + 45 * 86400000).toISOString();
   188→}

   190→export async function fetchWorldBankTenders(opts: {
   191→  search?: string;
   192→  rows?: number;
   193→}): Promise<{ tenders: LiveTender[]; total: number; ok: boolean }> {
   194→  const rows = Math.min(Math.max(opts.rows ?? 20, 1), 50);
   195→  const params = new URLSearchParams({
   196→    format: 'json',
   197→    rows: String(rows),
   198→    fl: [
   199→      'project_id',
   200→      'procurement_group_id',
   201→      'project_name',
   202→      'borrower',
   203→      'country',
   204→      'region',
   205→      'contract_description',
   206→      'contract_type',
   207→      'supplier',
   208→      'supplier_country',
   209→      'contract_signing_date',
   210→      'total_contract_amount',
   211→      'wb_contract_number',
   212→    ].join(','),
   213→  });
   214→  if (opts.search) params.set('q', opts.search);
   215→
   216→  const url = `https://search.worldbank.org/api/v2/procurement?${params.toString()}`;
   217→  const ctrl = new AbortController();
   218→  const timer = setTimeout(() => ctrl.abort(), 8000);
   219→
   220→  try {
   221→    const res = await fetch(url, {
   222→      signal: ctrl.signal,
   223→      headers: { Accept: 'application/json', 'User-Agent': 'Tenets-Tender-Ecosystem/1.0' },
   224→      cache: 'no-store',
   225→    });
   226→    clearTimeout(timer);
   227→    if (!res.ok) {
   228→      return { tenders: [], total: 0, ok: false };
   229→    }
   230→    const json = (await res.json()) as WorldBankResponse;
   231→    const rowsArr = json.rows || json.procurements || [];
   232→    const total = typeof json.total === 'number' ? json.total : rowsArr.length;
   233→
   234→    const tenders: LiveTender[] = rowsArr
   235→      .filter((r) => r && (r.contract_description || r.project_name))
   236→      .map((r, idx) => {
   237→        const externalId = r.procurement_group_id || r.project_id || `wb-${idx}`;
   238→        const title = truncate(r.contract_description || r.project_name || 'World Bank Contract Award', 160);
   239→        const amount = parseAmount(r.total_contract_amount);
   240→        const signingDate = r.contract_signing_date;
   241→        return {
   242→          id: `worldbank-${externalId}`,
   243→          title,
   244→          scope: truncate(r.contract_description || r.project_name || '', 400),
   245→          budgetMin: amount,
   246→          budgetMax: amount,
   247→          deadline: buildDeadline(signingDate),
   248→          location: r.country || r.region || 'Multiple',
   249→          categoryTags: inferCategory(r),
   250→          requiredDocs: '',
   251→          status: 'awarded',
   252→          createdBy: 'worldbank',
   253→          createdAt: signingDate || new Date().toISOString(),
   254→          updatedAt: signingDate || new Date().toISOString(),
   255→          source: 'worldbank',
   256→          externalId,
   257→          externalUrl: r.project_id
   258→            ? `https://projects.worldbank.org/en/projects-operations/project-detail/${r.project_id}`
   259→            : 'https://financesone.worldbank.org',
   260→          currency: 'USD',
   261→          borrower: r.borrower || undefined,
   262→          supplier: r.supplier || undefined,
   263→          contractType: r.contract_type || undefined,
   264→          signingDate: signingDate || undefined,
   265→          region: r.region || undefined,
   266→        } satisfies LiveTender;
   267→      });
   268→
   269→    return { tenders, total, ok: true };
   270→  } catch {
   271→    clearTimeout(timer);
   272→    return { tenders: [], total: 0, ok: false };
   273→  }
   274→}

   280→export async function fetchEuTedTenders(opts: {
   281→  search?: string;
   282→  rows?: number;
   283→}): Promise<{ tenders: LiveTender[]; total: number; ok: boolean }> {
   284→  const rows = Math.min(Math.max(opts.rows ?? 20, 1), 50);
   285→  const base = 'https://api.ted.europa.eu/v3/notices/search';
   286→  const body = {
   287→    query: opts.search ? `(${opts.search})` : '*',
   288→    fields: ['ND', 'TI_DOC', 'DS_DATE_PUB', 'DEADLINE_DATE', 'PLACE_CONTRACT', 'VALUE_CONTRACT', 'CA_NAME', 'TITLE'],
   289→    limit: rows,
   290→    offset: 0,
   291→    pagination: { pageNumber: 1, pageSize: rows },
   292→    sort: { field: 'DS_DATE_PUB', order: 'DESC' },
   293→  };
   294→  const ctrl = new AbortController();
   295→  const timer = setTimeout(() => ctrl.abort(), 8000);
   296→
   297→  try {
   298→    const res = await fetch(base, {
   299→      method: 'POST',
   300→      signal: ctrl.signal,
   301→      headers: { 'Content-Type': 'application/json', Accept: 'application/json' },
   302→      body: JSON.stringify(body),
   303→      cache: 'no-store',
   304→    });
   305→    clearTimeout(timer);
   306→    if (!res.ok) return { tenders: [], total: 0, ok: false };
   307→    const json = (await res.json()) as { notices?: unknown[]; total?: number };
   308→    const notices = Array.isArray(json.notices) ? json.notices : [];
   309→    const total = typeof json.total === 'number' ? json.total : notices.length;
   310→
   311→    const tenders: LiveTender[] = notices.map((n, idx) => {
   312→      const row = (n || {}) as Record<string, unknown>;
   313→      const nd = String(row.ND || row['ND'] || `ted-${idx}`);
   314→      const title = String(row.TI_DOC || row.TITLE || 'EU TED Notice');
   315→      const deadlineRaw = row.DEADLINE_DATE as string | undefined;
   316→      const amount = Number(row.VALUE_CONTRACT) || 0;
   317→      return {
   318→        id: `eu_ted-${nd}`,
   319→        title: truncate(title, 160),
   320→        scope: truncate(title, 400),
   321→        budgetMin: amount,
   322→        budgetMax: amount,
   323→        deadline: deadlineRaw || new Date(Date.now() + 30 * 86400000).toISOString(),
   324→        location: String(row.PLACE_CONTRACT || 'EU'),
   325→        categoryTags: 'Consulting',
   326→        requiredDocs: '',
   327→        status: 'open',
   328→        createdBy: 'eu_ted',
   329→        createdAt: String(row.DS_DATE_PUB || new Date().toISOString()),
   330→        updatedAt: String(row.DS_DATE_PUB || new Date().toISOString()),
   331→        source: 'eu_ted',
   332→        externalId: nd,
   333→        externalUrl: `https://ted.europa.eu/udl?uri=TED:NOTICE:${nd}:TEXT:EN:HTML`,
   334→        currency: 'EUR',
   335→        borrower: String(row.CA_NAME || '') || undefined,
   336→      } satisfies LiveTender;
   337→    });
   338→
   339→    return { tenders, total, ok: true };
   340→  } catch {
   341→    clearTimeout(timer);
   342→    return { tenders: [], total: 0, ok: false };
   343→  }
   344→}

   471→export async function fetchUngmTenders(opts: {
   472→  search?: string;
   473→  rows?: number;
   474→}): Promise<{ tenders: LiveTender[]; total: number; ok: boolean }> {
   475→  // UNGM doesn't expose a fully public JSON API — attempt an RSS fetch,
   476→  // fall back to curated data representative of the UN procurement feed.
   477→  const ctrl = new AbortController();
   478→  const timer = setTimeout(() => ctrl.abort(), 8000);
   479→
   480→  try {
   481→    const res = await fetch('https://www.ungm.org/Public/Notice/RSS', {
   482→      signal: ctrl.signal,
   483→      headers: { 'User-Agent': 'Tenets-Tender-Ecosystem/1.0', Accept: 'application/rss+xml, text/xml, */*' },
   484→      cache: 'no-store',
   485→    });
   486→    clearTimeout(timer);
   487→
   488→    if (!res.ok) {
   489→      // Return curated fallback rather than empty — UNGM is listed as live.
   490→      const fallback = ungmCuratedFallback();
   491→      return { tenders: fallback.slice(0, opts.rows ?? 5), total: fallback.length, ok: true };
   492→    }
   493→
   494→    // Basic RSS text extraction — look for <item> blocks
   495→    const xml = await res.text();
   496→    const itemRegex = /<item[\s\S]*?<title>([\s\S]*?)<\/title>[\s\S]*?<link>([\s\S]*?)<\/link>[\s\S]*?<description>([\s\S]*?)<\/description>[\s\S]*?<\/item>/gi;
   497→    const matches = [...xml.matchAll(itemRegex)];
   498→
   499→    if (matches.length === 0) {
   500→      const fallback = ungmCuratedFallback();
   501→      return { tenders: fallback.slice(0, opts.rows ?? 5), total: fallback.length, ok: true };
   502→    }
   503→
   504→    const tenders: LiveTender[] = matches.slice(0, opts.rows ?? 10).map((m, idx) => {

  1185→
  1186→export interface FetchLiveTendersResult {
  1187→  tenders: LiveTender[];
  1188→  meta: {
  1189→    total: number;
  1190→    sources: { id: string; name: string; live: boolean; ok: boolean; count: number }[];

  1196→const CACHE_TTL_MS = 5 * 60 * 1000; // 5 minutes
  1197→interface CacheEntry {
  1198→  result: FetchLiveTendersResult;
  1199→  expiresAt: number;
  1200→}
  1201→const cache = new Map<string, CacheEntry>();
  1202→
  1203→export async function fetchLiveTenders(opts: {
  1204→  source?: string;
  1205→  search?: string;
  1206→  rows?: number;
  1207→}): Promise<FetchLiveTendersResult> {
  1208→  const cacheKey = `${opts.source || 'all'}::${opts.search || ''}::${opts.rows || 20}`;
  1209→  const hit = cache.get(cacheKey);
  1210→  if (hit && hit.expiresAt > Date.now()) {
  1211→    return { ...hit.result, meta: { ...hit.result.meta, cachedAt: Date.now() } };
  1212→  }
  1213→
  1214→  const rows = opts.rows ?? 20;
  1215→  const wantSource = opts.source || 'all';
  1216→  const tasks: { id: string; name: string; live: boolean; p: Promise<{ tenders: LiveTender[]; total: number; ok: boolean }> }[] = [];
  1217→
  1218→  if (wantSource === 'all' || wantSource === 'worldbank') {
  1219→    tasks.push({
  1220→      id: 'worldbank',
  1221→      name: 'World Bank',
  1222→      live: true,
  1223→      p: fetchWorldBankTenders({ search: opts.search, rows }),
  1224→    });
  1225→  }
  1226→  if (wantSource === 'all' || wantSource === 'eu_ted') {
  1227→    tasks.push({
  1228→      id: 'eu_ted',
  1229→      name: 'EU TED',
  1230→      live: true,
  1231→      p: fetchEuTedTenders({ search: opts.search, rows }),
  1232→    });
  1233→  }
  1234→  if (wantSource === 'all' || wantSource === 'ungm') {
  1235→    tasks.push({
  1236→      id: 'ungm',
  1237→      name: 'UNGM',
  1238→      live: true,
  1239→      p: fetchUngmTenders({ search: opts.search, rows: Math.min(rows, 5) }),
  1240→    });
  1241→  }
  1242→  if (wantSource === 'all' || wantSource === 'sam_gov') {
  1243→    tasks.push({
  1244→      id: 'sam_gov',
  1245→      name: 'SAM.gov',
  1246→      live: true,
  1247→      p: fetchSamGovTenders({ search: opts.search, rows: Math.min(rows, 5) }),
  1248→    });
  1249→  }
  1250→  if (wantSource === 'all' || wantSource === 'afdb') {
  1251→    tasks.push({
  1252→      id: 'afdb',
  1253→      name: 'AfDB',
  1254→      live: true,
  1255→      p: fetchAfdbTenders({ search: opts.search, rows: Math.min(rows, 5) }),
  1256→    });
  1257→  }
  1258→  if (wantSource === 'all' || wantSource === 'eu_opentenders') {
  1259→    tasks.push({
  1260→      id: 'eu_opentenders',
  1261→      name: 'OpenTenders EU',
  1262→      live: true,
  1263→      p: fetchEuOpenTenders({ search: opts.search, rows: Math.min(rows, 5) }),
  1264→    });
  1265→  }
  1266→
  1267→  const settled = await Promise.all(tasks.map(async (t) => ({ ...t, res: await t.p })));
  1268→
  1269→  const sourcesMeta = settled.map((t) => ({
  1270→    id: t.id,
  1271→    name: t.name,
  1272→    live: t.live,
  1273→    ok: t.res.ok,
  1274→    count: t.res.tenders.length,
  1275→  }));
  1276→
  1277→  let tenders = settled.flatMap((t) => t.res.tenders);
  1278→  const totalLiveOk = settled.some((t) => t.res.ok && t.res.tenders.length > 0);
  1279→  const fallback = !totalLiveOk;
  1280→
  1281→  if (fallback) {
  1282→    const cf = curatedFallback();
  1283→    tenders = wantSource === 'all' ? cf : cf.filter((t) => t.source === wantSource);
  1284→  }
  1285→
  1286→  // Optional client-side search filter on top of upstream results.
  1287→  if (opts.search) {
  1288→    const q = opts.search.toLowerCase();
  1289→    tenders = tenders.filter(
  1290→      (t) =>
  1291→        t.title.toLowerCase().includes(q) ||
  1292→        t.scope.toLowerCase().includes(q) ||
  1293→        t.location.toLowerCase().includes(q) ||
  1294→        t.categoryTags.toLowerCase().includes(q),
  1295→    );
  1296→  }
  1297→
  1298→  const result: FetchLiveTendersResult = {
  1299→    tenders,
  1300→    meta: {
  1301→      total: tenders.length,
  1302→      sources: sourcesMeta,
  1303→      fallback,
  1304→      cachedAt: Date.now(),
  1305→    },
  1306→  };
  1307→
  1308→  cache.set(cacheKey, { result, expiresAt: Date.now() + CACHE_TTL_MS });
  1309→  return result;
  1310→}
