   174→function relativeTime(dateStr: string): string {
   175→  const now = Date.now();
   176→  const then = new Date(dateStr).getTime();
   177→  const diff = Math.max(0, now - then);
   178→  const seconds = Math.floor(diff / 1000);
   179→  if (seconds < 60) return 'just now';
   180→  const minutes = Math.floor(seconds / 60);
   181→  if (minutes < 60) return `${minutes}m ago`;
   182→  const hours = Math.floor(minutes / 60);
   183→  if (hours < 24) return `${hours}h ago`;
   184→  const days = Math.floor(hours / 24);
   185→  if (days < 7) return `${days}d ago`;
   186→  if (days < 30) return `${Math.floor(days / 7)}w ago`;
   187→  return new Date(dateStr).toLocaleDateString();
   188→}

   190→function getInitials(name: string): string {
   191→  return name
   192→    .split(' ')
   193→    .map((n) => n[0])
   194→    .join('')
   195→    .toUpperCase()
   196→    .slice(0, 2);
   197→}

   199→function parseImageUrls(imageUrls: string): string[] {
   200→  try {
   201→    const parsed = JSON.parse(imageUrls);
   202→    return Array.isArray(parsed) ? parsed : [];
   203→  } catch {
   204→    return [];
   205→  }
   206→}
   207→
   208→function parseTags(tags: string): string[] {
   209→  if (!tags) return [];
   210→  try {
   211→    const parsed = JSON.parse(tags);
   212→    return Array.isArray(parsed) ? parsed : tags.split(',').map((t) => t.trim()).filter(Boolean);
   213→  } catch {
   214→    return tags.split(',').map((t) => t.trim()).filter(Boolean);
   215→  }
   216→}
   217→
   218→function parseSkillTags(skillTags: string): string[] {
   219→  if (!skillTags) return [];
   220→  try {
   221→    const parsed = JSON.parse(skillTags);
   222→    return Array.isArray(parsed) ? parsed : skillTags.split(',').map((t) => t.trim()).filter(Boolean);
   223→  } catch {
   224→    return skillTags.split(',').map((t) => t.trim()).filter(Boolean);
   225→  }
   226→}

   384→function CreatePostBox({ onPostCreated }: { onPostCreated: () => void }) {
   385→  const { user } = useAuthStore();
   386→  const [content, setContent] = useState('');
   387→  const [tagInput, setTagInput] = useState('');
   388→  const [tags, setTags] = useState<string[]>([]);
   389→  const [visibility, setVisibility] = useState('public');
   390→  const [isSubmitting, setIsSubmitting] = useState(false);
   391→  const [isExpanded, setIsExpanded] = useState(false);
   392→
   393→  const addTag = () => {
   394→    const trimmed = tagInput.trim();
   395→    if (trimmed && !tags.includes(trimmed) && tags.length < 5) {
   396→      setTags([...tags, trimmed]);
   397→      setTagInput('');
   398→    }
   399→  };
   400→
   401→  const removeTag = (tag: string) => {
   402→    setTags(tags.filter((t) => t !== tag));
   403→  };
   404→
   405→  const handleSubmit = async () => {
   406→    if (!content.trim()) return;
   407→    setIsSubmitting(true);
   408→    try {
   409→      const res = await api.post('/social/posts', {
   410→        content: content.trim(),
   411→        tags: tags.join(','),
   412→        visibility,
   413→      });
   414→      if (res.success) {
   415→        toast.success('Post created!');
   416→        setContent('');
   417→        setTags([]);
   418→        setTagInput('');
   419→        setVisibility('public');
   420→        setIsExpanded(false);
   421→        onPostCreated();
   422→      } else {
   423→        toast.error(res.error || 'Failed to create post');
   424→      }
   425→    } catch {
   426→      toast.error('Failed to create post');
   427→    } finally {
   428→      setIsSubmitting(false);
   429→    }
   430→  };
   431→
   432→  const profile = user?.profile;
   433→  const fullName = profile?.fullName || user?.email || 'User';
   434→
   435→  return (
   436→    <Card className="mb-4">
   437→      <CardContent className="p-4">
   438→        <div className="flex gap-3">
   439→          <ProfileAvatar src={profile?.profilePhoto} name={fullName} className="h-10 w-10 flex-shrink-0" />
   440→          <div className="flex-1 min-w-0">
   441→            <Textarea
   442→              placeholder="Share an update, insight, or opportunity..."
   443→              value={content}
   444→              onChange={(e) => {
   445→                setContent(e.target.value);
   446→                if (!isExpanded && e.target.value) setIsExpanded(true);
   447→              }}
   448→              onFocus={() => setIsExpanded(true)}
   449→              className="min-h-[60px] resize-none border-0 bg-muted/50 focus-visible:ring-1 focus-visible:ring-emerald-500/30 p-3 text-sm"
   450→              maxLength={5000}
   451→            />
   452→            {isExpanded && (
   453→              <div className="mt-3 space-y-3">
   454→                {/* Tag Input */}
   455→                <div className="flex items-center gap-2">
   456→                  <Input
   457→                    placeholder="Add a tag (press Enter)"
   458→                    value={tagInput}
   459→                    onChange={(e) => setTagInput(e.target.value)}
   460→                    onKeyDown={(e) => {
   461→                      if (e.key === 'Enter') {
   462→                        e.preventDefault();
   463→                        addTag();
   464→                      }
   465→                    }}
   466→                    className="h-8 text-xs"
   467→                  />
   468→                  <Button variant="outline" size="sm" onClick={addTag} className="h-8 px-2 flex-shrink-0">
   469→                    <Plus className="h-3.5 w-3.5" />
   470→                  </Button>
   471→                </div>
   472→                {tags.length > 0 && (
   473→                  <div className="flex flex-wrap gap-1.5">
   474→                    {tags.map((tag) => (
   475→                      <Badge key={tag} variant="outline" className={`text-xs pr-1 ${getSkillColor(tag)}`}>
   476→                        {tag}
   477→                        <button onClick={() => removeTag(tag)} className="ml-1 hover:text-red-500">
   478→                          <X className="h-3 w-3" />
   479→                        </button>
   480→                      </Badge>
   481→                    ))}
   482→                  </div>
   483→                )}
   484→                {/* Visibility + Post */}
   485→                <div className="flex items-center justify-between">
   486→                  <div className="flex items-center gap-2">
   487→                    <Select value={visibility} onValueChange={setVisibility}>
   488→                      <SelectTrigger className="h-8 w-[130px] text-xs">
   489→                        <Globe2 className="h-3.5 w-3.5 mr-1" />
   490→                        <SelectValue />
   491→                      </SelectTrigger>
   492→                      <SelectContent>
   493→                        <SelectItem value="public">🌐 Public</SelectItem>
   494→                        <SelectItem value="connections">🔗 Connections</SelectItem>
   495→                        <SelectItem value="private">🔒 Private</SelectItem>
   496→                      </SelectContent>
   497→                    </Select>
   498→                  </div>
   499→                  <Button
   500→                    onClick={handleSubmit}
   501→                    disabled={!content.trim() || isSubmitting}
   502→                    className="h-8 px-4 bg-emerald-600 hover:bg-emerald-700 text-white text-xs"
   503→                  >
   504→                    {isSubmitting ? (
   505→                      <div className="h-3.5 w-3.5 border-2 border-white/30 border-t-white rounded-full animate-spin" />
   506→                    ) : (
   507→                      <>
   508→                        <Send className="h-3.5 w-3.5 mr-1" />
   509→                        Post
   510→                      </>
   511→                    )}
   512→                  </Button>
   513→                </div>
   514→              </div>
   515→            )}
   516→          </div>
   517→        </div>
   518→      </CardContent>
   519→    </Card>
   520→  );
   521→}

   846→function FeedTab() {
   847→  const { user } = useAuthStore();
   848→  const [posts, setPosts] = useState<SocialPost[]>([]);
   849→  const [loading, setLoading] = useState(true);
   850→  const [page, setPage] = useState(1);
   851→  const [hasMore, setHasMore] = useState(true);
   852→  const [loadingMore, setLoadingMore] = useState(false);
   853→  const observerRef = useRef<IntersectionObserver | null>(null);
   854→  const loadMoreRef = useRef<HTMLDivElement>(null);
   855→
   856→  const fetchPosts = useCallback(async (pageNum: number, append = false) => {
   857→    if (pageNum === 1) setLoading(true);
   858→    else setLoadingMore(true);
   859→    try {
   860→      const res = await api.get('/social/posts', { page: String(pageNum), limit: '10' });
   861→      if (res.success) {
   862→        const newPosts = res.data || [];
   863→        setPosts(append ? (prev) => [...prev, ...newPosts] : newPosts);
   864→        setHasMore(pageNum < (res.meta?.totalPages || 1));
   865→      }
   866→    } catch {
   867→      toast.error('Failed to load posts');
   868→    } finally {
   869→      setLoading(false);
   870→      setLoadingMore(false);
   871→    }
   872→  }, []);
   873→
   874→  useEffect(() => {
   875→    fetchPosts(1);
   876→  }, [fetchPosts]);
   877→
   878→  // Infinite scroll
   879→  useEffect(() => {
   880→    if (observerRef.current) observerRef.current.disconnect();
   881→    if (!loadMoreRef.current || !hasMore) return;
   882→
   883→    observerRef.current = new IntersectionObserver(
   884→      (entries) => {
   885→        if (entries[0].isIntersecting && hasMore && !loadingMore) {
   886→          const nextPage = page + 1;
   887→          setPage(nextPage);
   888→          fetchPosts(nextPage, true);
   889→        }
   890→      },
   891→      { threshold: 0.5 }
   892→    );
   893→    observerRef.current.observe(loadMoreRef.current);
   894→
   895→    return () => { if (observerRef.current) observerRef.current.disconnect(); };
   896→  }, [hasMore, loadingMore, page, fetchPosts]);
   897→
   898→  const handleReact = async (postId: string, emoji: string) => {
   899→    // Optimistic update
   900→    setPosts((prev) =>
   901→      prev.map((p) => {
   902→        if (p.id !== postId) return p;
   903→        const summary = [...(p.reactionSummary || [])];
   904→        const existingIdx = summary.findIndex((r) => r.emoji === emoji);
   905→        if (existingIdx >= 0) {
   906→          const existing = { ...summary[existingIdx] };
   907→          if (existing.hasReacted) {
   908→            existing.count = Math.max(0, existing.count - 1);
   909→            existing.hasReacted = false;
   910→          } else {
   911→            existing.count += 1;
   912→            existing.hasReacted = true;
   913→          }
   914→          summary[existingIdx] = existing;
   915→        } else {
   916→          summary.push({ emoji, count: 1, hasReacted: true, users: [] });
   917→        }
   918→        return { ...p, reactionSummary: summary };
   919→      })
   920→    );
   921→
   922→    try {
   923→      const res = await api.patch(`/social/posts/${postId}`, { emoji });
   924→      if (!res.success) {
   925→        // Revert on failure
   926→        fetchPosts(1);
   927→        toast.error(res.error || 'Failed to react');
   928→      }
   929→    } catch {
   930→      fetchPosts(1);
   931→      toast.error('Failed to react');
   932→    }
   933→  };
   934→
   935→  const handleDelete = async (postId: string) => {
   936→    try {
   937→      const res = await api.delete(`/social/posts/${postId}`);
   938→      if (res.success) {
   939→        setPosts((prev) => prev.filter((p) => p.id !== postId));
   940→        toast.success('Post deleted');
   941→      } else {
   942→        toast.error(res.error || 'Failed to delete post');
   943→      }
   944→    } catch {
   945→      toast.error('Failed to delete post');
   946→    }
   947→  };
   948→
   949→  const handlePostCreated = () => {
   950→    fetchPosts(1);
   951→  };
   952→
   953→  return (
   954→    <div>
   955→      <CreatePostBox onPostCreated={handlePostCreated} />
   956→
   957→      {loading ? (
   958→        <div className="space-y-4">
   959→          {[1, 2, 3].map((i) => (
   960→            <Card key={i} className="p-4">
   961→              <div className="flex gap-3">
   962→                <Skeleton className="h-10 w-10 rounded-full" />
   963→                <div className="flex-1 space-y-2">
   964→                  <Skeleton className="h-4 w-1/3" />
   965→                  <Skeleton className="h-3 w-1/4" />
   966→                  <Skeleton className="h-20 w-full mt-2" />
   967→                </div>
   968→              </div>
   969→            </Card>
   970→          ))}
   971→        </div>
   972→      ) : posts.length === 0 ? (
   973→        <Card className="p-8">
   974→          <div className="text-center">
   975→            <MessageCircle className="h-12 w-12 text-muted-foreground/40 mx-auto mb-3" />
   976→            <h3 className="font-semibold text-sm text-foreground">No posts yet</h3>
   977→            <p className="text-xs text-muted-foreground mt-1">Be the first to share something with your network!</p>
   978→          </div>
   979→        </Card>
   980→      ) : (
   981→        <div>
   982→          {posts.map((post) => (
   983→            <PostCard key={post.id} post={post} onReact={handleReact} onDelete={handleDelete} />
   984→          ))}
   985→          <div ref={loadMoreRef} className="py-4 text-center">
   986→            {loadingMore && (
   987→              <div className="flex items-center justify-center gap-2 text-xs text-muted-foreground">
   988→                <div className="h-4 w-4 border-2 border-emerald-500 border-t-transparent rounded-full animate-spin" />
   989→                Loading more...
   990→              </div>
   991→            )}
   992→            {!hasMore && posts.length > 0 && (
   993→              <p className="text-xs text-muted-foreground">You&apos;re all caught up!</p>
   994→            )}
   995→          </div>
   996→        </div>
   997→      )}
   998→    </div>
   999→  );
  1000→}

  1019→      const s = searchOverride !== undefined ? searchOverride : search;
  1020→      const ind = industryOverride !== undefined ? industryOverride : industry;
  1021→      if (s) params.search = s;
  1022→      if (ind) params.industry = ind;
  1023→

  1026→        const newProfiles = res.data || [];
  1027→        setProfiles(append ? (prev) => [...prev, ...newProfiles] : newProfiles);
  1028→        setHasMore(pageNum < (res.meta?.totalPages || 1));
  1029→      }
  1030→    } catch {
  1031→      toast.error('Failed to load profiles');
  1032→    } finally {
  1033→      setLoading(false);
  1034→    }
  1035→  }, [search, industry]);
  1036→
  1037→  useEffect(() => {
  1038→    fetchProfiles(1);
  1039→  }, [fetchProfiles]);

  1042→    setPage(1);
  1043→    fetchProfiles(1, false, search, industry);

  1047→    const key = profile.userId;
  1048→    setConnectingIds((prev) => new Set(prev).add(key));
  1049→
  1050→    // Optimistic
  1051→    setProfiles((prev) =>
  1052→      prev.map((p) =>
  1053→        p.userId === key ? { ...p, connectionStatus: 'pending', connectionDirection: 'sent' } : p
  1054→      )
  1055→    );

  1057→    try {
  1058→      const res = await api.post('/social/connections', { receiverId: key });
  1059→      if (res.success) {
  1060→        toast.success(`Connection request sent to ${profile.fullName}`);
  1061→      } else {
  1062→        // Revert
  1063→        setProfiles((prev) =>
  1064→          prev.map((p) => (p.userId === key ? { ...p, connectionStatus: 'none', connectionDirection: null } : p))
  1065→        );
  1066→        toast.error(res.error || 'Failed to send request');
  1067→      }
  1068→    } catch {
  1069→      setProfiles((prev) =>
  1070→        prev.map((p) => (p.userId === key ? { ...p, connectionStatus: 'none', connectionDirection: null } : p))
  1071→      );
  1072→      toast.error('Failed to send request');
  1073→    } finally {
  1074→      setConnectingIds((prev) => {
  1075→        const next = new Set(prev);
  1076→        next.delete(key);
  1077→        return next;
  1078→      });

  1092→          prev.map((p) => {
