   146→// ─── Content Formatting (similar to agent.tsx formatAIContent) ───────────────
   147→
   148→function formatInline(text: string): React.ReactNode {
   149→  const parts: React.ReactNode[] = [];
   150→  let remaining = text;
   151→  let partKey = 0;
   152→
   153→  while (remaining.length > 0) {
   154→    const boldMatch = remaining.match(/\*\*(.+?)\*\*/);
   155→    const italicMatch = remaining.match(/\*(.+?)\*/);
   156→
   157→    let firstMatch: { index: number; length: number; content: string; type: 'bold' | 'italic' } | null = null;
   158→
   159→    if (boldMatch && boldMatch.index !== undefined) {
   160→      firstMatch = { index: boldMatch.index, length: boldMatch[0].length, content: boldMatch[1], type: 'bold' };
   161→    }
   162→
   163→    if (italicMatch && italicMatch.index !== undefined) {
   164→      if (!firstMatch || italicMatch.index < firstMatch.index) {
   165→        firstMatch = { index: italicMatch.index, length: italicMatch[0].length, content: italicMatch[1], type: 'italic' };
   166→      }
   167→    }
   168→
   169→    if (firstMatch) {
   170→      if (firstMatch.index > 0) {
   171→        parts.push(<span key={`t-${partKey++}`}>{remaining.slice(0, firstMatch.index)}</span>);
   172→      }
   173→      if (firstMatch.type === 'bold') {
   174→        parts.push(<strong key={`b-${partKey++}`} className="font-semibold text-foreground">{firstMatch.content}</strong>);
   175→      } else {
   176→        parts.push(<em key={`i-${partKey++}`} className="italic text-muted-foreground">{firstMatch.content}</em>);
   177→      }
   178→      remaining = remaining.slice(firstMatch.index + firstMatch.length);
   179→    } else {
   180→      parts.push(<span key={`t-${partKey++}`}>{remaining}</span>);
   181→      break;
   182→    }
   183→  }
   184→
   185→  return <>{parts}</>;
   186→}

   188→function formatDocContent(content: string): React.ReactNode[] {
   189→  const lines = content.split('\n');
   190→  const result: React.ReactNode[] = [];
   191→
   192→  for (let i = 0; i < lines.length; i++) {
   193→    const line = lines[i];
   194→    const key = `doc-line-${i}`;
   195→
   196→    // Heading lines (## Title, ### Subtitle)
   197→    const h2Match = line.match(/^##\s+(.*)/);
   198→    if (h2Match) {
   199→      result.push(
   200→        <h2 key={key} className="text-xl font-bold text-foreground mt-6 mb-3 pb-2 border-b border-border/40">
   201→          {formatInline(h2Match[1])}
   202→        </h2>
   203→      );
   204→      continue;
   205→    }
   206→
   207→    const h3Match = line.match(/^###\s+(.*)/);
   208→    if (h3Match) {
   209→      result.push(
   210→        <h3 key={key} className="text-lg font-semibold text-foreground mt-4 mb-2">
   211→          {formatInline(h3Match[1])}
   212→        </h3>
   213→      );
   214→      continue;
   215→    }
   216→
   217→    const h1Match = line.match(/^#\s+(.*)/);
   218→    if (h1Match) {
   219→      result.push(
   220→        <h1 key={key} className="text-2xl font-bold text-foreground mt-4 mb-3">
   221→          {formatInline(h1Match[1])}
   222→        </h1>
   223→      );
   224→      continue;
   225→    }
   226→
   227→    // Horizontal rule
   228→    if (line.match(/^---+$/)) {
   229→      result.push(<Separator key={key} className="my-4" />);
   230→      continue;
   231→    }
   232→
   233→    // Bullet point lines
   234→    const bulletMatch = line.match(/^[\s]*[-*•]\s+(.*)/);
   235→    if (bulletMatch) {
   236→      const bulletContent = bulletMatch[1];
   237→      result.push(
   238→        <div key={key} className="flex items-start gap-2.5 ml-3 my-1">
   239→          <span className="text-emerald-500 mt-1 flex-shrink-0 text-xs">&#9679;</span>
   240→          <span className="text-sm leading-relaxed">{formatInline(bulletContent)}</span>
   241→        </div>
   242→      );
   243→      continue;
   244→    }
   245→
   246→    // Numbered list lines
   247→    const numberedMatch = line.match(/^[\s]*(\d+)[.)]\s+(.*)/);
   248→    if (numberedMatch) {
   249→      const num = numberedMatch[1];
   250→      const numberedContent = numberedMatch[2];
   251→      result.push(
   252→        <div key={key} className="flex items-start gap-2.5 ml-3 my-1">
   253→          <span className="text-emerald-600 font-semibold flex-shrink-0 text-sm min-w-[1.25rem]">{num}.</span>
   254→          <span className="text-sm leading-relaxed">{formatInline(numberedContent)}</span>
   255→        </div>
   256→      );
   257→      continue;
   258→    }
   259→
   260→    // Empty lines
   261→    if (line.trim() === '') {
   262→      result.push(<div key={key} className="h-2" />);
   263→      continue;
   264→    }
   265→
   266→    // Regular line
   267→    result.push(<div key={key} className="text-sm leading-relaxed">{formatInline(line)}</div>);
   268→  }
   269→
   270→  return result;
   271→}

   522→          {activeTemplate.fields.map(field => (
   523→            <div key={field.key} className="space-y-2">
   524→              <Label className="text-sm font-medium">{field.label}</Label>
   525→              {field.type === 'textarea' ? (
   526→                <Textarea
   527→                  placeholder={field.placeholder}
   528→                  rows={field.rows || 3}
   529→                  value={inputData[field.key] || ''}
   530→                  onChange={e => handleInputChange(field.key, e.target.value)}
   531→                  className="bg-muted/50 border-border/60 rounded-xl resize-none focus:border-emerald-300 focus:ring-emerald-200/40"
   532→                />
   533→              ) : field.type === 'select-tender' ? (
   534→                <Select
   535→                  value={inputData[field.key] || ''}
   536→                  onValueChange={val => handleInputChange(field.key, val)}
   537→                >
   538→                  <SelectTrigger className="bg-muted/50 border-border/60 rounded-xl w-full">
   539→                    <SelectValue placeholder={field.placeholder} />
   540→                  </SelectTrigger>
   541→                  <SelectContent>
   542→                    {tenders.length === 0 ? (
   543→                      <SelectItem value="_none" disabled>No open tenders available</SelectItem>
   544→                    ) : (
   545→                      tenders.map(tender => (
   546→                        <SelectItem key={tender.id} value={tender.id}>
   547→                          {tender.title}
   548→                        </SelectItem>
   549→                      ))
   550→                    )}
   551→                  </SelectContent>
   552→                </Select>
   553→              ) : field.type === 'select-project' ? (
   554→                <Select
   555→                  value={inputData[field.key] || ''}
   556→                  onValueChange={val => handleInputChange(field.key, val)}
   557→                >
   558→                  <SelectTrigger className="bg-muted/50 border-border/60 rounded-xl w-full">
   559→                    <SelectValue placeholder={field.placeholder} />
   560→                  </SelectTrigger>
   561→                  <SelectContent>
   562→                    {projects.filter(p => p.status === 'active').length === 0 ? (
   563→                      <SelectItem value="_none" disabled>No active projects available</SelectItem>
   564→                    ) : (
   565→                      projects
   566→                        .filter(p => p.status === 'active')
   567→                        .map(project => (
   568→                          <SelectItem key={project.id} value={project.id}>
   569→                            {project.tender?.title || `Project ${project.id.slice(0, 8)}`}
   570→                          </SelectItem>
   571→                        ))
   572→                    )}
   573→                  </SelectContent>
   574→                </Select>
   575→              ) : (
   576→                <Input
   577→                  type={field.type === 'number' ? 'number' : 'text'}
   578→                  placeholder={field.placeholder}
   579→                  value={inputData[field.key] || ''}
   580→                  onChange={e => handleInputChange(field.key, e.target.value)}
   581→                  className="bg-muted/50 border-border/60 rounded-xl focus:border-emerald-300 focus:ring-emerald-200/40"
   582→                />
   583→              )}
   584→            </div>
   585→          ))}
   586→        </div>
   587→
   588→        {/* Generate Button */}
   589→        <Button
   590→          onClick={handleGenerate}
   591→          disabled={generating}
   592→          className="w-full gradient-emerald hover:opacity-90 text-white rounded-xl h-12 premium-shadow transition-all hover:-translate-y-0.5 disabled:opacity-50 disabled:hover:translate-y-0 text-sm font-semibold"
   593→        >
   594→          {generating ? (
   595→            <>
   596→              <Loader2 className="h-4 w-4 mr-2 animate-spin" />
   597→              Generating Document...
   598→            </>
   599→          ) : (
   600→            <>
   601→              <Sparkles className="h-4 w-4 mr-2" />
   602→              Generate Document
   603→            </>
   604→          )}
   605→        </Button>
   606→      </div>
   607→    );
   608→  };
   609→
   610→  // ─── Render: Generating Animation ──────────────────────────────────────────
   611→
   612→  const renderGenerating = () => (
   613→    <div className="flex flex-col items-center justify-center py-16 gap-6 view-enter">
   614→      <div className="relative">
   615→        <div className="w-16 h-16 rounded-2xl bg-emerald-50 flex items-center justify-center animate-pulse">
   616→          <FileText className="h-8 w-8 text-emerald-500" />
   617→        </div>
   618→        <div className="absolute -top-1 -right-1 w-6 h-6 rounded-full gradient-emerald flex items-center justify-center">
   619→          <Sparkles className="h-3 w-3 text-white animate-spin" />
   620→        </div>
   621→      </div>
   622→      <div className="text-center">
   623→        <h4 className="font-semibold text-base mb-1">AI is writing your document...</h4>
   624→        <p className="text-sm text-muted-foreground">This may take a moment. Our AI is crafting your professional document.</p>
   625→      </div>
   626→      <div className="flex gap-1.5">
   627→        <span className="w-2.5 h-2.5 bg-emerald-400 rounded-full animate-bounce" style={{ animationDelay: '0ms' }} />
   628→        <span className="w-2.5 h-2.5 bg-emerald-400 rounded-full animate-bounce" style={{ animationDelay: '150ms' }} />
   629→        <span className="w-2.5 h-2.5 bg-emerald-400 rounded-full animate-bounce" style={{ animationDelay: '300ms' }} />
   630→      </div>
   631→    </div>
   632→  );
   633→
   634→  // ─── Render: Generated Document Preview ────────────────────────────────────
   635→
   636→  const renderGeneratedDoc = () => {
   637→    if (!generatedContent) return null;
   638→    const templateDef = TEMPLATES.find(t => t.id === generatedTemplateType);
   639→    const DocIcon = templateDef?.icon || FileText;
   640→
   641→    return (
   642→      <div className="space-y-4 view-enter">
   643→        {/* Document Actions */}
   644→        <div className="flex items-center justify-between">
   645→          <div className="flex items-center gap-2">
   646→            <div className={`w-8 h-8 rounded-lg ${templateDef?.gradientClass || 'gradient-emerald'} flex items-center justify-center`}>
   647→              <DocIcon className="h-4 w-4 text-white" />
   648→            </div>
   649→            <div>
   650→              <h4 className="font-semibold text-sm">{templateDef?.name || 'Generated Document'}</h4>
   651→              {generatedAt && (
   652→                <p className="text-[11px] text-muted-foreground">
   653→                  Generated {new Date(generatedAt).toLocaleDateString()} at {new Date(generatedAt).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })}
   654→                </p>
   655→              )}
   656→            </div>
   657→          </div>
   658→          <div className="flex items-center gap-1.5">
   659→            <Button variant="ghost" size="sm" onClick={handleCopy}
   660→              className="h-8 gap-1.5 text-xs text-muted-foreground hover:text-emerald-600 hover:bg-emerald-50 rounded-lg">
   661→              <Copy className="h-3.5 w-3.5" />
   662→              Copy
   663→            </Button>
   664→            <Button variant="ghost" size="sm" onClick={handlePrint}
   665→              className="h-8 gap-1.5 text-xs text-muted-foreground hover:text-emerald-600 hover:bg-emerald-50 rounded-lg">
   666→              <Printer className="h-3.5 w-3.5" />
   667→              Print
   668→            </Button>
   669→            <Button variant="ghost" size="sm" onClick={handleNewDocument}
   670→              className="h-8 gap-1.5 text-xs text-muted-foreground hover:text-emerald-600 hover:bg-emerald-50 rounded-lg">
   671→              <RotateCcw className="h-3.5 w-3.5" />
   672→              New
   673→            </Button>
   674→          </div>
   675→        </div>
   676→
   677→        {/* Translator for generated document */}
   678→        {generatedContent && (
   679→          <div className="mt-4">
   680→            <InlineTranslator text={generatedContent} />
   681→          </div>
   682→        )}
   683→
   684→        {/* Document Paper */}
   685→        <div className="bg-white rounded-xl border border-border/40 premium-shadow-lg overflow-hidden">
   686→          {/* Document Header Bar */}
   687→          <div className="px-6 py-3 bg-muted/30 border-b border-border/30 flex items-center justify-between">
   688→            <div className="flex items-center gap-2">
   689→              <div className={`w-2 h-2 rounded-full ${templateDef?.bgClass || 'bg-emerald-50'}`} style={{ backgroundColor: 'oklch(0.558 0.155 163)' }} />
   690→              <span className="text-[11px] font-medium text-muted-foreground uppercase tracking-wider">
   691→                {templateDef?.name || 'Document'}
   692→              </span>
   693→            </div>
   694→            <Badge className="text-[9px] px-1.5 py-0 gradient-emerald text-white border-0 font-medium">
   695→              AI Generated
   696→            </Badge>
   697→          </div>
   698→
   699→          {/* Document Content */}
   700→          <ScrollArea className="max-h-[60vh]">
   701→            <div className="p-6 md:p-8">
   702→              {/* Document Meta */}
   703→              <div className="flex items-center justify-between mb-6 pb-4 border-b border-border/30">
   704→                <div>
   705→                  <p className="text-[11px] text-muted-foreground uppercase tracking-wider mb-0.5">Prepared by</p>
   706→                  <p className="text-sm font-medium">
   707→                    {user?.company?.name || user?.profile?.fullName || user?.email || 'Unknown User'}
   708→                  </p>
   709→                </div>
   710→                <div className="text-right">
   711→                  <p className="text-[11px] text-muted-foreground uppercase tracking-wider mb-0.5">Date</p>
   712→                  <p className="text-sm font-medium">
   713→                    {generatedAt ? new Date(generatedAt).toLocaleDateString('en-US', { year: 'numeric', month: 'long', day: 'numeric' }) : new Date().toLocaleDateString('en-US', { year: 'numeric', month: 'long', day: 'numeric' })}
   714→                  </p>
   715→                </div>
   716→              </div>
   717→
   718→              {/* Formatted Content */}
   719→              <div className="prose-sm space-y-1">
   720→                {formatDocContent(generatedContent)}
   721→              </div>
   722→
   723→              {/* Document Footer */}
   724→              <div className="mt-8 pt-4 border-t border-border/30 flex items-center justify-between">
   725→                <p className="text-[10px] text-muted-foreground">
   726→                  Tenet Tender Ecosystem &middot; AI-Powered Document
   727→                </p>
   728→                <p className="text-[10px] text-muted-foreground">
   729→                  {generatedTemplateType?.replace(/-/g, ' ').replace(/\b\w/g, c => c.toUpperCase())}
   730→                </p>
   731→              </div>
   732→            </div>
   733→          </ScrollArea>
   734→        </div>
   735→      </div>
   736→    );
   737→  };
   738→
   739→  // ─── Render: Right Panel ───────────────────────────────────────────────────
   740→
   741→  const renderRightPanel = () => {
   742→    if (!selectedTemplate) return null;
   743→
   744→    return (
   745→      <Card className="premium-shadow rounded-xl border-0 bg-white">
   746→        <CardContent className="p-5 md:p-6">
   747→          {generating ? renderGenerating() : generatedContent ? renderGeneratedDoc() : renderTemplateForm()}
   748→        </CardContent>
   749→      </Card>
   750→    );
   751→  };
   752→
   753→  // ─── Main Render ───────────────────────────────────────────────────────────
   754→
   755→  return (
   756→    <div className="p-4 md:p-6 space-y-6 max-w-6xl mx-auto view-enter">
   757→      {/* Header */}
   758→      <div className="flex items-center gap-4">
   759→        <div className="p-3 rounded-2xl gradient-emerald shadow-md flex-shrink-0">
   760→          <BookOpen className="h-6 w-6 text-white" />
   761→        </div>
   762→        <div className="flex-1">
   763→          <h2 className="text-2xl md:text-3xl font-bold tracking-tight">
   764→            <span className="text-gradient-emerald">Document</span> Builder
   765→          </h2>
   766→          <p className="text-muted-foreground text-sm mt-0.5">AI-powered document preparation for the tender ecosystem</p>
   767→        </div>
   768→        {selectedTemplate && (
   769→          <Button
   770→            variant="outline"
   771→            size="sm"
   772→            onClick={handleNewDocument}
   773→            className="gap-1.5 text-xs rounded-xl border-border/60 hover:bg-emerald-50 hover:text-emerald-700 hover:border-emerald-300 transition-all"
   774→          >
   775→            <RotateCcw className="h-3.5 w-3.5" />
   776→            Start Over
   777→          </Button>
   778→        )}
   779→      </div>
   780→
   781→      {/* Body */}
   782→      {!selectedTemplate ? (
   783→        renderEmptyState()
   784→      ) : (
   785→        <div className="grid grid-cols-1 lg:grid-cols-12 gap-5">
   786→          {/* Left Panel */}
   787→          <div className="lg:col-span-4 xl:col-span-3">
   788→            <Card className="premium-shadow rounded-xl border-0 bg-white">
   789→              <CardContent className="p-4">
   790→                {renderTemplateList()}
   791→              </CardContent>
   792→            </Card>
   793→          </div>
   794→
   795→          {/* Right Panel */}
   796→          <div className="lg:col-span-8 xl:col-span-9">
   797→            {renderRightPanel()}
   798→          </div>
   799→        </div>
   800→      )}
   801→    </div>
   802→  );
   803→}
