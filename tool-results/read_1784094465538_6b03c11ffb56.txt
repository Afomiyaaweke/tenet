   122→function parseRequirements(requirements: string): string[] {
   123→  try {
   124→    const parsed = JSON.parse(requirements);
   125→    return Array.isArray(parsed) ? parsed : [];
   126→  } catch {
   127→    return [];
   128→  }
   129→}

   131→function formatDate(dateStr: string): string {
   132→  try {
   133→    return new Date(dateStr).toLocaleDateString('en-GB', {
   134→      day: 'numeric',
   135→      month: 'short',
   136→      year: 'numeric',
   137→    });
   138→  } catch {
   139→    return dateStr;
   140→  }
   141→}
   142→
   143→function isDeadlineSoon(deadline: string): boolean {
   144→  const d = new Date(deadline);
   145→  const now = new Date();
   146→  const diff = d.getTime() - now.getTime();
   147→  return diff > 0 && diff < 7 * 24 * 60 * 60 * 1000;
   148→}
   149→
   150→function isDeadlinePast(deadline: string): boolean {
   151→  return new Date(deadline).getTime() < Date.now();
   152→}

   156→async function apiFetch<T>(url: string, options?: RequestInit): Promise<T> {
   157→  const res = await fetch(url, {
   158→    headers: { 'Content-Type': 'application/json', ...options?.headers },
   159→    ...options,
   160→  });
   161→  const data = await res.json();
   162→  if (!res.ok) throw new Error(data.error || 'Request failed');
   163→  return data;
   164→}

   168→function AuthGate({ onAuth }: { onAuth: (user: User) => void }) {

   170→  const [loading, setLoading] = useState(false);
   171→  const [error, setError] = useState('');
   172→
   173→  // Login fields
   174→  const [loginEmail, setLoginEmail] = useState('');
   175→  const [loginPassword, setLoginPassword] = useState('');
   176→
   177→  // Register fields
   178→  const [regName, setRegName] = useState('');
   179→  const [regEmail, setRegEmail] = useState('');
   180→  const [regPassword, setRegPassword] = useState('');
   181→  const [regCompany, setRegCompany] = useState('');
   182→
   183→  const handleLogin = async (e: React.FormEvent) => {
   184→    e.preventDefault();
   185→    setError('');
   186→    setLoading(true);
   187→    try {
   188→      const data = await apiFetch<{ user: User }>('/api/auth/login', {
   189→        method: 'POST',
   190→        body: JSON.stringify({ email: loginEmail, password: loginPassword }),
   191→      });

   193→      onAuth(data.user);
   194→    } catch (err) {
   195→      setError(err instanceof Error ? err.message : 'Login failed');

   197→      setLoading(false);

   202→    e.preventDefault();

   205→    try {
   206→      const data = await apiFetch<{ user: User }>('/api/auth/register', {
   207→        method: 'POST',
   208→        body: JSON.stringify({
   209→          name: regName,
   210→          email: regEmail,
   211→          password: regPassword,
   212→          company: regCompany || undefined,
   213→        }),
   214→      });
   215→      toast.success('Account created successfully!');
   216→      onAuth(data.user);
   217→    } catch (err) {
   218→      setError(err instanceof Error ? err.message : 'Registration failed');
   219→    } finally {
   220→      setLoading(false);
   221→    }

   235→            </div>
   236→            <p className="text-muted-foreground text-sm">
   237→              Transforming procurement through technology
   238→            </p>

   241→          {/* Auth Card */}
   242→          <Card className="glass-card premium-shadow border-0">
   243→            <CardHeader className="pb-4">
   244→              <Tabs value={tab} onValueChange={(v) => { setTab(v as 'login' | 'register'); setError(''); }}>
   245→                <TabsList className="w-full">
   246→                  <TabsTrigger value="login" className="flex-1">Sign In</TabsTrigger>
   247→                  <TabsTrigger value="register" className="flex-1">Create Account</TabsTrigger>
   248→                </TabsList>
   249→              </Tabs>
   250→            </CardHeader>
   251→            <CardContent>
   252→              {error && (
   253→                <div className="mb-4 p-3 rounded-lg bg-destructive/10 text-destructive text-sm flex items-center gap-2">
   254→                  <X className="size-4 shrink-0" />
   255→                  {error}
   256→                </div>
   257→              )}
   258→
   259→              {tab === 'login' ? (
   260→                <form onSubmit={handleLogin} className="space-y-4">

   262→                    <Label htmlFor="login-email">Email</Label>

   269→                      required

   286→                    {loading && <Loader2 className="size-4 animate-spin" />}
   287→                    {loading ? 'Signing in...' : 'Sign In'}
   288→                  </Button>
   289→                </form>
   290→              ) : (

   361→function TenderReviewDialog({
   362→  tender,
   363→  open,
   364→  onOpenChange,
   365→}: {
   366→  tender: Tender | null;
   367→  open: boolean;
   368→  onOpenChange: (open: boolean) => void;
   369→}) {
   370→  const [exporting, setExporting] = useState(false);
   371→
   372→  if (!tender) return null;
   373→
   374→  const requirements = parseRequirements(tender.requirements);
   375→
   376→  const handleExportPdf = async () => {
   377→    setExporting(true);
   378→    try {
   379→      const res = await fetch(`/api/tenders/${tender.id}/export-pdf`);
   380→      if (!res.ok) throw new Error('Export failed');
   381→      const blob = await res.blob();
   382→      const url = window.URL.createObjectURL(blob);
   383→      const a = document.createElement('a');
   384→      a.href = url;
   385→      a.download = `${tender.title.replace(/[^a-zA-Z0-9]/g, '_')}.pdf`;
   386→      document.body.appendChild(a);
   387→      a.click();
   388→      a.remove();
   389→      window.URL.revokeObjectURL(url);
   390→      toast.success('PDF exported successfully');
   391→    } catch {
   392→      toast.error('Failed to export PDF');
   393→    } finally {
   394→      setExporting(false);
   395→    }
   396→  };
   397→
   398→  return (
   399→    <Dialog open={open} onOpenChange={onOpenChange}>
   400→      <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-hidden flex flex-col p-0">
   401→        <div className="flex-1 overflow-y-auto">
   402→          {/* Header Section */}
   403→          <div className="gradient-emerald p-6 pb-8 rounded-t-lg relative">
   404→            <DialogHeader>
   405→              <DialogTitle className="text-xl text-white font-bold leading-tight pr-8">
   406→                {tender.title}
   407→              </DialogTitle>
   408→              <DialogDescription className="text-white/80 mt-1">
   409→                Published by {tender.user.name}
   410→                {tender.user.company && ` · ${tender.user.company}`}
   411→              </DialogDescription>
   412→            </DialogHeader>
   413→            <div className="flex flex-wrap gap-2 mt-4">
   414→              <Badge className="bg-white/20 text-white border-white/30 hover:bg-white/30">
   415→                <Building2 className="size-3" />
   416→                {tender.organization}
   417→              </Badge>
   418→              <Badge className="bg-white/20 text-white border-white/30 hover:bg-white/30">
   419→                <Briefcase className="size-3" />
   420→                {tender.category}
   421→              </Badge>
   422→              <Badge
   423→                className={`border-0 ${
   424→                  tender.status === 'open'
   425→                    ? 'bg-emerald-600 text-white'
   426→                    : tender.status === 'closed'
   427→                      ? 'bg-red-600 text-white'
   428→                      : 'bg-amber-600 text-white'
   429→                }`}

   436→          <div className="p-6 space-y-6">
   437→            {/* Key Details Grid */}
   438→            <div className="grid grid-cols-2 sm:grid-cols-3 gap-4">
   439→              <div className="space-y-1">
   440→                <p className="text-xs font-medium text-muted-foreground uppercase tracking-wide">Budget</p>
   441→                <p className="text-sm font-semibold flex items-center gap-1.5">
   442→                  <DollarSign className="size-4 text-emerald-600" />
   443→                  {tender.budget}
   444→                </p>
   445→              </div>
   446→              <div className="space-y-1">
   447→                <p className="text-xs font-medium text-muted-foreground uppercase tracking-wide">Deadline</p>
