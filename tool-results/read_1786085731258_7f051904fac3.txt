    96→/* ───────────────────────── Password Strength Meter ───────────────────────── */
    97→function getPasswordStrength(password: string): { score: number; label: string; color: string } {
    98→  let score = 0;
    99→  if (password.length >= 8) score += 25;
   100→  if (password.length >= 12) score += 10;
   101→  if (/[A-Z]/.test(password)) score += 20;
   102→  if (/[a-z]/.test(password)) score += 20;
   103→  if (/[0-9]/.test(password)) score += 20;
   104→  if (/[^A-Za-z0-9]/.test(password)) score += 5;  // bonus only
   105→  if (new Set(password.toLowerCase()).size >= 8) score += 5;
   106→
   107→  if (score >= 70) return { score, label: 'Strong', color: 'bg-green-500' };
   108→  if (score >= 50) return { score, label: 'Good', color: 'bg-blue-500' };
   109→  if (score >= 30) return { score, label: 'Fair', color: 'bg-yellow-500' };
   110→  return { score, label: 'Weak', color: 'bg-red-500' };
   111→}
   112→
   113→function PasswordStrengthMeter({ password }: { password: string }) {
   114→  if (!password) return null;
   115→  const { score, label, color } = getPasswordStrength(password);
   116→  return (
   117→    <div className="space-y-1.5 mt-2">
   118→      <div className="flex gap-1">
   119→        {[25, 50, 75, 100].map(threshold => (
   120→          <div key={threshold} className="h-1.5 flex-1 rounded-full bg-muted">
   121→            <div
   122→              className={`h-full rounded-full transition-all duration-300 ${score >= threshold ? color : 'bg-transparent'}`}
   123→              style={{ width: score >= threshold ? '100%' : '0%' }}
   124→            />
   125→          </div>
   126→        ))}
   127→      </div>
   128→      <div className="flex justify-between items-center">
   129→        <span className="text-xs text-muted-foreground">Strength:</span>
   130→        <span className={`text-xs font-medium ${score >= 80 ? 'text-green-600' : score >= 60 ? 'text-blue-600' : score >= 40 ? 'text-yellow-600' : 'text-red-600'}`}>{label}</span>
   131→      </div>
   132→    </div>
   133→  );
   134→}

   159→/* ───────────────────────── Step Indicator ───────────────────────── */
   160→function StepIndicator({ currentStep, totalSteps = 4 }: { currentStep: RegStep; totalSteps?: number }) {
   161→  return (
   162→    <div className="flex items-center justify-center gap-1 mb-6">
   163→      {Array.from({ length: totalSteps }, (_, i) => (i + 1) as RegStep).map((step, i) => {
   164→        const meta = REG_STEP_META[step];
   165→        const isCompleted = currentStep > step;
   166→        const isCurrent = currentStep === step;
   167→        const Icon = meta.icon;
   168→        return (
   169→          <React.Fragment key={step}>
   170→            <div className="flex items-center gap-1.5">
   171→              <div
   172→                className={`h-7 w-7 rounded-full flex items-center justify-center text-xs font-bold transition-all ${
   173→                  isCompleted || isCurrent
   174→                    ? 'gradient-orange text-white shadow-md shadow-orange-500/30'
   175→                    : 'bg-muted text-muted-foreground'
   176→                }`}
   177→              >
   178→                {isCompleted ? <CheckCircle2 className="w-4 h-4" /> : <Icon className="w-3.5 h-3.5" />}
   179→              </div>
   180→              <span className={`text-xs font-medium hidden sm:inline ${isCompleted || isCurrent ? 'text-foreground' : 'text-muted-foreground'}`}>
   181→                {meta.label}
   182→              </span>
   183→            </div>
   184→            {i < totalSteps - 1 && (
   185→              <div className={`h-0.5 w-4 sm:w-8 rounded-full ${currentStep > step ? 'bg-orange-500' : 'bg-muted'}`} />
   186→            )}
   187→          </React.Fragment>
   188→        );
   189→      })}
   190→    </div>
   191→  );
   192→}

   203→    // Company

   206→    companyCountry: '', companyEmail: '', companyWebsite: '',
   207→    // Role
   208→    role: 'user' as string,
   209→  });

   211→
   212→  const [activeTab, setActiveTab] = useState<'login' | 'register'>('login');
   213→  const [showPassword, setShowPassword] = useState(false);
   214→  const [regStep, setRegStep] = useState<RegStep>(1);
   215→
   216→  // Forgot / Reset password state
   217→  const [authMode, setAuthMode] = useState<'login' | 'register' | 'forgot-password' | 'forgot-sent' | 'reset-password'>('login');
   218→  const [forgotEmail, setForgotEmail] = useState('');
   219→  const [resetToken, setResetToken] = useState('');
   220→  const [tokenValidation, setTokenValidation] = useState<'idle' | 'checking' | 'valid' | 'invalid'>('idle');
   221→  const [newPassword, setNewPassword] = useState('');
   222→  const [confirmPassword, setConfirmPassword] = useState('');
   223→  const [regConfirmPassword, setRegConfirmPassword] = useState('');
   224→  const [resetSuccess, setResetSuccess] = useState(false);
   225→
   226→  // Handle ?token=xxx in URL for email reset links
   227→  useEffect(() => {
   228→    const params = new URLSearchParams(window.location.search);
   229→    const tokenFromUrl = params.get('token');
   230→    if (tokenFromUrl) {
   231→      setResetToken(tokenFromUrl);
   232→      setAuthMode('reset-password');
   233→      // Clean up URL without reloading
   234→      window.history.replaceState({}, '', window.location.pathname);
   235→    }
   236→  }, []);
   237→
   238→  // Debounced token validation — checks the code against the server as user types
   239→  useEffect(() => {
   240→    if (!resetToken || resetToken.length < 10) {
   241→      setTokenValidation('idle');
   242→      return;
   243→    }
   244→
   245→    const timer = setTimeout(async () => {
   246→      setTokenValidation('checking');
   247→      try {
   248→        const { api } = await import('@/lib/api');
   249→        const res = await api.get(`/auth/validate-reset-token?token=${encodeURIComponent(resetToken)}`);
   250→        setTokenValidation(res.valid ? 'valid' : 'invalid');
   251→      } catch {
   252→        setTokenValidation('idle'); // Network error — don't block
   253→      }
   254→    }, 600); // 600ms debounce
   255→
   256→    return () => clearTimeout(timer);
   257→  }, [resetToken]);
   258→
   259→  const handleLoginSubmit = async (e: React.FormEvent) => {
   260→    e.preventDefault();
   261→    if (!loginData.email || !loginData.password) return;
   262→    setLoading(true);
   263→    try {
   264→      const ok = await login(loginData.email, loginData.password);
   265→      if (!ok) {
   266→        toast.error('Invalid email or password. Please try again.');
   267→      }
   268→    } catch {
   269→      toast.error('Login failed. Please check your connection and try again.');
   270→    }
   271→    setLoading(false);
   272→  };
   273→
   274→  const handleForgotPassword = async (e: React.FormEvent) => {
   275→    e.preventDefault();
   276→    if (!forgotEmail) return;
   277→    setLoading(true);
   278→    try {
   279→      const { api } = await import('@/lib/api');
   280→      const res = await api.post('/auth/forgot-password', { email: forgotEmail });
   281→      if (res.success) {
   282→        // Always show the confirmation screen to prevent email enumeration
   283→        // The reset token is sent ONLY to the user's email - never in the API response
   284→        setResetToken(''); // Clear any previous token
   285→        setTokenValidation('idle');
   286→        setAuthMode('forgot-sent');
   287→      } else {
   288→        toast.error(res.error || 'Failed to process request');
   289→      }
   290→    } catch {
   291→      toast.error('Network error. Please try again.');
   292→    }
   293→    setLoading(false);
   294→  };
   295→
   296→  const handleResetPassword = async (e: React.FormEvent) => {
   297→    e.preventDefault();
   298→    if (!resetToken || !newPassword || !confirmPassword) return;
   299→    if (newPassword !== confirmPassword) {
   300→      toast.error('Passwords do not match');
   301→      return;
   302→    }
   303→    if (newPassword.length < 8) {
   304→      toast.error('Password must be at least 8 characters');
   305→      return;
   306→    }
   307→    if (!/[A-Z]/.test(newPassword) || !/[a-z]/.test(newPassword) || !/[0-9]/.test(newPassword)) {
   308→      toast.error('Password must include uppercase, lowercase, and number');
   309→      return;
   310→    }
   311→    setLoading(true);
   312→    try {
   313→      const { api } = await import('@/lib/api');
   314→      const res = await api.post('/auth/reset-password', {
   315→        token: resetToken,
   316→        newPassword,
   317→        confirmPassword,
   318→      });
   319→      if (res.success) {
   320→        setResetSuccess(true);
   321→        toast.success('Password reset successfully! You can now sign in.');
   322→      } else {
   323→        toast.error(res.error || 'Failed to reset password');
   324→      }
   325→    } catch {
   326→      toast.error('Network error. Please try again.');
   327→    }
   328→    setLoading(false);
   329→  };
   330→
   331→  /* ─── Registration step navigation ─── */
   332→  const canGoNext = (): boolean => {
   333→    switch (regStep) {
   334→      case 1:
   335→        return !!regData.email &&
   336→          !!regData.password &&
   337→          regData.password.length >= 8 &&
   338→          /[A-Z]/.test(regData.password) &&
   339→          /[a-z]/.test(regData.password) &&
   340→          /[0-9]/.test(regData.password) &&
   341→          !!regConfirmPassword &&
   342→          regData.password === regConfirmPassword;
   343→      case 2:
   344→        return !!regData.companyName;
   345→      case 3:
   346→        return !!regData.fullName;
   347→      case 4:
   348→        return true;
   349→      default:
   350→        return false;
   351→    }
   352→  };
   353→
   354→  const goNext = () => {
   355→    if (!canGoNext()) return;
   356→    if (regStep < 4) setRegStep((regStep + 1) as RegStep);
   357→  };
   358→
   359→  const goBack = () => {
   360→    if (regStep > 1) setRegStep((regStep - 1) as RegStep);
   361→  };
   362→

   365→    try {
   366→      const ok = await register(regData);
   367→      if (!ok) {
   368→        toast.error('This email is already registered. Please sign in instead.', {
   369→          action: {
   370→            label: 'Sign In',

   388→
   389→        {/* ═══════════ LEFT PANEL (Desktop) ═══════════ */}
