   176→function relativeTime(dateStr: string): string {
   177→  const now = Date.now();
   178→  const then = new Date(dateStr).getTime();
   179→  const diff = Math.max(0, now - then);
   180→  const seconds = Math.floor(diff / 1000);
   181→  if (seconds < 60) return 'just now';
   182→  const minutes = Math.floor(seconds / 60);
   183→  if (minutes < 60) return `${minutes}m ago`;
   184→  const hours = Math.floor(minutes / 60);
   185→  if (hours < 24) return `${hours}h ago`;
   186→  const days = Math.floor(hours / 24);
   187→  if (days < 7) return `${days}d ago`;
   188→  return new Date(dateStr).toLocaleDateString();
   189→}

   348→/* ──────────────────── NotificationDropdown ──────────────────── */
   349→
   350→function NotificationDropdown({
   351→  notifications,
   352→  onMarkAsRead,
   353→  onMarkAllRead,
   354→}: {
   355→  notifications: { id: string; title: string; message: string; type: string; read: boolean; createdAt: string }[];
   356→  onMarkAsRead: (id: string) => void;
   357→  onMarkAllRead: () => void;
   358→}) {
   359→  const [open, setOpen] = useState(false);
   360→  const unread = useMemo(() => notifications.filter((n) => !n.read), [notifications]);
   361→  const displayItems = useMemo(() => unread.slice(0, 6), [unread]);
   362→
   363→  return (
   364→    <Popover open={open} onOpenChange={setOpen}>
   365→      <PopoverTrigger asChild>
   366→        <Button variant="ghost" size="icon" className="relative h-9 w-9 hover:bg-muted/80 transition-colors">
   367→          <Bell className="h-[18px] w-[18px] text-muted-foreground" />
   368→          {unread.length > 0 && (
   369→            <span className="absolute -top-0.5 -right-0.5 min-w-[18px] h-[18px] px-1 bg-red-500 text-white text-[10px] font-bold rounded-full flex items-center justify-center shadow-sm shadow-red-200 animate-in zoom-in-50 duration-200">
   370→              {unread.length > 99 ? '99+' : unread.length}
   371→            </span>
   372→          )}
   373→        </Button>
   374→      </PopoverTrigger>
   375→      <PopoverContent
   376→        className="w-80 p-0 glass-card rounded-xl overflow-hidden animate-in slide-in-from-top-2 fade-in-0 duration-200"
   377→        align="end"
   378→        sideOffset={8}
   379→      >
   380→        {/* Header */}
   381→        <div className="flex items-center justify-between px-4 py-3 border-b border-border/50">
   382→          <div className="flex items-center gap-2">
   383→            <h4 className="text-sm font-bold text-foreground">Notifications</h4>
   384→            {unread.length > 0 && (
   385→              <Badge className="text-[10px] px-1.5 py-0 bg-red-500 text-white font-bold border-0">
   386→                {unread.length}
   387→              </Badge>
   388→            )}
   389→          </div>
   390→          {unread.length > 0 && (
   391→            <Button
   392→              variant="ghost"
   393→              size="sm"
   394→              className="text-[11px] h-7 px-2 text-primary hover:text-primary/80 hover:bg-primary/10 font-semibold"
   395→              onClick={onMarkAllRead}
   396→            >
   397→              <Check className="h-3 w-3 mr-1" /> Mark all read
   398→            </Button>
   399→          )}
   400→        </div>
   401→
   402→        {/* Items */}
   403→        <ScrollArea className="max-h-[340px]">
   404→          {displayItems.length === 0 ? (
   405→            <div className="py-8 text-center">
   406→              <div className="w-12 h-12 mx-auto mb-3 rounded-full bg-orange-50 dark:bg-orange-950/30 flex items-center justify-center">
   407→                <CheckCircle className="h-6 w-6 text-orange-400" />
   408→              </div>
   409→              <p className="text-sm font-medium text-muted-foreground">All caught up!</p>
   410→              <p className="text-xs text-muted-foreground/60 mt-0.5">No new notifications</p>
   411→            </div>
   412→          ) : (
   413→            <div className="divide-y divide-border/40">
   414→              {displayItems.map((n, i) => {
   415→                const Icon = NOTIFICATION_ICONS[n.type] || Info;
   416→                const colorClass = NOTIFICATION_COLORS[n.type] || 'text-gray-500';
   417→                const bgClass = NOTIFICATION_BG[n.type] || 'bg-gray-50';
   418→                return (
   419→                  <button
   420→                    key={n.id}
   421→                    onClick={() => {
   422→                      onMarkAsRead(n.id);
   423→                      setOpen(false);
   424→                    }}
   425→                    className="w-full flex items-start gap-3 px-4 py-3 hover:bg-muted/50 transition-colors text-left group"
   426→                    style={{ animationDelay: `${i * 40}ms` }}
   427→                  >
   428→                    <div
   429→                      className={`w-8 h-8 rounded-lg ${bgClass} flex items-center justify-center flex-shrink-0 mt-0.5 transition-transform group-hover:scale-110`}
   430→                    >
   431→                      <Icon className={`h-4 w-4 ${colorClass}`} />
   432→                    </div>
   433→                    <div className="min-w-0 flex-1">
   434→                      <p className="text-[13px] font-semibold truncate text-foreground">{n.title}</p>
   435→                      <p className="text-[11px] text-muted-foreground truncate mt-0.5 leading-relaxed">{n.message}</p>
   436→                      <p className="text-[10px] text-muted-foreground/50 mt-1 font-medium">{relativeTime(n.createdAt)}</p>
   437→                    </div>
   438→                    <div className="w-2 h-2 rounded-full bg-orange-500 flex-shrink-0 mt-2 ring-2 ring-orange-100 dark:ring-orange-900/40" />
   439→                  </button>
   440→                );
   441→              })}
   442→            </div>
   443→          )}
   444→        </ScrollArea>
   445→
   446→        {/* Footer */}
   447→        {notifications.length > 6 && (
   448→          <div className="px-3 py-2 border-t border-border/40">
   449→            <Button
   450→              variant="ghost"
   451→              size="sm"
   452→              className="w-full text-[11px] text-muted-foreground hover:text-foreground font-medium"
   453→              onClick={() => setOpen(false)}
   454→            >
   455→              View all notifications
   456→            </Button>
   457→          </div>
   458→        )}
   459→      </PopoverContent>
   460→    </Popover>
   461→  );
   462→}

   467→  const { user, company, logout } = useAuthStore();
   468→  const { view, viewParams, setView } = useNavStore();
   469→  const { fetchNotifications, notifications } = useDataStore();
   470→  const role = user?.role || 'user';
   471→  const navSections = useMemo(() => getNavItemsForRole(role), [role]);
   472→  const unreadCount = notifications.filter((n) => !n.read).length;
   473→
   474→  // Flat list for title lookup
   475→  const allNavItems = useMemo(
   476→    () => navSections.flatMap((s) => s.items),
   477→    [navSections]
   478→  );
   479→
   480→  useEffect(() => {
   481→    fetchNotifications();
   482→  }, [fetchNotifications]);
   483→
   484→  const handleMarkAsRead = async (id: string) => {
   485→    await api.patch(`/notifications/${id}`, { read: true });
   486→    fetchNotifications();
   487→  };
   488→
   489→  const handleMarkAllRead = async () => {
   490→    const unread = notifications.filter((n) => !n.read);
   491→    await Promise.all(unread.map((n) => api.patch(`/notifications/${n.id}`, { read: true })));
   492→    fetchNotifications();
   493→  };
   494→
   495→  const renderView = () => {
   496→    switch (view) {
   497→      case 'dashboard':
   498→        return <DashboardView />;
   499→      case 'tenders':
   500→        return <TendersView />;
   501→      case 'live-tenders':
   502→        return <LiveTendersView />;
   503→      case 'tender-detail':
   504→        return <TenderDetailView tenderId={viewParams.id} initialTab={viewParams.tab as any} />;
   505→      case 'tender-compare':
   506→        return <TenderCompareView tenderIds={viewParams.ids} />;
   507→      case 'bid-compare':
   508→        return <BidCompareView tenderId={viewParams.tenderId} />;
   509→      case 'bid-analysis':
   510→        return <BidsView />;
   511→      case 'bids':
   512→        return <BidsView />;
   513→      case 'applicants':
   514→        return <ApplicantsView />;
   515→      case 'projects':
   516→        return <ProjectsView />;
   517→      case 'project-detail':
   518→        return <ProjectDetailView projectId={viewParams.id} />;
   519→      case 'chat':
   520→        return <ChatView chatId={viewParams.id} />;
   521→      case 'finance':
   522→        return <ProjectDetailView projectId={viewParams.id} />;
   523→      case 'events':
   524→        return <EventsView />;
   525→      case 'profile':
   526→        return <ProfileView />;
   527→      case 'company-settings':
   528→        return <ProfileView />;
   529→      case 'documents':
   530→        return <DocumentsView />;
   531→      case 'ai-doc-studio':
   532→        return <AIDocStudioView />;
   533→      case 'doc-builder':
   534→        return <DocBuilderView />;
   535→      case 'tender-analyzer':
   536→        return <TenderAnalyzerView />;
   537→      case 'agent':
   538→        return <AgentView />;
   539→      case 'staff':
   540→        return <StaffView />;
   541→      case 'contact-us':
   542→        return <ContactUsView />;
   543→      case 'privacy-policy':
   544→        return <PrivacyPolicyView />;
   545→      case 'social-circle':
   546→        return <SocialCircleView />;
   547→      default:
   548→        return <DashboardView />;
   549→    }
   550→  };
   551→
   552→  const pageTitle = allNavItems.find((i) => i.id === view)?.label
   553→    || (view === 'tender-compare' ? 'Compare Tenders' : view === 'bid-compare' ? 'Compare Bids' : view === 'bid-analysis' ? 'Bid Analysis' : 'Dashboard');
   554→  const breadcrumb = view === 'tender-detail' || view === 'bid-compare' || view === 'bid-analysis' ? 'Tenders' : view === 'tender-compare' ? 'Tenders' : view === 'project-detail' ? 'Projects' : null;
   555→
   556→  const badgeConfig = ROLE_BADGE_CONFIG[role] || ROLE_BADGE_CONFIG.user;
   557→  const companyName = company?.name || user?.company?.name;
   558→
   559→  const sidebarProps = { user, role, company, navSections, view, setView, unreadCount, logout };
   560→
   561→  return (
   562→    <div className="min-h-screen flex bg-background">
   563→      {/* ── Desktop Sidebar ── */}
   564→      <aside className="hidden md:flex w-[260px] bg-card border-r border-border/50 flex-col flex-shrink-0 sidebar-shadow">
   565→        <SidebarContent {...sidebarProps} />
   566→      </aside>
   567→
   568→      {/* ── Main Area ── */}
   569→      <div className="flex-1 flex flex-col min-w-0">
   570→        {/* ── Top Bar ── */}
   571→        <header className="h-14 bg-background/80 backdrop-blur-md border-b border-border/50 flex items-center px-4 gap-3 flex-shrink-0 sticky top-0 z-30">
   572→          {/* Mobile hamburger */}
   573→          <Sheet>
   574→            <SheetTrigger asChild>
   575→              <Button variant="ghost" size="icon" className="md:hidden h-9 w-9 hover:bg-muted/80">
   576→                <Menu className="h-5 w-5 text-muted-foreground" />
   577→              </Button>
   578→            </SheetTrigger>
   579→            <SheetContent side="left" className="p-0 w-[260px]">
   580→              <SidebarContent {...sidebarProps} />
   581→            </SheetContent>
   582→          </Sheet>
   583→
   584→          {/* Page Title & Breadcrumb */}
   585→          <div className="flex-1 min-w-0 flex items-center gap-1.5">
   586→            {breadcrumb && (
   587→              <>
   588→                <span className="text-sm text-muted-foreground/60 font-medium hidden sm:inline">
   589→                  {breadcrumb}
   590→                </span>
   591→                <ChevronRight className="h-3 w-3 text-muted-foreground/40 hidden sm:inline" />
   592→              </>
   593→            )}
   594→            <h1 className="text-base font-bold truncate text-foreground">{pageTitle}</h1>
   595→          </div>
   596→
   597→          {/* Right Actions */}
   598→          <div className="flex items-center gap-1">
   599→            {/* Search */}
   600→            <Button variant="ghost" size="icon" className="h-9 w-9 hover:bg-muted/80 transition-colors">
   601→              <Search className="h-[18px] w-[18px] text-muted-foreground" />
   602→            </Button>
   603→
   604→            {/* Theme Toggle */}
   605→            <ThemeToggle />
   606→
   607→            {/* Notifications */}
   608→            <NotificationDropdown
   609→              notifications={notifications}
   610→              onMarkAsRead={handleMarkAsRead}
   611→              onMarkAllRead={handleMarkAllRead}
   612→            />
   613→
   614→            {/* User Avatar Dropdown */}
   615→            <DropdownMenu>
   616→              <DropdownMenuTrigger asChild>
   617→                <Button variant="ghost" size="icon" className="h-9 w-9 hover:bg-muted/80 transition-colors ml-0.5">
   618→                  <div className="w-7 h-7 rounded-full bg-slate-700 flex items-center justify-center shadow-sm shadow-slate-200">
   619→                    <span className="text-white font-bold text-[11px]">{getUserInitial(user)}</span>
   620→                  </div>
   621→                </Button>
   622→              </DropdownMenuTrigger>
   623→              <DropdownMenuContent align="end" className="w-56">
   624→                <DropdownMenuLabel className="font-normal">
   625→                  <div className="flex flex-col space-y-1">
   626→                    <div className="flex items-center gap-2">
   627→                      <p className="text-sm font-medium leading-none">{user?.profile?.fullName || user?.email || 'User'}</p>
   628→                      <Badge variant="secondary" className={`text-[9px] px-1.5 py-0 font-semibold border-0 ${badgeConfig.className}`}>
   629→                        {badgeConfig.label}
   630→                      </Badge>
   631→                    </div>
   632→                    <p className="text-xs leading-none text-muted-foreground">{user?.email}</p>
   633→                    {companyName && (
   634→                      <p className="text-[11px] leading-none text-muted-foreground/70 flex items-center gap-1">
   635→                        <Building2 className="w-3 h-3" /> {companyName}
   636→                      </p>
   637→                    )}
   638→                  </div>
   639→                </DropdownMenuLabel>
   640→                <DropdownMenuSeparator />
   641→                <DropdownMenuItem onClick={() => setView('profile' as View)} className="cursor-pointer">
   642→                  <User className="mr-2 h-4 w-4" />
   643→                  <span>Profile</span>
   644→                </DropdownMenuItem>
   645→                <DropdownMenuItem onClick={() => setView('documents' as View)} className="cursor-pointer">
   646→                  <FileText className="mr-2 h-4 w-4" />
   647→                  <span>Documents</span>
   648→                </DropdownMenuItem>
   649→                <DropdownMenuSeparator />
   650→                <DropdownMenuItem onClick={logout} className="cursor-pointer text-red-600 focus:text-red-600 focus:bg-red-50 dark:focus:bg-red-950/30">
   651→                  <LogOut className="mr-2 h-4 w-4" />
   652→                  <span>Sign Out</span>
   653→                </DropdownMenuItem>
   654→              </DropdownMenuContent>
   655→            </DropdownMenu>
   656→          </div>
   657→        </header>
   658→
   659→        {/* ── Content Area ── */}
   660→        <main className="flex-1 overflow-auto">
   661→          <div key={view} className="view-enter">
   662→            <Suspense fallback={<ViewLoader />}>
   663→              {renderView()}
   664→            </Suspense>
   665→          </div>
   666→        </main>
   667→      </div>
   668→    </div>
   669→  );
   670→}
