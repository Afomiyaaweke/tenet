   143→   143→function formatCurrency(amount: number): string {
   144→   144→  return new Intl.NumberFormat('en-ET', { style: 'currency', currency: 'ETB', maximumFractionDigits: 0 }).format(amount);

   147→   147→function formatDate(dateStr: string): string {
   148→   148→  if (!dateStr) return '—';
   149→   149→  return new Date(dateStr).toLocaleDateString('en-GB', { day: '2-digit', month: 'short', year: 'numeric' });
   150→   150→}

   152→   152→function daysUntil(dateStr: string): number {
   153→   153→  return Math.ceil((new Date(dateStr).getTime() - Date.now()) / (1000 * 60 * 60 * 24));
   154→   154→}

   218→   218→      setLoading(false);

   229→   229→  }, [closedTenders]);

   232→   232→  const sortedRows = useMemo(() => {
   233→   233→    const copy = [...rows];
   234→   234→    copy.sort((a, b) => {
   235→   235→      const aVal = a[sortField];
   236→   236→      const bVal = b[sortField];
   237→   237→      if (aVal === bVal) return 0;
   238→   238→      if (aVal == null) return 1;
   239→   239→      if (bVal == null) return -1;
   240→   240→      let cmp = 0;
   241→   241→      if (typeof aVal === 'number' && typeof bVal === 'number') {
   242→   242→        cmp = aVal - bVal;
   243→   243→      } else if (typeof aVal === 'boolean' && typeof bVal === 'boolean') {
   244→   244→        cmp = Number(aVal) - Number(bVal);
   245→   245→      } else {
   246→   246→        cmp = String(aVal).localeCompare(String(bVal));
   247→   247→      }
   248→   248→      return sortDir === 'asc' ? cmp : -cmp;
   249→   249→    });
   250→   250→    return copy;
   251→   251→  }, [rows, sortField, sortDir]);
   252→   252→
   253→   253→  // Visible columns

   255→   255→    () => COLUMNS.filter(c => visibleColumns.has(c.key)),

   285→   285→      setSortDir('asc');

   295→   295→  };

   299→   299→    if (!rows.length) return;
   300→   300→    const headers = activeColumns.map(c => c.label);

   306→   306→        val = String(val || '');

   351→   351→            </p>
   352→   352→          </div>
   353→   353→        </div>
   354→   354→        <div className="flex items-center gap-2">
   355→   355→          <Button variant="outline" size="sm" onClick={fetchData}>
   356→   356→            <RefreshCw className={`h-4 w-4 mr-1 ${loading ? 'animate-spin' : ''}`} />
   357→   357→            Refresh

   359→   359→          <Button variant="outline" size="sm" onClick={exportCSV} disabled={!rows.length}>

   366→   366→      {/* Stats bar */}
   367→   367→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   368→   368→        <Card className="bg-gradient-to-br from-orange-50 to-white dark:from-orange-950/30 dark:to-background border-orange-200/50 dark:border-orange-800/30">
   369→   369→          <CardContent className="p-3 flex items-center gap-3">

   378→   378→        </Card>
