   295→
   296→function fmtMoney(amount: number, currency: string): string {
   297→  if (!amount) return '—';
   298→  try {
   299→    return new Intl.NumberFormat('en-US', {
   300→      style: 'currency',
   301→      currency: currency || 'USD',
   302→      maximumFractionDigits: 0,
   303→    }).format(amount);
   304→  } catch {
   305→    return `${currency} ${amount.toLocaleString()}`;
   306→  }
   307→}
   308→
   309→function daysUntil(dateStr: string): number {
   310→  if (!dateStr) return 999;
   311→  return Math.ceil((new Date(dateStr).getTime() - Date.now()) / (1000 * 60 * 60 * 24));
   312→}
   313→
   314→function deadlineBadge(days: number): string {
   315→  if (days <= 0) return 'bg-rose-50 text-rose-700 dark:bg-rose-950/40 dark:text-rose-300';
   316→  if (days <= 7) return 'bg-amber-50 text-amber-700 dark:bg-amber-950/40 dark:text-amber-300';
   317→  if (days <= 21) return 'bg-sky-50 text-sky-700 dark:bg-sky-950/40 dark:text-sky-300';
   318→  return 'bg-emerald-50 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300';
   319→}
   320→
   321→function deadlineLabel(days: number): string {
   322→  if (days <= 0) return 'Closed';
   323→  if (days <= 7) return `${days}d left`;
   324→  if (days <= 30) return `${days}d left`;
   325→  return `${days}d left`;
   326→}
   327→
   328→function relativeTime(dateStr: string): string {
   329→  if (!dateStr) return '';
   330→  const diff = Date.now() - new Date(dateStr).getTime();
   331→  const day = 86400000;
   332→  if (diff < day) return 'today';
   333→  if (diff < 2 * day) return 'yesterday';
   334→  if (diff < 30 * day) return `${Math.floor(diff / day)}d ago`;
   335→  return new Date(dateStr).toLocaleDateString();
   336→}
   337→
   338→function riskBadgeColor(level: string): string {
   339→  if (level === 'low') return 'bg-emerald-50 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300';
   340→  if (level === 'medium') return 'bg-amber-50 text-amber-700 dark:bg-amber-950/40 dark:text-amber-300';
   341→  return 'bg-rose-50 text-rose-700 dark:bg-rose-950/40 dark:text-rose-300';
   342→}
   343→
   344→function readinessColor(score: number): string {
   345→  if (score >= 8) return 'bg-emerald-500';
   346→  if (score >= 5) return 'bg-amber-500';
   347→  return 'bg-rose-500';

   350→/* ─────────────────────────────────────────────────────────────────────
   351→ * Inline Document Viewer Component
   352→ * ───────────────────────────────────────────────────────────────────── */
   353→
   354→function InlineDocumentViewer({ doc, onClose }: { doc: InlineDocument; onClose: () => void }) {
   355→  const copyContent = () => {
   356→    const text = doc.sections
   357→      ? doc.sections.map((s) => `${s.heading}\n${s.content}`).join('\n\n')
   358→      : doc.content;
   359→    navigator.clipboard.writeText(text);
   360→    toast.success('Content copied to clipboard');
   361→  };
   362→
   363→  return (
   364→    <div className="overflow-hidden animate-[fadeIn_0.3s_ease-out]">
   365→      <div className="border-t border-border bg-gradient-to-b from-muted/30 to-background">
   366→        <div className="p-4 md:p-6 space-y-4">
   367→          {/* Document header */}
   368→          <div className="flex items-start justify-between gap-3">
   369→            <div className="min-w-0 flex-1">
   370→              <div className="flex items-center gap-2 text-xs text-muted-foreground mb-1">
   371→                <BookOpen className="h-3.5 w-3.5" />
   372→                <span>Full tender content loaded from {new URL(doc.url).hostname}</span>
   373→                <span>·</span>
   374→                <span>{new Date(doc.fetchedAt).toLocaleTimeString()}</span>
   375→              </div>
   376→              <h3 className="text-lg font-semibold text-foreground leading-snug">
   377→                {doc.title || 'Tender Document'}
   378→              </h3>
   379→              {doc.metaDescription && (
   380→                <p className="text-sm text-muted-foreground mt-1 line-clamp-2">
   381→                  {doc.metaDescription}
   382→                </p>
   383→              )}
   384→            </div>
   385→            <div className="flex items-center gap-2 shrink-0">
   386→              <Button variant="ghost" size="sm" className="gap-1.5 text-xs" onClick={copyContent}>
   387→                <Copy className="h-3.5 w-3.5" />
   388→                Copy
   389→              </Button>
   390→              <a
   391→                href={doc.url}
   392→                target="_blank"
   393→                rel="noopener noreferrer"
   394→                className="inline-flex items-center gap-1.5 text-xs font-medium text-primary hover:underline"
   395→              >
   396→                <ExternalLink className="h-3.5 w-3.5" />
   397→                Open original
   398→              </a>
   399→              <Button variant="ghost" size="sm" onClick={onClose} className="gap-1.5 text-xs text-muted-foreground">
   400→                <ChevronUp className="h-3.5 w-3.5" />
   401→                Collapse
   402→              </Button>
   403→            </div>
   404→          </div>
   405→
   406→          {/* Extracted metadata pills */}
   407→          {(doc.deadlines && doc.deadlines.length > 0) || (doc.budgets && doc.budgets.length > 0) ? (
   408→            <div className="flex flex-wrap gap-2">
   409→              {doc.deadlines?.map((d, i) => (
   410→                <Badge key={`dl-${i}`} variant="outline" className="gap-1.5 text-xs border-amber-300 text-amber-700 dark:border-amber-800 dark:text-amber-300">
   411→                  <Calendar className="h-3 w-3" />
   412→                  {d}
   413→                </Badge>
   414→              ))}
   415→              {doc.budgets?.map((b, i) => (
   416→                <Badge key={`bg-${i}`} variant="outline" className="gap-1.5 text-xs border-emerald-300 text-emerald-700 dark:border-emerald-800 dark:text-emerald-300">
   417→                  <DollarSign className="h-3 w-3" />
   418→                  {b}
   419→                </Badge>
   420→              ))}
   421→            </div>
   422→          ) : null}
   423→
   424→          {/* Sections view */}
   425→          {doc.sections && doc.sections.length > 0 ? (
   426→            <div className="max-h-96 overflow-y-auto rounded-lg border border-border bg-background p-4 space-y-4 scrollbar-thin">
   427→              {doc.sections.map((section, i) => (
   428→                <div key={i}>
   429→                  <h4 className="text-sm font-semibold text-foreground mb-1.5">{section.heading}</h4>
   430→                  <p className="text-sm text-muted-foreground leading-relaxed whitespace-pre-line">
   431→                    {section.content}
   432→                  </p>
   433→                  {i < doc.sections!.length - 1 && <Separator className="mt-4 bg-border" />}
   434→                </div>
   435→              ))}
   436→            </div>
   437→          ) : (
   438→            <div className="max-h-96 overflow-y-auto rounded-lg border border-border bg-background p-4 scrollbar-thin">
   439→              <p className="text-sm text-muted-foreground leading-relaxed whitespace-pre-line">
   440→                {doc.content || 'No content could be extracted from this page.'}
   441→              </p>
   442→            </div>
   443→          )}
   444→        </div>
   445→      </div>
   446→    </div>
   447→  );
   448→}
