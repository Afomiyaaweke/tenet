    27→// ─── Helpers ────────────────────────────────────────────────────────
    28→function formatETB(amount: number): string {
    29→  if (amount >= 1_000_000) return `ETB ${(amount / 1_000_000).toFixed(1)}M`;
    30→  if (amount >= 1_000) return `ETB ${(amount / 1_000).toFixed(0)}K`;
    31→  return `ETB ${amount.toLocaleString()}`;
    32→}
    33→
    34→function statusColorMap(status: string) {
    35→  switch (status) {
    36→    case 'active': return { bg: 'bg-emerald-100 dark:bg-emerald-900/30', text: 'text-emerald-700 dark:text-emerald-300', dot: 'bg-emerald-500', bar: '#10b981', label: 'Active' };
    37→    case 'completed': return { bg: 'bg-teal-100 dark:bg-teal-900/30', text: 'text-teal-700 dark:text-teal-300', dot: 'bg-teal-500', bar: '#14b8a6', label: 'Completed' };
    38→    case 'on_hold': return { bg: 'bg-amber-100 dark:bg-amber-900/30', text: 'text-amber-700 dark:text-amber-300', dot: 'bg-amber-500', bar: '#f59e0b', label: 'On Hold' };
    39→    case 'cancelled': return { bg: 'bg-rose-100 dark:bg-rose-900/30', text: 'text-rose-700 dark:text-rose-300', dot: 'bg-rose-500', bar: '#f43f5e', label: 'Cancelled' };
    40→    default: return { bg: 'bg-muted', text: 'text-muted-foreground', dot: 'bg-muted-foreground/50', bar: '#94a3b8', label: status };
    41→  }
    42→}

    45→function getTimelineRange(project: Project) {
    46→  const now = new Date();
    47→  const start = new Date(now.getFullYear(), now.getMonth() - 2, 1);
    48→  const end = new Date(now.getFullYear(), now.getMonth() + 8, 0);
    49→  return { start, end };
    50→}
    51→
    52→function monthRange(start: Date, end: Date): Date[] {
    53→  const months: Date[] = [];
    54→  const d = new Date(start.getFullYear(), start.getMonth(), 1);
    55→  while (d <= end) {
    56→    months.push(new Date(d));
    57→    d.setMonth(d.getMonth() + 1);
    58→  }
    59→  return months;
    60→}
    61→
    62→function dateToX(date: Date, start: Date, end: Date, width: number): number {
    63→  const total = end.getTime() - start.getTime();
    64→  if (total === 0) return 0;
    65→  return ((date.getTime() - start.getTime()) / total) * width;
    66→}

   186→
   187→  const [showAddPayment, setShowAddPayment] = useState(false);

   193→    if (res.success) setProject(res.data);
   194→    setLoading(false);

   196→
   197→  /* eslint-disable react-hooks/set-state-in-effect */

   202→

   217→    if (res.success) {
   218→      toast.success('Task updated');
   219→      loadProject();
   220→    } else toast.error(res.error || 'Failed to update task');
   221→  };
   222→

   230→

   235→  };

   238→    if (!editingTask || !editTaskForm.title.trim()) return;
   239→    const res = await api.patch(`/projects/${projectId}/tasks/${editingTask.id}`, editTaskForm);
   240→    if (res.success) {
   241→      toast.success('Task updated');
   242→      setShowEditTask(false);
   243→      setEditingTask(null);
   244→      loadProject();
   245→    } else toast.error(res.error || 'Failed to update task');
   246→  };
   247→
   248→  // ── Milestone Handlers ──
   249→  const handleAddMilestone = async () => {
   250→    if (!newMilestone.title.trim()) { toast.error('Milestone title is required'); return; }
   251→    const res = await api.post(`/projects/${projectId}/milestones`, newMilestone);
   252→    if (res.success) {
   253→      toast.success('Milestone added');
   254→      setShowAddMilestone(false);
   255→      setNewMilestone({ title: '', dueDate: '' });
   256→      loadProject();
   257→    } else toast.error(res.error || 'Failed to add milestone');
   258→  };

   263→      toast.success(completed ? 'Milestone reopened' : 'Milestone completed');
   264→      loadProject();
   265→    }

   268→  // ── Payment Handlers ──
   269→  const handleAddPayment = async () => {

   271→      ...newPayment, amount: parseFloat(newPayment.amount),

   314→  const payments = project.payments || [];

   318→  return (

   839→// ─── Project Timeline Sub-Component ──────────────────────────────────
   840→function ProjectTimeline({
   841→  project,
   842→  onToggleMilestone,
   843→}: {
   844→  project: Project;
   845→  onToggleMilestone: (id: string, completed: boolean) => void;
   846→}) {
   847→  const timelineWidth = 800;
   848→  const headerHeight = 36;
   849→  const milestones = project.milestones || [];
   850→  const rowHeight = 44;
   851→
   852→  const { start, end } = useMemo(() => getTimelineRange(project), [project]);
   853→  const months = useMemo(() => monthRange(start, end), [start, end]);
   854→
   855→  const createdAt = new Date(project.createdAt);
   856→  const lastMilestone = milestones.sort((a, b) => new Date(b.dueDate).getTime() - new Date(a.dueDate).getTime())[0];
   857→  const projectEnd = lastMilestone ? new Date(lastMilestone.dueDate) : new Date(createdAt.getTime() + 90 * 86400000);
   858→
   859→  const sc = statusColorMap(project.status);
   860→  const taskProgress = (project.tasks || []).length > 0
   861→    ? Math.round((project.tasks!.filter(t => t.status === 'done').length / project.tasks!.length) * 100)
   862→    : 0;
   863→
   864→  const nowX = dateToX(new Date(), start, end, timelineWidth);
   865→
   866→  const milestoneData = milestones.map(m => ({
   867→    ...m,
   868→    x: dateToX(new Date(m.dueDate), start, end, timelineWidth),
   869→  }));
   870→
   871→  const svgHeight = headerHeight + rowHeight + 20;
   872→
   873→  return (
   874→    <div className="space-y-3">
   875→      <Card className="rounded-xl border border-border/60 bg-card overflow-hidden">
   876→        <div className="overflow-x-auto">
   877→          <svg width={timelineWidth + 180} height={svgHeight} className="min-w-full">
   878→            {/* Month columns */}
   879→            {months.map((m, i) => {
   880→              const x = dateToX(m, start, end, timelineWidth) + 180;
   881→              const nextM = new Date(m);
   882→              nextM.setMonth(nextM.getMonth() + 1);
   883→              const x2 = dateToX(nextM, start, end, timelineWidth) + 180;
   884→              return (
   885→                <g key={i}>
   886→                  <rect x={x} y={0} width={x2 - x} height={svgHeight} fill={i % 2 === 0 ? 'transparent' : 'rgba(128,128,128,0.03)'} />
   887→                  <text x={x + (x2 - x) / 2} y={22} textAnchor="middle" className="text-[10px] fill-muted-foreground font-medium">
   888→                    {m.toLocaleDateString('en-US', { month: 'short', year: '2-digit' })}
   889→                  </text>
   890→                  <line x1={x} y1={headerHeight} x2={x} y2={svgHeight} stroke="currentColor" strokeOpacity={0.06} />
   891→                </g>
   892→              );
   893→            })}
   894→
   895→            {/* Today line */}
   896→            <line x1={nowX + 180} y1={0} x2={nowX + 180} y2={svgHeight} stroke="#f43f5e" strokeWidth={1.5} strokeDasharray="4 3" />
   897→            <text x={nowX + 180} y={12} textAnchor="middle" className="text-[9px] fill-rose-500 font-semibold">TODAY</text>
   898→
   899→            {/* Project bar */}
   900→            {(() => {
   901→              const x1 = dateToX(createdAt, start, end, timelineWidth) + 180;
   902→              const x2 = dateToX(projectEnd, start, end, timelineWidth) + 180;
   903→              const y = headerHeight + 12;
   904→              const barH = 20;
   905→              const barW = Math.max(x2 - x1, 8);
   906→              return (
   907→                <g>
   908→                  <rect x={x1} y={y} width={barW} height={barH} rx={4} fill={sc.bar} fillOpacity={0.2} />
   909→                  <rect x={x1} y={y} width={barW * (taskProgress / 100)} height={barH} rx={4} fill={sc.bar} />
   910→                </g>
   911→              );
   912→            })()}
   913→
   914→            {/* Milestone diamonds */}
   915→            {milestoneData.map((m, i) => {
   916→              const y = headerHeight + rowHeight / 2;
   917→              return (
   918→                <g key={m.id} className="cursor-pointer" onClick={() => onToggleMilestone(m.id, m.completed)}>
   919→                  <polygon
   920→                    points={`${m.x + 180},${y - 6} ${m.x + 186},${y} ${m.x + 180},${y + 6} ${m.x + 174},${y}`}
   921→                    fill={m.completed ? '#10b981' : sc.bar}
   922→                    stroke="white"
   923→                    strokeWidth={1.5}
   924→                  />
   925→                  <text x={m.x + 180} y={y + 18} textAnchor="middle" className={`text-[9px] ${m.completed ? 'fill-emerald-500' : 'fill-muted-foreground'} font-medium`}>
   926→                    {m.title.slice(0, 12)}
