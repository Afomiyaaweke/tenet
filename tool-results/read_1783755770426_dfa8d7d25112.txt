   143→function formatCurrency(amount: number): string {
   144→  return new Intl.NumberFormat('en-ET', { style: 'currency', currency: 'ETB', maximumFractionDigits: 0 }).format(amount);
   145→}
   146→
   147→function formatDate(dateStr: string): string {
   148→  if (!dateStr) return '—';
   149→  return new Date(dateStr).toLocaleDateString('en-GB', { day: '2-digit', month: 'short', year: 'numeric' });
   150→}
   151→
   152→function daysUntil(dateStr: string): number {
   153→  return Math.ceil((new Date(dateStr).getTime() - Date.now()) / (1000 * 60 * 60 * 24));
   154→}

   173→export function ApplicantsView() {
   174→  const { user } = useAuthStore();
   175→  const { setView } = useNavStore();
   176→  const [rows, setRows] = useState<ApplicantRow[]>([]);
   177→  const [meta, setMeta] = useState<ApplicantMeta | null>(null);
   178→  const [openTenders, setOpenTenders] = useState<TenderInfo[]>([]);
   179→  const [closedTenders, setClosedTenders] = useState<TenderInfo[]>([]);
   180→  const [loading, setLoading] = useState(true);
   181→  const [search, setSearch] = useState('');
   182→  const [statusFilter, setStatusFilter] = useState<string>('all');
   183→  const [tenderFilter, setTenderFilter] = useState<string>('all');
   184→  const [sortField, setSortField] = useState<SortField>('submittedAt');
   185→  const [sortDir, setSortDir] = useState<SortDir>('desc');
   186→  const [viewMode, setViewMode] = useState<ViewMode>('table');
   187→  const [expandedRow, setExpandedRow] = useState<string | null>(null);
   188→  const [visibleColumns, setVisibleColumns] = useState<Set<string>>(() =>
   189→    new Set(COLUMNS.filter(c => c.defaultVisible).map(c => c.key))
   190→  );
   191→  const [columnMenuOpen, setColumnMenuOpen] = useState(false);
   192→  const [page, setPage] = useState(1);
   193→  const PAGE_SIZE = 50;
   194→
   195→  // Fetch data
   196→  const fetchData = useCallback(async () => {
   197→    setLoading(true);
   198→    try {
   199→      const params: Record<string, string> = {
   200→        page: String(page),
   201→        limit: String(PAGE_SIZE),
   202→      };
   203→      if (search) params.search = search;
   204→      if (statusFilter && statusFilter !== 'all') params.status = statusFilter;
   205→      if (tenderFilter && tenderFilter !== 'all') params.tenderId = tenderFilter;
   206→
   207→      const res = await api.get('/applicants', params);
   208→      if (res.success) {
   209→        setRows(res.data);
   210→        setMeta(res.meta);
   211→        setOpenTenders(res.openTenders || []);
   212→        setClosedTenders(res.closedTenders || []);
   213→      }
   214→    } catch (err) {
   215→      console.error('Fetch applicants error:', err);
   216→      toast.error('Failed to load applicants');
   217→    } finally {
   218→      setLoading(false);
   219→    }
   220→  }, [page, search, statusFilter, tenderFilter]);
   221→
   222→  useEffect(() => {
   223→    fetchData();
   224→  }, [fetchData]);
   225→
   226→  // Get unique tenders for filter dropdown (from closedTenders API data)
   227→  const tenderOptions = useMemo(() => {
   228→    return closedTenders.map(t => [t.id, t.title] as [string, string]);
   229→  }, [closedTenders]);
   230→
   231→  // Sorted rows (client-side sorting on current page)
   232→  const sortedRows = useMemo(() => {
   233→    const copy = [...rows];
   234→    copy.sort((a, b) => {
   235→      const aVal = a[sortField];
   236→      const bVal = b[sortField];
   237→      if (aVal === bVal) return 0;
   238→      if (aVal == null) return 1;
   239→      if (bVal == null) return -1;
   240→      let cmp = 0;
   241→      if (typeof aVal === 'number' && typeof bVal === 'number') {
   242→        cmp = aVal - bVal;
   243→      } else if (typeof aVal === 'boolean' && typeof bVal === 'boolean') {
   244→        cmp = Number(aVal) - Number(bVal);
   245→      } else {
   246→        cmp = String(aVal).localeCompare(String(bVal));
   247→      }
   248→      return sortDir === 'asc' ? cmp : -cmp;
   249→    });
   250→    return copy;
   251→  }, [rows, sortField, sortDir]);
   252→
   253→  // Visible columns
   254→  const activeColumns = useMemo(
   255→    () => COLUMNS.filter(c => visibleColumns.has(c.key)),
   256→    [visibleColumns]
   257→  );
   258→
   259→  // Column groups
   260→  const columnGroups = useMemo(() => {
   261→    const groups: Record<string, ColumnDef[]> = {};
   262→    activeColumns.forEach(c => {
   263→      if (!groups[c.group]) groups[c.group] = [];
   264→      groups[c.group].push(c);
   265→    });
   266→    return groups;
   267→  }, [activeColumns]);
   268→
   269→  // Toggle column visibility
   270→  const toggleColumn = (key: string) => {
   271→    setVisibleColumns(prev => {
   272→      const next = new Set(prev);
   273→      if (next.has(key)) next.delete(key);
   274→      else next.add(key);
   275→      return next;
   276→    });
   277→  };
   278→
   279→  // Toggle sort
   280→  const handleSort = (field: SortField) => {
   281→    if (sortField === field) {
   282→      setSortDir(prev => prev === 'asc' ? 'desc' : 'asc');
   283→    } else {
   284→      setSortField(field);
   285→      setSortDir('asc');
   286→    }
   287→  };
   288→
   289→  // Sort icon
   290→  const SortIcon = ({ field }: { field: SortField }) => {
   291→    if (sortField !== field) return <ArrowUpDown className="h-3 w-3 text-muted-foreground/40" />;
   292→    return sortDir === 'asc'
   293→      ? <ArrowUp className="h-3 w-3 text-primary" />
   294→      : <ArrowDown className="h-3 w-3 text-primary" />;
   295→  };
   296→
   297→  // Export CSV
   298→  const exportCSV = () => {
   299→    if (!rows.length) return;
   300→    const headers = activeColumns.map(c => c.label);
   301→    const csvRows = rows.map(row =>
   302→      activeColumns.map(c => {
   303→        let val = row[c.key];
   304→        if (typeof val === 'boolean') val = val ? 'Yes' : 'No';
   305→        if (typeof val === 'number') val = val.toLocaleString();
   306→        val = String(val || '');
   307→        // Escape CSV
   308→        if (val.includes(',') || val.includes('"') || val.includes('\n')) {
   309→          val = `"${val.replace(/"/g, '""')}"`;
   310→        }
   311→        return val;
   312→      }).join(',')
   313→    );
   314→    const csv = [headers.join(','), ...csvRows].join('\n');
   315→    const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });
   316→    const url = URL.createObjectURL(blob);
   317→    const link = document.createElement('a');
   318→    link.href = url;
   319→    link.download = `applicants-export-${new Date().toISOString().slice(0, 10)}.csv`;
   320→    link.click();
   321→    URL.revokeObjectURL(url);
   322→    toast.success('Spreadsheet exported successfully');
   323→  };
   324→
   325→  // Stats
   326→  const statusCounts = meta?.statusCounts || {};
   327→  const totalApplicants = meta?.total || 0;
   328→  const totalBudget = meta?.totalBudgetSum || 0;
   329→  const uniqueTenders = meta?.uniqueTenders || 0;
   330→  const uniqueCompanies = meta?.uniqueCompanies || 0;
   331→
   332→  // ─── Render ───────────────────────────────────────────────────────
   333→
   334→  return (
   335→    <div className="flex flex-col gap-4 p-4 md:p-6">
   336→      {/* Header */}
   337→      <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-3">
   338→        <div className="flex items-center gap-3">
   339→          <div className="p-2 rounded-xl bg-orange-100 dark:bg-orange-900/30">
   340→            <FileSpreadsheet className="h-6 w-6 text-orange-600 dark:text-orange-400" />
   341→          </div>
   342→          <div>
   343→            <h1 className="text-xl font-bold">My Applicants</h1>
   344→            <p className="text-sm text-muted-foreground">
   345→              {totalApplicants > 0
   346→                ? `${totalApplicants} applicant${totalApplicants !== 1 ? 's' : ''} across ${uniqueTenders} closed tender${uniqueTenders !== 1 ? 's' : ''}`
   347→                : openTenders.length > 0
   348→                  ? `${openTenders.length} tender${openTenders.length !== 1 ? 's' : ''} still accepting bids`
   349→                  : 'Applicants for your published tenders'
   350→              }
   351→            </p>
   352→          </div>
   353→        </div>
   354→        <div className="flex items-center gap-2">
   355→          <Button variant="outline" size="sm" onClick={fetchData}>
   356→            <RefreshCw className={`h-4 w-4 mr-1 ${loading ? 'animate-spin' : ''}`} />
   357→            Refresh
   358→          </Button>
   359→          <Button variant="outline" size="sm" onClick={exportCSV} disabled={!rows.length}>
   360→            <Download className="h-4 w-4 mr-1" />
   361→            Export CSV
   362→          </Button>
   363→        </div>
   364→      </div>
   365→
   366→      {/* Stats bar */}
   367→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   368→        <Card className="bg-gradient-to-br from-orange-50 to-white dark:from-orange-950/30 dark:to-background border-orange-200/50 dark:border-orange-800/30">
   369→          <CardContent className="p-3 flex items-center gap-3">
   370→            <div className="p-2 rounded-lg bg-orange-100 dark:bg-orange-900/30">
   371→              <Users className="h-5 w-5 text-orange-600 dark:text-orange-400" />
   372→            </div>
   373→            <div>
   374→              <p className="text-xs text-muted-foreground">Total Applicants</p>
   375→              <p className="text-lg font-bold">{totalApplicants}</p>
   376→            </div>
   377→          </CardContent>
   378→        </Card>
   379→        <Card className="bg-gradient-to-br from-emerald-50 to-white dark:from-emerald-950/30 dark:to-background border-emerald-200/50 dark:border-emerald-800/30">
   380→          <CardContent className="p-3 flex items-center gap-3">
   381→            <div className="p-2 rounded-lg bg-emerald-100 dark:bg-emerald-900/30">
   382→              <DollarSign className="h-5 w-5 text-emerald-600 dark:text-emerald-400" />
   383→            </div>
   384→            <div>
   385→              <p className="text-xs text-muted-foreground">Total Bid Value</p>

   387→            </div>
   388→          </CardContent>
   389→        </Card>

   391→          <CardContent className="p-3 flex items-center gap-3">
   392→            <div className="p-2 rounded-lg bg-blue-100 dark:bg-blue-900/30">
   393→              <FileSearch className="h-5 w-5 text-blue-600 dark:text-blue-400" />
   394→            </div>
   395→            <div>
   396→              <p className="text-xs text-muted-foreground">Tenders</p>
   397→              <p className="text-lg font-bold">{uniqueTenders}</p>
   398→            </div>
   399→          </CardContent>
   400→        </Card>

   411→        </Card>
   412→      </div>
   413→
   414→      {/* Info banner: explain visibility rules */}
   415→      <Card className="bg-amber-50 dark:bg-amber-950/20 border-amber-200/60 dark:border-amber-800/30">
   416→        <CardContent className="p-3 flex items-start gap-3">
   417→          <Lock className="h-5 w-5 text-amber-600 dark:text-amber-400 shrink-0 mt-0.5" />
   418→          <div>
   419→            <p className="text-sm font-semibold text-amber-800 dark:text-amber-300">Applicants are visible only after the tender deadline closes</p>
   420→            <p className="text-xs text-amber-700/70 dark:text-amber-400/60 mt-0.5">
   421→              To protect bid integrity, applicant details remain sealed while your tender is still accepting bids.
   422→              {closedTenders.length > 0 && ` You have ${closedTenders.length} closed tender${closedTenders.length !== 1 ? 's' : ''} with visible applicants.`}
   423→              {openTenders.length > 0 && ` ${openTenders.length} tender${openTenders.length !== 1 ? 's' : ''} still open.`}

   425→          </div>
   426→        </CardContent>
   427→      </Card>
   428→
   429→      {/* Open tenders still accepting bids */}
   430→      {openTenders.length > 0 && (
   431→        <Card className="border-amber-200/60 dark:border-amber-800/30">
   432→          <CardContent className="p-4">
   433→            <div className="flex items-center gap-2 mb-3">

   745→                <ChevronRight className="h-4 w-4" />

   832→                      variant="ghost"

   841→              </Card>

   843→          })}
   844→        </div>
   845→      )}
   846→    </div>
   847→  );

   851→
   852→function CellRenderer({ row, col, setView }: { row: ApplicantRow; col: ColumnDef; setView: (v: string, p?: Record<string, string>) => void }) {
   853→  const val = row[col.key];
   854→
   855→  switch (col.key) {
   856→    case 'applicantName':
   857→      return (
   858→        <div className="flex items-center gap-2 min-w-0">
   859→          <div className="h-7 w-7 rounded-full bg-gradient-to-br from-orange-200 to-orange-400 dark:from-orange-800 dark:to-orange-600 flex items-center justify-center text-white font-bold text-[10px] shrink-0">
   860→            {String(val).charAt(0).toUpperCase()}
   861→          </div>
   862→          <div className="min-w-0">
   863→            <div className="flex items-center gap-1">
   864→              <span className="font-medium truncate">{String(val)}</span>
   865→              {row.applicantVerified && <ShieldCheck className="h-3 w-3 text-emerald-500 shrink-0" />}
   866→            </div>
   867→            <span className="text-[10px] text-muted-foreground truncate block">{row.applicantJobTitle}</span>
   868→          </div>
   869→        </div>
   870→      );
   871→
   872→    case 'applicantEmail':
   873→      return (
   874→        <div className="flex items-center gap-1.5 min-w-0">
   875→          <Mail className="h-3 w-3 text-muted-foreground shrink-0" />
   876→          <span className="truncate text-muted-foreground">{String(val)}</span>
   877→        </div>
   878→      );
   879→
   880→    case 'applicantPhone':
   881→      return (
   882→        <div className="flex items-center gap-1.5 min-w-0">
   883→          <Phone className="h-3 w-3 text-muted-foreground shrink-0" />
   884→          <span className="truncate">{String(val) || '—'}</span>
   885→        </div>
   886→      );
   887→
   888→    case 'applicantVerified':
   889→    case 'companyVerified':
   890→      return val
   891→        ? <Badge className="text-[9px] px-1.5 py-0 bg-emerald-100 text-emerald-700 border-0 rounded-md"><ShieldCheck className="h-3 w-3 mr-0.5" />Yes</Badge>
   892→        : <Badge className="text-[9px] px-1.5 py-0 bg-gray-100 text-gray-500 border-0 rounded-md">No</Badge>;
   893→
   894→    case 'companyName':
   895→      return (
   896→        <div className="flex items-center gap-1.5 min-w-0">
   897→          <Building2 className="h-3 w-3 text-violet-500 shrink-0" />
   898→          <span className="font-medium truncate">{String(val) || 'Individual'}</span>
   899→        </div>
   900→      );
   901→
   902→    case 'companyIndustry':
   903→      return (
   904→        <Badge variant="secondary" className="text-[10px] px-1.5 py-0">
   905→          <Briefcase className="h-2.5 w-2.5 mr-0.5" />
   906→          {String(val) || '—'}
   907→        </Badge>
   908→      );
   909→
   910→    case 'tenderTitle':
   911→      return (
   912→        <span
   913→          className="text-primary font-medium truncate cursor-pointer hover:underline block"
