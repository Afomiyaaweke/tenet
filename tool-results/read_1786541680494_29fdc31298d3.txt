    95→/* ───────────────────────── Password Strength Meter ───────────────────────── */
    96→function getPasswordStrength(password: string): { score: number; label: string; color: string } {
    97→  let score = 0;
    98→  if (password.length >= 8) score += 25;
    99→  if (password.length >= 12) score += 10;
   100→  if (/[A-Z]/.test(password)) score += 20;
   101→  if (/[a-z]/.test(password)) score += 20;
   102→  if (/[0-9]/.test(password)) score += 20;
   103→  if (/[^A-Za-z0-9]/.test(password)) score += 5;  // bonus only
   104→  if (new Set(password.toLowerCase()).size >= 8) score += 5;
   105→
   106→  if (score >= 70) return { score, label: 'Strong', color: 'bg-green-500' };
   107→  if (score >= 50) return { score, label: 'Good', color: 'bg-blue-500' };
   108→  if (score >= 30) return { score, label: 'Fair', color: 'bg-yellow-500' };
   109→  return { score, label: 'Weak', color: 'bg-red-500' };
   110→}
   111→
   112→function PasswordStrengthMeter({ password }: { password: string }) {
   113→  if (!password) return null;
   114→  const { score, label, color } = getPasswordStrength(password);
   115→  return (
   116→    <div className="space-y-1.5 mt-2">
   117→      <div className="flex gap-1">
   118→        {[25, 50, 75, 100].map(threshold => (
   119→          <div key={threshold} className="h-1.5 flex-1 rounded-full bg-muted">
   120→            <div
   121→              className={`h-full rounded-full transition-all duration-300 ${score >= threshold ? color : 'bg-transparent'}`}
   122→              style={{ width: score >= threshold ? '100%' : '0%' }}
   123→            />
   124→          </div>
   125→        ))}
   126→      </div>
   127→      <div className="flex justify-between items-center">
   128→        <span className="text-xs text-muted-foreground">Strength:</span>
   129→        <span className={`text-xs font-medium ${score >= 80 ? 'text-green-600' : score >= 60 ? 'text-blue-600' : score >= 40 ? 'text-yellow-600' : 'text-red-600'}`}>{label}</span>
   130→      </div>
   131→    </div>
   132→  );
   133→}

   158→/* ───────────────────────── Step Indicator ───────────────────────── */
   159→function StepIndicator({ currentStep, totalSteps = 4 }: { currentStep: RegStep; totalSteps?: number }) {
   160→  return (
   161→    <div className="flex items-center justify-center gap-1 mb-6">
   162→      {Array.from({ length: totalSteps }, (_, i) => (i + 1) as RegStep).map((step, i) => {
   163→        const meta = REG_STEP_META[step];
   164→        const isCompleted = currentStep > step;
   165→        const isCurrent = currentStep === step;
   166→        const Icon = meta.icon;
   167→        return (
   168→          <React.Fragment key={step}>
   169→            <div className="flex items-center gap-1.5">
   170→              <div
   171→                className={`h-7 w-7 rounded-full flex items-center justify-center text-xs font-bold transition-all ${
   172→                  isCompleted || isCurrent
   173→                    ? 'gradient-orange text-white shadow-md shadow-orange-500/30'
   174→                    : 'bg-muted text-muted-foreground'
   175→                }`}
   176→              >
   177→                {isCompleted ? <CheckCircle2 className="w-4 h-4" /> : <Icon className="w-3.5 h-3.5" />}
   178→              </div>
   179→              <span className={`text-xs font-medium hidden sm:inline ${isCompleted || isCurrent ? 'text-foreground' : 'text-muted-foreground'}`}>
   180→                {meta.label}
   181→              </span>
   182→            </div>
   183→            {i < totalSteps - 1 && (
   184→              <div className={`h-0.5 w-4 sm:w-8 rounded-full ${currentStep > step ? 'bg-orange-500' : 'bg-muted'}`} />
   185→            )}
   186→          </React.Fragment>
   187→        );
   188→      })}
   189→    </div>
   190→  );
   191→}

   195→export function AuthGate({ onBack }: { onBack?: () => void }) {
   196→  const { login, register } = useAuthStore();
   197→  const [loginData, setLoginData] = useState({ email: '', password: '' });
   198→  const [regData, setRegData] = useState({
   199→    email: '', password: '',
   200→    // Personal
   201→    fullName: '', jobTitle: '', phone: '', location: '',
   202→    // Company
   203→    companyName: '', companyIndustry: '', companyTinNumber: '',
   204→    companyRegistrationNo: '', companyPhone: '', companyCity: '',
   205→    companyCountry: '', companyEmail: '', companyWebsite: '',
   206→
   207→  });
   208→  const [loading, setLoading] = useState(false);
   209→  const [emailChecking, setEmailChecking] = useState(false);
   210→
   211→  const [activeTab, setActiveTab] = useState<'login' | 'register'>('login');
   212→  const [showPassword, setShowPassword] = useState(false);
   213→  const [regStep, setRegStep] = useState<RegStep>(1);
   214→
   215→  // Forgot / Reset password state
   216→  const [authMode, setAuthMode] = useState<'login' | 'register' | 'forgot-password' | 'forgot-sent' | 'reset-password'>('login');
   217→  const [forgotEmail, setForgotEmail] = useState('');
   218→  const [resetToken, setResetToken] = useState('');
   219→  const [tokenValidation, setTokenValidation] = useState<'idle' | 'checking' | 'valid' | 'invalid'>('idle');
   220→  const [newPassword, setNewPassword] = useState('');
   221→  const [confirmPassword, setConfirmPassword] = useState('');
   222→  const [regConfirmPassword, setRegConfirmPassword] = useState('');
   223→  const [resetSuccess, setResetSuccess] = useState(false);
   224→
   225→  // Handle ?token=xxx in URL for email reset links
   226→  useEffect(() => {
   227→    const params = new URLSearchParams(window.location.search);
   228→    const tokenFromUrl = params.get('token');
   229→    if (tokenFromUrl) {
   230→      setResetToken(tokenFromUrl);
   231→      setAuthMode('reset-password');
   232→      // Clean up URL without reloading
   233→      window.history.replaceState({}, '', window.location.pathname);
   234→    }
   235→  }, []);
   236→
   237→  // Debounced token validation — checks the code against the server as user types
   238→  useEffect(() => {
   239→    if (!resetToken || resetToken.length < 10) {
   240→      setTokenValidation('idle');
   241→      return;
   242→    }
   243→
   244→    const timer = setTimeout(async () => {
   245→      setTokenValidation('checking');
   246→      try {
   247→        const { api } = await import('@/lib/api');
   248→        const res = await api.get(`/auth/validate-reset-token?token=${encodeURIComponent(resetToken)}`);
   249→        setTokenValidation(res.valid ? 'valid' : 'invalid');
   250→      } catch {
   251→        setTokenValidation('idle'); // Network error — don't block
   252→      }
   253→    }, 600); // 600ms debounce
   254→
   255→    return () => clearTimeout(timer);
   256→  }, [resetToken]);
   257→
   258→  const handleLoginSubmit = async (e: React.FormEvent) => {
   259→    e.preventDefault();
   260→    if (!loginData.email || !loginData.password) return;
   261→    setLoading(true);
   262→    try {
   263→      const result = await login(loginData.email, loginData.password);
   264→      if (!result.success) {
   265→        toast.error(result.error || 'Invalid email or password. Please try again.');
   266→      }
   267→    } catch {
   268→      toast.error('Login failed. Please check your connection and try again.');
   269→    }
   270→    setLoading(false);
   271→  };
   272→
   273→  const handleForgotPassword = async (e: React.FormEvent) => {
   274→    e.preventDefault();
   275→    if (!forgotEmail) return;
   276→    setLoading(true);
   277→    try {
   278→      const { api } = await import('@/lib/api');
   279→      const res = await api.post('/auth/forgot-password', { email: forgotEmail });
   280→      if (res.success) {
   281→        // Always show the confirmation screen to prevent email enumeration
   282→        // The reset token is sent ONLY to the user's email - never in the API response
   283→        setResetToken(''); // Clear any previous token
   284→        setTokenValidation('idle');
   285→        setAuthMode('forgot-sent');
   286→      } else {
   287→        toast.error(res.error || 'Failed to process request');
   288→      }
   289→    } catch {
   290→      toast.error('Network error. Please try again.');
   291→    }
   292→    setLoading(false);
   293→  };
   294→
   295→  const handleResetPassword = async (e: React.FormEvent) => {
   296→    e.preventDefault();
   297→    if (!resetToken || !newPassword || !confirmPassword) return;
   298→    if (newPassword !== confirmPassword) {
   299→      toast.error('Passwords do not match');
   300→      return;
   301→    }
   302→    if (newPassword.length < 8) {
   303→      toast.error('Password must be at least 8 characters');
   304→      return;
   305→    }
   306→    if (!/[A-Z]/.test(newPassword) || !/[a-z]/.test(newPassword) || !/[0-9]/.test(newPassword)) {
   307→      toast.error('Password must include uppercase, lowercase, and number');
   308→      return;
   309→    }
   310→    setLoading(true);
   311→    try {
   312→      const { api } = await import('@/lib/api');
   313→      const res = await api.post('/auth/reset-password', {
   314→        token: resetToken,
   315→        newPassword,
   316→        confirmPassword,
   317→      });
   318→      if (res.success) {
   319→        setResetSuccess(true);
   320→        toast.success('Password reset successfully! You can now sign in.');
   321→      } else {
   322→        toast.error(res.error || 'Failed to reset password');
   323→      }
   324→    } catch {
   325→      toast.error('Network error. Please try again.');
   326→    }
   327→    setLoading(false);
   328→  };
   329→
   330→  /* ─── Registration step navigation ─── */
   331→  const canGoNext = (): boolean => {
   332→    switch (regStep) {
   333→      case 1:
   334→        return !!regData.email &&
   335→          !!regData.password &&
   336→          regData.password.length >= 8 &&
   337→          /[A-Z]/.test(regData.password) &&
   338→          /[a-z]/.test(regData.password) &&
   339→          /[0-9]/.test(regData.password) &&
   340→          !!regConfirmPassword &&
   341→          regData.password === regConfirmPassword;
   342→      case 2:
   343→        return !!regData.companyName;
   344→      case 3:
   345→        return !!regData.fullName;
   346→      case 4:
   347→        return true;
   348→      default:
   349→        return false;
   350→    }
   351→  };
   352→
   353→  const goNext = async () => {
   354→    if (!canGoNext()) return;
   355→    // At Step 1, check if email is already registered → auto-switch to login
   356→    if (regStep === 1 && regData.email) {
   357→      setEmailChecking(true);
   358→      try {
   359→        const { api } = await import('@/lib/api');
   360→        const res = await api.post('/auth/check-email', { email: regData.email });
   361→        if (res.success && res.exists) {
   362→          setEmailChecking(false);
   363→          // Auto-switch to login with pre-filled email
   364→          toast.success('Welcome back! Switching to Sign In...', { duration: 4000 });
   365→          setActiveTab('login');
   366→          setLoginData(d => ({ ...d, email: regData.email }));
   367→          return;
   368→        }
   369→      } catch {
   370→        // If check fails, just proceed — the register endpoint will catch it later
   371→      }
   372→      setEmailChecking(false);
   373→    }
   374→    if (regStep < 4) setRegStep((regStep + 1) as RegStep);
   375→  };
   376→
   377→  const goBack = () => {
   378→    if (regStep > 1) setRegStep((regStep - 1) as RegStep);
   379→  };
   380→
   381→  const handleRegister = async () => {
   382→    setLoading(true);
   383→    try {
   384→      const result = await register(regData);
   385→      if (!result.success) {
   386→        // If the email is already registered, auto-switch to login form
   387→        if (result.error?.toLowerCase().includes('already') || result.error?.toLowerCase().includes('registered')) {
   388→          toast.success('Welcome back! Switching to Sign In...', { duration: 4000 });
   389→          setActiveTab('login');
   390→          setLoginData(d => ({ ...d, email: regData.email || d.email }));
   391→          setRegStep(1);
   392→        } else {
   393→          toast.error(result.error || 'Registration failed. Please try again.');
   394→        }
   395→      }
   396→      else toast.success('Welcome to TenetBid!');
   397→    } catch {
   398→      toast.error('Registration failed. Please check your connection and try again.');
   399→    }
   400→    setLoading(false);
   401→  };
   402→
   403→
   404→
   405→  return (
   406→    <div className="min-h-screen flex flex-col">
   407→      {/* ─── Main content area ─── */}
   408→      <div className="flex-1 flex flex-col lg:flex-row">
   409→
   410→        {/* ═══════════ LEFT PANEL (Desktop) ═══════════ */}
   411→        <div className="hidden lg:flex lg:w-[52%] relative overflow-hidden gradient-slate flex-col">
   412→          <FloatingDots />
