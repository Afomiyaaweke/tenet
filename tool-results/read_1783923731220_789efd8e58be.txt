    58→function daysUntil(dateStr: string): number {
    59→  return Math.ceil((new Date(dateStr).getTime() - Date.now()) / (1000 * 60 * 60 * 24));
    60→}
    61→
    62→function deadlineColor(days: number) {
    63→  if (days <= 0) return 'rose';
    64→  if (days <= 3) return 'rose';
    65→  if (days <= 7) return 'amber';
    66→  return 'emerald';
    67→}
    68→
    69→function deadlineBg(days: number) {
    70→  const c = deadlineColor(days);
    71→  const map: Record<string, string> = {
    72→    rose: 'bg-rose-50 text-rose-700',
    73→    amber: 'bg-amber-50 text-amber-700',
    74→    emerald: 'bg-emerald-50 text-emerald-700',
    75→  };
    76→  return map[c] || map.emerald;
    77→}
    78→
    79→function formatBudget(min: number, max: number) {
    80→  const fmt = (n: number) => n >= 1_000_000 ? `${(n / 1_000_000).toFixed(1)}M` : n >= 1_000 ? `${(n / 1_000).toFixed(0)}K` : n.toLocaleString();
    81→  return `ETB ${fmt(min)} – ${fmt(max)}`;
    82→}

  1191→// ─── Shared helpers ──────────────────────────────────────────────
  1192→function statusColor(status: string) {
  1193→  switch (status) {
  1194→    case 'open': return 'bg-emerald-100 text-emerald-700 hover:bg-emerald-100';
  1195→    case 'closed': return 'bg-rose-100 text-rose-700 hover:bg-rose-100';
  1196→    case 'awarded': return 'bg-teal-100 text-teal-700 hover:bg-teal-100';
  1197→    case 'cancelled': return 'bg-muted text-muted-foreground hover:bg-muted';
  1198→    default: return 'bg-muted text-muted-foreground hover:bg-muted';
  1199→  }
  1200→}
  1201→
  1202→function statusAccent(status: string) {
  1203→  switch (status) {
  1204→    case 'open': return 'from-emerald-400 to-emerald-600';
  1205→    case 'closed': return 'from-rose-400 to-rose-600';
  1206→    case 'awarded': return 'from-teal-400 to-teal-600';
  1207→    case 'cancelled': return 'from-gray-300 to-gray-400';
  1208→    default: return 'from-gray-300 to-gray-400';
  1209→  }
  1210→}
  1211→
  1212→function matchBarColor(score: number) {
  1213→  if (score >= 70) return 'from-emerald-400 to-emerald-600';
  1214→  if (score >= 40) return 'from-amber-400 to-amber-600';
  1215→  return 'from-gray-300 to-gray-400';
  1216→}
