   145→function getProfileCompleteness(profile: Record<string, unknown> | undefined | null): { pct: number; missing: string[] } {
   146→  if (!profile) return { pct: 0, missing: ['Full Name', 'Phone', 'Location', 'Skills', 'Bio'] };
   147→  const checks: [string, boolean][] = [
   148→    ['Full Name', !!(profile.fullName as string)],
   149→    ['Phone', !!(profile.phone as string)],
   150→    ['Location', !!(profile.location as string)],
   151→    ['Skills', !!(profile.skillTags as string)],
   152→    ['Bio', !!(profile.bio as string)],
   153→  ];
   154→  const completed = checks.filter(([, v]) => v).length;
   155→  const missing = checks.filter(([, v]) => !v).map(([label]) => label);
   156→  return { pct: Math.round((completed / checks.length) * 100), missing };
   157→}
