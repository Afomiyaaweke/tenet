   177→   177→function fmtMoney(amount: number, currency: string): string {
   178→   178→  if (!amount) return '—';
   179→   179→  try {
   180→   180→    return new Intl.NumberFormat('en-US', {
   181→   181→      style: 'currency',
   182→   182→      currency: currency || 'USD',
   183→   183→      maximumFractionDigits: 0,
   184→   184→    }).format(amount);
   185→   185→  } catch {
   186→   186→    return `${currency} ${amount.toLocaleString()}`;
   187→   187→  }

   190→   190→function daysUntil(dateStr: string): number {
   191→   191→  return Math.ceil((new Date(dateStr).getTime() - Date.now()) / (1000 * 60 * 60 * 24));
   192→   192→}
   193→   193→
   194→   194→function deadlineBadge(days: number): string {
   195→   195→  if (days <= 0) return 'bg-rose-50 text-rose-700 dark:bg-rose-950/40 dark:text-rose-300';
   196→   196→  if (days <= 7) return 'bg-amber-50 text-amber-700 dark:bg-amber-950/40 dark:text-amber-300';
   197→   197→  if (days <= 21) return 'bg-sky-50 text-sky-700 dark:bg-sky-950/40 dark:text-sky-300';
   198→   198→  return 'bg-emerald-50 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300';
   199→   199→}

   201→   201→function relativeTime(dateStr: string): string {
   202→   202→  const diff = Date.now() - new Date(dateStr).getTime();
   203→   203→  const day = 86400000;
   204→   204→  if (diff < day) return 'today';
   205→   205→  if (diff < 2 * day) return 'yesterday';
   206→   206→  if (diff < 30 * day) return `${Math.floor(diff / day)}d ago`;
   207→   207→  return new Date(dateStr).toLocaleDateString();
   208→   208→}

   317→   317→
   318→   318→export function LiveTendersView() {
   319→   319→  const [tenders, setTenders] = useState<LiveTender[]>([]);
   320→   320→  const [dataSources, setDataSources] = useState<DataSource[]>([]);
   321→   321→  const [sourceMeta, setSourceMeta] = useState<
   322→   322→    { id: string; name: string; live: boolean; ok: boolean; count: number }[]
   323→   323→  >([]);
   324→   324→  const [loading, setLoading] = useState(true);
   325→   325→  const [refreshing, setRefreshing] = useState(false);
   326→   326→  const [fallback, setFallback] = useState(false);
   327→   327→  const [search, setSearch] = useState('');
   328→   328→  const [sourceFilter, setSourceFilter] = useState('all');
   329→   329→  const [sectorFilter, setSectorFilter] = useState('');
   330→   330→  const [sectorCounts, setSectorCounts] = useState<{ id: string; label: string; count: number }[]>([]);
   331→   331→
   332→   332→  // Inline document state
   333→   333→  const [expandedId, setExpandedId] = useState<string | null>(null);
   334→   334→  const [docLoading, setDocLoading] = useState<string | null>(null);
   335→   335→  const [docData, setDocData] = useState<Record<string, InlineDocument>>({});
   336→   336→  const [docError, setDocError] = useState<Record<string, string>>({});
   337→   337→
   338→   338→  // Import state
   339→   339→  const [importing, setImporting] = useState<string | null>(null);
   340→   340→
   341→   341→  const load = useCallback(
   342→   342→    async (isRefresh = false) => {
   343→   343→      if (isRefresh) setRefreshing(true);
   344→   344→      else setLoading(true);
   345→   345→      const params: Record<string, string> = { rows: '50' };
   346→   346→      if (search) params.search = search;
   347→   347→      if (sourceFilter && sourceFilter !== 'all') params.source = sourceFilter;
   348→   348→      if (sectorFilter) params.sector = sectorFilter;
   349→   349→      const res = await api.get('/tenders/live', params);
   350→   350→      if (res.success) {
   351→   351→        setTenders(res.data as LiveTender[]);
   352→   352→        setSourceMeta(res.meta?.sources || []);
   353→   353→        setFallback(Boolean(res.meta?.fallback));
   354→   354→        if (Array.isArray(res.meta?.dataSources)) setDataSources(res.meta.dataSources);
   355→   355→        if (Array.isArray(res.meta?.sectors)) setSectorCounts(res.meta.sectors);
   356→   356→      } else {
   357→   357→        toast.error(res.error || 'Failed to load live tenders');
   358→   358→      }
   359→   359→      setLoading(false);
   360→   360→      setRefreshing(false);
   361→   361→    },
   362→   362→    [search, sourceFilter, sectorFilter],
   363→   363→  );
   364→   364→
   365→   365→  useEffect(() => {
   366→   366→    const t = setTimeout(() => load(), 250);
   367→   367→    return () => clearTimeout(t);
   368→   368→  }, [load]);
   369→   369→
   370→   370→  const liveSourcesCount = useMemo(
   371→   371→    () => sourceMeta.filter((s) => s.live && s.ok).length,
   372→   372→    [sourceMeta],
   373→   373→  );
   374→   374→
   375→   375→  const totalTenderValue = useMemo(() => {
   376→   376→    return tenders.reduce((sum, t) => sum + (t.budgetMax || 0), 0);
   377→   377→  }, [tenders]);
   378→   378→
   379→   379→  /* ────── Inline document loading ────── */
   380→   380→  const loadDocument = useCallback(async (tender: LiveTender) => {
   381→   381→    const id = tender.id;
   382→   382→    if (expandedId === id) {
   383→   383→      setExpandedId(null);
   384→   384→      return;
   385→   385→    }
   386→   386→
   387→   387→    // Already fetched?
   388→   388→    if (docData[id]) {
   389→   389→      setExpandedId(id);
   390→   390→      return;
   391→   391→    }
   392→   392→
   393→   393→    setExpandedId(id);
   394→   394→    setDocLoading(id);
   395→   395→    setDocError((prev) => { const n = { ...prev }; delete n[id]; return n; });
   396→   396→
   397→   397→    try {
   398→   398→      const res = await api.get(`/tenders/${encodeURIComponent(id)}/documents`, { url: tender.externalUrl });
   399→   399→      if (res.success && res.data) {
   400→   400→        setDocData((prev) => ({ ...prev, [id]: res.data as InlineDocument }));
   401→   401→      } else {
   402→   402→        setDocError((prev) => ({ ...prev, [id]: res.error || 'Failed to load document' }));
   403→   403→      }
   404→   404→    } catch {
   405→   405→      setDocError((prev) => ({ ...prev, [id]: 'Network error' }));
   406→   406→    }
   407→   407→    setDocLoading(null);
   408→   408→  }, [expandedId, docData]);
   409→   409→
   410→   410→  /* ────── Import to local tenders ────── */
   411→   411→  const importTender = useCallback(async (tender: LiveTender) => {
   412→   412→    setImporting(tender.id);
   413→   413→    try {
   414→   414→      const res = await api.post('/tenders', {
   415→   415→        title: tender.title,
   416→   416→        scope: tender.scope,
