   168→function relativeTime(dateStr: string): string {
   169→  const now = Date.now();
   170→  const then = new Date(dateStr).getTime();
   171→  const diff = Math.max(0, now - then);
   172→  const seconds = Math.floor(diff / 1000);
   173→  if (seconds < 60) return 'just now';
   174→  const minutes = Math.floor(seconds / 60);
   175→  if (minutes < 60) return `${minutes}m ago`;
   176→  const hours = Math.floor(minutes / 60);
   177→  if (hours < 24) return `${hours}h ago`;
   178→  const days = Math.floor(hours / 24);
   179→  if (days < 7) return `${days}d ago`;
   180→  return new Date(dateStr).toLocaleDateString();
   181→}

   426→                      <p className="text-[13px] font-semibold truncate text-foreground">{n.title}</p>
   427→                      <p className="text-[11px] text-muted-foreground truncate mt-0.5 leading-relaxed">{n.message}</p>

   429→                    </div>
   430→                    <div className="w-2 h-2 rounded-full bg-orange-500 flex-shrink-0 mt-2 ring-2 ring-orange-100 dark:ring-orange-900/40" />

   438→        {/* Footer */}

   442→              variant="ghost"
   443→              size="sm"
   444→              className="w-full text-[11px] text-muted-foreground hover:text-foreground font-medium"
   445→              onClick={() => setOpen(false)}

   447→              View all notifications
   448→            </Button>
   449→          </div>
   450→        )}
   451→      </PopoverContent>
   452→    </Popover>

   458→export function AppShell() {
   459→  const { user, company, logout } = useAuthStore();
   460→  const { view, viewParams, setView } = useNavStore();
   461→  const { fetchNotifications, notifications } = useDataStore();
   462→  const role = user?.role || 'user';
   463→  const navSections = useMemo(() => getNavItemsForRole(role), [role]);
   464→  const unreadCount = notifications.filter((n) => !n.read).length;
   465→
   466→  // Flat list for title lookup
   467→  const allNavItems = useMemo(
   468→    () => navSections.flatMap((s) => s.items),
   469→    [navSections]
   470→  );
   471→
   472→  useEffect(() => {
   473→    fetchNotifications();
   474→  }, [fetchNotifications]);
   475→
   476→  const handleMarkAsRead = async (id: string) => {
   477→    await api.patch(`/notifications/${id}`, { read: true });
   478→    fetchNotifications();
   479→  };
   480→
