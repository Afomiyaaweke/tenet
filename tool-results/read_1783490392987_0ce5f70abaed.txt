   172→function relativeTime(dateStr: string): string {
   173→  const now = Date.now();
   174→  const then = new Date(dateStr).getTime();
   175→  const diff = Math.max(0, now - then);
   176→  const seconds = Math.floor(diff / 1000);
   177→  if (seconds < 60) return 'just now';
   178→  const minutes = Math.floor(seconds / 60);
   179→  if (minutes < 60) return `${minutes}m ago`;
   180→  const hours = Math.floor(minutes / 60);
   181→  if (hours < 24) return `${hours}h ago`;
   182→  const days = Math.floor(hours / 24);
   183→  if (days < 7) return `${days}d ago`;
   184→  return new Date(dateStr).toLocaleDateString();
   185→}

   463→/* ──────────────────────────── AppShell ──────────────────────────── */
   464→
   465→export function AppShell() {
   466→  const { user, company, logout } = useAuthStore();
   467→  const { view, viewParams, setView } = useNavStore();
   468→  const { fetchNotifications, notifications } = useDataStore();
   469→  const role = user?.role || 'user';
   470→  const navSections = useMemo(() => getNavItemsForRole(role), [role]);
   471→  const unreadCount = notifications.filter((n) => !n.read).length;
   472→
   473→  // Flat list for title lookup
   474→  const allNavItems = useMemo(
   475→    () => navSections.flatMap((s) => s.items),
   476→    [navSections]
   477→  );
   478→
   479→  useEffect(() => {
   480→    fetchNotifications();
   481→  }, [fetchNotifications]);
   482→
   483→  const handleMarkAsRead = async (id: string) => {
   484→    await api.patch(`/notifications/${id}`, { read: true });
   485→    fetchNotifications();
   486→  };
   487→
   488→  const handleMarkAllRead = async () => {
   489→    const unread = notifications.filter((n) => !n.read);
   490→    await Promise.all(unread.map((n) => api.patch(`/notifications/${n.id}`, { read: true })));
   491→    fetchNotifications();
   492→  };
