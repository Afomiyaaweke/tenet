    79→function formatETB(amount: number): string {
    80→  if (amount >= 1_000_000) return `ETB ${(amount / 1_000_000).toFixed(1)}M`;
    81→  if (amount >= 1_000) return `ETB ${(amount / 1_000).toFixed(0)}K`;
    82→  return `ETB ${amount.toLocaleString()}`;
    83→}

    85→function timeAgo(dateStr: string): string {
    86→  const diff = Date.now() - new Date(dateStr).getTime();
    87→  const mins = Math.floor(diff / 60000);
    88→  if (mins < 1) return 'just now';
    89→  if (mins < 60) return `${mins}m ago`;
    90→  const hrs = Math.floor(mins / 60);
    91→  if (hrs < 24) return `${hrs}h ago`;
    92→  const days = Math.floor(hrs / 24);
    93→  return `${days}d ago`;
    94→}
    95→
    96→function daysUntil(dateStr: string): number {
    97→  return Math.ceil((new Date(dateStr).getTime() - Date.now()) / 86400000);
    98→}

   100→// ─── Sparkline SVG Component ────────────────────────────────────────
   101→function SparklineBars({ data, color = '#10b981', height = 32 }: { data: number[]; color?: string; height?: number }) {
   102→  if (!data.length) return null;
   103→  const max = Math.max(...data, 1);
   104→  const barW = 4;
   105→  const gap = 3;
   106→  const totalW = data.length * (barW + gap) - gap;
   107→
   108→  return (
   109→    <svg width={totalW} height={height} className="flex-shrink-0">
   110→      {data.map((v, i) => {
   111→        const barH = Math.max((v / max) * (height - 4), 3);
   112→        return (
   113→          <rect
   114→            key={i}
   115→            x={i * (barW + gap)}
   116→            y={height - barH}
   117→            width={barW}
   118→            height={barH}
   119→            rx={2}
   120→            fill={color}
   121→            opacity={0.35 + (i / data.length) * 0.45}
   122→          />
   123→        );
   124→      })}
   125→    </svg>
   126→  );
   127→}

   227→  return 'bg-emerald-50';

   230→function deadlineBadge(days: number): string {
   231→  if (days < 0) return 'bg-gray-100 text-gray-600 hover:bg-gray-100';
