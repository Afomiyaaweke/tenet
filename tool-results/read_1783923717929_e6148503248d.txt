    41→export function BidsView() {
    42→  const { user } = useAuthStore();
    43→  const { setView } = useNavStore();
    44→  const [bids, setBids] = useState<Bid[]>([]);
    45→  const [loading, setLoading] = useState(true);
    46→  const [statusFilter, setStatusFilter] = useState('');
    47→  const [expandedId, setExpandedId] = useState<string | null>(null);
    48→  const [activeTab, setActiveTab] = useState<BidTab>('all');
    49→  const [stampSelectorOpen, setStampSelectorOpen] = useState(false);
    50→  const [selectedBidId, setSelectedBidId] = useState<string | null>(null);
    51→  const [bidSignatures, setBidSignatures] = useState<Record<string, SavedSignature>>({});
    52→  const stampSigHook = useStampSignature();
    53→  const [visibleCount, setVisibleCount] = useState(10);
    54→  const BID_PAGE_SIZE = 10;
    55→
    56→  // Document/OCR/AI state
    57→  const [ocrLoading, setOcrLoading] = useState<Set<string>>(new Set());
    58→  const [reviewLoading, setReviewLoading] = useState<Set<string>>(new Set());
    59→  const [docOcrText, setDocOcrText] = useState<Record<string, string>>({});
    60→  const [docReview, setDocReview] = useState<Record<string, AIReviewData>>({});
    61→  const [expandedDocId, setExpandedDocId] = useState<string | null>(null);
    62→  const [reviewDialogOpen, setReviewDialogOpen] = useState(false);
    63→  const [reviewDialogDocId, setReviewDialogDocId] = useState<string | null>(null);
    64→  const [docUploadBidId, setDocUploadBidId] = useState<string | null>(null);
    65→  const docFileRef = useRef<HTMLInputElement>(null);
    66→  const [docUploadType, setDocUploadType] = useState('bid_attachment');
    67→  const [extractPrompt, setExtractPrompt] = useState<Record<string, string>>({});
    68→  const [extractLoading, setExtractLoading] = useState<Set<string>>(new Set());
    69→  const [extractResults, setExtractResults] = useState<Record<string, string>>({});
    70→  const [showExtract, setShowExtract] = useState<Set<string>>(new Set());
    71→  const [submitUrlForBid, setSubmitUrlForBid] = useState<Record<string, string>>({});
    72→  const [reviewPromptForBid, setReviewPromptForBid] = useState<Record<string, string>>({});
    73→  const fileInputRefs = useRef<Record<string, HTMLInputElement | null>>({});
    74→
    75→  // Saved tenders state
    76→  const [savedTenders, setSavedTenders] = useState<any[]>([]);
    77→  const [savedLoading, setSavedLoading] = useState(false);
    78→
    79→  const loadSavedTenders = useCallback(async () => {
    80→    setSavedLoading(true);
    81→    const res = await api.get('/tenders/saved');
    82→    if (res.success) setSavedTenders(res.data);
    83→    setSavedLoading(false);
    84→  }, []);
    85→
    86→  const handleRemoveSaved = async (id: string) => {
    87→    const res = await api.delete(`/tenders/saved/${id}`);
    88→    if (res.success) {
    89→      toast.success('Removed from saved tenders');
    90→      loadSavedTenders();
    91→    } else {
    92→      toast.error('Failed to remove');
    93→    }
    94→  };
    95→
    96→  const loadBids = useCallback(async () => {
    97→    setLoading(true);
    98→    const params: Record<string, string> = {};
    99→    if (statusFilter && statusFilter !== 'all') params.status = statusFilter;
   100→    const res = await api.get('/bids', params);
   101→    if (res.success) setBids(res.data);
   102→    setLoading(false);
   103→  }, [statusFilter]);
   104→
   105→  useEffect(() => { loadBids(); loadSavedTenders(); }, [loadBids, loadSavedTenders]);
   106→
   107→  const handleStatusUpdate = async (bidId: string, status: string) => {
   108→    const res = await api.patch(`/bids/${bidId}/status`, { status });
   109→    if (res.success) {
   110→      toast.success(`Bid ${status}`);
   111→      loadBids();
   112→    } else {
   113→      toast.error(res.error || 'Failed to update bid');
   114→    }
   115→  };
   116→
   117→  const handleMarkAsDraft = async (bidId: string) => {
   118→    const res = await api.patch(`/bids/${bidId}/status`, { status: 'drafted' });
   119→    if (res.success) {
   120→      toast.success('Bid saved as draft');
   121→      loadBids();
   122→    } else {
   123→      toast.error(res.error || 'Failed to update bid');
   124→    }
   125→  };
   126→
   127→  const handleSubmitDraft = async (bidId: string) => {
   128→    const res = await api.patch(`/bids/${bidId}/status`, { status: 'pending_review' });
   129→    if (res.success) {
   130→      toast.success('Bid submitted for review');
   131→      loadBids();
   132→    } else {
   133→      toast.error(res.error || 'Failed to submit bid');
   134→    }
   135→  };
   136→
   137→  const [withdrawingBidId, setWithdrawingBidId] = useState<string | null>(null);
   138→
   139→  const handleWithdrawBid = async (bidId: string) => {
   140→    setWithdrawingBidId(bidId);
   141→    try {
   142→      const res = await api.patch(`/bids/${bidId}/status`, { status: 'withdrawn' });
   143→      if (res.success) {
   144→        toast.success('Bid withdrawn successfully');
   145→        loadBids();
   146→      } else {
   147→        toast.error(res.error || 'Failed to withdraw bid');
   148→      }
   149→    } catch {
   150→      toast.error('Failed to withdraw bid');
   151→    } finally {
   152→      setWithdrawingBidId(null);
   153→    }
   154→  };
   155→
   156→  // ── Document Upload (with auto-OCR + auto-AI Review) ──
   157→  const handleDocUpload = useCallback(async (bidId: string, docType?: string, fileInputRef?: React.RefObject<HTMLInputElement | null>) => {
   158→    const ref = fileInputRef || docFileRef;
   159→    const file = ref.current?.files?.[0];
   160→    if (!file) return;
   161→    const type = docType || docUploadType;
   162→    const formData = new FormData();
   163→    formData.append('file', file);
   164→    formData.append('docType', type);
   165→    formData.append('autoOcr', 'true');
   166→    formData.append('autoReview', 'true');
   167→    formData.append('submitUrl', submitUrlForBid[bidId] || '');
   168→    formData.append('reviewPrompt', reviewPromptForBid[bidId] || '');
   169→    try {
   170→      const res = await api.upload(`/bids/${bidId}/documents`, formData);
   171→      if (res.success) {
   172→        toast.success('Document uploaded — OCR & AI Review started automatically');
   173→        if (ref.current) ref.current.value = '';
   174→        loadBids();
   175→        // Start polling for the newly uploaded document
   176→        const newDocId = res.data?.id;
   177→        if (newDocId) {
   178→          setOcrLoading(prev => new Set(prev).add(newDocId));
   179→          pollOcrThenReview(newDocId);
   180→        }
   181→      } else {
   182→        toast.error(res.error || 'Upload failed');
   183→      }
   184→    } catch {
   185→      toast.error('Upload failed');
   186→    } finally {
   187→      setDocUploadBidId(null);
   188→    }
   189→  }, [docUploadType, loadBids, submitUrlForBid, reviewPromptForBid]);
   190→
   191→  // ── Ref for AI Review handler (avoids circular dep with pollOcrThenReview) ──
   192→  const runReviewRef = useRef<(docId: string) => void>(() => {});
   193→
   194→  // ── AI Review Processing ──
   195→  const handleRunReview = useCallback(async (docId: string) => {
   196→    setReviewLoading(prev => new Set(prev).add(docId));
   197→    try {
   198→      const res = await api.post(`/document-review/${docId}`);
   199→      if (!res.success && res.error?.includes('OCR must be completed')) {
   200→        setReviewLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   201→        toast.error('Run OCR first before AI review');
   202→        return;
   203→      }
   204→      const poll = async (attempts = 0): Promise<void> => {
   205→        if (attempts > 60) {
   206→          setReviewLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   207→          toast.error('AI Review processing timed out');
   208→          return;
   209→        }
   210→        await new Promise(r => setTimeout(r, 2000));
   211→        const pollRes = await api.get(`/document-review/${docId}`);
   212→        if (pollRes.success && pollRes.data?.aiReviewStatus === 'completed') {
   213→          setReviewLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   214→          let reviewData = pollRes.data.aiReview || {};
   215→          if (typeof reviewData === 'string') {
   216→            try { reviewData = JSON.parse(reviewData); } catch { reviewData = { summary: reviewData }; }
   217→          }
   218→          setDocReview(prev => ({ ...prev, [docId]: reviewData as AIReviewData }));
   219→          loadBids();
   220→          toast.success('AI Review completed — Document fully processed');
   221→        } else if (pollRes.success && pollRes.data?.aiReviewStatus === 'failed') {
   222→          setReviewLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   223→          toast.error('AI Review processing failed');
   224→          loadBids();
   225→        } else {
   226→          await poll(attempts + 1);
   227→        }
   228→      };
   229→      poll();
   230→    } catch {
   231→      setReviewLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   232→      toast.error('Failed to start AI Review');
   233→    }
   234→  }, [loadBids]);
   235→
   236→  // Keep ref updated
   237→  runReviewRef.current = handleRunReview;
   238→
   239→  // ── Poll OCR then auto-trigger AI Review ──
   240→  const pollOcrThenReview = useCallback(async (docId: string) => {
   241→    const poll = async (attempts = 0): Promise<void> => {
   242→      if (attempts > 60) {
   243→        setOcrLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   244→        toast.error('OCR processing timed out');
   245→        return;
   246→      }
   247→      await new Promise(r => setTimeout(r, 2000));
   248→      const res = await api.get(`/document-ocr/${docId}`);
   249→      if (res.success && res.data?.ocrStatus === 'completed') {
   250→        setOcrLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   251→        setDocOcrText(prev => ({ ...prev, [docId]: res.data.ocrText || '' }));
   252→        loadBids();
   253→        toast.success('OCR completed — Starting AI Review...');
   254→        // Auto-chain: trigger AI Review after OCR completes
   255→        runReviewRef.current(docId);
   256→      } else if (res.success && res.data?.ocrStatus === 'failed') {
   257→        setOcrLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   258→        toast.error('OCR processing failed');
   259→        loadBids();
   260→      } else {
   261→        await poll(attempts + 1);
   262→      }
   263→    };
   264→    poll();
   265→  }, [loadBids]);
   266→
   267→  // ── Drag & Drop Upload ──
   268→  const handleDocDrop = useCallback(async (bidId: string, files: FileList | null, docType?: string) => {
   269→    if (!files || files.length === 0) return;
   270→    const file = files[0];
   271→    const ext = file.name.split('.').pop()?.toLowerCase() || '';
   272→    const allowed = ['pdf', 'jpg', 'jpeg', 'png', 'docx', 'doc', 'txt'];
   273→    if (!allowed.includes(ext)) {
   274→      toast.error('Invalid file type. Allowed: PDF, JPEG, PNG, DOCX, DOC, TXT');
   275→      return;
   276→    }
   277→    if (file.size > 10 * 1024 * 1024) {
   278→      toast.error('File too large. Maximum 10MB.');
   279→      return;
   280→    }
   281→    const formData = new FormData();
   282→    formData.append('file', file);
   283→    formData.append('docType', docType || docUploadType);
   284→    formData.append('autoOcr', 'true');
   285→    formData.append('autoReview', 'true');
   286→    formData.append('submitUrl', submitUrlForBid[bidId] || '');
   287→    formData.append('reviewPrompt', reviewPromptForBid[bidId] || '');
   288→    try {
   289→      const res = await api.upload(`/bids/${bidId}/documents`, formData);
   290→      if (res.success) {
   291→        toast.success('Document dropped — OCR & AI Review started');
   292→        loadBids();
   293→        const newDocId = res.data?.id;
   294→        if (newDocId) {
   295→          setOcrLoading(prev => new Set(prev).add(newDocId));
   296→          pollOcrThenReview(newDocId);
   297→        }
   298→      } else {
   299→        toast.error(res.error || 'Upload failed');
   300→      }
   301→    } catch {
   302→      toast.error('Upload failed');
   303→    }
   304→  }, [docUploadType, loadBids, pollOcrThenReview, submitUrlForBid, reviewPromptForBid]);
   305→
   306→  // ── OCR Processing (manual trigger) ──
   307→  const handleRunOcr = useCallback(async (docId: string) => {
   308→    setOcrLoading(prev => new Set(prev).add(docId));
   309→    try {
   310→      await api.post(`/document-ocr/${docId}`);
   311→      pollOcrThenReview(docId);
   312→    } catch {
   313→      setOcrLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   314→      toast.error('Failed to start OCR');
   315→    }
   316→  }, [pollOcrThenReview]);
   317→
   318→  // ── View OCR text inline ──
   319→  const handleViewOcr = useCallback(async (docId: string) => {
   320→    if (expandedDocId === docId) {
   321→      setExpandedDocId(null);
   322→      return;
   323→    }
   324→    setExpandedDocId(docId);
   325→    if (!docOcrText[docId]) {
   326→      const res = await api.get(`/document-ocr/${docId}`);
   327→      if (res.success) {
   328→        setDocOcrText(prev => ({ ...prev, [docId]: res.data?.ocrText || '' }));
   329→      }
   330→    }
   331→  }, [expandedDocId, docOcrText]);
   332→
   333→  // ── AI Extract: extract specific info from document via prompt ──
   334→  const handleAiExtract = useCallback(async (docId: string) => {
   335→    const prompt = extractPrompt[docId]?.trim();
   336→    if (!prompt) {
   337→      toast.error('Please enter a prompt describing what to extract');
   338→      return;
   339→    }
   340→    setExtractLoading(prev => new Set(prev).add(docId));
   341→    try {
   342→      const res = await api.post('/documents/ai-extract', { documentId: docId, prompt });
   343→      if (res.success) {
   344→        setExtractResults(prev => ({ ...prev, [docId]: res.data.extractedInfo }));
   345→        toast.success('Information extracted successfully');
   346→      } else {
   347→        toast.error(res.error || 'AI extraction failed');
   348→      }
   349→    } catch {
   350→      toast.error('AI extraction failed');
   351→    } finally {
   352→      setExtractLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   353→    }
   354→  }, [extractPrompt]);
   355→
   356→  const isDrafted = (b: Bid) => {
   357→    if (b.status === 'drafted') return true;
   358→    // Backward compatibility: treat pending_review bids with no real data as drafted
   359→    if (b.status !== 'pending_review') return false;
   360→    return (
   361→      !b.financialProposal ||
   362→      b.financialProposal === 0 ||
   363→      b.technicalProposal?.includes('Pending document upload') ||
   364→      b.timeline?.includes('Pending document upload')
   365→    );
   366→  };
   367→
   368→  const stats = useMemo(() => ({
   369→    pending: bids.filter(b => b.status === 'pending_review').length,
   370→    drafted: bids.filter(b => isDrafted(b)).length,
   371→    shortlisted: bids.filter(b => b.status === 'shortlisted').length,
   372→    awarded: bids.filter(b => b.status === 'awarded').length,
   373→    rejected: bids.filter(b => b.status === 'rejected').length,
   374→    total: bids.length,
   375→  }), [bids]);
   376→
   377→  const filteredBids = useMemo(() => {
   378→    if (activeTab === 'saved') return [] as Bid[];
   379→    if (activeTab === 'all') return bids;
   380→    if (activeTab === 'drafted') return bids.filter(b => isDrafted(b));
   381→    return bids.filter(b => b.status === activeTab);
   382→  }, [bids, activeTab]);
   383→
   384→  const statusBadge = (status: string) => {
   385→    switch (status) {
   386→      case 'drafted': return 'bg-sky-100 text-sky-700 hover:bg-sky-100';
   387→      case 'pending_review': return 'bg-amber-100 text-amber-700 hover:bg-amber-100';
   388→      case 'shortlisted': return 'bg-teal-100 text-teal-700 hover:bg-teal-100';
   389→      case 'awarded': return 'bg-emerald-100 text-emerald-700 hover:bg-emerald-100';
   390→      case 'rejected': return 'bg-rose-100 text-rose-700 hover:bg-rose-100';
   391→      default: return 'bg-muted text-muted-foreground hover:bg-muted';
   392→    }
   393→  };
   394→
   395→  const statusIcon = (status: string) => {
   396→    switch (status) {
   397→      case 'drafted': return { icon: PenLine, bg: 'bg-sky-50', color: 'text-sky-600' };
   398→      case 'pending_review': return { icon: Clock, bg: 'bg-amber-50', color: 'text-amber-600' };
   399→      case 'shortlisted': return { icon: Award, bg: 'bg-teal-50', color: 'text-teal-600' };
   400→      case 'awarded': return { icon: CheckCircle, bg: 'bg-emerald-50', color: 'text-emerald-600' };
   401→      case 'rejected': return { icon: AlertCircle, bg: 'bg-rose-50', color: 'text-rose-600' };
   402→      default: return { icon: Gavel, bg: 'bg-muted/50', color: 'text-muted-foreground' };
   403→    }
   404→  };
   405→
   406→  const isAdminOrOwner = user?.role === 'team_admin';
   407→
   408→  const getScoreColor = (score: number) => {
   409→    if (score >= 80) return 'text-emerald-600';
   410→    if (score >= 60) return 'text-amber-600';
   411→    return 'text-rose-600';
   412→  };
   413→
   414→  const getScoreBg = (score: number) => {
   415→    if (score >= 80) return 'bg-emerald-50';
   416→    if (score >= 60) return 'bg-amber-50';
   417→    return 'bg-rose-50';
   418→  };
   419→
   420→  const getRiskColor = (risk: string) => {
   421→    switch (risk?.toLowerCase()) {
   422→      case 'low': return 'bg-emerald-100 text-emerald-700';
   423→      case 'medium': return 'bg-amber-100 text-amber-700';
   424→      case 'high': case 'critical': return 'bg-rose-100 text-rose-700';
   425→      default: return 'bg-gray-100 text-gray-700';
   426→    }
   427→  };
   428→
   429→  const tabs: { key: BidTab; label: string; icon: typeof Clock; count: number; color: string }[] = [
   430→    { key: 'all', label: 'All', icon: Gavel, count: stats.total, color: 'emerald' },
   431→    { key: 'pending_review', label: 'Pending', icon: Clock, count: stats.pending, color: 'amber' },
   432→    { key: 'drafted', label: 'Drafted', icon: PenLine, count: stats.drafted, color: 'sky' },
   433→    { key: 'shortlisted', label: 'Shortlisted', icon: Award, count: stats.shortlisted, color: 'teal' },
   434→    { key: 'awarded', label: 'Awarded', icon: CheckCircle, count: stats.awarded, color: 'emerald' },
   435→    { key: 'rejected', label: 'Rejected', icon: AlertCircle, count: stats.rejected, color: 'rose' },
   436→    { key: 'saved', label: 'Saved Tenders', icon: Bookmark, count: savedTenders.length, color: 'violet' },
   437→  ];
   438→
   439→  // Review dialog data
   440→  const reviewDialogData = reviewDialogDocId ? docReview[reviewDialogDocId] : null;
   441→
   442→  return (
   443→    <div className="p-4 md:p-6 space-y-6 max-w-5xl mx-auto view-enter">
   444→      {/* Header */}
   445→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4 animate-[fadeIn_0.3s_ease-out]">
   446→        <div className="flex items-center gap-4">
   447→          <div className="p-3 rounded-2xl gradient-amber shadow-md flex-shrink-0">
   448→            <Gavel className="h-6 w-6 text-white" />
   449→          </div>
   450→          <div>
   451→            <h2 className="text-2xl md:text-3xl font-bold tracking-tight">
   452→              <span className="text-gradient-emerald">{isAdminOrOwner ? 'Review' : 'My'}</span> Bids
   453→            </h2>
   454→            <p className="text-muted-foreground text-sm mt-0.5">
   455→              {isAdminOrOwner ? 'Manage and evaluate submitted bids' : 'Track your bid submissions'}
   456→            </p>
   457→          </div>
   458→        </div>
   459→      </div>
   460→
   461→      {/* Stats Summary */}
   462→      {!loading && (bids.length > 0 || savedTenders.length > 0) && (
   463→        <div className="grid grid-cols-2 sm:grid-cols-4 gap-4 animate-[fadeIn_0.3s_ease-out]">
