   176→function relativeTime(dateStr: string): string {
   177→  const now = Date.now();
   178→  const then = new Date(dateStr).getTime();
   179→  const diff = Math.max(0, now - then);
   180→  const seconds = Math.floor(diff / 1000);
   181→  if (seconds < 60) return 'just now';
   182→  const minutes = Math.floor(seconds / 60);
   183→  if (minutes < 60) return `${minutes}m ago`;
   184→  const hours = Math.floor(minutes / 60);
   185→  if (hours < 24) return `${hours}h ago`;
   186→  const days = Math.floor(hours / 24);
   187→  if (days < 7) return `${days}d ago`;
   188→  return new Date(dateStr).toLocaleDateString();
   189→}

   350→/* ──────────────────── NotificationDropdown ──────────────────── */
   351→
   352→function NotificationDropdown({
   353→  notifications,
   354→  onMarkAsRead,
   355→  onMarkAllRead,
   356→}: {
   357→  notifications: { id: string; title: string; message: string; type: string; read: boolean; createdAt: string }[];
   358→  onMarkAsRead: (id: string) => void;
   359→  onMarkAllRead: () => void;
   360→}) {
   361→  const [open, setOpen] = useState(false);
   362→  const unread = useMemo(() => notifications.filter((n) => !n.read), [notifications]);
   363→  const displayItems = useMemo(() => unread.slice(0, 6), [unread]);
   364→
   365→  return (
   366→    <Popover open={open} onOpenChange={setOpen}>
   367→      <PopoverTrigger asChild>
   368→        <Button variant="ghost" size="icon" className="relative h-9 w-9 hover:bg-muted/80 transition-colors">
   369→          <Bell className="h-[18px] w-[18px] text-muted-foreground" />
   370→          {unread.length > 0 && (
   371→            <span className="absolute -top-0.5 -right-0.5 min-w-[18px] h-[18px] px-1 bg-red-500 text-white text-[10px] font-bold rounded-full flex items-center justify-center shadow-sm shadow-red-200 animate-in zoom-in-50 duration-200">
   372→              {unread.length > 99 ? '99+' : unread.length}
   373→            </span>
   374→          )}
   375→        </Button>
   376→      </PopoverTrigger>
   377→      <PopoverContent
   378→        className="w-80 p-0 glass-card rounded-xl overflow-hidden animate-in slide-in-from-top-2 fade-in-0 duration-200"
   379→        align="end"
   380→        sideOffset={8}
   381→      >
   382→        {/* Header */}
   383→        <div className="flex items-center justify-between px-4 py-3 border-b border-border/50">
   384→          <div className="flex items-center gap-2">
   385→            <h4 className="text-sm font-bold text-foreground">Notifications</h4>
   386→            {unread.length > 0 && (
   387→              <Badge className="text-[10px] px-1.5 py-0 bg-red-500 text-white font-bold border-0">
   388→                {unread.length}
   389→              </Badge>
   390→            )}
   391→          </div>
   392→          {unread.length > 0 && (
   393→            <Button
   394→              variant="ghost"
   395→              size="sm"
   396→              className="text-[11px] h-7 px-2 text-primary hover:text-primary/80 hover:bg-primary/10 font-semibold"
   397→              onClick={onMarkAllRead}
   398→            >
   399→              <Check className="h-3 w-3 mr-1" /> Mark all read
   400→            </Button>
   401→          )}
   402→        </div>
   403→
   404→        {/* Items */}
   405→        <ScrollArea className="max-h-[340px]">
   406→          {displayItems.length === 0 ? (
   407→            <div className="py-8 text-center">
   408→              <div className="w-12 h-12 mx-auto mb-3 rounded-full bg-orange-50 dark:bg-orange-950/30 flex items-center justify-center">
   409→                <CheckCircle className="h-6 w-6 text-orange-400" />
   410→              </div>
   411→              <p className="text-sm font-medium text-muted-foreground">All caught up!</p>
   412→              <p className="text-xs text-muted-foreground/60 mt-0.5">No new notifications</p>
   413→            </div>
   414→          ) : (
   415→            <div className="divide-y divide-border/40">
   416→              {displayItems.map((n, i) => {
   417→                const Icon = NOTIFICATION_ICONS[n.type] || Info;
   418→                const colorClass = NOTIFICATION_COLORS[n.type] || 'text-gray-500';
   419→                const bgClass = NOTIFICATION_BG[n.type] || 'bg-gray-50';
   420→                return (
   421→                  <button
   422→                    key={n.id}
   423→                    onClick={() => {
   424→                      onMarkAsRead(n.id);
   425→                      setOpen(false);
   426→                    }}
   427→                    className="w-full flex items-start gap-3 px-4 py-3 hover:bg-muted/50 transition-colors text-left group"
   428→                    style={{ animationDelay: `${i * 40}ms` }}
   429→                  >
   430→                    <div
   431→                      className={`w-8 h-8 rounded-lg ${bgClass} flex items-center justify-center flex-shrink-0 mt-0.5 transition-transform group-hover:scale-110`}
   432→                    >
   433→                      <Icon className={`h-4 w-4 ${colorClass}`} />
   434→                    </div>
   435→                    <div className="min-w-0 flex-1">
   436→                      <p className="text-[13px] font-semibold truncate text-foreground">{n.title}</p>
   437→                      <p className="text-[11px] text-muted-foreground truncate mt-0.5 leading-relaxed">{n.message}</p>
   438→                      <p className="text-[10px] text-muted-foreground/50 mt-1 font-medium">{relativeTime(n.createdAt)}</p>
   439→                    </div>
   440→                    <div className="w-2 h-2 rounded-full bg-orange-500 flex-shrink-0 mt-2 ring-2 ring-orange-100 dark:ring-orange-900/40" />
   441→                  </button>
   442→                );
   443→              })}
   444→            </div>
   445→          )}
   446→        </ScrollArea>
   447→
   448→        {/* Footer */}
   449→        {notifications.length > 6 && (
   450→          <div className="px-3 py-2 border-t border-border/40">
   451→            <Button
   452→              variant="ghost"
   453→              size="sm"
   454→              className="w-full text-[11px] text-muted-foreground hover:text-foreground font-medium"
   455→              onClick={() => setOpen(false)}
   456→            >
   457→              View all notifications
   458→            </Button>
   459→          </div>
   460→        )}
   461→      </PopoverContent>
   462→    </Popover>
   463→  );
   464→}

   467→
   468→export function AppShell() {
   469→  const { user, company, logout } = useAuthStore();
   470→  const { view, viewParams, setView } = useNavStore();
   471→  const { fetchNotifications, notifications } = useDataStore();
   472→  const role = user?.role || 'user';
   473→  const navSections = useMemo(() => getNavItemsForRole(role), [role]);
   474→  const unreadCount = notifications.filter((n) => !n.read).length;
   475→
   476→  // Flat list for title lookup
   477→  const allNavItems = useMemo(
   478→    () => navSections.flatMap((s) => s.items),
   479→    [navSections]
   480→  );
   481→
   482→  useEffect(() => {
   483→    fetchNotifications();
   484→  }, [fetchNotifications]);
   485→
   486→  const handleMarkAsRead = async (id: string) => {
   487→    await api.patch(`/notifications/${id}`, { read: true });
   488→    fetchNotifications();
   489→  };
   490→
   491→  const handleMarkAllRead = async () => {
   492→    const unread = notifications.filter((n) => !n.read);
   493→    await Promise.all(unread.map((n) => api.patch(`/notifications/${n.id}`, { read: true })));
   494→    fetchNotifications();
   495→  };
   496→
   497→  const renderView = () => {
   498→    switch (view) {
   499→      case 'dashboard':
   500→        return <DashboardView />;
   501→      case 'tenders':
   502→        return <TendersView />;
   503→      case 'live-tenders':
   504→        return <LiveTendersView />;
   505→      case 'tender-detail':
   506→        return <TenderDetailView tenderId={viewParams.id} initialTab={viewParams.tab as any} />;
   507→      case 'tender-compare':
   508→        return <TenderCompareView tenderIds={viewParams.ids} />;
   509→      case 'bid-compare':
   510→        return <BidCompareView tenderId={viewParams.tenderId} />;
   511→      case 'bid-analysis':
   512→        return <BidsView />;
   513→      case 'bids':
   514→        return <BidsView />;
   515→      case 'applicants':
   516→        return <ApplicantsView />;
   517→      case 'projects':
   518→        return <ProjectsView />;
   519→      case 'project-detail':
   520→        return <ProjectDetailView projectId={viewParams.id} />;
   521→      case 'chat':
   522→        return <ChatView chatId={viewParams.id} />;
   523→      case 'finance':
   524→        return <ProjectDetailView projectId={viewParams.id} />;
   525→      case 'events':
   526→        return <EventsView />;
   527→      case 'profile':
   528→        return <ProfileView />;
   529→      case 'company-settings':
   530→        return <ProfileView />;
   531→      case 'documents':
   532→        return <DocumentsView />;
   533→      case 'ai-doc-studio':
   534→        return <AIDocStudioView />;
   535→      case 'doc-builder':
   536→        return <DocBuilderView />;
   537→      case 'tender-analyzer':
   538→        return <TenderAnalyzerView />;
   539→      case 'agent':
   540→        return <AgentView />;
   541→      case 'staff':
   542→        return <StaffView />;
   543→      case 'contact-us':
   544→        return <ContactUsView />;
   545→      case 'privacy-policy':
   546→        return <PrivacyPolicyView />;
   547→      case 'social-circle':
   548→        return <SocialCircleView />;
   549→      default:
   550→        return <DashboardView />;
   551→    }
   552→  };
   553→
   554→  const pageTitle = allNavItems.find((i) => i.id === view)?.label
   555→    || (view === 'tender-compare' ? 'Compare Tenders' : view === 'bid-compare' ? 'Compare Bids' : view === 'bid-analysis' ? 'Bid Analysis' : 'Dashboard');
   556→  const breadcrumb = view === 'tender-detail' || view === 'bid-compare' || view === 'bid-analysis' ? 'Tenders' : view === 'tender-compare' ? 'Tenders' : view === 'project-detail' ? 'Projects' : null;
   557→
   558→  const badgeConfig = ROLE_BADGE_CONFIG[role] || ROLE_BADGE_CONFIG.user;
   559→  const companyName = company?.name || user?.company?.name;
   560→
   561→  const sidebarProps = { user, role, company, navSections, view, setView, unreadCount, logout };
   562→
   563→  return (
   564→    <div className="min-h-screen flex bg-background">
   565→      {/* ── Desktop Sidebar ── */}
   566→      <aside className="hidden md:flex w-[260px] bg-card border-r border-border/50 flex-col flex-shrink-0 sidebar-shadow">
   567→        <SidebarContent {...sidebarProps} />
   568→      </aside>
   569→
   570→      {/* ── Main Area ── */}
   571→      <div className="flex-1 flex flex-col min-w-0">
   572→        {/* ── Top Bar ── */}
   573→        <header className="h-14 bg-background/80 backdrop-blur-md border-b border-border/50 flex items-center px-4 gap-3 flex-shrink-0 sticky top-0 z-30">
   574→          {/* Mobile hamburger */}
   575→          <Sheet>
   576→            <SheetTrigger asChild>
   577→              <Button variant="ghost" size="icon" className="md:hidden h-9 w-9 hover:bg-muted/80">
   578→                <Menu className="h-5 w-5 text-muted-foreground" />
   579→              </Button>
   580→            </SheetTrigger>
   581→            <SheetContent side="left" className="p-0 w-[260px]">
   582→              <SidebarContent {...sidebarProps} />
   583→            </SheetContent>
   584→          </Sheet>
   585→
   586→          {/* Page Title & Breadcrumb */}
   587→          <div className="flex-1 min-w-0 flex items-center gap-1.5">
   588→            {breadcrumb && (
   589→              <>
   590→                <span className="text-sm text-muted-foreground/60 font-medium hidden sm:inline">
   591→                  {breadcrumb}
   592→                </span>
   593→                <ChevronRight className="h-3 w-3 text-muted-foreground/40 hidden sm:inline" />
   594→              </>
   595→            )}
   596→            <h1 className="text-base font-bold truncate text-foreground">{pageTitle}</h1>
   597→          </div>
   598→
   599→          {/* Right Actions */}
   600→          <div className="flex items-center gap-1">
   601→            {/* Search */}
   602→            <Button variant="ghost" size="icon" className="h-9 w-9 hover:bg-muted/80 transition-colors">
   603→              <Search className="h-[18px] w-[18px] text-muted-foreground" />
   604→            </Button>
   605→
   606→            {/* Theme Toggle */}
   607→            <ThemeToggle />
   608→
   609→            {/* Notifications */}
   610→            <NotificationDropdown
   611→              notifications={notifications}
   612→              onMarkAsRead={handleMarkAsRead}
   613→              onMarkAllRead={handleMarkAllRead}
   614→            />
   615→
   616→            {/* User Avatar Dropdown */}
   617→            <DropdownMenu>
   618→              <DropdownMenuTrigger asChild>
   619→                <Button variant="ghost" size="icon" className="h-9 w-9 hover:bg-muted/80 transition-colors ml-0.5">
   620→                  <div className="w-7 h-7 rounded-full bg-slate-700 flex items-center justify-center shadow-sm shadow-slate-200">
   621→                    <span className="text-white font-bold text-[11px]">{getUserInitial(user)}</span>
   622→                  </div>
   623→                </Button>
   624→              </DropdownMenuTrigger>
   625→              <DropdownMenuContent align="end" className="w-56">
   626→                <DropdownMenuLabel className="font-normal">
   627→                  <div className="flex flex-col space-y-1">
   628→                    <div className="flex items-center gap-2">
   629→                      <p className="text-sm font-medium leading-none">{user?.profile?.fullName || user?.email || 'User'}</p>
   630→                      <Badge variant="secondary" className={`text-[9px] px-1.5 py-0 font-semibold border-0 ${badgeConfig.className}`}>
   631→                        {badgeConfig.label}
   632→                      </Badge>
   633→                    </div>
   634→                    <p className="text-xs leading-none text-muted-foreground">{user?.email}</p>
   635→                    {companyName && (
   636→                      <p className="text-[11px] leading-none text-muted-foreground/70 flex items-center gap-1">
   637→                        <Building2 className="w-3 h-3" /> {companyName}
   638→                      </p>
   639→                    )}
   640→                  </div>
   641→                </DropdownMenuLabel>
   642→                <DropdownMenuSeparator />
   643→                <DropdownMenuItem onClick={() => setView('profile' as View)} className="cursor-pointer">
   644→                  <User className="mr-2 h-4 w-4" />
   645→                  <span>Profile</span>
   646→                </DropdownMenuItem>
   647→                <DropdownMenuItem onClick={() => setView('documents' as View)} className="cursor-pointer">
   648→                  <FileText className="mr-2 h-4 w-4" />
   649→                  <span>Documents</span>
   650→                </DropdownMenuItem>
   651→                <DropdownMenuSeparator />
   652→                <DropdownMenuItem onClick={logout} className="cursor-pointer text-red-600 focus:text-red-600 focus:bg-red-50 dark:focus:bg-red-950/30">
   653→                  <LogOut className="mr-2 h-4 w-4" />
   654→                  <span>Sign Out</span>
   655→                </DropdownMenuItem>
   656→              </DropdownMenuContent>
   657→            </DropdownMenu>
   658→          </div>
   659→        </header>
   660→
   661→        {/* ── Content Area ── */}
   662→        <main className="flex-1 overflow-auto">
   663→          <div key={view} className="view-enter">
   664→            <Suspense fallback={<ViewLoader />}>
   665→              {renderView()}
   666→            </Suspense>
   667→          </div>
   668→        </main>
   669→      </div>
   670→    </div>
   671→  );
   672→}
