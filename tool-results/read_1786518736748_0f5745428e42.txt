   189→function relativeTime(dateStr: string): string {
   190→  const now = Date.now();
   191→  const then = new Date(dateStr).getTime();
   192→  const diff = Math.max(0, now - then);
   193→  const seconds = Math.floor(diff / 1000);
   194→  if (seconds < 60) return 'just now';
   195→  const minutes = Math.floor(seconds / 60);
   196→  if (minutes < 60) return `${minutes}m ago`;
   197→  const hours = Math.floor(minutes / 60);
   198→  if (hours < 24) return `${hours}h ago`;
   199→  const days = Math.floor(hours / 24);
   200→  if (days < 7) return `${days}d ago`;
   201→  return new Date(dateStr).toLocaleDateString();
   202→}

   363→/* ──────────────────── NotificationDropdown ──────────────────── */
   364→
   365→function NotificationDropdown({
   366→  notifications,
   367→  onMarkAsRead,
   368→  onMarkAllRead,
   369→}: {
   370→  notifications: { id: string; title: string; message: string; type: string; read: boolean; createdAt: string }[];
   371→  onMarkAsRead: (id: string) => void;
   372→  onMarkAllRead: () => void;
   373→}) {
   374→  const [open, setOpen] = useState(false);
   375→  const unread = useMemo(() => notifications.filter((n) => !n.read), [notifications]);
   376→  const displayItems = useMemo(() => unread.slice(0, 6), [unread]);
   377→
   378→  return (
   379→    <Popover open={open} onOpenChange={setOpen}>
   380→      <PopoverTrigger asChild>
   381→        <Button variant="ghost" size="icon" className="relative h-9 w-9 hover:bg-muted/80 transition-colors">
   382→          <Bell className="h-[18px] w-[18px] text-muted-foreground" />
   383→          {unread.length > 0 && (
   384→            <span className="absolute -top-0.5 -right-0.5 min-w-[18px] h-[18px] px-1 bg-red-500 text-white text-[10px] font-bold rounded-full flex items-center justify-center shadow-sm shadow-red-200 animate-in zoom-in-50 duration-200">
   385→              {unread.length > 99 ? '99+' : unread.length}
   386→            </span>
   387→          )}
   388→        </Button>
   389→      </PopoverTrigger>
   390→      <PopoverContent
   391→        className="w-80 p-0 glass-card rounded-xl overflow-hidden animate-in slide-in-from-top-2 fade-in-0 duration-200"
   392→        align="end"
   393→        sideOffset={8}
   394→      >
   395→        {/* Header */}
   396→        <div className="flex items-center justify-between px-4 py-3 border-b border-border/50">
   397→          <div className="flex items-center gap-2">
   398→            <h4 className="text-sm font-bold text-foreground">Notifications</h4>
   399→            {unread.length > 0 && (
   400→              <Badge className="text-[10px] px-1.5 py-0 bg-red-500 text-white font-bold border-0">
   401→                {unread.length}
   402→              </Badge>
   403→            )}
   404→          </div>
   405→          {unread.length > 0 && (
   406→            <Button
   407→              variant="ghost"
   408→              size="sm"
   409→              className="text-[11px] h-7 px-2 text-primary hover:text-primary/80 hover:bg-primary/10 font-semibold"
   410→              onClick={onMarkAllRead}
   411→            >
   412→              <Check className="h-3 w-3 mr-1" /> Mark all read
   413→            </Button>
   414→          )}
   415→        </div>
   416→
   417→        {/* Items */}
   418→        <ScrollArea className="max-h-[340px]">
   419→          {displayItems.length === 0 ? (
   420→            <div className="py-8 text-center">
   421→              <div className="w-12 h-12 mx-auto mb-3 rounded-full bg-orange-50 dark:bg-orange-950/30 flex items-center justify-center">
   422→                <CheckCircle className="h-6 w-6 text-orange-400" />
   423→              </div>
   424→              <p className="text-sm font-medium text-muted-foreground">All caught up!</p>
   425→              <p className="text-xs text-muted-foreground/60 mt-0.5">No new notifications</p>
   426→            </div>
   427→          ) : (
   428→            <div className="divide-y divide-border/40">
   429→              {displayItems.map((n, i) => {
   430→                const Icon = NOTIFICATION_ICONS[n.type] || Info;
   431→                const colorClass = NOTIFICATION_COLORS[n.type] || 'text-gray-500';
   432→                const bgClass = NOTIFICATION_BG[n.type] || 'bg-gray-50';
   433→                return (
   434→                  <button
   435→                    key={n.id}
   436→                    onClick={() => {
   437→                      onMarkAsRead(n.id);
   438→                      setOpen(false);
   439→                    }}
   440→                    className="w-full flex items-start gap-3 px-4 py-3 hover:bg-muted/50 transition-colors text-left group"
   441→                    style={{ animationDelay: `${i * 40}ms` }}
   442→                  >
   443→                    <div
   444→                      className={`w-8 h-8 rounded-lg ${bgClass} flex items-center justify-center flex-shrink-0 mt-0.5 transition-transform group-hover:scale-110`}
   445→                    >
   446→                      <Icon className={`h-4 w-4 ${colorClass}`} />
   447→                    </div>
   448→                    <div className="min-w-0 flex-1">
   449→                      <p className="text-[13px] font-semibold truncate text-foreground">{n.title}</p>
   450→                      <p className="text-[11px] text-muted-foreground truncate mt-0.5 leading-relaxed">{n.message}</p>
   451→                      <p className="text-[10px] text-muted-foreground/50 mt-1 font-medium">{relativeTime(n.createdAt)}</p>
   452→                    </div>

   454→                  </button>
   455→                );
   456→              })}
   457→            </div>
   458→          )}
   459→        </ScrollArea>
   460→
   461→        {/* Footer */}
   462→        {notifications.length > 6 && (

   466→              size="sm"

   472→          </div>
   473→        )}
   474→      </PopoverContent>
   475→    </Popover>
   476→  );
