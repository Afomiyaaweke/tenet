    96→/* ───────────────────────── Password Strength Meter ───────────────────────── */
    97→function getPasswordStrength(password: string): { score: number; label: string; color: string } {
    98→  let score = 0;
    99→  if (password.length >= 8) score += 25;
   100→  if (password.length >= 12) score += 10;
   101→  if (/[A-Z]/.test(password)) score += 20;
   102→  if (/[a-z]/.test(password)) score += 20;
   103→  if (/[0-9]/.test(password)) score += 20;
   104→  if (/[^A-Za-z0-9]/.test(password)) score += 5;  // bonus only
   105→  if (new Set(password.toLowerCase()).size >= 8) score += 5;
   106→
   107→  if (score >= 70) return { score, label: 'Strong', color: 'bg-green-500' };
   108→  if (score >= 50) return { score, label: 'Good', color: 'bg-blue-500' };
   109→  if (score >= 30) return { score, label: 'Fair', color: 'bg-yellow-500' };
   110→  return { score, label: 'Weak', color: 'bg-red-500' };
   111→}
   112→
   113→function PasswordStrengthMeter({ password }: { password: string }) {
   114→  if (!password) return null;
   115→  const { score, label, color } = getPasswordStrength(password);
   116→  return (
   117→    <div className="space-y-1.5 mt-2">
   118→      <div className="flex gap-1">
   119→        {[25, 50, 75, 100].map(threshold => (
   120→          <div key={threshold} className="h-1.5 flex-1 rounded-full bg-muted">
   121→            <div
   122→              className={`h-full rounded-full transition-all duration-300 ${score >= threshold ? color : 'bg-transparent'}`}
   123→              style={{ width: score >= threshold ? '100%' : '0%' }}
   124→            />
   125→          </div>
   126→        ))}
   127→      </div>
   128→      <div className="flex justify-between items-center">
   129→        <span className="text-xs text-muted-foreground">Strength:</span>
   130→        <span className={`text-xs font-medium ${score >= 80 ? 'text-green-600' : score >= 60 ? 'text-blue-600' : score >= 40 ? 'text-yellow-600' : 'text-red-600'}`}>{label}</span>
   131→      </div>
   132→    </div>
   133→  );
   134→}

   159→/* ───────────────────────── Step Indicator ───────────────────────── */
   160→function StepIndicator({ currentStep, totalSteps = 4 }: { currentStep: RegStep; totalSteps?: number }) {
   161→  return (
   162→    <div className="flex items-center justify-center gap-1 mb-6">
   163→      {Array.from({ length: totalSteps }, (_, i) => (i + 1) as RegStep).map((step, i) => {
   164→        const meta = REG_STEP_META[step];
   165→        const isCompleted = currentStep > step;
   166→        const isCurrent = currentStep === step;
   167→        const Icon = meta.icon;
   168→        return (
   169→          <React.Fragment key={step}>
   170→            <div className="flex items-center gap-1.5">
   171→              <div
   172→                className={`h-7 w-7 rounded-full flex items-center justify-center text-xs font-bold transition-all ${
   173→                  isCompleted || isCurrent
   174→                    ? 'gradient-orange text-white shadow-md shadow-orange-500/30'
   175→                    : 'bg-muted text-muted-foreground'
   176→                }`}
   177→              >
   178→                {isCompleted ? <CheckCircle2 className="w-4 h-4" /> : <Icon className="w-3.5 h-3.5" />}
   179→              </div>
   180→              <span className={`text-xs font-medium hidden sm:inline ${isCompleted || isCurrent ? 'text-foreground' : 'text-muted-foreground'}`}>
   181→                {meta.label}
   182→              </span>
   183→            </div>
   184→            {i < totalSteps - 1 && (
   185→              <div className={`h-0.5 w-4 sm:w-8 rounded-full ${currentStep > step ? 'bg-orange-500' : 'bg-muted'}`} />
   186→            )}
   187→          </React.Fragment>
   188→        );
   189→      })}
   190→    </div>
   191→  );
   192→}

   196→export function AuthGate({ onBack }: { onBack?: () => void }) {
   197→  const { login, register } = useAuthStore();
   198→  const [loginData, setLoginData] = useState({ email: '', password: '' });
   199→  const [regData, setRegData] = useState({
   200→    email: '', password: '',
   201→    // Personal
   202→    fullName: '', jobTitle: '', phone: '', location: '',
   203→    // Company
   204→    companyName: '', companyIndustry: '', companyTinNumber: '',
   205→    companyRegistrationNo: '', companyPhone: '', companyCity: '',
   206→    companyCountry: '', companyEmail: '', companyWebsite: '',
   207→    // Role
   208→    role: 'user' as string,
   209→  });
   210→  const [loading, setLoading] = useState(false);
   211→  const [emailChecking, setEmailChecking] = useState(false);
   212→
   213→  const [activeTab, setActiveTab] = useState<'login' | 'register'>('login');
   214→  const [showPassword, setShowPassword] = useState(false);
   215→  const [regStep, setRegStep] = useState<RegStep>(1);
   216→
   217→  // Forgot / Reset password state
   218→  const [authMode, setAuthMode] = useState<'login' | 'register' | 'forgot-password' | 'forgot-sent' | 'reset-password'>('login');
   219→  const [forgotEmail, setForgotEmail] = useState('');
   220→  const [resetToken, setResetToken] = useState('');
   221→  const [tokenValidation, setTokenValidation] = useState<'idle' | 'checking' | 'valid' | 'invalid'>('idle');
   222→  const [newPassword, setNewPassword] = useState('');
   223→  const [confirmPassword, setConfirmPassword] = useState('');
   224→  const [regConfirmPassword, setRegConfirmPassword] = useState('');
   225→  const [resetSuccess, setResetSuccess] = useState(false);
   226→
   227→  // Handle ?token=xxx in URL for email reset links
   228→  useEffect(() => {
   229→    const params = new URLSearchParams(window.location.search);
   230→    const tokenFromUrl = params.get('token');
   231→    if (tokenFromUrl) {
   232→      setResetToken(tokenFromUrl);
   233→      setAuthMode('reset-password');
   234→      // Clean up URL without reloading
   235→      window.history.replaceState({}, '', window.location.pathname);
   236→    }
   237→  }, []);
   238→
   239→  // Debounced token validation — checks the code against the server as user types
   240→  useEffect(() => {
   241→    if (!resetToken || resetToken.length < 10) {
   242→      setTokenValidation('idle');
   243→      return;
   244→    }
   245→
   246→    const timer = setTimeout(async () => {
   247→      setTokenValidation('checking');
   248→      try {
   249→        const { api } = await import('@/lib/api');
   250→        const res = await api.get(`/auth/validate-reset-token?token=${encodeURIComponent(resetToken)}`);
   251→        setTokenValidation(res.valid ? 'valid' : 'invalid');
   252→      } catch {
   253→        setTokenValidation('idle'); // Network error — don't block
   254→      }
   255→    }, 600); // 600ms debounce
   256→
   257→    return () => clearTimeout(timer);
   258→  }, [resetToken]);
   259→
   260→  const handleLoginSubmit = async (e: React.FormEvent) => {
   261→    e.preventDefault();
   262→    if (!loginData.email || !loginData.password) return;
   263→    setLoading(true);
   264→    try {
   265→      const result = await login(loginData.email, loginData.password);
   266→      if (!result.success) {
   267→        toast.error(result.error || 'Invalid email or password. Please try again.');
   268→      }
   269→    } catch {
   270→      toast.error('Login failed. Please check your connection and try again.');
   271→    }
   272→    setLoading(false);
   273→  };
   274→
   275→  const handleForgotPassword = async (e: React.FormEvent) => {
   276→    e.preventDefault();
   277→    if (!forgotEmail) return;
   278→    setLoading(true);
   279→    try {
   280→      const { api } = await import('@/lib/api');
   281→      const res = await api.post('/auth/forgot-password', { email: forgotEmail });
   282→      if (res.success) {
   283→        // Always show the confirmation screen to prevent email enumeration
   284→        // The reset token is sent ONLY to the user's email - never in the API response
   285→        setResetToken(''); // Clear any previous token
   286→        setTokenValidation('idle');
   287→        setAuthMode('forgot-sent');
   288→      } else {
   289→        toast.error(res.error || 'Failed to process request');
   290→      }
   291→    } catch {
   292→      toast.error('Network error. Please try again.');
   293→    }
   294→    setLoading(false);
   295→  };
   296→
   297→  const handleResetPassword = async (e: React.FormEvent) => {
   298→    e.preventDefault();
   299→    if (!resetToken || !newPassword || !confirmPassword) return;
   300→    if (newPassword !== confirmPassword) {
   301→      toast.error('Passwords do not match');
   302→      return;
   303→    }
   304→    if (newPassword.length < 8) {
   305→      toast.error('Password must be at least 8 characters');
   306→      return;
   307→    }
   308→    if (!/[A-Z]/.test(newPassword) || !/[a-z]/.test(newPassword) || !/[0-9]/.test(newPassword)) {
   309→      toast.error('Password must include uppercase, lowercase, and number');
   310→      return;
   311→    }
   312→    setLoading(true);
   313→    try {
   314→      const { api } = await import('@/lib/api');
   315→      const res = await api.post('/auth/reset-password', {
   316→        token: resetToken,
   317→        newPassword,
   318→        confirmPassword,
   319→      });
   320→      if (res.success) {
   321→        setResetSuccess(true);
   322→        toast.success('Password reset successfully! You can now sign in.');
   323→      } else {
   324→        toast.error(res.error || 'Failed to reset password');
   325→      }
   326→    } catch {
   327→      toast.error('Network error. Please try again.');
   328→    }
   329→    setLoading(false);
   330→  };
   331→
   332→  /* ─── Registration step navigation ─── */
   333→  const canGoNext = (): boolean => {
   334→    switch (regStep) {
   335→      case 1:
   336→        return !!regData.email &&
   337→          !!regData.password &&
   338→          regData.password.length >= 8 &&
   339→          /[A-Z]/.test(regData.password) &&
   340→          /[a-z]/.test(regData.password) &&
   341→          /[0-9]/.test(regData.password) &&
   342→          !!regConfirmPassword &&
   343→          regData.password === regConfirmPassword;
   344→      case 2:
   345→        return !!regData.companyName;
   346→      case 3:
   347→        return !!regData.fullName;
   348→      case 4:
   349→        return true;
   350→      default:
   351→        return false;
   352→    }
   353→  };
   354→
   355→  const goNext = async () => {
   356→    if (!canGoNext()) return;
   357→    // At Step 1, check if email is already registered → auto-switch to login
   358→    if (regStep === 1 && regData.email) {
   359→      setEmailChecking(true);
   360→      try {
   361→        const { api } = await import('@/lib/api');
   362→        const res = await api.post('/auth/check-email', { email: regData.email });
   363→        if (res.success && res.exists) {
   364→          setEmailChecking(false);
   365→          // Auto-switch to login with pre-filled email
   366→          toast.success('Welcome back! Switching to Sign In...', { duration: 4000 });
   367→          setActiveTab('login');
   368→          setLoginData(d => ({ ...d, email: regData.email }));
   369→          return;
   370→        }
   371→      } catch {
   372→        // If check fails, just proceed — the register endpoint will catch it later
   373→      }
   374→      setEmailChecking(false);
   375→    }
   376→    if (regStep < 4) setRegStep((regStep + 1) as RegStep);
   377→  };
   378→
   379→  const goBack = () => {
   380→    if (regStep > 1) setRegStep((regStep - 1) as RegStep);
   381→  };
   382→
   383→  const handleRegister = async () => {
   384→    setLoading(true);
   385→    try {
   386→      const result = await register(regData);
   387→      if (!result.success) {
   388→        // If the email is already registered, auto-switch to login form
   389→        if (result.error?.toLowerCase().includes('already') || result.error?.toLowerCase().includes('registered')) {
   390→          toast.success('Welcome back! Switching to Sign In...', { duration: 4000 });
   391→          setActiveTab('login');
   392→          setLoginData(d => ({ ...d, email: regData.email || d.email }));
   393→          setRegStep(1);
   394→        } else {
   395→          toast.error(result.error || 'Registration failed. Please try again.');
   396→        }
   397→      }
   398→      else toast.success('Welcome to TenetBid!');
   399→    } catch {
   400→      toast.error('Registration failed. Please check your connection and try again.');
   401→    }
   402→    setLoading(false);
   403→  };
   404→
   405→
   406→
   407→  return (
   408→    <div className="min-h-screen flex flex-col">
   409→      {/* ─── Main content area ─── */}
   410→      <div className="flex-1 flex flex-col lg:flex-row">
   411→
   412→        {/* ═══════════ LEFT PANEL (Desktop) ═══════════ */}
   413→        <div className="hidden lg:flex lg:w-[52%] relative overflow-hidden gradient-slate flex-col">
   414→          <FloatingDots />
