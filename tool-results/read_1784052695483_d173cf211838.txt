    95→/* ───────────────────────── Password Strength Meter ───────────────────────── */
    96→function scorePassword(pw: string): { score: number; label: string; color: string } {
    97→  let score = 0;
    98→  if (pw.length >= 8) score++;
    99→  if (pw.length >= 12) score++;
   100→  if (/[a-z]/.test(pw) && /[A-Z]/.test(pw)) score++;
   101→  if (/[0-9]/.test(pw)) score++;
   102→  if (/[^A-Za-z0-9]/.test(pw)) score++;
   103→  const labels = ['Too weak', 'Weak', 'Fair', 'Good', 'Strong', 'Very strong'];
   104→  const colors = ['bg-rose-500', 'bg-rose-500', 'bg-amber-500', 'bg-yellow-500', 'bg-orange-500', 'bg-orange-600'];
   105→  return { score, label: labels[score], color: colors[score] };
   106→}

   108→/* ───────────────────────── Step Indicator ───────────────────────── */
   109→function StepIndicator({ currentStep, totalSteps = 4 }: { currentStep: RegStep; totalSteps?: number }) {
   110→  return (
   111→    <div className="flex items-center justify-center gap-1 mb-6">
   112→      {Array.from({ length: totalSteps }, (_, i) => (i + 1) as RegStep).map((step, i) => {
   113→        const meta = REG_STEP_META[step];
   114→        const isCompleted = currentStep > step;
   115→        const isCurrent = currentStep === step;
   116→        const Icon = meta.icon;
   117→        return (
   118→          <React.Fragment key={step}>
   119→            <div className="flex items-center gap-1.5">
   120→              <div
   121→                className={`h-7 w-7 rounded-full flex items-center justify-center text-xs font-bold transition-all ${
   122→                  isCompleted || isCurrent
   123→                    ? 'gradient-orange text-white shadow-md shadow-orange-500/30'
   124→                    : 'bg-muted text-muted-foreground'
   125→                }`}
   126→              >
   127→                {isCompleted ? <CheckCircle2 className="w-4 h-4" /> : <Icon className="w-3.5 h-3.5" />}
   128→              </div>
   129→              <span className={`text-xs font-medium hidden sm:inline ${isCompleted || isCurrent ? 'text-foreground' : 'text-muted-foreground'}`}>
   130→                {meta.label}
   131→              </span>
   132→            </div>
   133→            {i < totalSteps - 1 && (
   134→              <div className={`h-0.5 w-4 sm:w-8 rounded-full ${currentStep > step ? 'bg-orange-500' : 'bg-muted'}`} />
   135→            )}
   136→          </React.Fragment>
   137→        );
   138→      })}
   139→    </div>
   140→  );
   141→}
