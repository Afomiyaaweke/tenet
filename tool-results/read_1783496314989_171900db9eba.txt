   109→   109→function generateSeal(text: string, date: string): string {
   110→   110→  const svg = `<svg xmlns="http://www.w3.org/2000/svg" width="120" height="120" viewBox="0 0 120 120">
   111→   111→    <circle cx="60" cy="60" r="55" fill="none" stroke="#dc2626" stroke-width="3"/>
   112→   112→    <circle cx="60" cy="60" r="48" fill="none" stroke="#dc2626" stroke-width="1.5"/>
   113→   113→    <path id="topArc" d="M 15,60 A 45,45 0 0,1 105,60" fill="none"/>
   114→   114→    <text font-size="10" fill="#dc2626" font-weight="bold" letter-spacing="3">
   115→   115→      <textPath href="#topArc" startOffset="50%" text-anchor="middle">${text}</textPath>
   116→   116→    </text>
   117→   117→    <text x="60" y="68" text-anchor="middle" font-size="9" fill="#dc2626">${date}</text>
   118→   118→    <text x="60" y="80" text-anchor="middle" font-size="7" fill="#dc2626">TENETS</text>
   119→   119→  </svg>`;
   120→   120→  return 'data:image/svg+xml;base64,' + btoa(svg);
   121→   121→}

   123→   123→function formatAIResultToHTML(data: Record<string, unknown>): string {
   124→   124→  let html = '';
   125→   125→  for (const [key, value] of Object.entries(data)) {
   126→   126→    const title = key.replace(/([A-Z])/g, ' $1').replace(/^./, s => s.toUpperCase());
   127→   127→    const content = typeof value === 'string' ? value : JSON.stringify(value, null, 2);
   128→   128→    const formattedContent = content
   129→   129→      .replace(/&/g, '&amp;')
   130→   130→      .replace(/</g, '&lt;')
   131→   131→      .replace(/>/g, '&gt;')
   132→   132→      .replace(/\*\*(.+?)\*\*/g, '<strong>$1</strong>')
   133→   133→      .replace(/^[-*•]\s+(.*)/gm, '<li>$1</li>')
   134→   134→      .replace(/^(\d+)[.)]\s+(.*)/gm, '<li>$2</li>')
   135→   135→      .replace(/^#{1,3}\s+(.*)/gm, '<strong>$1</strong>')
   136→   136→      .replace(/\n\n/g, '</p><p>')
   137→   137→      .replace(/\n/g, '<br/>');
   138→   138→    html += `<h2 style="font-size:16px;font-weight:700;margin:16px 0 8px;color:#059669;border-bottom:1px solid #d1fae5;padding-bottom:4px;">${title}</h2>`;
   139→   139→    html += `<p>${formattedContent}</p>`;
   140→   140→  }
   141→   141→  return html;
   142→   142→}

   161→   161→function SkillTagSelector({ selected, onChange }: { selected: string[]; onChange: (s: string[]) => void }) {
   162→   162→  const toggle = (skill: string) => {
   163→   163→    if (selected.includes(skill)) onChange(selected.filter(s => s !== skill));
   164→   164→    else onChange([...selected, skill]);
   165→   165→  };

   167→   167→    <div className="flex flex-wrap gap-1">
   168→   168→      {COMMON_SKILLS.map(skill => (
   169→   169→        <button key={skill} type="button" onClick={() => toggle(skill)}
   170→   170→          className={`px-2 py-0.5 rounded-full text-[10px] font-medium transition-all border ${
   171→   171→            selected.includes(skill)
   172→   172→              ? 'bg-emerald-50 text-emerald-700 border-emerald-300'
   173→   173→              : 'bg-muted/50 text-muted-foreground border-border hover:border-emerald-300'
   174→   174→          }`}>
   175→   175→          {skill}
   176→   176→        </button>
   177→   177→      ))}
   178→   178→    </div>
   179→   179→  );
   180→   180→}
