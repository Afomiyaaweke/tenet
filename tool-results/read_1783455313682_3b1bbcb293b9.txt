   167→
   168→function parseAmount(value: number | string | undefined): number {
   169→  if (value === undefined || value === null) return 0;
   170→  if (typeof value === 'number') return value;
   171→  const n = Number(String(value).replace(/[^0-9.-]/g, ''));
   172→  return Number.isFinite(n) ? n : 0;
   173→}
   174→
   175→function truncate(text: string, max = 280): string {
   176→  if (!text) return '';
   177→  return text.length > max ? text.slice(0, max).trimEnd() + '…' : text;
   178→}

   206→
   207→function inferCategory(row: WorldBankProcurementRow): string {
   208→  const t = (row.contract_type || '').toLowerCase();
   209→  if (t.includes('work') || t.includes('civil') || t.includes('construction')) return 'Construction';
   210→  if (t.includes('consult')) return 'Consulting';
   211→  if (t.includes('good') || t.includes('supply') || t.includes('equipment')) return 'Supply';
   212→  if (t.includes('service')) return 'Consulting';
   213→  const p = (row.project_name || '').toLowerCase();
   214→  if (p.includes('road') || p.includes('bridge') || p.includes('infrastructure')) return 'Construction';
   215→  if (p.includes('agric') || p.includes('farm') || p.includes('irrigat')) return 'Agriculture';
   216→  if (p.includes('health') || p.includes('clinic') || p.includes('hospital')) return 'Healthcare';
   217→  if (p.includes('energy') || p.includes('power') || p.includes('electric')) return 'Engineering';
   218→  if (p.includes('edu') || p.includes('school')) return 'Education';
   219→  if (p.includes('water') || p.includes('sanitation')) return 'Engineering';
   220→  return 'Consulting';
   221→}
   222→
   223→function buildDeadline(signingDate?: string): string {
   224→  if (!signingDate) return new Date(Date.now() + 45 * 86400000).toISOString();
   225→  const base = new Date(signingDate);
   226→  if (Number.isNaN(base.getTime())) return new Date(Date.now() + 45 * 86400000).toISOString();
   227→  return new Date(base.getTime() + 45 * 86400000).toISOString();
   228→}

   230→export async function fetchWorldBankTenders(opts: {
   231→  search?: string;
   232→  rows?: number;
   233→}): Promise<{ tenders: LiveTender[]; total: number; ok: boolean }> {
   234→  const rows = Math.min(Math.max(opts.rows ?? 20, 1), 50);
   235→  const params = new URLSearchParams({
   236→    format: 'json',
   237→    rows: String(rows),
   238→    fl: [
   239→      'project_id',
   240→      'procurement_group_id',
   241→      'project_name',
   242→      'borrower',
   243→      'country',
   244→      'region',
   245→      'contract_description',
   246→      'contract_type',
   247→      'supplier',
   248→      'supplier_country',
   249→      'contract_signing_date',
   250→      'total_contract_amount',
   251→      'wb_contract_number',
   252→    ].join(','),
   253→  });
   254→  if (opts.search) params.set('q', opts.search);
   255→
   256→  const url = `https://search.worldbank.org/api/v2/procurement?${params.toString()}`;
   257→  const ctrl = new AbortController();
   258→  const timer = setTimeout(() => ctrl.abort(), 8000);
   259→
   260→  try {
   261→    const res = await fetch(url, {
   262→      signal: ctrl.signal,
   263→      headers: { Accept: 'application/json', 'User-Agent': 'Tenets-Tender-Ecosystem/1.0' },
   264→      cache: 'no-store',
   265→    });
   266→    clearTimeout(timer);
   267→    if (!res.ok) {
   268→      return { tenders: [], total: 0, ok: false };
   269→    }
   270→    const json = (await res.json()) as WorldBankResponse;
   271→    const rowsArr = json.rows || json.procurements || [];
   272→    const total = typeof json.total === 'number' ? json.total : rowsArr.length;
   273→
   274→    const tenders: LiveTender[] = rowsArr
   275→      .filter((r) => r && (r.contract_description || r.project_name))
   276→      .map((r, idx) => {
   277→        const externalId = r.procurement_group_id || r.project_id || `wb-${idx}`;
   278→        const title = truncate(r.contract_description || r.project_name || 'World Bank Contract Award', 160);
   279→        const amount = parseAmount(r.total_contract_amount);
   280→        const signingDate = r.contract_signing_date;
   281→        return {
   282→          id: `worldbank-${externalId}`,
   283→          title,
   284→          scope: truncate(r.contract_description || r.project_name || '', 400),
   285→          budgetMin: amount,
   286→          budgetMax: amount,
   287→          deadline: buildDeadline(signingDate),
   288→          location: r.country || r.region || 'Multiple',
   289→          categoryTags: inferCategory(r),
   290→          requiredDocs: '',
   291→          status: 'awarded',
   292→          createdBy: 'worldbank',
   293→          createdAt: signingDate || new Date().toISOString(),
   294→          updatedAt: signingDate || new Date().toISOString(),
   295→          source: 'worldbank',
   296→          externalId,
   297→          externalUrl: r.project_id
   298→            ? `https://projects.worldbank.org/en/projects-operations/project-detail/${r.project_id}`
   299→            : 'https://financesone.worldbank.org',
   300→          currency: 'USD',
   301→          borrower: r.borrower || undefined,
   302→          supplier: r.supplier || undefined,
   303→          contractType: r.contract_type || undefined,
   304→          signingDate: signingDate || undefined,
   305→          region: r.region || undefined,
   306→        } satisfies LiveTender;
   307→      });
   308→
   309→    return { tenders, total, ok: true };
   310→  } catch {
   311→    clearTimeout(timer);
   312→    return { tenders: [], total: 0, ok: false };
   313→  }
   314→}

   320→export async function fetchEuTedTenders(opts: {
   321→  search?: string;
   322→  rows?: number;
   323→}): Promise<{ tenders: LiveTender[]; total: number; ok: boolean }> {
   324→  const rows = Math.min(Math.max(opts.rows ?? 20, 1), 50);
   325→  const base = 'https://api.ted.europa.eu/v3/notices/search';
   326→  const body = {
   327→    query: opts.search ? `(${opts.search})` : '*',
   328→    fields: ['ND', 'TI_DOC', 'DS_DATE_PUB', 'DEADLINE_DATE', 'PLACE_CONTRACT', 'VALUE_CONTRACT', 'CA_NAME', 'TITLE'],
   329→    limit: rows,
   330→    offset: 0,
   331→    pagination: { pageNumber: 1, pageSize: rows },
   332→    sort: { field: 'DS_DATE_PUB', order: 'DESC' },
   333→  };
   334→  const ctrl = new AbortController();
   335→  const timer = setTimeout(() => ctrl.abort(), 8000);
   336→
   337→  try {
   338→    const res = await fetch(base, {
   339→      method: 'POST',
   340→      signal: ctrl.signal,
   341→      headers: { 'Content-Type': 'application/json', Accept: 'application/json' },
   342→      body: JSON.stringify(body),
   343→      cache: 'no-store',
   344→    });
   345→    clearTimeout(timer);
   346→    if (!res.ok) return { tenders: [], total: 0, ok: false };
   347→    const json = (await res.json()) as { notices?: unknown[]; total?: number };
   348→    const notices = Array.isArray(json.notices) ? json.notices : [];
   349→    const total = typeof json.total === 'number' ? json.total : notices.length;
   350→
   351→    const tenders: LiveTender[] = notices.map((n, idx) => {
   352→      const row = (n || {}) as Record<string, unknown>;
   353→      const nd = String(row.ND || row['ND'] || `ted-${idx}`);
   354→      const title = String(row.TI_DOC || row.TITLE || 'EU TED Notice');
   355→      const deadlineRaw = row.DEADLINE_DATE as string | undefined;
   356→      const amount = Number(row.VALUE_CONTRACT) || 0;
   357→      return {
   358→        id: `eu_ted-${nd}`,
   359→        title: truncate(title, 160),
   360→        scope: truncate(title, 400),
   361→        budgetMin: amount,
   362→        budgetMax: amount,
   363→        deadline: deadlineRaw || new Date(Date.now() + 30 * 86400000).toISOString(),
   364→        location: String(row.PLACE_CONTRACT || 'EU'),
   365→        categoryTags: 'Consulting',
   366→        requiredDocs: '',
   367→        status: 'open',
   368→        createdBy: 'eu_ted',
   369→        createdAt: String(row.DS_DATE_PUB || new Date().toISOString()),
   370→        updatedAt: String(row.DS_DATE_PUB || new Date().toISOString()),
   371→        source: 'eu_ted',
   372→        externalId: nd,
   373→        externalUrl: `https://ted.europa.eu/udl?uri=TED:NOTICE:${nd}:TEXT:EN:HTML`,
   374→        currency: 'EUR',
   375→        borrower: String(row.CA_NAME || '') || undefined,
   376→      } satisfies LiveTender;
   377→    });
   378→
   379→    return { tenders, total, ok: true };
   380→  } catch {
   381→    clearTimeout(timer);
   382→    return { tenders: [], total: 0, ok: false };
   383→  }
   384→}

   386→/* ─────────────────────────────────────────────────────────────────────
   387→ * UNGM — UN Global Procurement adapter (live via RSS)
   388→ * ───────────────────────────────────────────────────────────────────── */
   389→
   390→export async function fetchUngmTenders(opts: {
   391→  search?: string;
   392→  rows?: number;
   393→}): Promise<{ tenders: LiveTender[]; total: number; ok: boolean }> {
   394→  const ctrl = new AbortController();
   395→  const timer = setTimeout(() => ctrl.abort(), 8000);
   396→
   397→  try {
   398→    const res = await fetch('https://www.ungm.org/Public/Notice/RSS', {
   399→      signal: ctrl.signal,
   400→      headers: { 'User-Agent': 'Tenets-Tender-Ecosystem/1.0', Accept: 'application/rss+xml, text/xml, */*' },
   401→      cache: 'no-store',
   402→    });
   403→    clearTimeout(timer);
   404→
   405→    if (!res.ok) return { tenders: [], total: 0, ok: false };
   406→
   407→    const xml = await res.text();
   408→    const itemRegex = /<item[\s\S]*?<title>([\s\S]*?)<\/title>[\s\S]*?<link>([\s\S]*?)<\/link>[\s\S]*?<description>([\s\S]*?)<\/description>[\s\S]*?<\/item>/gi;
   409→    const matches = [...xml.matchAll(itemRegex)];
   410→
   411→    if (matches.length === 0) return { tenders: [], total: 0, ok: false };
   412→
   413→    const tenders: LiveTender[] = matches.slice(0, opts.rows ?? 10).map((m, idx) => {
   414→      const rawTitle = m[1]?.replace(/<!\[CDATA\[([\s\S]*?)\]\]>/, '$1').trim() || `UNGM Notice ${idx + 1}`;
   415→      const rawLink = m[2]?.trim() || 'https://www.ungm.org';
   416→      const rawDesc = m[3]?.replace(/<!\[CDATA\[([\s\S]*?)\]\]>/, '$1').trim() || '';
   417→      return {
   418→        id: `ungm-rss-${idx}`,
   419→        title: truncate(rawTitle, 160),
   420→        scope: truncate(rawDesc, 400),
   421→        budgetMin: 0,
   422→        budgetMax: 0,
   423→        deadline: new Date(Date.now() + 30 * 86400000).toISOString(),
   424→        location: 'International',
   425→        categoryTags: 'Consulting',
   426→        requiredDocs: '',
   427→        status: 'open',
   428→        createdBy: 'ungm',
   429→        createdAt: new Date().toISOString(),
   430→        updatedAt: new Date().toISOString(),
   431→        source: 'ungm',
   432→        externalId: `ungm-rss-${idx}`,
   433→        externalUrl: rawLink,
   434→        currency: 'USD',
   435→        region: 'Global',
   436→      } satisfies LiveTender;
   437→    });
   438→
   439→    return { tenders, total: matches.length, ok: true };
   440→  } catch {
   441→    clearTimeout(timer);
   442→    return { tenders: [], total: 0, ok: false };
   443→  }
   444→}

   625→/* ─────────────────────────────────────────────────────────────────────
   626→ * Aggregator — called by the API route
   627→ * ───────────────────────────────────────────────────────────────────── */
   628→
   629→export interface FetchLiveTendersResult {
   630→  tenders: LiveTender[];
   631→  meta: {
   632→    total: number;
   633→    sources: { id: string; name: string; live: boolean; ok: boolean; count: number }[];
   634→    fallback: boolean;
   635→    cachedAt: number;
   636→  };
   637→}
   638→
   639→const CACHE_TTL_MS = 5 * 60 * 1000; // 5 minutes
   640→interface CacheEntry {
   641→  result: FetchLiveTendersResult;
   642→  expiresAt: number;
   643→}
   644→const cache = new Map<string, CacheEntry>();
   645→
   646→export async function fetchLiveTenders(opts: {
   647→  source?: string;
   648→  search?: string;
   649→  rows?: number;
   650→}): Promise<FetchLiveTendersResult> {
   651→  const cacheKey = `${opts.source || 'all'}::${opts.search || ''}::${opts.rows || 20}`;
   652→  const hit = cache.get(cacheKey);
   653→  if (hit && hit.expiresAt > Date.now()) {
   654→    return { ...hit.result, meta: { ...hit.result.meta, cachedAt: Date.now() } };
   655→  }
   656→
   657→  const rows = opts.rows ?? 20;
   658→  const wantSource = opts.source || 'all';
   659→  const tasks: { id: string; name: string; live: boolean; p: Promise<{ tenders: LiveTender[]; total: number; ok: boolean }> }[] = [];
   660→
   661→  if (wantSource === 'all' || wantSource === 'worldbank') {
   662→    tasks.push({
   663→      id: 'worldbank',
   664→      name: 'World Bank',
   665→      live: true,
   666→      p: fetchWorldBankTenders({ search: opts.search, rows }),
   667→    });
   668→  }
   669→  if (wantSource === 'all' || wantSource === 'eu_ted') {
   670→    tasks.push({
   671→      id: 'eu_ted',
   672→      name: 'EU TED',
   673→      live: true,
   674→      p: fetchEuTedTenders({ search: opts.search, rows }),
   675→    });
   676→  }
   677→  if (wantSource === 'all' || wantSource === 'ungm') {
   678→    tasks.push({
   679→      id: 'ungm',
   680→      name: 'UNGM',
   681→      live: true,
   682→      p: fetchUngmTenders({ search: opts.search, rows: Math.min(rows, 5) }),
   683→    });
   684→  }
   685→  if (wantSource === 'all' || wantSource === 'sam_gov') {
   686→    tasks.push({
   687→      id: 'sam_gov',
   688→      name: 'SAM.gov',
   689→      live: true,
   690→      p: fetchSamGovTenders({ search: opts.search, rows: Math.min(rows, 5) }),
   691→    });
   692→  }
   693→  if (wantSource === 'all' || wantSource === 'afdb') {
   694→    tasks.push({
   695→      id: 'afdb',
   696→      name: 'AfDB',
   697→      live: true,
   698→      p: fetchAfdbTenders({ search: opts.search, rows: Math.min(rows, 5) }),
   699→    });
   700→  }
   701→  if (wantSource === 'all' || wantSource === 'eu_opentenders') {
   702→    tasks.push({
   703→      id: 'eu_opentenders',
   704→      name: 'OpenTenders EU',
   705→      live: true,
   706→      p: fetchEuOpenTenders({ search: opts.search, rows: Math.min(rows, 5) }),
   707→    });
   708→  }
   709→  if (wantSource === 'all' || wantSource === 'jica') {
   710→    tasks.push({
   711→      id: 'jica',
   712→      name: 'JICA',
   713→      live: true,
   714→      p: fetchJicaTenders({ search: opts.search, rows: Math.min(rows, 5) }),
   715→    });
   716→  }
   717→  if (wantSource === 'all' || wantSource === 'adb') {
   718→    tasks.push({
   719→      id: 'adb',
   720→      name: 'ADB',
   721→      live: true,
   722→      p: fetchAdbTenders({ search: opts.search, rows: Math.min(rows, 5) }),
   723→    });
   724→  }
   725→  if (wantSource === 'all' || wantSource === 'uk_contracts') {
   726→    tasks.push({
   727→      id: 'uk_contracts',
   728→      name: 'UK Contracts Finder',
   729→      live: true,
   730→      p: fetchUkContractsTenders({ search: opts.search, rows: Math.min(rows, 5) }),
   731→    });
   732→  }
   733→  if (wantSource === 'all' || wantSource === 'dgmarket') {
   734→    tasks.push({
   735→      id: 'dgmarket',
   736→      name: 'DgMarket',
   737→      live: true,
   738→      p: fetchDgMarketTenders({ search: opts.search, rows: Math.min(rows, 5) }),
   739→    });
   740→  }
   741→
   742→  const settled = await Promise.all(tasks.map(async (t) => ({ ...t, res: await t.p })));
   743→
   744→  const sourcesMeta = settled.map((t) => ({
   745→    id: t.id,
   746→    name: t.name,
   747→    live: t.live,
   748→    ok: t.res.ok,
   749→    count: t.res.tenders.length,
   750→  }));
   751→
   752→  const tenders = settled.flatMap((t) => t.res.tenders);
   753→  const totalLiveOk = settled.some((t) => t.res.ok && t.res.tenders.length > 0);
   754→  const fallback = !totalLiveOk;
   755→
   756→  // Client-side search filter on top of upstream results
   757→  let filteredTenders = tenders;
   758→  if (opts.search) {
   759→    const q = opts.search.toLowerCase();
   760→    filteredTenders = tenders.filter(
   761→      (t) =>
   762→        t.title.toLowerCase().includes(q) ||
   763→        t.scope.toLowerCase().includes(q) ||
   764→        t.location.toLowerCase().includes(q) ||
   765→        t.categoryTags.toLowerCase().includes(q),
   766→    );
   767→  }
   768→
   769→  const result: FetchLiveTendersResult = {
   770→    tenders: filteredTenders,
   771→    meta: {
   772→      total: filteredTenders.length,
   773→      sources: sourcesMeta,
   774→      fallback,
   775→      cachedAt: Date.now(),
   776→    },
   777→  };
   778→
   779→  cache.set(cacheKey, { result, expiresAt: Date.now() + CACHE_TTL_MS });
   780→  return result;
   781→}
