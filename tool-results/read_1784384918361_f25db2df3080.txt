   178→function relativeTime(dateStr: string): string {
   179→  const now = Date.now();
   180→  const then = new Date(dateStr).getTime();
   181→  const diff = Math.max(0, now - then);
   182→  const seconds = Math.floor(diff / 1000);
   183→  if (seconds < 60) return 'just now';
   184→  const minutes = Math.floor(seconds / 60);
   185→  if (minutes < 60) return `${minutes}m ago`;
   186→  const hours = Math.floor(minutes / 60);
   187→  if (hours < 24) return `${hours}h ago`;
   188→  const days = Math.floor(hours / 24);
   189→  if (days < 7) return `${days}d ago`;
   190→  return new Date(dateStr).toLocaleDateString();
   191→}

   354→  onMarkAsRead,
   355→  onMarkAllRead,

   362→  const unread = useMemo(() => notifications.filter((n) => !n.read), [notifications]);
   363→  const displayItems = useMemo(() => unread.slice(0, 6), [unread]);
   364→
   365→  return (
   366→    <Popover open={open} onOpenChange={setOpen}>

   452→              variant="ghost"
   453→              size="sm"
   454→              className="w-full text-[11px] text-muted-foreground hover:text-foreground font-medium"

   457→              View all notifications
