    95→    95→/* ───────────────────────── Password Strength Meter ───────────────────────── */
    96→    96→function scorePassword(pw: string): { score: number; label: string; color: string } {
    97→    97→  let score = 0;
    98→    98→  if (pw.length >= 8) score++;
    99→    99→  if (pw.length >= 12) score++;
   100→   100→  if (/[a-z]/.test(pw) && /[A-Z]/.test(pw)) score++;
   101→   101→  if (/[0-9]/.test(pw)) score++;
   102→   102→  if (/[^A-Za-z0-9]/.test(pw)) score++;
   103→   103→  const labels = ['Too weak', 'Weak', 'Fair', 'Good', 'Strong', 'Very strong'];
   104→   104→  const colors = ['bg-rose-500', 'bg-rose-500', 'bg-amber-500', 'bg-yellow-500', 'bg-orange-500', 'bg-orange-600'];
   105→   105→  return { score, label: labels[score], color: colors[score] };
   106→   106→}
