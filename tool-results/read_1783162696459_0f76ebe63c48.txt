   127→   127→
   128→   128→function parseAmount(value: number | string | undefined): number {
   129→   129→  if (value === undefined || value === null) return 0;
   130→   130→  if (typeof value === 'number') return value;
   131→   131→  const n = Number(String(value).replace(/[^0-9.-]/g, ''));
   132→   132→  return Number.isFinite(n) ? n : 0;
   133→   133→}
   134→   134→
   135→   135→function truncate(text: string, max = 280): string {
   136→   136→  if (!text) return '';
   137→   137→  return text.length > max ? text.slice(0, max).trimEnd() + '…' : text;
   138→   138→}

   161→   161→  total?: number;

   166→   166→
   167→   167→function inferCategory(row: WorldBankProcurementRow): string {
   168→   168→  const t = (row.contract_type || '').toLowerCase();
   169→   169→  if (t.includes('work') || t.includes('civil') || t.includes('construction')) return 'Construction';
   170→   170→  if (t.includes('consult')) return 'Consulting';
   171→   171→  if (t.includes('good') || t.includes('supply') || t.includes('equipment')) return 'Supply';
   172→   172→  if (t.includes('service')) return 'Consulting';
   173→   173→  const p = (row.project_name || '').toLowerCase();
   174→   174→  if (p.includes('road') || p.includes('bridge') || p.includes('infrastructure')) return 'Construction';
   175→   175→  if (p.includes('agric') || p.includes('farm') || p.includes('irrigat')) return 'Agriculture';
   176→   176→  if (p.includes('health') || p.includes('clinic') || p.includes('hospital')) return 'Healthcare';
   177→   177→  if (p.includes('energy') || p.includes('power') || p.includes('electric')) return 'Engineering';
   178→   178→  if (p.includes('edu') || p.includes('school')) return 'Education';
   179→   179→  if (p.includes('water') || p.includes('sanitation')) return 'Engineering';
   180→   180→  return 'Consulting';
   181→   181→}
   182→   182→
   183→   183→function buildDeadline(signingDate?: string): string {
   184→   184→  if (!signingDate) return new Date(Date.now() + 45 * 86400000).toISOString();
   185→   185→  const base = new Date(signingDate);
   186→   186→  if (Number.isNaN(base.getTime())) return new Date(Date.now() + 45 * 86400000).toISOString();
   187→   187→  return new Date(base.getTime() + 45 * 86400000).toISOString();
   188→   188→}

   190→   190→export async function fetchWorldBankTenders(opts: {
   191→   191→  search?: string;
   192→   192→  rows?: number;
   193→   193→}): Promise<{ tenders: LiveTender[]; total: number; ok: boolean }> {
   194→   194→  const rows = Math.min(Math.max(opts.rows ?? 20, 1), 50);
   195→   195→  const params = new URLSearchParams({
   196→   196→    format: 'json',
   197→   197→    rows: String(rows),
   198→   198→    fl: [
   199→   199→      'project_id',
   200→   200→      'procurement_group_id',
   201→   201→      'project_name',
   202→   202→      'borrower',
   203→   203→      'country',
   204→   204→      'region',
   205→   205→      'contract_description',
   206→   206→      'contract_type',
   207→   207→      'supplier',
   208→   208→      'supplier_country',
   209→   209→      'contract_signing_date',
   210→   210→      'total_contract_amount',
   211→   211→      'wb_contract_number',
   212→   212→    ].join(','),
   213→   213→  });
   214→   214→  if (opts.search) params.set('q', opts.search);
   215→   215→
   216→   216→  const url = `https://search.worldbank.org/api/v2/procurement?${params.toString()}`;
   217→   217→  const ctrl = new AbortController();
   218→   218→  const timer = setTimeout(() => ctrl.abort(), 8000);
   219→   219→
   220→   220→  try {
   221→   221→    const res = await fetch(url, {
   222→   222→      signal: ctrl.signal,
   223→   223→      headers: { Accept: 'application/json', 'User-Agent': 'Tenets-Tender-Ecosystem/1.0' },
   224→   224→      cache: 'no-store',
   225→   225→    });
   226→   226→    clearTimeout(timer);
   227→   227→    if (!res.ok) {
   228→   228→      return { tenders: [], total: 0, ok: false };
   229→   229→    }
   230→   230→    const json = (await res.json()) as WorldBankResponse;
   231→   231→    const rowsArr = json.rows || json.procurements || [];
   232→   232→    const total = typeof json.total === 'number' ? json.total : rowsArr.length;
   233→   233→
   234→   234→    const tenders: LiveTender[] = rowsArr
   235→   235→      .filter((r) => r && (r.contract_description || r.project_name))
   236→   236→      .map((r, idx) => {
   237→   237→        const externalId = r.procurement_group_id || r.project_id || `wb-${idx}`;
   238→   238→        const title = truncate(r.contract_description || r.project_name || 'World Bank Contract Award', 160);
   239→   239→        const amount = parseAmount(r.total_contract_amount);
   240→   240→        const signingDate = r.contract_signing_date;
   241→   241→        return {
   242→   242→          id: `worldbank-${externalId}`,
   243→   243→          title,
   244→   244→          scope: truncate(r.contract_description || r.project_name || '', 400),
   245→   245→          budgetMin: amount,
   246→   246→          budgetMax: amount,
   247→   247→          deadline: buildDeadline(signingDate),
   248→   248→          location: r.country || r.region || 'Multiple',
   249→   249→          categoryTags: inferCategory(r),
   250→   250→          requiredDocs: '',
   251→   251→          status: 'awarded',
   252→   252→          createdBy: 'worldbank',
   253→   253→          createdAt: signingDate || new Date().toISOString(),
   254→   254→          updatedAt: signingDate || new Date().toISOString(),
   255→   255→          source: 'worldbank',
   256→   256→          externalId,
   257→   257→          externalUrl: r.project_id
   258→   258→            ? `https://projects.worldbank.org/en/projects-operations/project-detail/${r.project_id}`
   259→   259→            : 'https://financesone.worldbank.org',
   260→   260→          currency: 'USD',
   261→   261→          borrower: r.borrower || undefined,
   262→   262→          supplier: r.supplier || undefined,
   263→   263→          contractType: r.contract_type || undefined,
   264→   264→          signingDate: signingDate || undefined,
   265→   265→          region: r.region || undefined,
   266→   266→        } satisfies LiveTender;
   267→   267→      });
   268→   268→
   269→   269→    return { tenders, total, ok: true };
   270→   270→  } catch {
   271→   271→    clearTimeout(timer);
   272→   272→    return { tenders: [], total: 0, ok: false };
   273→   273→  }
   274→   274→}

   279→   279→
   280→   280→export async function fetchEuTedTenders(opts: {
   281→   281→  search?: string;
   282→   282→  rows?: number;
   283→   283→}): Promise<{ tenders: LiveTender[]; total: number; ok: boolean }> {
   284→   284→  const rows = Math.min(Math.max(opts.rows ?? 20, 1), 50);
   285→   285→  const base = 'https://api.ted.europa.eu/v3/notices/search';
   286→   286→  const body = {
   287→   287→    query: opts.search ? `(${opts.search})` : '*',
   288→   288→    fields: ['ND', 'TI_DOC', 'DS_DATE_PUB', 'DEADLINE_DATE', 'PLACE_CONTRACT', 'VALUE_CONTRACT', 'CA_NAME', 'TITLE'],
   289→   289→    limit: rows,
   290→   290→    offset: 0,
   291→   291→    pagination: { pageNumber: 1, pageSize: rows },
   292→   292→    sort: { field: 'DS_DATE_PUB', order: 'DESC' },
   293→   293→  };
   294→   294→  const ctrl = new AbortController();
   295→   295→  const timer = setTimeout(() => ctrl.abort(), 8000);
   296→   296→
   297→   297→  try {
   298→   298→    const res = await fetch(base, {
   299→   299→      method: 'POST',
   300→   300→      signal: ctrl.signal,
   301→   301→      headers: { 'Content-Type': 'application/json', Accept: 'application/json' },
   302→   302→      body: JSON.stringify(body),
   303→   303→      cache: 'no-store',
   304→   304→    });
   305→   305→    clearTimeout(timer);
   306→   306→    if (!res.ok) return { tenders: [], total: 0, ok: false };
   307→   307→    const json = (await res.json()) as { notices?: unknown[]; total?: number };
   308→   308→    const notices = Array.isArray(json.notices) ? json.notices : [];
   309→   309→    const total = typeof json.total === 'number' ? json.total : notices.length;
   310→   310→
   311→   311→    const tenders: LiveTender[] = notices.map((n, idx) => {
   312→   312→      const row = (n || {}) as Record<string, unknown>;
   313→   313→      const nd = String(row.ND || row['ND'] || `ted-${idx}`);
   314→   314→      const title = String(row.TI_DOC || row.TITLE || 'EU TED Notice');
   315→   315→      const deadlineRaw = row.DEADLINE_DATE as string | undefined;
   316→   316→      const amount = Number(row.VALUE_CONTRACT) || 0;
   317→   317→      return {
   318→   318→        id: `eu_ted-${nd}`,
   319→   319→        title: truncate(title, 160),
   320→   320→        scope: truncate(title, 400),
   321→   321→        budgetMin: amount,
   322→   322→        budgetMax: amount,
   323→   323→        deadline: deadlineRaw || new Date(Date.now() + 30 * 86400000).toISOString(),
   324→   324→        location: String(row.PLACE_CONTRACT || 'EU'),
   325→   325→        categoryTags: 'Consulting',
   326→   326→        requiredDocs: '',
   327→   327→        status: 'open',
   328→   328→        createdBy: 'eu_ted',
   329→   329→        createdAt: String(row.DS_DATE_PUB || new Date().toISOString()),
   330→   330→        updatedAt: String(row.DS_DATE_PUB || new Date().toISOString()),
   331→   331→        source: 'eu_ted',
   332→   332→        externalId: nd,
   333→   333→        externalUrl: `https://ted.europa.eu/udl?uri=TED:NOTICE:${nd}:TEXT:EN:HTML`,
   334→   334→        currency: 'EUR',
   335→   335→        borrower: String(row.CA_NAME || '') || undefined,
   336→   336→      } satisfies LiveTender;
   337→   337→    });
   338→   338→
   339→   339→    return { tenders, total, ok: true };
   340→   340→  } catch {
   341→   341→    clearTimeout(timer);
   342→   342→    return { tenders: [], total: 0, ok: false };
