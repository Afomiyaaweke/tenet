     1→'use client';
     2→
     3→import { useState, useEffect, useCallback } from 'react';
     4→import { LiveTender } from '@/lib/api';
     5→import { api } from '@/lib/api';
     6→import { useAuthStore } from '@/store';
     7→import {
     8→  Dialog, DialogContent, DialogHeader, DialogTitle,
     9→} from '@/components/ui/dialog';
    10→import {
    11→  Tabs, TabsContent, TabsList, TabsTrigger,
    12→} from '@/components/ui/tabs';
    13→import { Card, CardContent } from '@/components/ui/card';
    14→import { Badge } from '@/components/ui/badge';
    15→import { Button } from '@/components/ui/button';
    16→import { Input } from '@/components/ui/input';
    17→import { Separator } from '@/components/ui/separator';
    18→import { ScrollArea } from '@/components/ui/scroll-area';
    19→import { Skeleton } from '@/components/ui/skeleton';
    20→import { toast } from 'sonner';
    21→import {
    22→  FileSearch, FileText, ExternalLink, Download, Copy,
    23→  Loader2, Calendar, DollarSign, FileDown, FileSpreadsheet,
    24→  Search, ChevronDown, ChevronUp, X, Globe2, ScanSearch,
    25→} from 'lucide-react';
    26→
    27→// ── Types ──────────────────────────────────────────────────────────────
    28→interface DiscoverData {
    29→  title: string;
    30→  metaDescription?: string;
    31→  metaKeywords?: string;
    32→  content: string;
    33→  sections: { heading: string; content: string }[];
    34→  deadlines: string[];
    35→  budgets: string[];
    36→  documentLinks: { url: string; label: string; type: string }[];
    37→  pageLinks: { url: string; label: string; relevance: string }[];
    38→  url: string;
    39→  publishedTime?: string | null;
    40→  contentType: string;
    41→  fetchedAt: string;
    42→  stats: {
    43→    totalChars: number;
    44→    sectionsFound: number;
    45→    docLinksFound: number;
    46→    pageLinksFound: number;
    47→    deadlinesFound: number;
    48→    budgetsFound: number;
    49→  };
    50→}
    51→
    52→interface FullDocumentViewerProps {
    53→  tender: LiveTender;
    54→  open: boolean;
    55→  onOpenChange: (open: boolean) => void;
    56→}
    57→
    58→// ── Component ──────────────────────────────────────────────────────────
    59→export function FullDocumentViewer({ tender, open, onOpenChange }: FullDocumentViewerProps) {
    60→  const [loading, setLoading] = useState(false);
    61→  const [data, setData] = useState<DiscoverData | null>(null);
    62→  const [activeTab, setActiveTab] = useState('structured');
    63→  const [searchQuery, setSearchQuery] = useState('');
    64→  const [expandedSections, setExpandedSections] = useState<Set<number>>(new Set());
    65→  const [copySuccess, setCopySuccess] = useState(false);
    66→
    67→  const fetchDiscovery = useCallback(async () => {
    68→    const docUrl =
    69→      tender.documentUrl ||
    70→      (tender.requiredDocs && tender.requiredDocs.startsWith('http') ? tender.requiredDocs : null) ||
    71→      tender.externalUrl;
    72→
    73→    if (!docUrl) {
    74→      toast.error('No document URL available for this tender');
    75→      return;
    76→    }
    77→
    78→    setLoading(true);
    79→    try {
    80→      const res = await api.post('/tenders/fetch-doc/discover', { url: docUrl });
    81→      if (res.success && res.data) {
    82→        setData(res.data as DiscoverData);
    83→        // Expand all sections by default
    84→        setExpandedSections(new Set(res.data.sections.map((_: { heading: string; content: string }, i: number) => i)));
    85→      } else {
    86→        toast.error(res.error || 'Failed to discover document');
    87→      }
    88→    } catch {
    89→      toast.error('Failed to discover document');
    90→    }
    91→    setLoading(false);
    92→  }, [tender]);
    93→
    94→  // Reset state when dialog opens
    95→  useEffect(() => {
    96→    if (open) {
    97→      setData(null);
    98→      setActiveTab('structured');
    99→      setSearchQuery('');
   100→      setExpandedSections(new Set());
   101→      fetchDiscovery();
   102→    }
   103→  }, [open, fetchDiscovery]);
   104→
   105→  const toggleSection = (idx: number) => {
   106→    setExpandedSections((prev) => {
   107→      const next = new Set(prev);
   108→      if (next.has(idx)) next.delete(idx);
   109→      else next.add(idx);
   110→      return next;
   111→    });
   112→  };
   113→
   114→  const handleCopy = async () => {
   115→    if (!data) return;
   116→    try {
   117→      await navigator.clipboard.writeText(data.content);
   118→      setCopySuccess(true);
   119→      toast.success('Content copied to clipboard');
   120→      setTimeout(() => setCopySuccess(false), 2000);
   121→    } catch {
   122→      toast.error('Failed to copy content');
   123→    }
   124→  };
   125→
   126→  const handleExportPdf = async () => {
   127→    if (!data) return;
   128→    try {
   129→      const token = useAuthStore.getState().token;
   130→      const res = await fetch('/api/tenders/fetch-doc/export-pdf', {
   131→        method: 'POST',
   132→        headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${token}` },
   133→        body: JSON.stringify({ url: data.url, title: tender.title }),
   134→      });
   135→      if (!res.ok) throw new Error('Export failed');
   136→      const blob = await res.blob();
   137→      const blobUrl = window.URL.createObjectURL(blob);
   138→      const a = document.createElement('a');
   139→      a.href = blobUrl;
   140→      a.download = `Full_Doc_${(tender.title || 'Tender').replace(/[^a-zA-Z0-9]/g, '_')}.pdf`;
   141→      document.body.appendChild(a);
   142→      a.click();
   143→      document.body.removeChild(a);
   144→      window.URL.revokeObjectURL(blobUrl);
   145→      toast.success('PDF exported!');
   146→    } catch {
   147→      toast.error('Failed to export PDF');
   148→    }
   149→  };
   150→
   151→  const handleExportCsv = async () => {
   152→    if (!data) return;
   153→    try {
   154→      const token = useAuthStore.getState().token;
   155→      const res = await fetch('/api/tenders/fetch-doc/export-csv', {
   156→        method: 'POST',
   157→        headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${token}` },
   158→        body: JSON.stringify({ url: data.url, title: tender.title }),
   159→      });
   160→      if (!res.ok) throw new Error('Export failed');
   161→      const blob = await res.blob();
   162→      const blobUrl = window.URL.createObjectURL(blob);
   163→      const a = document.createElement('a');
   164→      a.href = blobUrl;
   165→      a.download = `Full_Doc_${(tender.title || 'Tender').replace(/[^a-zA-Z0-9]/g, '_')}.csv`;
   166→      document.body.appendChild(a);
   167→      a.click();
   168→      document.body.removeChild(a);
   169→      window.URL.revokeObjectURL(blobUrl);
   170→      toast.success('CSV exported!');
   171→    } catch {
   172→      toast.error('Failed to export CSV');
   173→    }
   174→  };
   175→
   176→  // Filter full text content based on search
   177→  const filteredContent = data?.content
   178→    ? searchQuery
   179→      ? data.content
   180→          .split('\n')
   181→          .filter((line) => line.toLowerCase().includes(searchQuery.toLowerCase()))
   182→          .join('\n')
   183→      : data.content
   184→    : '';
   185→
   186→  const sourceHost = (() => {
   187→    try {
   188→      return new URL(data?.url || tender.externalUrl || '').hostname;
   189→    } catch {
   190→      return 'external source';
   191→    }
   192→  })();
   193→
   194→  const docIcon = (type: string) => {
   195→    const t = type.toUpperCase();
   196→    if (['XLS', 'XLSX', 'CSV'].includes(t)) return <FileSpreadsheet className="h-4 w-4 text-emerald-600 dark:text-emerald-400" />;
   197→    if (['PDF'].includes(t)) return <FileDown className="h-4 w-4 text-rose-600 dark:text-rose-400" />;
   198→    return <FileText className="h-4 w-4 text-sky-600 dark:text-sky-400" />;
   199→  };
   200→
   201→  return (
   202→    <Dialog open={open} onOpenChange={onOpenChange}>
   203→      <DialogContent className="max-w-5xl h-[90vh] flex flex-col p-0 gap-0">
   204→        {/* Header */}
   205→        <DialogHeader className="px-6 pt-6 pb-3 space-y-2 border-b shrink-0">
   206→          <div className="flex items-start justify-between gap-3">
   207→            <div className="min-w-0 flex-1">
   208→              <DialogTitle className="text-lg font-bold leading-snug line-clamp-2">
   209→                {tender.title}
   210→              </DialogTitle>
   211→              <div className="flex items-center gap-2 mt-1.5 flex-wrap">
   212→                <Badge variant="outline" className="text-[10px] px-1.5 py-0 border-emerald-300 dark:border-emerald-700 text-emerald-700 dark:text-emerald-300 bg-emerald-50 dark:bg-emerald-950/30">
   213→                  <Globe2 className="h-2.5 w-2.5 mr-0.5" />
   214→                  {sourceHost}
   215→                </Badge>
   216→                {data?.publishedTime && (
   217→                  <Badge variant="outline" className="text-[10px] px-1.5 py-0">
   218→                    <Calendar className="h-2.5 w-2.5 mr-0.5" />
   219→                    {new Date(data.publishedTime).toLocaleDateString()}
   220→                  </Badge>
   221→                )}
   222→                {loading ? (
   223→                  <Badge variant="outline" className="text-[10px] px-1.5 py-0 text-amber-600 border-amber-300 dark:border-amber-700 dark:text-amber-400">
   224→                    <Loader2 className="h-2.5 w-2.5 mr-0.5 animate-spin" />
   225→                    Discovering…
   226→                  </Badge>
   227→                ) : data ? (
   228→                  <Badge variant="outline" className="text-[10px] px-1.5 py-0 text-emerald-600 border-emerald-300 dark:border-emerald-700 dark:text-emerald-400">
   229→                    <ScanSearch className="h-2.5 w-2.5 mr-0.5" />
   230→                    Discovered
   231→                  </Badge>
   232→                ) : null}
   233→              </div>
   234→            </div>
   235→            {/* Export buttons */}
   236→            <div className="flex items-center gap-1 shrink-0">
   237→              <Button
   238→                variant="ghost"
   239→                size="sm"
   240→                className="h-7 gap-1 text-[10px] px-2"
   241→                onClick={handleExportPdf}
   242→                disabled={!data}
   243→              >
   244→                <FileDown className="h-3 w-3" /> PDF
   245→              </Button>
   246→              <Button
   247→                variant="ghost"
   248→                size="sm"
   249→                className="h-7 gap-1 text-[10px] px-2"
   250→                onClick={handleExportCsv}
   251→                disabled={!data}
   252→              >
   253→                <FileSpreadsheet className="h-3 w-3" /> CSV
   254→              </Button>
   255→              <Button
   256→                variant="ghost"
   257→                size="sm"
   258→                className="h-7 gap-1 text-[10px] px-2"
   259→                onClick={handleCopy}
   260→                disabled={!data}
   261→              >
   262→                {copySuccess ? <span className="text-emerald-600">Copied!</span> : <><Copy className="h-3 w-3" /> Copy</>}
   263→              </Button>
   264→            </div>
   265→          </div>
   266→
   267→          {/* Stats bar */}
   268→          {data?.stats && (
   269→            <div className="flex items-center gap-3 flex-wrap text-[10px] text-muted-foreground">
   270→              <span>{data.stats.totalChars.toLocaleString()} chars</span>
   271→              <span className="text-border">|</span>
   272→              <span>{data.stats.sectionsFound} sections</span>
   273→              <span className="text-border">|</span>
   274→              <span>{data.stats.docLinksFound} documents</span>
   275→              <span className="text-border">|</span>
   276→              <span>{data.stats.pageLinksFound} related pages</span>
   277→              {data.stats.deadlinesFound > 0 && (
   278→                <>
   279→                  <span className="text-border">|</span>
   280→                  <span className="text-amber-600 dark:text-amber-400">{data.stats.deadlinesFound} deadline(s)</span>
   281→                </>
   282→              )}
   283→              {data.stats.budgetsFound > 0 && (
   284→                <>
   285→                  <span className="text-border">|</span>
   286→                  <span className="text-emerald-600 dark:text-emerald-400">{data.stats.budgetsFound} budget ref(s)</span>
   287→                </>
   288→              )}
   289→            </div>
   290→          )}
   291→        </DialogHeader>
   292→
   293→        {/* Tabs + Content */}
   294→        <div className="flex-1 min-h-0 flex flex-col">
   295→          {loading ? (
   296→            <LoadingSkeleton />
   297→          ) : data ? (
   298→            <Tabs value={activeTab} onValueChange={setActiveTab} className="flex-1 flex flex-col min-h-0">
   299→              <div className="px-6 pt-2 shrink-0">
   300→                <TabsList className="h-8 bg-muted/50">
   301→                  <TabsTrigger value="structured" className="text-xs gap-1 px-3 h-7">
   302→                    <FileSearch className="h-3 w-3" /> Structured View
   303→                  </TabsTrigger>
   304→                  <TabsTrigger value="fulltext" className="text-xs gap-1 px-3 h-7">
   305→                    <FileText className="h-3 w-3" /> Full Text
   306→                  </TabsTrigger>
   307→                  <TabsTrigger value="documents" className="text-xs gap-1 px-3 h-7">
   308→                    <Download className="h-3 w-3" /> Linked Documents
   309→                    {data.documentLinks.length > 0 && (
   310→                      <Badge className="ml-0.5 text-[9px] px-1 py-0 h-3.5 bg-emerald-100 text-emerald-700 dark:bg-emerald-900/40 dark:text-emerald-300 border-0">
   311→                        {data.documentLinks.length}
   312→                      </Badge>
   313→                    )}
   314→                  </TabsTrigger>
   315→                </TabsList>
   316→              </div>
   317→
   318→              {/* Structured View Tab */}
   319→              <TabsContent value="structured" className="flex-1 min-h-0 m-0 mt-2 px-6 pb-6">
   320→                <ScrollArea className="h-full">
   321→                  <div className="space-y-4 pr-3">
   322→                    {/* Meta description */}
   323→                    {data.metaDescription && (
   324→                      <Card className="border-emerald-200/60 dark:border-emerald-800/40 bg-emerald-50/30 dark:bg-emerald-950/10">
   325→                        <CardContent className="p-4">
   326→                          <p className="text-xs text-muted-foreground mb-1 font-medium uppercase tracking-wide">Description</p>
   327→                          <p className="text-sm leading-relaxed">{data.metaDescription}</p>
   328→                        </CardContent>
   329→                      </Card>
   330→                    )}
   331→
   332→                    {/* Deadlines & Budgets row */}
   333→                    {(data.deadlines.length > 0 || data.budgets.length > 0) && (
   334→                      <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   335→                        {data.deadlines.length > 0 && (
   336→                          <Card className="border-amber-200/60 dark:border-amber-800/40 bg-amber-50/30 dark:bg-amber-950/10">
   337→                            <CardContent className="p-4 space-y-2">
   338→                              <div className="flex items-center gap-1.5 text-xs font-semibold text-amber-700 dark:text-amber-300 uppercase tracking-wide">
   339→                                <Calendar className="h-3.5 w-3.5" />
   340→                                Deadlines
   341→                              </div>
   342→                              {data.deadlines.map((d, i) => (
   343→                                <div key={i} className="flex items-center gap-2 text-sm">
   344→                                  <div className="w-1.5 h-1.5 rounded-full bg-amber-500 shrink-0" />
   345→                                  <span>{d}</span>
   346→                                </div>
   347→                              ))}
   348→                            </CardContent>
   349→                          </Card>
   350→                        )}
   351→                        {data.budgets.length > 0 && (
   352→                          <Card className="border-emerald-200/60 dark:border-emerald-800/40 bg-emerald-50/30 dark:bg-emerald-950/10">
   353→                            <CardContent className="p-4 space-y-2">
   354→                              <div className="flex items-center gap-1.5 text-xs font-semibold text-emerald-700 dark:text-emerald-300 uppercase tracking-wide">
   355→                                <DollarSign className="h-3.5 w-3.5" />
   356→                                Budget References
   357→                              </div>
   358→                              {data.budgets.map((b, i) => (
   359→                                <div key={i} className="flex items-center gap-2 text-sm font-medium">
   360→                                  <div className="w-1.5 h-1.5 rounded-full bg-emerald-500 shrink-0" />
   361→                                  <span>{b}</span>
   362→                                </div>
   363→                              ))}
   364→                            </CardContent>
   365→                          </Card>
   366→                        )}
   367→                      </div>
   368→                    )}
   369→
   370→                    {/* Sections */}
   371→                    {data.sections.length > 0 ? (
   372→                      <div className="space-y-2">
   373→                        <h4 className="text-xs font-semibold text-muted-foreground uppercase tracking-wide flex items-center gap-1.5">
   374→                          <FileSearch className="h-3.5 w-3.5" />
   375→                          Sections ({data.sections.length})
   376→                        </h4>
   377→                        {data.sections.map((sec, idx) => (
   378→                          <Card key={idx} className="border-border/60">
   379→                            <CardContent className="p-0">
   380→                              <button
   381→                                className="w-full flex items-center justify-between p-3 text-left hover:bg-muted/40 transition-colors rounded-lg"
   382→                                onClick={() => toggleSection(idx)}
   383→                              >
   384→                                <span className="text-sm font-semibold text-foreground">{sec.heading}</span>
   385→                                {expandedSections.has(idx) ? (
   386→                                  <ChevronUp className="h-4 w-4 text-muted-foreground shrink-0" />
   387→                                ) : (
   388→                                  <ChevronDown className="h-4 w-4 text-muted-foreground shrink-0" />
   389→                                )}
   390→                              </button>
   391→                              {expandedSections.has(idx) && (
   392→                                <div className="px-3 pb-3 pt-0 animate-[fadeIn_0.2s_ease-out]">
   393→                                  <Separator className="mb-2" />
   394→                                  <p className="text-sm leading-relaxed whitespace-pre-wrap">{sec.content}</p>
   395→                                </div>
   396→                              )}
   397→                            </CardContent>
   398→                          </Card>
   399→                        ))}
   400→                      </div>
   401→                    ) : (
   402→                      <Card className="border-dashed">
   403→                        <CardContent className="p-6 text-center">
   404→                          <FileSearch className="h-8 w-8 text-muted-foreground/50 mx-auto mb-2" />
   405→                          <p className="text-sm text-muted-foreground">No structured sections detected.</p>
   406→                          <p className="text-xs text-muted-foreground mt-1">Try the Full Text tab to see the complete content.</p>
   407→                        </CardContent>
   408→                      </Card>
   409→                    )}
   410→
   411→                    {/* Related pages */}
   412→                    {data.pageLinks.length > 0 && (
   413→                      <div className="space-y-2">
   414→                        <h4 className="text-xs font-semibold text-muted-foreground uppercase tracking-wide flex items-center gap-1.5">
   415→                          <ExternalLink className="h-3.5 w-3.5" />
   416→                          Related Pages ({data.pageLinks.length})
   417→                        </h4>
   418→                        <div className="space-y-1">
   419→                          {data.pageLinks.map((p, i) => (
   420→                            <a
   421→                              key={i}
   422→                              href={p.url}
   423→                              target="_blank"
   424→                              rel="noopener noreferrer"
   425→                              className="flex items-center gap-2 text-sm p-2 rounded-lg hover:bg-muted/50 transition-colors group"
   426→                            >
   427→                              <Globe2 className="h-3.5 w-3.5 text-emerald-600 dark:text-emerald-400 shrink-0" />
   428→                              <span className="truncate flex-1 group-hover:text-emerald-700 dark:group-hover:text-emerald-300 transition-colors">
   429→                                {p.label}
   430→                              </span>
   431→                              <Badge variant="outline" className="text-[9px] px-1 py-0 h-4 shrink-0">
   432→                                {p.relevance}
   433→                              </Badge>
   434→                              <ExternalLink className="h-3 w-3 text-muted-foreground shrink-0" />
   435→                            </a>
   436→                          ))}
   437→                        </div>
   438→                      </div>
   439→                    )}
   440→                  </div>
   441→                </ScrollArea>
   442→              </TabsContent>
   443→
   444→              {/* Full Text Tab */}
   445→              <TabsContent value="fulltext" className="flex-1 min-h-0 m-0 mt-2 px-6 pb-6 flex flex-col">
   446→                <div className="relative mb-2 shrink-0">
   447→                  <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 h-3.5 w-3.5 text-muted-foreground" />
   448→                  <Input
   449→                    placeholder="Search in full text…"
   450→                    value={searchQuery}
   451→                    onChange={(e) => setSearchQuery(e.target.value)}
   452→                    className="h-8 text-sm pl-8 pr-8"
   453→                  />
   454→                  {searchQuery && (
   455→                    <button
   456→                      className="absolute right-2 top-1/2 -translate-y-1/2"
   457→                      onClick={() => setSearchQuery('')}
   458→                    >
   459→                      <X className="h-3.5 w-3.5 text-muted-foreground hover:text-foreground" />
   460→                    </button>
   461→                  )}
   462→                </div>
   463→                <ScrollArea className="flex-1 min-h-0">
   464→                  <div className="rounded-lg border bg-muted/20 p-4">
   465→                    <pre className="text-xs leading-relaxed whitespace-pre-wrap font-mono break-words">
   466→                      {searchQuery ? (
   467→                        <HighlightedText text={filteredContent} query={searchQuery} />
   468→                      ) : (
   469→                        data.content
   470→                      )}
   471→                    </pre>
   472→                  </div>
   473→                </ScrollArea>
   474→              </TabsContent>
   475→
   476→              {/* Linked Documents Tab */}
   477→              <TabsContent value="documents" className="flex-1 min-h-0 m-0 mt-2 px-6 pb-6">
   478→                <ScrollArea className="h-full">
   479→                  <div className="space-y-3 pr-3">
   480→                    {data.documentLinks.length > 0 ? (
   481→                      <>
   482→                        <h4 className="text-xs font-semibold text-muted-foreground uppercase tracking-wide flex items-center gap-1.5">
   483→                          <Download className="h-3.5 w-3.5" />
   484→                          Document Files ({data.documentLinks.length})
   485→                        </h4>
   486→                        <div className="grid grid-cols-1 sm:grid-cols-2 gap-2">
   487→                          {data.documentLinks.map((doc, i) => (
   488→                            <Card key={i} className="border-border/60 hover:border-emerald-300/60 dark:hover:border-emerald-700/40 transition-colors">
   489→                              <CardContent className="p-3 flex items-center gap-3">
   490→                                <div className="shrink-0 p-2 rounded-lg bg-muted/50">
   491→                                  {docIcon(doc.type)}
   492→                                </div>
   493→                                <div className="min-w-0 flex-1">
   494→                                  <p className="text-sm font-medium truncate">{doc.label}</p>
   495→                                  <div className="flex items-center gap-1.5 mt-0.5">

   563→                      </Card>

   565→                  </div>
   566→                </ScrollArea>

   580→        </div>

   586→// ── Highlighted text sub-component ────────────────────────────────────
   587→function HighlightedText({ text, query }: { text: string; query: string }) {
   588→  if (!query) return <>{text}</>;
   589→  const parts: React.ReactNode[] = [];
   590→  const lowerText = text.toLowerCase();
   591→  const lowerQuery = query.toLowerCase();
   592→  let lastIndex = 0;
   593→  let idx = lowerText.indexOf(lowerQuery, lastIndex);
   594→
   595→  let key = 0;
   596→  while (idx !== -1) {
   597→    if (idx > lastIndex) {
   598→      parts.push(<span key={key++}>{text.slice(lastIndex, idx)}</span>);
   599→    }
   600→    parts.push(
   601→      <mark key={key++} className="bg-emerald-200/70 dark:bg-emerald-800/50 rounded px-0.5">
   602→        {text.slice(idx, idx + query.length)}
   603→      </mark>,
   604→    );
   605→    lastIndex = idx + query.length;
   606→    idx = lowerText.indexOf(lowerQuery, lastIndex);
   607→  }
   608→  if (lastIndex < text.length) {
   609→    parts.push(<span key={key++}>{text.slice(lastIndex)}</span>);
   610→  }
   611→  return <>{parts}</>;
   612→}
