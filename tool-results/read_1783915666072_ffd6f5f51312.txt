    41→export function BidsView() {
    42→  const { user } = useAuthStore();
    43→  const { setView } = useNavStore();
    44→  const [bids, setBids] = useState<Bid[]>([]);
    45→  const [loading, setLoading] = useState(true);
    46→  const [statusFilter, setStatusFilter] = useState('');
    47→  const [expandedId, setExpandedId] = useState<string | null>(null);
    48→  const [activeTab, setActiveTab] = useState<BidTab>('all');
    49→  const [stampSelectorOpen, setStampSelectorOpen] = useState(false);
    50→  const [selectedBidId, setSelectedBidId] = useState<string | null>(null);
    51→  const [bidSignatures, setBidSignatures] = useState<Record<string, SavedSignature>>({});
    52→  const stampSigHook = useStampSignature();
    53→  const [visibleCount, setVisibleCount] = useState(10);
    54→  const BID_PAGE_SIZE = 10;
    55→
    56→  // Document/OCR/AI state
    57→  const [ocrLoading, setOcrLoading] = useState<Set<string>>(new Set());
    58→  const [reviewLoading, setReviewLoading] = useState<Set<string>>(new Set());
    59→  const [docOcrText, setDocOcrText] = useState<Record<string, string>>({});
    60→  const [docReview, setDocReview] = useState<Record<string, AIReviewData>>({});
    61→  const [expandedDocId, setExpandedDocId] = useState<string | null>(null);
    62→  const [reviewDialogOpen, setReviewDialogOpen] = useState(false);
    63→  const [reviewDialogDocId, setReviewDialogDocId] = useState<string | null>(null);
    64→  const [docUploadBidId, setDocUploadBidId] = useState<string | null>(null);
    65→  const docFileRef = useRef<HTMLInputElement>(null);
    66→  const [docUploadType, setDocUploadType] = useState('bid_attachment');
    67→  const [extractPrompt, setExtractPrompt] = useState<Record<string, string>>({});
    68→  const [extractLoading, setExtractLoading] = useState<Set<string>>(new Set());
    69→  const [extractResults, setExtractResults] = useState<Record<string, string>>({});
    70→  const [showExtract, setShowExtract] = useState<Set<string>>(new Set());
    71→  const [submitUrlForBid, setSubmitUrlForBid] = useState<Record<string, string>>({});
    72→  const [reviewPromptForBid, setReviewPromptForBid] = useState<Record<string, string>>({});
    73→  const fileInputRefs = useRef<Record<string, HTMLInputElement | null>>({});
    74→
    75→  // Saved tenders state
    76→  const [savedTenders, setSavedTenders] = useState<any[]>([]);
    77→  const [savedLoading, setSavedLoading] = useState(false);
    78→
    79→  const loadSavedTenders = useCallback(async () => {
    80→    setSavedLoading(true);
    81→    const res = await api.get('/tenders/saved');
    82→    if (res.success) setSavedTenders(res.data);
    83→    setSavedLoading(false);
    84→  }, []);
    85→
    86→  const handleRemoveSaved = async (id: string) => {
    87→    const res = await api.delete(`/tenders/saved/${id}`);
    88→    if (res.success) {
    89→      toast.success('Removed from saved tenders');
    90→      loadSavedTenders();
    91→    } else {
    92→      toast.error('Failed to remove');
    93→    }
    94→  };
    95→
    96→  const loadBids = useCallback(async () => {
    97→    setLoading(true);
    98→    const params: Record<string, string> = {};
    99→    if (statusFilter && statusFilter !== 'all') params.status = statusFilter;
   100→    const res = await api.get('/bids', params);
   101→    if (res.success) setBids(res.data);
   102→    setLoading(false);
   103→  }, [statusFilter]);
   104→
   105→  useEffect(() => { loadBids(); loadSavedTenders(); }, [loadBids, loadSavedTenders]);
   106→
   107→  const handleStatusUpdate = async (bidId: string, status: string) => {
   108→    const res = await api.patch(`/bids/${bidId}/status`, { status });
   109→    if (res.success) {
   110→      toast.success(`Bid ${status}`);
   111→      loadBids();
   112→    } else {
   113→      toast.error(res.error || 'Failed to update bid');
   114→    }
   115→  };
   116→
   117→  // ── Document Upload (with auto-OCR + auto-AI Review) ──
   118→  const handleDocUpload = useCallback(async (bidId: string, docType?: string, fileInputRef?: React.RefObject<HTMLInputElement | null>) => {
   119→    const ref = fileInputRef || docFileRef;
   120→    const file = ref.current?.files?.[0];
   121→    if (!file) return;
   122→    const type = docType || docUploadType;
   123→    const formData = new FormData();
   124→    formData.append('file', file);
   125→    formData.append('docType', type);
   126→    formData.append('autoOcr', 'true');
   127→    formData.append('autoReview', 'true');
   128→    formData.append('submitUrl', submitUrlForBid[bidId] || '');
   129→    formData.append('reviewPrompt', reviewPromptForBid[bidId] || '');
   130→    try {
   131→      const res = await api.upload(`/bids/${bidId}/documents`, formData);
   132→      if (res.success) {
   133→        toast.success('Document uploaded — OCR & AI Review started automatically');
   134→        if (ref.current) ref.current.value = '';
   135→        loadBids();
   136→        // Start polling for the newly uploaded document
   137→        const newDocId = res.data?.id;
   138→        if (newDocId) {
   139→          setOcrLoading(prev => new Set(prev).add(newDocId));
   140→          pollOcrThenReview(newDocId);
   141→        }
   142→      } else {
   143→        toast.error(res.error || 'Upload failed');
   144→      }
   145→    } catch {
   146→      toast.error('Upload failed');
   147→    } finally {
   148→      setDocUploadBidId(null);
   149→    }
   150→  }, [docUploadType, loadBids, submitUrlForBid, reviewPromptForBid]);
   151→
   152→  // ── Ref for AI Review handler (avoids circular dep with pollOcrThenReview) ──
   153→  const runReviewRef = useRef<(docId: string) => void>(() => {});
   154→
   155→  // ── AI Review Processing ──
   156→  const handleRunReview = useCallback(async (docId: string) => {
   157→    setReviewLoading(prev => new Set(prev).add(docId));
   158→    try {
   159→      const res = await api.post(`/document-review/${docId}`);
   160→      if (!res.success && res.error?.includes('OCR must be completed')) {
   161→        setReviewLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   162→        toast.error('Run OCR first before AI review');
   163→        return;
   164→      }
   165→      const poll = async (attempts = 0): Promise<void> => {
   166→        if (attempts > 60) {
   167→          setReviewLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   168→          toast.error('AI Review processing timed out');
   169→          return;
   170→        }
   171→        await new Promise(r => setTimeout(r, 2000));
   172→        const pollRes = await api.get(`/document-review/${docId}`);
   173→        if (pollRes.success && pollRes.data?.aiReviewStatus === 'completed') {
   174→          setReviewLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   175→          let reviewData = pollRes.data.aiReview || {};
   176→          if (typeof reviewData === 'string') {
   177→            try { reviewData = JSON.parse(reviewData); } catch { reviewData = { summary: reviewData }; }
   178→          }
   179→          setDocReview(prev => ({ ...prev, [docId]: reviewData as AIReviewData }));
   180→          loadBids();
   181→          toast.success('AI Review completed — Document fully processed');
   182→        } else if (pollRes.success && pollRes.data?.aiReviewStatus === 'failed') {
   183→          setReviewLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   184→          toast.error('AI Review processing failed');
   185→          loadBids();
   186→        } else {
   187→          await poll(attempts + 1);
   188→        }
   189→      };
   190→      poll();
   191→    } catch {
   192→      setReviewLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   193→      toast.error('Failed to start AI Review');
   194→    }
   195→  }, [loadBids]);
   196→
   197→  // Keep ref updated
   198→  runReviewRef.current = handleRunReview;
   199→
   200→  // ── Poll OCR then auto-trigger AI Review ──
   201→  const pollOcrThenReview = useCallback(async (docId: string) => {
   202→    const poll = async (attempts = 0): Promise<void> => {
   203→      if (attempts > 60) {
   204→        setOcrLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   205→        toast.error('OCR processing timed out');
   206→        return;
   207→      }
   208→      await new Promise(r => setTimeout(r, 2000));
   209→      const res = await api.get(`/document-ocr/${docId}`);
   210→      if (res.success && res.data?.ocrStatus === 'completed') {
   211→        setOcrLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   212→        setDocOcrText(prev => ({ ...prev, [docId]: res.data.ocrText || '' }));
   213→        loadBids();
   214→        toast.success('OCR completed — Starting AI Review...');
   215→        // Auto-chain: trigger AI Review after OCR completes
   216→        runReviewRef.current(docId);
   217→      } else if (res.success && res.data?.ocrStatus === 'failed') {
   218→        setOcrLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   219→        toast.error('OCR processing failed');
   220→        loadBids();
   221→      } else {
   222→        await poll(attempts + 1);
   223→      }
   224→    };
   225→    poll();
   226→  }, [loadBids]);
   227→
   228→  // ── Drag & Drop Upload ──
   229→  const handleDocDrop = useCallback(async (bidId: string, files: FileList | null, docType?: string) => {
   230→    if (!files || files.length === 0) return;
   231→    const file = files[0];
   232→    const ext = file.name.split('.').pop()?.toLowerCase() || '';
   233→    const allowed = ['pdf', 'jpg', 'jpeg', 'png', 'docx', 'doc', 'txt'];
   234→    if (!allowed.includes(ext)) {
   235→      toast.error('Invalid file type. Allowed: PDF, JPEG, PNG, DOCX, DOC, TXT');
   236→      return;
   237→    }
   238→    if (file.size > 10 * 1024 * 1024) {
   239→      toast.error('File too large. Maximum 10MB.');
   240→      return;
   241→    }
   242→    const formData = new FormData();
   243→    formData.append('file', file);
   244→    formData.append('docType', docType || docUploadType);
   245→    formData.append('autoOcr', 'true');
   246→    formData.append('autoReview', 'true');
   247→    formData.append('submitUrl', submitUrlForBid[bidId] || '');
   248→    formData.append('reviewPrompt', reviewPromptForBid[bidId] || '');
   249→    try {
   250→      const res = await api.upload(`/bids/${bidId}/documents`, formData);
   251→      if (res.success) {
   252→        toast.success('Document dropped — OCR & AI Review started');
   253→        loadBids();
   254→        const newDocId = res.data?.id;
   255→        if (newDocId) {
   256→          setOcrLoading(prev => new Set(prev).add(newDocId));
   257→          pollOcrThenReview(newDocId);
   258→        }
   259→      } else {
   260→        toast.error(res.error || 'Upload failed');
   261→      }
   262→    } catch {
   263→      toast.error('Upload failed');
   264→    }
   265→  }, [docUploadType, loadBids, pollOcrThenReview, submitUrlForBid, reviewPromptForBid]);
   266→
   267→  // ── OCR Processing (manual trigger) ──
   268→  const handleRunOcr = useCallback(async (docId: string) => {
   269→    setOcrLoading(prev => new Set(prev).add(docId));
   270→    try {
   271→      await api.post(`/document-ocr/${docId}`);
   272→      pollOcrThenReview(docId);
   273→    } catch {
   274→      setOcrLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   275→      toast.error('Failed to start OCR');
   276→    }
   277→  }, [pollOcrThenReview]);
   278→
   279→  // ── View OCR text inline ──
   280→  const handleViewOcr = useCallback(async (docId: string) => {
   281→    if (expandedDocId === docId) {
   282→      setExpandedDocId(null);
   283→      return;
   284→    }
   285→    setExpandedDocId(docId);
   286→    if (!docOcrText[docId]) {
   287→      const res = await api.get(`/document-ocr/${docId}`);
   288→      if (res.success) {
   289→        setDocOcrText(prev => ({ ...prev, [docId]: res.data?.ocrText || '' }));
   290→      }
   291→    }
   292→  }, [expandedDocId, docOcrText]);
   293→
   294→  // ── AI Extract: extract specific info from document via prompt ──
   295→  const handleAiExtract = useCallback(async (docId: string) => {
   296→    const prompt = extractPrompt[docId]?.trim();
   297→    if (!prompt) {
   298→      toast.error('Please enter a prompt describing what to extract');
   299→      return;
   300→    }
   301→    setExtractLoading(prev => new Set(prev).add(docId));
   302→    try {
   303→      const res = await api.post('/documents/ai-extract', { documentId: docId, prompt });
   304→      if (res.success) {
   305→        setExtractResults(prev => ({ ...prev, [docId]: res.data.extractedInfo }));
   306→        toast.success('Information extracted successfully');
   307→      } else {
   308→        toast.error(res.error || 'AI extraction failed');
   309→      }
   310→    } catch {
   311→      toast.error('AI extraction failed');
   312→    } finally {
   313→      setExtractLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   314→    }
   315→  }, [extractPrompt]);
   316→
   317→  const isDrafted = (b: Bid) => {
   318→    if (b.status !== 'pending_review') return false;
   319→    return (
   320→      !b.financialProposal ||
   321→      b.financialProposal === 0 ||
   322→      b.technicalProposal?.includes('Pending document upload') ||
   323→      b.timeline?.includes('Pending document upload')
   324→    );
   325→  };
   326→
   327→  const stats = useMemo(() => ({
   328→    pending: bids.filter(b => b.status === 'pending_review').length,
   329→    drafted: bids.filter(b => isDrafted(b)).length,
   330→    shortlisted: bids.filter(b => b.status === 'shortlisted').length,
   331→    awarded: bids.filter(b => b.status === 'awarded').length,
   332→    rejected: bids.filter(b => b.status === 'rejected').length,
   333→    total: bids.length,
   334→  }), [bids]);
   335→
   336→  const filteredBids = useMemo(() => {
   337→    if (activeTab === 'saved') return [] as Bid[];
   338→    if (activeTab === 'all') return bids;
   339→    if (activeTab === 'drafted') return bids.filter(b => isDrafted(b));
   340→    return bids.filter(b => b.status === activeTab);
   341→  }, [bids, activeTab]);
   342→
   343→  const statusBadge = (status: string) => {
   344→    switch (status) {
   345→      case 'pending_review': return 'bg-amber-100 text-amber-700 hover:bg-amber-100';
   346→      case 'shortlisted': return 'bg-teal-100 text-teal-700 hover:bg-teal-100';
   347→      case 'awarded': return 'bg-emerald-100 text-emerald-700 hover:bg-emerald-100';
   348→      case 'rejected': return 'bg-rose-100 text-rose-700 hover:bg-rose-100';
   349→      default: return 'bg-muted text-muted-foreground hover:bg-muted';
   350→    }
   351→  };
   352→
   353→  const statusIcon = (status: string) => {
   354→    switch (status) {
   355→      case 'pending_review': return { icon: Clock, bg: 'bg-amber-50', color: 'text-amber-600' };
   356→      case 'shortlisted': return { icon: Award, bg: 'bg-teal-50', color: 'text-teal-600' };
   357→      case 'awarded': return { icon: CheckCircle, bg: 'bg-emerald-50', color: 'text-emerald-600' };
   358→      case 'rejected': return { icon: AlertCircle, bg: 'bg-rose-50', color: 'text-rose-600' };
   359→      default: return { icon: Gavel, bg: 'bg-muted/50', color: 'text-muted-foreground' };
   360→    }
   361→  };
   362→
   363→  const isAdminOrOwner = user?.role === 'team_admin';
   364→
   365→  const getScoreColor = (score: number) => {
   366→    if (score >= 80) return 'text-emerald-600';
   367→    if (score >= 60) return 'text-amber-600';
   368→    return 'text-rose-600';
   369→  };
   370→
   371→  const getScoreBg = (score: number) => {
   372→    if (score >= 80) return 'bg-emerald-50';
   373→    if (score >= 60) return 'bg-amber-50';
   374→    return 'bg-rose-50';
   375→  };
   376→
   377→  const getRiskColor = (risk: string) => {
   378→    switch (risk?.toLowerCase()) {
   379→      case 'low': return 'bg-emerald-100 text-emerald-700';
   380→      case 'medium': return 'bg-amber-100 text-amber-700';
   381→      case 'high': case 'critical': return 'bg-rose-100 text-rose-700';
   382→      default: return 'bg-gray-100 text-gray-700';
   383→    }
   384→  };
   385→
   386→  const tabs: { key: BidTab; label: string; icon: typeof Clock; count: number; color: string }[] = [
   387→    { key: 'all', label: 'All', icon: Gavel, count: stats.total, color: 'emerald' },
   388→    { key: 'pending_review', label: 'Pending', icon: Clock, count: stats.pending, color: 'amber' },
   389→    { key: 'drafted', label: 'Drafted', icon: PenLine, count: stats.drafted, color: 'sky' },
   390→    { key: 'shortlisted', label: 'Shortlisted', icon: Award, count: stats.shortlisted, color: 'teal' },
   391→    { key: 'awarded', label: 'Awarded', icon: CheckCircle, count: stats.awarded, color: 'emerald' },
   392→    { key: 'rejected', label: 'Rejected', icon: AlertCircle, count: stats.rejected, color: 'rose' },
   393→    { key: 'saved', label: 'Saved Tenders', icon: Bookmark, count: savedTenders.length, color: 'violet' },
   394→  ];
   395→
   396→  // Review dialog data
   397→  const reviewDialogData = reviewDialogDocId ? docReview[reviewDialogDocId] : null;
   398→
   399→  return (
   400→    <div className="p-4 md:p-6 space-y-6 max-w-5xl mx-auto view-enter">
   401→      {/* Header */}
   402→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4 animate-[fadeIn_0.3s_ease-out]">
   403→        <div className="flex items-center gap-4">
   404→          <div className="p-3 rounded-2xl gradient-amber shadow-md flex-shrink-0">
   405→            <Gavel className="h-6 w-6 text-white" />
   406→          </div>
   407→          <div>
   408→            <h2 className="text-2xl md:text-3xl font-bold tracking-tight">
   409→              <span className="text-gradient-emerald">{isAdminOrOwner ? 'Review' : 'My'}</span> Bids
   410→            </h2>
   411→            <p className="text-muted-foreground text-sm mt-0.5">
   412→              {isAdminOrOwner ? 'Manage and evaluate submitted bids' : 'Track your bid submissions'}
   413→            </p>
   414→          </div>
   415→        </div>
   416→      </div>
   417→
   418→      {/* Stats Summary */}
   419→      {!loading && (bids.length > 0 || savedTenders.length > 0) && (
   420→        <div className="grid grid-cols-2 sm:grid-cols-4 gap-4 animate-[fadeIn_0.3s_ease-out]">
   421→          {[
   422→            { label: 'Pending', count: stats.pending, icon: Clock, bg: 'bg-amber-50', color: 'text-amber-600' },
   423→            { label: 'Drafted', count: stats.drafted, icon: PenLine, bg: 'bg-sky-50', color: 'text-sky-600' },
   424→            { label: 'Shortlisted', count: stats.shortlisted, icon: Award, bg: 'bg-teal-50', color: 'text-teal-600' },
   425→            { label: 'Awarded', count: stats.awarded, icon: CheckCircle, bg: 'bg-emerald-50', color: 'text-emerald-600' },
   426→          ].map(stat => (
   427→            <div key={stat.label}>
   428→              <Card className="premium-shadow rounded-xl border-0 bg-card hover:-translate-y-0.5 transition-all duration-200">
   429→                <CardContent className="p-4 flex items-center gap-3">
   430→                  <div className={`p-2 rounded-lg ${stat.bg} flex-shrink-0`}>
   431→                    <stat.icon className={`h-4 w-4 ${stat.color}`} />
   432→                  </div>
   433→                  <div>
   434→                    <p className="text-xl font-bold">{stat.count}</p>
   435→                    <p className="text-xs text-muted-foreground">{stat.label}</p>
   436→                  </div>
   437→                </CardContent>
   438→              </Card>
   439→            </div>
   440→          ))}
   441→        </div>
   442→      )}
   443→
   444→      {/* Tab Navigation */}
   445→      {!loading && (
   446→        <div className="animate-[fadeIn_0.3s_ease-out]">
   447→          <Card className="premium-shadow rounded-xl border-0 bg-card">
   448→            <CardContent className="p-1.5">
   449→              <div className="flex gap-1 overflow-x-auto">
   450→                {tabs.map(tab => (
   451→                  <button
   452→                    key={tab.key}
   453→                    onClick={() => setActiveTab(tab.key)}
   454→                    className={`flex items-center gap-1.5 px-3 py-2 rounded-lg text-sm font-medium transition-all duration-200 whitespace-nowrap ${
