   176→
   177→function fmtMoney(amount: number, currency: string): string {
   178→  if (!amount) return '—';
   179→  try {
   180→    return new Intl.NumberFormat('en-US', {
   181→      style: 'currency',
   182→      currency: currency || 'USD',
   183→      maximumFractionDigits: 0,
   184→    }).format(amount);
   185→  } catch {
   186→    return `${currency} ${amount.toLocaleString()}`;
   187→  }
   188→}
   189→
   190→function daysUntil(dateStr: string): number {
   191→  return Math.ceil((new Date(dateStr).getTime() - Date.now()) / (1000 * 60 * 60 * 24));
   192→}
   193→
   194→function deadlineBadge(days: number): string {
   195→  if (days <= 0) return 'bg-rose-50 text-rose-700 dark:bg-rose-950/40 dark:text-rose-300';
   196→  if (days <= 7) return 'bg-amber-50 text-amber-700 dark:bg-amber-950/40 dark:text-amber-300';
   197→  if (days <= 21) return 'bg-sky-50 text-sky-700 dark:bg-sky-950/40 dark:text-sky-300';
   198→  return 'bg-emerald-50 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300';
   199→}
   200→
   201→function relativeTime(dateStr: string): string {
   202→  const diff = Date.now() - new Date(dateStr).getTime();
   203→  const day = 86400000;
   204→  if (diff < day) return 'today';
   205→  if (diff < 2 * day) return 'yesterday';
   206→  if (diff < 30 * day) return `${Math.floor(diff / day)}d ago`;
   207→  return new Date(dateStr).toLocaleDateString();
   208→}

   211→ * Inline Document Viewer Component

   213→
   214→function InlineDocumentViewer({ doc, onClose }: { doc: InlineDocument; onClose: () => void }) {
   215→  const scrollRef = useRef<HTMLDivElement>(null);
   216→
   217→  const copyContent = () => {
   218→    const text = doc.sections
   219→      ? doc.sections.map((s) => `${s.heading}\n${s.content}`).join('\n\n')
   220→      : doc.content;
   221→    navigator.clipboard.writeText(text);
   222→    toast.success('Content copied to clipboard');
   223→  };
   224→
   225→  return (
   226→    <div
   227→ className="overflow-hidden animate-[fadeIn_0.3s_ease-out]"
   228→ >
   229→      <div className="border-t border-border bg-gradient-to-b from-muted/30 to-background">
   230→        <div className="p-4 md:p-6 space-y-4">
   231→          {/* Document header */}
   232→          <div className="flex items-start justify-between gap-3">
   233→            <div className="min-w-0 flex-1">
   234→              <div className="flex items-center gap-2 text-xs text-muted-foreground mb-1">
   235→                <BookOpen className="h-3.5 w-3.5" />
   236→                <span>Document loaded from {new URL(doc.url).hostname}</span>
   237→                <span>·</span>
   238→                <span>{new Date(doc.fetchedAt).toLocaleTimeString()}</span>
   239→              </div>
   240→              <h3 className="text-lg font-semibold text-foreground leading-snug">
   241→                {doc.title || 'Tender Document'}
   242→              </h3>
   243→              {doc.metaDescription && (
   244→                <p className="text-sm text-muted-foreground mt-1 line-clamp-2">
   245→                  {doc.metaDescription}
   246→                </p>
   247→              )}
   248→            </div>
   249→            <div className="flex items-center gap-2 shrink-0">
   250→              <Button variant="ghost" size="sm" className="gap-1.5 text-xs" onClick={copyContent}>
   251→                <Copy className="h-3.5 w-3.5" />
   252→                Copy
   253→              </Button>
   254→              <a
   255→                href={doc.url}
   256→                target="_blank"
   257→                rel="noopener noreferrer"
   258→                className="inline-flex items-center gap-1.5 text-xs font-medium text-primary hover:underline"
   259→              >
   260→                <ExternalLink className="h-3.5 w-3.5" />
   261→                Open original
   262→              </a>
   263→              <Button variant="ghost" size="sm" onClick={onClose} className="gap-1.5 text-xs text-muted-foreground">
   264→                <ChevronUp className="h-3.5 w-3.5" />
   265→                Collapse
   266→              </Button>
   267→            </div>
   268→          </div>
   269→
   270→          {/* Extracted metadata pills */}
   271→          {(doc.deadlines && doc.deadlines.length > 0) || (doc.budgets && doc.budgets.length > 0) ? (
   272→            <div className="flex flex-wrap gap-2">
   273→              {doc.deadlines?.map((d, i) => (
   274→                <Badge key={`dl-${i}`} variant="outline" className="gap-1.5 text-xs border-amber-300 text-amber-700 dark:border-amber-800 dark:text-amber-300">
   275→                  <Calendar className="h-3 w-3" />
   276→                  {d}
   277→                </Badge>
   278→              ))}
   279→              {doc.budgets?.map((b, i) => (
   280→                <Badge key={`bg-${i}`} variant="outline" className="gap-1.5 text-xs border-emerald-300 text-emerald-700 dark:border-emerald-800 dark:text-emerald-300">
   281→                  <DollarSign className="h-3 w-3" />
   282→                  {b}
   283→                </Badge>
   284→              ))}
   285→            </div>
   286→          ) : null}
   287→
   288→          {/* Sections view */}
   289→          {doc.sections && doc.sections.length > 0 ? (
   290→            <div ref={scrollRef} className="max-h-80 overflow-y-auto rounded-lg border border-border bg-background p-4 space-y-4 scrollbar-thin">
   291→              {doc.sections.map((section, i) => (
   292→                <div key={i}>
   293→                  <h4 className="text-sm font-semibold text-foreground mb-1.5">{section.heading}</h4>
   294→                  <p className="text-sm text-muted-foreground leading-relaxed whitespace-pre-line">
   295→                    {section.content}
   296→                  </p>
   297→                  {i < doc.sections!.length - 1 && <Separator className="mt-4 bg-border" />}
   298→                </div>
   299→              ))}
   300→            </div>
   301→          ) : (
   302→            <div ref={scrollRef} className="max-h-80 overflow-y-auto rounded-lg border border-border bg-background p-4 scrollbar-thin">
   303→              <p className="text-sm text-muted-foreground leading-relaxed whitespace-pre-line">
   304→                {doc.content || 'No content could be extracted from this page.'}
   305→              </p>
   306→            </div>
   307→          )}
   308→        </div>
   309→      </div>
   310→    </div>
   311→  );
   312→}

   314→/* ─────────────────────────────────────────────────────────────────────
   315→ * Main LiveTendersView component — Notion-style
   316→ * ───────────────────────────────────────────────────────────────────── */
   317→
   318→export function LiveTendersView() {
   319→  const [tenders, setTenders] = useState<LiveTender[]>([]);
   320→  const [dataSources, setDataSources] = useState<DataSource[]>([]);
   321→  const [sourceMeta, setSourceMeta] = useState<
   322→    { id: string; name: string; live: boolean; ok: boolean; count: number; error?: string }[]
   323→  >([]);
   324→  const [loading, setLoading] = useState(true);
   325→  const [refreshing, setRefreshing] = useState(false);
   326→  const [fallback, setFallback] = useState(false);
   327→  const [search, setSearch] = useState('');
   328→  const [sourceFilter, setSourceFilter] = useState('all');
   329→  const [sectorFilter, setSectorFilter] = useState('');
   330→  const [sectorCounts, setSectorCounts] = useState<{ id: string; label: string; count: number }[]>([]);
   331→
   332→  // Inline document state
   333→  const [expandedId, setExpandedId] = useState<string | null>(null);
   334→  const [docLoading, setDocLoading] = useState<string | null>(null);
   335→  const [docData, setDocData] = useState<Record<string, InlineDocument>>({});
   336→  const [docError, setDocError] = useState<Record<string, string>>({});
   337→
   338→  // Import state
   339→  const [importing, setImporting] = useState<string | null>(null);
   340→
   341→  const load = useCallback(
   342→    async (isRefresh = false) => {
   343→      if (isRefresh) setRefreshing(true);
   344→      else setLoading(true);
   345→      const params: Record<string, string> = { rows: '50' };
   346→      if (search) params.search = search;
   347→      if (sourceFilter && sourceFilter !== 'all') params.source = sourceFilter;
   348→      if (sectorFilter) params.sector = sectorFilter;
   349→      const res = await api.get('/tenders/live', params);
   350→      if (res.success) {
   351→        setTenders(res.data as LiveTender[]);
   352→        setSourceMeta(res.meta?.sources || []);
   353→        setFallback(Boolean(res.meta?.fallback));
   354→        if (Array.isArray(res.meta?.dataSources)) setDataSources(res.meta.dataSources);
   355→        if (Array.isArray(res.meta?.sectors)) setSectorCounts(res.meta.sectors);
   356→      } else {
   357→        toast.error(res.error || 'Failed to load live tenders');
   358→      }
   359→      setLoading(false);
   360→      setRefreshing(false);
   361→    },
   362→    [search, sourceFilter, sectorFilter],
   363→  );
   364→
   365→  useEffect(() => {
   366→    const t = setTimeout(() => load(), 250);
   367→    return () => clearTimeout(t);
   368→  }, [load]);
   369→
   370→  const liveSourcesCount = useMemo(
   371→    () => sourceMeta.filter((s) => s.live && s.ok).length,
   372→    [sourceMeta],
   373→  );
   374→
   375→  const totalTenderValue = useMemo(() => {
   376→    return tenders.reduce((sum, t) => sum + (t.budgetMax || 0), 0);
   377→  }, [tenders]);
   378→
   379→  /* ────── Inline document loading ────── */
   380→  const loadDocument = useCallback(async (tender: LiveTender) => {
   381→    const id = tender.id;
   382→    if (expandedId === id) {
   383→      setExpandedId(null);
   384→      return;
   385→    }
   386→
   387→    // Already fetched?
   388→    if (docData[id]) {
   389→      setExpandedId(id);
   390→      return;
   391→    }
   392→
   393→    setExpandedId(id);
   394→    setDocLoading(id);
   395→    setDocError((prev) => { const n = { ...prev }; delete n[id]; return n; });
   396→
   397→    try {
   398→      const res = await api.get(`/tenders/${encodeURIComponent(id)}/documents`, { url: tender.externalUrl });
   399→      if (res.success && res.data) {
   400→        setDocData((prev) => ({ ...prev, [id]: res.data as InlineDocument }));
   401→      } else {
   402→        setDocError((prev) => ({ ...prev, [id]: res.error || 'Failed to load document' }));
   403→      }
   404→    } catch {
   405→      setDocError((prev) => ({ ...prev, [id]: 'Network error' }));
   406→    }
   407→    setDocLoading(null);
   408→  }, [expandedId, docData]);
   409→
   410→  /* ────── Import to local tenders ────── */
   411→  const importTender = useCallback(async (tender: LiveTender) => {
   412→    setImporting(tender.id);
   413→    try {
   414→      const res = await api.post('/tenders', {
   415→        title: tender.title,
   416→        scope: tender.scope,
   417→        budgetMin: tender.budgetMin,
   418→        budgetMax: tender.budgetMax,
   419→        deadline: tender.deadline,
   420→        location: tender.location,
   421→        categoryTags: tender.categoryTags,
   422→        requiredDocs: `Source: ${SOURCE_LABELS[tender.source] || tender.source} | External ID: ${tender.externalId} | URL: ${tender.externalUrl}`,
   423→        status: 'open',
   424→        currency: tender.currency,
   425→      });
   426→      if (res.success) {
   427→        toast.success('Tender imported successfully', {
   428→          description: `"${tender.title}" is now in your tenders list.`,
   429→        });
   430→      } else {
   431→        toast.error(res.error || 'Failed to import tender');
   432→      }
   433→    } catch {
   434→      toast.error('Failed to import tender');
   435→    }
   436→    setImporting(null);
   437→  }, []);
   438→
   439→  /* ────── Render ────── */
   440→  return (
   441→    <div className="view-enter">
   442→      {/* ───────────────── Notion-style Cover ───────────────── */}
   443→      <div className="relative h-32 md:h-40 bg-gradient-to-br from-emerald-600 via-teal-700 to-cyan-800 overflow-hidden">
   444→        <div className="absolute inset-0 bg-[url('data:image/svg+xml;base64,PHN2ZyB3aWR0aD0iNDAiIGhlaWdodD0iNDAiIHZpZXdCb3g9IjAgMCA0MCA0MCIgeG1sbnM9Imh0dHA6Ly93d3cudzMub3JnLzIwMDAvc3ZnIj48Y2lyY2xlIGN4PSIyMCIgY3k9IjIwIiByPSIxIiBmaWxsPSJyZ2JhKDI1NSwyNTUsMjU1LDAuMDUpIi8+PC9zdmc+')] opacity-60" />
   445→        <div className="absolute bottom-0 left-0 right-0 h-16 bg-gradient-to-t from-background to-transparent" />
   446→        <div className="absolute top-4 right-4 md:right-6">
   447→          <Button
   448→            variant="secondary"
   449→            size="sm"
   450→            onClick={() => load(true)}
   451→            disabled={refreshing}
   452→            className="gap-1.5 shadow-lg bg-white/90 dark:bg-gray-800/90 backdrop-blur-sm"
   453→          >
   454→            <RefreshCw className={`h-3.5 w-3.5 ${refreshing ? 'animate-spin' : ''}`} />
   455→            {refreshing ? 'Refreshing…' : 'Refresh'}
   456→          </Button>
   457→        </div>
   458→      </div>
   459→
   460→      <div className="px-4 md:px-6 max-w-5xl mx-auto -mt-8 relative z-10 space-y-6">
   461→        {/* ───────────────── Notion-style Icon + Title ───────────────── */}
   462→        <div className="flex items-end gap-4">
   463→          <div className="h-14 w-14 md:h-16 md:w-16 rounded-xl bg-gradient-to-br from-emerald-500 to-teal-600 flex items-center justify-center shadow-lg shadow-emerald-500/20 border-4 border-background shrink-0">
