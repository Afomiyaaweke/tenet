    33→function parseBidAnalysis(analysis: BidAnalysis): BidAnalysisResult | null {
    34→  try {

    38→      budgetAnalysis: analysis.budgetAnalysis || '',
    39→      riskSummary: analysis.riskSummary || '',

    47→function ScoreBar({ score, label }: { score: number; label?: string }) {
    48→  const colorClass = score >= 80 ? 'bg-emerald-500' : score >= 60 ? 'bg-amber-500' : 'bg-red-500';
    49→  const textColor = score >= 80 ? 'text-emerald-700' : score >= 60 ? 'text-amber-700' : 'text-red-700';
    50→  return (
    51→    <div className="flex items-center gap-2 min-w-[120px]">
    52→      {label && <span className="text-[10px] text-muted-foreground w-12 shrink-0">{label}</span>}
    53→      <div className="flex-1 h-2 bg-muted/50 rounded-full overflow-hidden">
    54→        <div
    55→ className={`h-full rounded-full ${colorClass} transition-[width] duration-700`} style={{ width: `${Math.min(score, 100)}%` }}
    56→ />
    57→      </div>
    58→      <span className={`text-xs font-semibold ${textColor} w-8 text-right`}>{score}</span>
    59→    </div>
    60→  );
    61→}

    63→function RiskBadge({ level }: { level: 'low' | 'medium' | 'high' }) {
    64→  const config = {
    65→    low: { className: 'bg-emerald-100 text-emerald-700', icon: ShieldCheck },
    66→    medium: { className: 'bg-amber-100 text-amber-700', icon: ShieldQuestion },
    67→    high: { className: 'bg-red-100 text-red-700', icon: ShieldAlert },
    68→  };
    69→  const c = config[level];
    70→  const Icon = c.icon;
    71→  return (
    72→    <Badge className={`text-[10px] px-2 py-0.5 border-0 rounded-lg ${c.className}`}>
    73→      <Icon className="h-3 w-3 mr-1" />
    74→      {level.charAt(0).toUpperCase() + level.slice(1)}
    75→    </Badge>
    76→  );
    77→}
