   171→   171→function formatDate(dateStr: string): string {
   172→   172→  if (!dateStr) return '—';
   173→   173→  return new Date(dateStr).toLocaleDateString('en-GB', { day: '2-digit', month: 'short', year: 'numeric' });
   174→   174→}

   176→   176→function daysUntil(dateStr: string): number {
   177→   177→  return Math.ceil((new Date(dateStr).getTime() - Date.now()) / (1000 * 60 * 60 * 24));

   339→   339→          fetchData();
   340→   340→        } else {
   341→   341→          await poll(attempts + 1);
   342→   342→        }
   343→   343→      };
   344→   344→      poll();
   345→   345→    } catch {
   346→   346→      setOcrLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });

   359→   359→          return;

   372→   372→        } else if (res.success && res.data?.aiReviewStatus === 'failed') {
   373→   373→          setReviewLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   374→   374→          toast.error('AI Review processing failed');
   375→   375→          fetchData();
   376→   376→        } else {
   377→   377→          await poll(attempts + 1);
   378→   378→        }
   379→   379→      };
   380→   380→      poll();
   381→   381→    } catch {
   382→   382→      setReviewLoading(prev => { const s = new Set(prev); s.delete(docId); return s; });
   383→   383→      toast.error('Failed to start AI Review');
   384→   384→    }
   385→   385→  }, [fetchData]);
   386→   386→
   387→   387→  const handleViewDocDetail = useCallback(async (docId: string, type: 'ocr' | 'review') => {

   397→   397→        if (res.success) {
