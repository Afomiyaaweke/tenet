   443→   443→function relativeTime(dateStr: string): string {
   444→   444→  if (!dateStr) return '';
   445→   445→  const diff = Date.now() - new Date(dateStr).getTime();
   446→   446→  const day = 86400000;
   447→   447→  if (diff < day) return 'today';
   448→   448→  if (diff < 2 * day) return 'yesterday';
   449→   449→  if (diff < 30 * day) return `${Math.floor(diff / day)}d ago`;
   450→   450→  return new Date(dateStr).toLocaleDateString();
   451→   451→}
