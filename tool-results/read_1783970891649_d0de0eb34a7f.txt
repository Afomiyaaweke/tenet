   411→function fmtMoney(amount: number, currency: string): string {
   412→  if (!amount) return '—';
   413→  try {
   414→    return new Intl.NumberFormat('en-US', {
   415→      style: 'currency',
   416→      currency: currency || 'USD',
   417→      maximumFractionDigits: 0,
   418→    }).format(amount);
   419→  } catch {
   420→    return `${currency} ${amount.toLocaleString()}`;
   421→  }
   422→}
   423→
   424→function daysUntil(dateStr: string): number {
   425→  if (!dateStr) return 999;
   426→  return Math.ceil((new Date(dateStr).getTime() - Date.now()) / (1000 * 60 * 60 * 24));
   427→}
   428→
   429→function deadlineBadge(days: number): string {
   430→  if (days <= 0) return 'bg-rose-50 text-rose-700 dark:bg-rose-950/40 dark:text-rose-300';
   431→  if (days <= 7) return 'bg-amber-50 text-amber-700 dark:bg-amber-950/40 dark:text-amber-300';
   432→  if (days <= 21) return 'bg-sky-50 text-sky-700 dark:bg-sky-950/40 dark:text-sky-300';
   433→  return 'bg-emerald-50 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300';
   434→}
   435→
   436→function deadlineLabel(days: number): string {
   437→  if (days <= 0) return 'Closed';
   438→  if (days <= 7) return `${days}d left`;
   439→  if (days <= 30) return `${days}d left`;
   440→  return `${days}d left`;
   441→}
   442→
   443→function relativeTime(dateStr: string): string {
   444→  if (!dateStr) return '';
   445→  const diff = Date.now() - new Date(dateStr).getTime();
   446→  const day = 86400000;
   447→  if (diff < day) return 'today';
   448→  if (diff < 2 * day) return 'yesterday';
   449→  if (diff < 30 * day) return `${Math.floor(diff / day)}d ago`;
   450→  return new Date(dateStr).toLocaleDateString();
   451→}

   456→  return 'bg-rose-50 text-rose-700 dark:bg-rose-950/40 dark:text-rose-300';
   457→}

   465→/* ─────────────────────────────────────────────────────────────────────
   466→ * Inline Document Viewer Component
   467→ * ───────────────────────────────────────────────────────────────────── */
   468→
   469→function InlineDocumentViewer({ doc, onClose }: { doc: InlineDocument; onClose: () => void }) {
   470→  const copyContent = () => {
   471→    const text = doc.sections
   472→      ? doc.sections.map((s) => `${s.heading}\n${s.content}`).join('\n\n')
   473→      : doc.content;
   474→    navigator.clipboard.writeText(text);
   475→    toast.success('Content copied to clipboard');
   476→  };
   477→
   478→  return (
