     9→
    10→const DB_PATH = process.env.DATABASE_URL?.replace('file:', '') || join(dirname(fileURLToPath(import.meta.url)), '..', '..', 'db', 'custom.db');
    11→
    12→let db: Database;
    13→
    14→function getDb(): Database {
    15→  if (!db) {
    16→    if (!existsSync(DB_PATH)) {
    17→      console.error(`[InfraService] ❌ Database not found at: ${DB_PATH}`);
    18→      process.exit(1);
    19→    }
    20→    db = new Database(DB_PATH);
    21→    db.exec('PRAGMA journal_mode = WAL');
    22→    console.log(`[InfraService] ✅ Connected to database: ${DB_PATH}`);
    23→  }
    24→  return db;
    25→}

    31→interface UserStats {
    32→  totalUsers: number;
    33→  activeUsers: number;
    34→  suspendedUsers: number;
    35→  bannedUsers: number;
    36→  verifiedUsers: number;
    37→  unverifiedUsers: number;
    38→  newUsersToday: number;
    39→  newUsersThisWeek: number;
    40→  newUsersThisMonth: number;
    41→  usersWithCompany: number;
    42→  usersWithoutCompany: number;
    43→  teamAdmins: number;
    44→  regularUsers: number;
    45→  recentRegistrations: Array<{ id: string; email: string; role: string; status: string; createdAt: string; companyId: string | null }>;
    46→  dailyRegistrations: Array<{ date: string; count: number }>;
    47→  monthlyRegistrations: Array<{ month: string; count: number }>;
    48→  roleDistribution: Array<{ role: string; count: number }>;
    49→  statusDistribution: Array<{ status: string; count: number }>;
    50→}
    51→
    52→function getUserStats(): UserStats {
    53→  const d = getDb();
    54→
    55→  const totalUsers = ((d.prepare('SELECT COUNT(*) as c FROM User').get() as any)?.c) ?? 0;
    56→  const activeUsers = ((d.prepare("SELECT COUNT(*) as c FROM User WHERE status = 'active'").get() as any)?.c) ?? 0;
    57→  const suspendedUsers = ((d.prepare("SELECT COUNT(*) as c FROM User WHERE status = 'suspended'").get() as any)?.c) ?? 0;
    58→  const bannedUsers = ((d.prepare("SELECT COUNT(*) as c FROM User WHERE status = 'banned'").get() as any)?.c) ?? 0;
    59→  const verifiedUsers = ((d.prepare('SELECT COUNT(*) as c FROM User WHERE emailVerified = 1').get() as any)?.c) ?? 0;
    60→  const unverifiedUsers = ((d.prepare('SELECT COUNT(*) as c FROM User WHERE emailVerified = 0').get() as any)?.c) ?? 0;
    61→  const usersWithCompany = ((d.prepare('SELECT COUNT(*) as c FROM User WHERE companyId IS NOT NULL').get() as any)?.c) ?? 0;
    62→  const usersWithoutCompany = totalUsers - usersWithCompany;
    63→  const teamAdmins = ((d.prepare("SELECT COUNT(*) as c FROM User WHERE role = 'team_admin'").get() as any)?.c) ?? 0;
    64→  const regularUsers = totalUsers - teamAdmins;
    65→
    66→  // Prisma stores DateTime as Unix ms timestamps; convert to Unix seconds for SQLite date functions
    67→  const nowMs = Date.now();
    68→  const todayStart = new Date().setHours(0, 0, 0, 0);
    69→  const weekAgoMs = nowMs - 7 * 86400000;
    70→  const monthAgoMs = nowMs - 30 * 86400000;
    71→
    72→  const newUsersToday = ((d.prepare('SELECT COUNT(*) as c FROM User WHERE createdAt >= ?').get(todayStart) as any)?.c) ?? 0;
    73→  const newUsersThisWeek = ((d.prepare('SELECT COUNT(*) as c FROM User WHERE createdAt >= ?').get(weekAgoMs) as any)?.c) ?? 0;
    74→  const newUsersThisMonth = ((d.prepare('SELECT COUNT(*) as c FROM User WHERE createdAt >= ?').get(monthAgoMs) as any)?.c) ?? 0;
    75→
    76→  const recentRegistrations = d.prepare(
    77→    'SELECT id, email, role, status, createdAt, companyId FROM User ORDER BY createdAt DESC LIMIT 20'
    78→  ).all() as Array<{ id: string; email: string; role: string; status: string; createdAt: number; companyId: string | null }>;
    79→
    80→  const dailyRegistrations = d.prepare(
    81→    "SELECT DATE(createdAt / 1000, 'unixepoch') as date, COUNT(*) as count FROM User WHERE createdAt >= ? GROUP BY DATE(createdAt / 1000, 'unixepoch') ORDER BY date DESC"
    82→  ).all(nowMs - 30 * 86400000) as Array<{ date: string; count: number }>;
    83→
    84→  const monthlyRegistrations = d.prepare(
    85→    "SELECT STRFTIME('%Y-%m', createdAt / 1000, 'unixepoch') as month, COUNT(*) as count FROM User WHERE createdAt >= ? GROUP BY STRFTIME('%Y-%m', createdAt / 1000, 'unixepoch') ORDER BY month DESC"
    86→  ).all(nowMs - 365 * 86400000) as Array<{ month: string; count: number }>;
    87→
    88→  const roleDistribution = d.prepare(
    89→    'SELECT role, COUNT(*) as count FROM User GROUP BY role ORDER BY count DESC'
    90→  ).all() as Array<{ role: string; count: number }>;
    91→
    92→  const statusDistribution = d.prepare(
    93→    'SELECT status, COUNT(*) as count FROM User GROUP BY status ORDER BY count DESC'
    94→  ).all() as Array<{ status: string; count: number }>;
    95→
    96→  return {
    97→    totalUsers, activeUsers, suspendedUsers, bannedUsers,
    98→    verifiedUsers, unverifiedUsers,
    99→    newUsersToday, newUsersThisWeek, newUsersThisMonth,
   100→    usersWithCompany, usersWithoutCompany,
   101→    teamAdmins, regularUsers,
   102→    recentRegistrations, dailyRegistrations, monthlyRegistrations,
   103→    roleDistribution, statusDistribution,
   104→  };
   105→}

   181→    fail_tolerance: 'healthy',
   182→    backup: 'warning',
   183→    data_modeling: 'healthy',
   184→    rate_limiting: 'healthy',
   185→    caching: 'healthy',
   186→    edge_computing: 'warning',
   187→    web_performance: 'healthy',
   188→    cdn: 'healthy',
   189→    monitoring: 'healthy',
   190→    network_security: 'healthy',
   191→    api_integration: 'warning',
   192→    idempotency: 'healthy',
   193→    automation: 'healthy',
   194→    webhooks: 'healthy',
   195→    secret_management: 'healthy',
   196→    audits: 'healthy',
   197→    stateless: 'not_configured',
   198→  };
   199→
   200→  // Try to check DB connectivity
   201→  try {
   202→    getDb().prepare('SELECT 1').get();
   203→    concernStatuses.data_modeling = 'healthy';
   204→  } catch {
   205→    concernStatuses.data_modeling = 'critical';
   206→  }
   207→
   208→  const weights: Record<string, number> = { healthy: 1, warning: 0.6, critical: 0.2, not_configured: 0.4 };
   209→  const values = Object.values(concernStatuses);
   210→  const score = Math.round((values.reduce((sum, s) => sum + (weights[s] ?? 0.5), 0) / values.length) * 100);
   211→
   212→  return { score, concerns: concernStatuses, lastChecked: new Date().toISOString(), uptime: process.uptime() };
   213→}

   331→function getCacheStats() {
   332→  const d = getDb();
   333→  try {
   334→    const total = ((d.prepare('SELECT COUNT(*) as c FROM CacheEntry').get() as any)?.c) ?? 0;
   335→    const totalHits = ((d.prepare('SELECT SUM(hits) as c FROM CacheEntry').get() as any)?.c) ?? 0;
   336→    const totalMisses = ((d.prepare('SELECT SUM(misses) as c FROM CacheEntry').get() as any)?.c) ?? 0;
   337→    const totalSize = ((d.prepare('SELECT SUM(sizeBytes) as c FROM CacheEntry').get() as any)?.c) ?? 0;
   338→    const expired = ((d.prepare('SELECT COUNT(*) as c FROM CacheEntry WHERE expiresAt < ?').get(Date.now()) as any)?.c) ?? 0;
   339→    const hitRate = (totalHits + totalMisses) > 0 ? Math.round((totalHits / (totalHits + totalMisses)) * 100) : 0;
   340→    return { total, totalHits, totalMisses, totalSize, expired, hitRate };
   341→  } catch {
   342→    return { total: 0, totalHits: 0, totalMisses: 0, totalSize: 0, expired: 0, hitRate: 0 };
   343→  }
   344→}

   686→function formatBytes(bytes) {
   687→  if (!bytes || bytes === 0) return '0 B';
   688→  const k = 1024;
   689→  const sizes = ['B', 'KB', 'MB', 'GB'];
   690→  const i = Math.floor(Math.log(bytes) / Math.log(k));
   691→  return parseFloat((bytes / Math.pow(k, i)).toFixed(1)) + ' ' + sizes[i];
   692→}

   694→function timeAgo(dateStr) {
   695→  if (!dateStr) return 'N/A';
   696→  const diff = Date.now() - new Date(dateStr).getTime();
   697→  const mins = Math.floor(diff / 60000);
   698→  if (mins < 1) return 'just now';
   699→  if (mins < 60) return mins + 'm ago';
   700→  const hrs = Math.floor(mins / 60);
   701→  if (hrs < 24) return hrs + 'h ago';
   702→  const days = Math.floor(hrs / 24);
   703→  return days + 'd ago';
   704→}

  1073→// ─────────────────────────────────────────────────────────────────────────────
  1074→// HTTP Server – using Bun.serve() for stability
  1075→// ─────────────────────────────────────────────────────────────────────────────
  1076→
  1077→const PORT = 3004;
  1078→
  1079→// Initialize DB connection on startup
  1080→getDb();
  1081→
  1082→Bun.serve({
  1083→  port: PORT,
  1084→  async fetch(req) {
  1085→    const url = new URL(req.url);
  1086→    const path = url.pathname;
  1087→    const method = req.method;
  1088→
  1089→    // CORS headers
  1090→    const corsHeaders = {
  1091→      'Access-Control-Allow-Origin': '*',
  1092→      'Access-Control-Allow-Methods': 'GET, POST, PUT, DELETE, OPTIONS',
  1093→      'Access-Control-Allow-Headers': 'Content-Type, Authorization',
  1094→    };
  1095→
  1096→    if (method === 'OPTIONS') {
  1097→      return new Response(null, { status: 204, headers: corsHeaders });
  1098→    }
  1099→
  1100→    try {
  1101→      const headers = { ...corsHeaders };
  1102→
  1103→      // ── Serve Dashboard HTML ──
  1104→      if (path === '/' || path === '/index.html') {
  1105→        return new Response(getDashboardHTML(), { headers: { ...headers, 'Content-Type': 'text/html; charset=utf-8' } });
  1106→      }
  1107→
  1108→      // ── API: User Statistics ──
  1109→      if (path === '/api/users' && method === 'GET') {
  1110→        return Response.json({ success: true, data: getUserStats() }, { headers });
  1111→      }
  1112→
  1113→      // ── API: Health Check ──
  1114→      if (path === '/api/health' && method === 'GET') {
  1115→        return Response.json({ success: true, data: getHealthCheck() }, { headers });
  1116→      }
  1117→
  1118→      // ── API: System Metrics ──
  1119→      if (path === '/api/metrics' && method === 'GET') {
  1120→        return Response.json({ success: true, data: getSystemMetrics() }, { headers });
  1121→      }
  1122→
  1123→      // ── API: Alerts ──
  1124→      if (path === '/api/alerts' && method === 'GET') {
  1125→        return Response.json({ success: true, data: getAlerts() }, { headers });
  1126→      }
  1127→
  1128→      if (path === '/api/alerts' && method === 'POST') {
  1129→        const data = await req.json();
  1130→        const result = createAlert(data);
  1131→        return Response.json(result, { status: result.success ? 201 : 400, headers });
  1132→      }
  1133→
  1134→      // Acknowledge alert
  1135→      const ackMatch = path.match(/^\/api\/alerts\/([^/]+)\/acknowledge$/);
  1136→      if (ackMatch && method === 'POST') {
  1137→        const result = acknowledgeAlert(ackMatch[1]);
  1138→        return Response.json(result, { status: result.success ? 200 : 400, headers });
  1139→      }
  1140→
  1141→      // Resolve alert
  1142→      const resolveMatch = path.match(/^\/api\/alerts\/([^/]+)\/resolve$/);
  1143→      if (resolveMatch && method === 'POST') {
  1144→        const result = resolveAlert(resolveMatch[1]);
  1145→        return Response.json(result, { status: result.success ? 200 : 400, headers });
  1146→      }
  1147→
  1148→      // ── API: Webhooks ──
  1149→      if (path === '/api/webhooks' && method === 'GET') {
  1150→        return Response.json({ success: true, data: getWebhooks() }, { headers });
  1151→      }
  1152→
  1153→      if (path === '/api/webhooks' && method === 'POST') {
  1154→        const data = await req.json();
  1155→        const result = createWebhook(data);
  1156→        return Response.json(result, { status: result.success ? 201 : 400, headers });
  1157→      }
  1158→
  1159→      if (path === '/api/webhooks' && method === 'DELETE') {
  1160→        const id = url.searchParams.get('id');
  1161→        if (id) {
  1162→          const result = deleteWebhook(id);
