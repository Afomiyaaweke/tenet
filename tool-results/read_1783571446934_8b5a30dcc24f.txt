   195→   195→function relativeTime(dateStr: string): string {
   196→   196→  const now = Date.now();
   197→   197→  const then = new Date(dateStr).getTime();
   198→   198→  const diff = Math.max(0, now - then);
   199→   199→  const seconds = Math.floor(diff / 1000);
   200→   200→  if (seconds < 60) return 'just now';
   201→   201→  const minutes = Math.floor(seconds / 60);
   202→   202→  if (minutes < 60) return `${minutes}m ago`;
   203→   203→  const hours = Math.floor(minutes / 60);
   204→   204→  if (hours < 24) return `${hours}h ago`;
   205→   205→  const days = Math.floor(hours / 24);
   206→   206→  if (days < 7) return `${days}d ago`;
   207→   207→  return new Date(dateStr).toLocaleDateString();
   208→   208→}
