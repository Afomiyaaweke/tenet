    35→  try {

    48→function ScoreBar({ score, label }: { score: number; label?: string }) {
    49→  const colorClass = score >= 80 ? 'bg-emerald-500' : score >= 60 ? 'bg-amber-500' : 'bg-red-500';
    50→  const textColor = score >= 80 ? 'text-emerald-700' : score >= 60 ? 'text-amber-700' : 'text-red-700';
    51→  return (
    52→    <div className="flex items-center gap-2 min-w-[120px]">
    53→      {label && <span className="text-[10px] text-muted-foreground w-12 shrink-0">{label}</span>}
    54→      <div className="flex-1 h-2 bg-muted/50 rounded-full overflow-hidden">
    55→        <div
    56→ className={`h-full rounded-full ${colorClass} transition-[width] duration-700`} style={{ width: `${Math.min(score, 100)}%` }}
    57→ />
    58→      </div>
    59→      <span className={`text-xs font-semibold ${textColor} w-8 text-right`}>{score}</span>
    60→    </div>
    61→  );
    62→}

    64→function RiskBadge({ level }: { level: 'low' | 'medium' | 'high' }) {
    65→  const config = {
    66→    low: { className: 'bg-emerald-100 text-emerald-700', icon: ShieldCheck },
    67→    medium: { className: 'bg-amber-100 text-amber-700', icon: ShieldQuestion },
    68→    high: { className: 'bg-red-100 text-red-700', icon: ShieldAlert },
    69→  };
    70→  const c = config[level];
    71→  const Icon = c.icon;
    72→  return (
    73→    <Badge className={`text-[10px] px-2 py-0.5 border-0 rounded-lg ${c.className}`}>
    74→      <Icon className="h-3 w-3 mr-1" />
    75→      {level.charAt(0).toUpperCase() + level.slice(1)}
    76→    </Badge>
    77→  );
    78→}
