   146→function getProfileCompleteness(profile: Record<string, unknown> | undefined): { pct: number; missing: string[] } {
   147→  if (!profile) return { pct: 0, missing: ['Full Name', 'Phone', 'Location', 'Skills', 'Bio'] };
   148→  const checks: [string, boolean][] = [
   149→    ['Full Name', !!(profile.fullName as string)],
   150→    ['Phone', !!(profile.phone as string)],
   151→    ['Location', !!(profile.location as string)],
   152→    ['Skills', !!(profile.skillTags as string)],
   153→    ['Bio', !!(profile.bio as string)],
   154→  ];
   155→  const completed = checks.filter(([, v]) => v).length;
   156→  const missing = checks.filter(([, v]) => !v).map(([label]) => label);
   157→  return { pct: Math.round((completed / checks.length) * 100), missing };
   158→}
