   367→   367→
   368→   368→function parseAmount(value: number | string | undefined): number {
   369→   369→  if (value === undefined || value === null) return 0;
   370→   370→  if (typeof value === 'number') return value;
   371→   371→  const n = Number(String(value).replace(/[^0-9.-]/g, ''));
   372→   372→  return Number.isFinite(n) ? n : 0;
   373→   373→}

   375→   375→function truncate(text: string, max = 280): string {
   376→   376→  if (!text) return '';
   377→   377→  return text.length > max ? text.slice(0, max).trimEnd() + '…' : text;
   378→   378→}

   407→   407→function inferCategory(row: WorldBankProcurementRow): string {
   408→   408→  const t = (row.contract_type || '').toLowerCase();
   409→   409→  if (t.includes('work') || t.includes('civil') || t.includes('construction')) return 'Construction';
   410→   410→  if (t.includes('consult')) return 'Consulting';
   411→   411→  if (t.includes('good') || t.includes('supply') || t.includes('equipment')) return 'Supply';
   412→   412→  if (t.includes('service')) return 'Consulting';
   413→   413→  const p = (row.project_name || '').toLowerCase();
   414→   414→  if (p.includes('road') || p.includes('bridge') || p.includes('infrastructure')) return 'Construction';
   415→   415→  if (p.includes('agric') || p.includes('farm') || p.includes('irrigat')) return 'Agriculture';
   416→   416→  if (p.includes('health') || p.includes('clinic') || p.includes('hospital')) return 'Healthcare';
   417→   417→  if (p.includes('energy') || p.includes('power') || p.includes('electric')) return 'Engineering';
   418→   418→  if (p.includes('edu') || p.includes('school')) return 'Education';
   419→   419→  if (p.includes('water') || p.includes('sanitation')) return 'Engineering';
   420→   420→  return 'Consulting';
   421→   421→}
   422→   422→
   423→   423→function buildDeadline(signingDate?: string): string {
   424→   424→  if (!signingDate) return new Date(Date.now() + 45 * 86400000).toISOString();
   425→   425→  const base = new Date(signingDate);
   426→   426→  if (Number.isNaN(base.getTime())) return new Date(Date.now() + 45 * 86400000).toISOString();
   427→   427→  return new Date(base.getTime() + 45 * 86400000).toISOString();
   428→   428→}

   430→   430→export async function fetchWorldBankTenders(opts: {

   436→   436→  const offset = opts.offset || 0;
   437→   437→  const params = new URLSearchParams({
   438→   438→    format: 'json',
   439→   439→    rows: String(rows),
   440→   440→    fl: [
   441→   441→      'project_id',
   442→   442→      'procurement_group_id',
   443→   443→      'project_name',
   444→   444→      'borrower',
   445→   445→      'country',
   446→   446→      'region',
   447→   447→      'contract_description',
   448→   448→      'contract_type',
   449→   449→      'supplier',
   450→   450→      'supplier_country',
