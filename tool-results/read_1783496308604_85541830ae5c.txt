   144→function getProfileCompleteness(profile: Record<string, unknown> | undefined | null): { pct: number; missing: string[] } {
   145→  if (!profile) return { pct: 0, missing: ['Full Name', 'Phone', 'Location', 'Skills', 'Bio'] };
   146→  const checks: [string, boolean][] = [
   147→    ['Full Name', !!(profile.fullName as string)],
   148→    ['Phone', !!(profile.phone as string)],
   149→    ['Location', !!(profile.location as string)],
   150→    ['Skills', !!(profile.skillTags as string)],
   151→    ['Bio', !!(profile.bio as string)],
   152→  ];
   153→  const completed = checks.filter(([, v]) => v).length;
   154→  const missing = checks.filter(([, v]) => !v).map(([label]) => label);
   155→  return { pct: Math.round((completed / checks.length) * 100), missing };
   156→}
