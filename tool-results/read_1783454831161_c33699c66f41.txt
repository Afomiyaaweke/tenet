   172→   172→function relativeTime(dateStr: string): string {
   173→   173→  const now = Date.now();
   174→   174→  const then = new Date(dateStr).getTime();
   175→   175→  const diff = Math.max(0, now - then);
   176→   176→  const seconds = Math.floor(diff / 1000);
   177→   177→  if (seconds < 60) return 'just now';
   178→   178→  const minutes = Math.floor(seconds / 60);
   179→   179→  if (minutes < 60) return `${minutes}m ago`;
   180→   180→  const hours = Math.floor(minutes / 60);
   181→   181→  if (hours < 24) return `${hours}h ago`;
   182→   182→  const days = Math.floor(hours / 24);
   183→   183→  if (days < 7) return `${days}d ago`;
   184→   184→  return new Date(dateStr).toLocaleDateString();
   185→   185→}
