   184→function relativeTime(dateStr: string): string {
   185→  const now = Date.now();
   186→  const then = new Date(dateStr).getTime();
   187→  const diff = Math.max(0, now - then);
   188→  const seconds = Math.floor(diff / 1000);
   189→  if (seconds < 60) return 'just now';
   190→  const minutes = Math.floor(seconds / 60);
   191→  if (minutes < 60) return `${minutes}m ago`;
   192→  const hours = Math.floor(minutes / 60);
   193→  if (hours < 24) return `${hours}h ago`;
   194→  const days = Math.floor(hours / 24);
   195→  if (days < 7) return `${days}d ago`;
   196→  return new Date(dateStr).toLocaleDateString();
   197→}

   365→/* ──────────────────── NotificationDropdown ──────────────────── */

   368→  notifications,
   369→  onMarkAsRead,
   370→  onMarkAllRead,
   371→}: {
   372→  notifications: { id: string; title: string; message: string; type: string; read: boolean; createdAt: string }[];

   374→  onMarkAllRead: () => void;

   376→  const [open, setOpen] = useState(false);
   377→  const unread = useMemo(() => notifications.filter((n) => !n.read), [notifications]);
   378→  const displayItems = useMemo(() => unread.slice(0, 6), [unread]);
   379→
   380→  return (
   381→    <Popover open={open} onOpenChange={setOpen}>

   383→        <Button variant="ghost" size="icon" className="relative h-9 w-9 hover:bg-muted/80 transition-colors">

   389→          )}
