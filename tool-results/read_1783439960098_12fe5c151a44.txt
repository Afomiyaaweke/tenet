   154→   154→function relativeTime(dateStr: string): string {
   155→   155→  const now = Date.now();
   156→   156→  const then = new Date(dateStr).getTime();
   157→   157→  const diff = Math.max(0, now - then);
   158→   158→  const seconds = Math.floor(diff / 1000);
   159→   159→  if (seconds < 60) return 'just now';
   160→   160→  const minutes = Math.floor(seconds / 60);
   161→   161→  if (minutes < 60) return `${minutes}m ago`;
   162→   162→  const hours = Math.floor(minutes / 60);
   163→   163→  if (hours < 24) return `${hours}h ago`;
   164→   164→  const days = Math.floor(hours / 24);
   165→   165→  if (days < 7) return `${days}d ago`;
   166→   166→  return new Date(dateStr).toLocaleDateString();
   167→   167→}
