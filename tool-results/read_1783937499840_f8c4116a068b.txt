   343→
   344→function fmtMoney(amount: number, currency: string): string {
   345→  if (!amount) return '—';
   346→  try {
   347→    return new Intl.NumberFormat('en-US', {
   348→      style: 'currency',
   349→      currency: currency || 'USD',
   350→      maximumFractionDigits: 0,
   351→    }).format(amount);
   352→  } catch {
   353→    return `${currency} ${amount.toLocaleString()}`;
   354→  }
   355→}
   356→
   357→function daysUntil(dateStr: string): number {
   358→  if (!dateStr) return 999;
   359→  return Math.ceil((new Date(dateStr).getTime() - Date.now()) / (1000 * 60 * 60 * 24));
   360→}
   361→
   362→function deadlineBadge(days: number): string {
   363→  if (days <= 0) return 'bg-rose-50 text-rose-700 dark:bg-rose-950/40 dark:text-rose-300';
   364→  if (days <= 7) return 'bg-amber-50 text-amber-700 dark:bg-amber-950/40 dark:text-amber-300';
   365→  if (days <= 21) return 'bg-sky-50 text-sky-700 dark:bg-sky-950/40 dark:text-sky-300';
   366→  return 'bg-emerald-50 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300';
   367→}
   368→
   369→function deadlineLabel(days: number): string {
   370→  if (days <= 0) return 'Closed';
   371→  if (days <= 7) return `${days}d left`;
   372→  if (days <= 30) return `${days}d left`;
   373→  return `${days}d left`;
   374→}
   375→
   376→function relativeTime(dateStr: string): string {
   377→  if (!dateStr) return '';
   378→  const diff = Date.now() - new Date(dateStr).getTime();
   379→  const day = 86400000;
   380→  if (diff < day) return 'today';
   381→  if (diff < 2 * day) return 'yesterday';
   382→  if (diff < 30 * day) return `${Math.floor(diff / day)}d ago`;
   383→  return new Date(dateStr).toLocaleDateString();
   384→}
   385→
   386→function riskBadgeColor(level: string): string {
   387→  if (level === 'low') return 'bg-emerald-50 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300';

   389→  return 'bg-rose-50 text-rose-700 dark:bg-rose-950/40 dark:text-rose-300';
   390→}

   419→                <BookOpen className="h-3.5 w-3.5" />
   420→                <span>Full tender content loaded from {new URL(doc.url).hostname}</span>
   421→                <span>·</span>
   422→                <span>{new Date(doc.fetchedAt).toLocaleTimeString()}</span>
   423→              </div>
   424→              <h3 className="text-lg font-semibold text-foreground leading-snug">
   425→                {doc.title || 'Tender Document'}
   426→              </h3>
   427→              {doc.metaDescription && (
   428→                <p className="text-sm text-muted-foreground mt-1 line-clamp-2">
   429→                  {doc.metaDescription}
   430→                </p>
   431→              )}
   432→            </div>
   433→            <div className="flex items-center gap-2 shrink-0">
   434→              <Button variant="ghost" size="sm" className="gap-1.5 text-xs" onClick={copyContent}>
   435→                <Copy className="h-3.5 w-3.5" />
   436→                Copy

   439→                href={doc.url}
   440→                target="_blank"

   452→          </div>
   453→
   454→          {/* Extracted metadata pills */}
   455→          {(doc.deadlines && doc.deadlines.length > 0) || (doc.budgets && doc.budgets.length > 0) ? (
   456→            <div className="flex flex-wrap gap-2">
   457→              {doc.deadlines?.map((d, i) => (
