    44→function formatETB(amount: number): string {
    45→  if (amount >= 1_000_000) return `ETB ${(amount / 1_000_000).toFixed(1)}M`;
    46→  if (amount >= 1_000) return `ETB ${(amount / 1_000).toFixed(0)}K`;
    47→  return `ETB ${amount.toLocaleString()}`;
    48→}
    49→
    50→function daysUntil(dateStr: string): number {
    51→  return Math.ceil((new Date(dateStr).getTime() - Date.now()) / 86400000);
    52→}
    53→
    54→function isOverdue(dateStr: string): boolean {
    55→  return new Date(dateStr).getTime() < Date.now();
    56→}
