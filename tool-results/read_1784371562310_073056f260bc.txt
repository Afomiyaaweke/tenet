   459→function fmtMoney(amount: number, currency: string): string {
   460→  if (!amount) return '-';
   461→  try {
   462→    return new Intl.NumberFormat('en-US', {
   463→      style: 'currency',
   464→      currency: currency || 'USD',
   465→      maximumFractionDigits: 0,
   466→    }).format(amount);
   467→  } catch {
   468→    return `${currency} ${amount.toLocaleString()}`;
   469→  }
   470→}

   472→function daysUntil(dateStr: string): number {
   473→  if (!dateStr) return 999;
   474→  return Math.ceil((new Date(dateStr).getTime() - Date.now()) / (1000 * 60 * 60 * 24));
   475→}

   477→function deadlineBadge(days: number): string {
   478→  if (days <= 0) return 'bg-rose-50 text-rose-700 dark:bg-rose-950/40 dark:text-rose-300';
   479→  if (days <= 7) return 'bg-amber-50 text-amber-700 dark:bg-amber-950/40 dark:text-amber-300';
   480→  if (days <= 21) return 'bg-sky-50 text-sky-700 dark:bg-sky-950/40 dark:text-sky-300';
   481→  return 'bg-emerald-50 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300';
   482→}
   483→
   484→function deadlineLabel(days: number): string {
   485→  if (days <= 0) return 'Closed';
   486→  if (days <= 7) return `${days}d left`;
   487→  if (days <= 30) return `${days}d left`;
   488→  return `${days}d left`;
   489→}
   490→
   491→function relativeTime(dateStr: string): string {
   492→  if (!dateStr) return '';
   493→  const diff = Date.now() - new Date(dateStr).getTime();
   494→  const day = 86400000;
   495→  if (diff < day) return 'today';
   496→  if (diff < 2 * day) return 'yesterday';
   497→  if (diff < 30 * day) return `${Math.floor(diff / day)}d ago`;
   498→  return new Date(dateStr).toLocaleDateString();
   499→}
   500→
   501→function riskBadgeColor(level: string): string {
   502→  if (level === 'low') return 'bg-emerald-50 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300';
   503→  if (level === 'medium') return 'bg-amber-50 text-amber-700 dark:bg-amber-950/40 dark:text-amber-300';
   504→  return 'bg-rose-50 text-rose-700 dark:bg-rose-950/40 dark:text-rose-300';
   505→}
   506→
   507→function readinessColor(score: number): string {
   508→  if (score >= 8) return 'bg-emerald-500';
   509→  if (score >= 5) return 'bg-amber-500';
   510→  return 'bg-rose-500';
   511→}

   523→      ? doc.sections.map((s) => `${s.heading}\n${s.content}`).join('\n\n')

   534→        method: 'POST',
