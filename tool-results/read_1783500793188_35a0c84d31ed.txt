   127→/* ───────────────────────── Password Strength Meter ───────────────────────── */
   128→function scorePassword(pw: string): { score: number; label: string; color: string } {
   129→  let score = 0;
   130→  if (pw.length >= 8) score++;
   131→  if (pw.length >= 12) score++;
   132→  if (/[a-z]/.test(pw) && /[A-Z]/.test(pw)) score++;
   133→  if (/[0-9]/.test(pw)) score++;
   134→  if (/[^A-Za-z0-9]/.test(pw)) score++;
   135→  const labels = ['Too weak', 'Weak', 'Fair', 'Good', 'Strong', 'Very strong'];
   136→  const colors = ['bg-rose-500', 'bg-rose-500', 'bg-amber-500', 'bg-yellow-500', 'bg-orange-500', 'bg-orange-600'];
   137→  return { score, label: labels[score], color: colors[score] };
   138→}

   140→/* ───────────────────────── Slide-to-Verify Captcha ───────────────────────── */
   141→function SlideCaptcha({ onVerified }: { onVerified: () => void }) {
   142→  const [offset, setOffset] = useState(0);
   143→  const [status, setStatus] = useState<'idle' | 'dragging' | 'success' | 'fail'>('idle');
   144→  const [target] = useState(() => 180 + Math.floor(Math.random() * 80));
   145→  const trackWidth = 320;
   146→  const handleWidth = 44;
   147→  const tolerance = 8;
   148→  const dragRef = React.useRef<{ start: number; startOffset: number } | null>(null);
   149→
   150→  const onPointerDown = (e: React.PointerEvent) => {
   151→    if (status === 'success') return;
   152→    setStatus('dragging');
   153→    dragRef.current = { start: e.clientX, startOffset: offset };
   154→    (e.target as HTMLElement).setPointerCapture(e.pointerId);
   155→  };
   156→
   157→  const onPointerMove = (e: React.PointerEvent) => {
   158→    if (!dragRef.current || status !== 'dragging') return;
   159→    const delta = e.clientX - dragRef.current.start;
   160→    const next = Math.max(0, Math.min(trackWidth - handleWidth, dragRef.current.startOffset + delta));
   161→    setOffset(next);
   162→  };
   163→
   164→  const onPointerUp = () => {
   165→    if (status !== 'dragging') return;
   166→    dragRef.current = null;
   167→    if (Math.abs(offset - target) <= tolerance) {
   168→      setStatus('success');
   169→      setTimeout(() => onVerified(), 450);
   170→    } else {
   171→      setStatus('fail');
   172→      setTimeout(() => {
   173→        setStatus('idle');
   174→        setOffset(0);
   175→      }, 700);
   176→    }
   177→  };
   178→
   179→  const pieceColor = status === 'success' ? 'bg-orange-500' : status === 'fail' ? 'bg-rose-500' : 'bg-primary';
   180→  const ringColor = status === 'success' ? 'ring-orange-300' : status === 'fail' ? 'ring-rose-300' : 'ring-primary/40';
   181→
   182→  return (
   183→    <div className="space-y-3">
   184→      <div className="text-xs font-medium text-muted-foreground flex items-center gap-1.5">
   185→        <Puzzle className="w-3.5 h-3.5" /> Security verification — slide to complete
   186→      </div>
   187→      {/* Puzzle canvas */}
   188→      <div className="relative h-12 rounded-xl overflow-hidden border border-border bg-muted/40 select-none">
   189→        {/* pattern background */}
   190→        <div
   191→          className="absolute inset-0 opacity-50"
   192→          style={{
   193→            backgroundImage:
   194→              'repeating-linear-gradient(45deg, hsl(var(--muted-foreground) / 0.08) 0 6px, transparent 6px 12px), repeating-linear-gradient(-45deg, hsl(var(--muted-foreground) / 0.06) 0 6px, transparent 6px 12px)',
   195→          }}
   196→        />
   197→        {/* target cutout (notch) */}
   198→        <div
   199→          className="absolute top-1/2 -translate-y-1/2 h-9 w-9 rounded-lg border-2 border-dashed border-muted-foreground/40 bg-background/60"
   200→          style={{ left: target + 4 }}
   201→        />
   202→        {/* moving puzzle piece */}
   203→        <div
   204→          className={`absolute top-1/2 -translate-y-1/2 h-9 w-9 rounded-lg ${pieceColor} shadow-lg ring-2 ${ringColor} flex items-center justify-center text-white transition-shadow ${status === 'fail' ? 'animate-[shake_0.4s_ease-in-out]' : ''}`}
   205→          style={{ left: 4 + offset, transition: status === 'dragging' ? 'none' : 'left 0.25s ease' }}
   206→        >
   207→          {status === 'success' ? <CheckCircle2 className="w-5 h-5" /> : status === 'fail' ? <XCircle className="w-5 h-5" /> : <Puzzle className="w-4 h-4" />}
   208→        </div>
   209→        {/* hint text */}
   210→        {status === 'idle' && (
   211→          <span className="absolute inset-0 flex items-center justify-center text-xs text-muted-foreground pointer-events-none">
   212→            Drag the slider below to fit the puzzle
   213→          </span>
   214→        )}
   215→        {status === 'success' && (
   216→          <span className="absolute inset-0 flex items-center justify-center text-xs font-medium text-orange-600 bg-orange-50/60 pointer-events-none">
   217→            Verified
   218→          </span>
   219→        )}
   220→      </div>
   221→      {/* Slider track */}
   222→      <div className="relative h-11 rounded-xl bg-muted border border-border overflow-hidden">
   223→        <div
   224→          className={`absolute inset-y-0 left-0 ${status === 'success' ? 'bg-orange-100 dark:bg-orange-950/40' : status === 'fail' ? 'bg-rose-100 dark:bg-rose-950/40' : 'bg-primary/10'} transition-all`}
   225→          style={{ width: offset + handleWidth }}
   226→        />
   227→        <div
   228→          className="absolute top-0 h-full flex items-center justify-center text-xs font-medium text-muted-foreground pointer-events-none"
   229→          style={{ left: handleWidth, right: 0 }}
   230→        >
   231→          {status === 'success' ? 'Verification complete' : status === 'fail' ? 'Verification failed, retrying…' : 'Slide right to verify'}
   232→        </div>
   233→        <div
   234→          onPointerDown={onPointerDown}
   235→          onPointerMove={onPointerMove}
   236→          onPointerUp={onPointerUp}
   237→          onPointerCancel={onPointerUp}
   238→          className={`absolute top-1/2 -translate-y-1/2 h-9 w-9 rounded-lg cursor-grab active:cursor-grabbing flex items-center justify-center text-white shadow-md ${pieceColor} ${status === 'success' ? 'cursor-default' : ''} touch-none`}
   239→          style={{ left: 4 + offset, transition: status === 'dragging' ? 'none' : 'left 0.25s ease' }}
   240→        >
   241→          {status === 'success' ? <CheckCircle2 className="w-5 h-5" /> : <ArrowRight className="w-4 h-4" />}
   242→        </div>
   243→      </div>
   244→    </div>
   245→  );
   246→}

   248→/* ───────────────────────── Step Indicator ───────────────────────── */
   249→function StepIndicator({ currentStep, totalSteps = 5 }: { currentStep: RegStep; totalSteps?: number }) {
   250→  return (
   251→    <div className="flex items-center justify-center gap-1 mb-6">
   252→      {Array.from({ length: totalSteps }, (_, i) => (i + 1) as RegStep).map((step, i) => {
   253→        const meta = REG_STEP_META[step];
   254→        const isCompleted = currentStep > step;
   255→        const isCurrent = currentStep === step;
   256→        const Icon = meta.icon;
   257→        return (
   258→          <React.Fragment key={step}>
   259→            <div className="flex items-center gap-1.5">
   260→              <div
   261→                className={`h-7 w-7 rounded-full flex items-center justify-center text-xs font-bold transition-all ${
   262→                  isCompleted || isCurrent
   263→                    ? 'gradient-orange text-white shadow-md shadow-orange-500/30'
   264→                    : 'bg-muted text-muted-foreground'
   265→                }`}
   266→              >
   267→                {isCompleted ? <CheckCircle2 className="w-4 h-4" /> : <Icon className="w-3.5 h-3.5" />}
   268→              </div>
   269→              <span className={`text-xs font-medium hidden sm:inline ${isCompleted || isCurrent ? 'text-foreground' : 'text-muted-foreground'}`}>
   270→                {meta.label}
   271→              </span>
   272→            </div>
   273→            {i < totalSteps - 1 && (
   274→              <div className={`h-0.5 w-4 sm:w-8 rounded-full ${currentStep > step ? 'bg-orange-500' : 'bg-muted'}`} />
   275→            )}
   276→          </React.Fragment>
   277→        );
   278→      })}
   279→    </div>
   280→  );
   281→}
