   119→function getAvatarGradient(seed: string): string {
   120→  let hash = 0;
   121→  for (let i = 0; i < seed.length; i++) {
   122→    hash = seed.charCodeAt(i) + ((hash << 5) - hash);
   123→  }
   124→  return GRADIENT_OPTIONS[Math.abs(hash) % GRADIENT_OPTIONS.length];
   125→}
   126→
   127→function getSenderColor(seed: string): string {
   128→  return SENDER_COLOR_MAP[getAvatarGradient(seed)] || 'text-primary';
   129→}

   131→function getUserInitial(name?: string): string {
   132→  if (!name) return '?';

   138→function formatRelativeTime(dateStr: string): string {
   139→  const now = new Date();
   140→  const date = new Date(dateStr);
   141→  const diffMs = now.getTime() - date.getTime();
   142→  const diffMin = Math.floor(diffMs / 60000);
   143→  const diffHour = Math.floor(diffMs / 3600000);
   144→  const diffDay = Math.floor(diffMs / 86400000);
   145→  if (diffMin < 1) return 'now';
   146→  if (diffMin < 60) return `${diffMin}m`;
   147→  if (diffHour < 24) return `${diffHour}h`;
   148→  if (diffDay === 1) return 'Yesterday';
   149→  if (diffDay < 7) return `${diffDay}d`;
   150→  return date.toLocaleDateString([], { month: 'short', day: 'numeric' });
   151→}

   160→function formatDateSeparator(dateStr: string): string {
   161→  const now = new Date();
   162→  const date = new Date(dateStr);
   163→  const diffMs = now.getTime() - date.getTime();
   164→  const diffDay = Math.floor(diffMs / 86400000);
   165→  if (diffDay === 0) return 'Today';
   166→  if (diffDay === 1) return 'Yesterday';
   167→  if (diffDay < 7) return date.toLocaleDateString([], { weekday: 'long' });
   168→  return date.toLocaleDateString([], {
   169→    month: 'long',
   170→    day: 'numeric',
   171→    year: 'numeric',
   172→  });
   173→}

   175→function shouldShowDateSeparator(
   176→  messages: ChatMessageItem[],
   177→  index: number
   178→): boolean {
   179→  if (index === 0) return true;
   180→  const prev = new Date(messages[index - 1].createdAt).toDateString();
   181→  const curr = new Date(messages[index].createdAt).toDateString();
   182→  return prev !== curr;
   183→}
   184→
   185→function isGroupedWithPrev(
   186→  messages: ChatMessageItem[],
   187→  index: number
   188→): boolean {
   189→  if (index === 0) return false;
   190→  const prev = messages[index - 1];
   191→  const curr = messages[index];
   192→  if (prev.userId !== curr.userId) return false;
   193→  if (shouldShowDateSeparator(messages, index)) return false;
   194→  const diff =
   195→    new Date(curr.createdAt).getTime() - new Date(prev.createdAt).getTime();
   196→  return diff < 5 * 60 * 1000; // 5 minutes
   197→}

   343→function MemberPicker({
   344→  excludedIds = [],
   345→  onSelect,
   346→  placeholder = 'Search users by name or email…',
   347→}: {
   348→  excludedIds?: string[];
   349→  onSelect: (user: UserSearchResult) => void;
   350→  placeholder?: string;
   351→}) {
   352→  const [query, setQuery] = useState('');
   353→  const [results, setResults] = useState<UserSearchResult[]>([]);
   354→  const [loading, setLoading] = useState(false);
   355→  const debounceRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   356→  const excludedKey = excludedIds.join(',');
   357→
   358→  useEffect(() => {
   359→    if (debounceRef.current) clearTimeout(debounceRef.current);
   360→    if (!query.trim()) {
   361→      setResults([]);
   362→      setLoading(false);
   363→      return;
   364→    }
   365→    setLoading(true);
   366→    debounceRef.current = setTimeout(async () => {
   367→      try {
   368→        const res = await api.get('/users/search', { q: query.trim() });
   369→        if (res.success) {
   370→          const excluded = new Set(excludedIds);
   371→          setResults(
   372→            (res.data || []).filter(
   373→              (u: UserSearchResult) => !excluded.has(u.id)
   374→            )
   375→          );
   376→        } else {
   377→          setResults([]);
   378→        }
   379→      } catch {
   380→        setResults([]);
   381→      } finally {
   382→        setLoading(false);
   383→      }
   384→    }, 250);
   385→    return () => {
   386→      if (debounceRef.current) clearTimeout(debounceRef.current);
   387→    };
   388→  }, [query, excludedKey, excludedIds]);
   389→
   390→  return (
   391→    <div className="space-y-2">
   392→      <div className="relative">
   393→        <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground/60" />
   394→        <Input
   395→          placeholder={placeholder}
   396→          value={query}
   397→          onChange={(e) => setQuery(e.target.value)}
   398→          className="h-9 text-sm pl-8 bg-muted/50 border-border/40 rounded-lg"
   399→        />
   400→      </div>
   401→      {loading && <p className="text-xs text-muted-foreground px-1">Searching…</p>}
   402→      {!loading && query.trim() && results.length === 0 && (
   403→        <p className="text-xs text-muted-foreground px-1">No users found.</p>
   404→      )}
   405→      {!loading && results.length > 0 && (
   406→        <div className="border border-border/40 rounded-lg max-h-56 overflow-y-auto divide-y divide-border/20">
   407→          {results.map((u) => (
   408→            <button
   409→              key={u.id}
   410→              onClick={() => onSelect(u)}
   411→              className="w-full flex items-center gap-3 p-2 hover:bg-muted/50 transition-colors text-left"
   412→            >
   413→              <UserAvatar
   414→                userId={u.id}
   415→                name={u.profile?.fullName}
   416→                photoUrl={u.profile?.profilePhoto}
   417→                size={32}
   418→                rounded="rounded-full"
   419→              />
   420→              <div className="min-w-0 flex-1">
   421→                <p className="text-sm font-medium truncate">
   422→                  {u.profile?.fullName || u.email}
   423→                </p>
   424→                <p className="text-xs text-muted-foreground truncate">{u.email}</p>
   425→              </div>
   426→              <Plus className="h-4 w-4 text-primary flex-shrink-0" />
   427→            </button>
   428→          ))}
   429→        </div>
   430→      )}
   431→    </div>
   432→  );
   433→}

   439→function NewGroupDialog({
   440→  open,
   441→  onOpenChange,
   442→  onCreated,
   443→  currentUserId,
   444→}: {
   445→  open: boolean;
   446→  onOpenChange: (o: boolean) => void;
   447→  onCreated: (conv: Conversation) => void;
   448→  currentUserId: string;
   449→}) {
   450→  const [title, setTitle] = useState('');
   451→  const [description, setDescription] = useState('');
   452→  const [selected, setSelected] = useState<UserSearchResult[]>([]);
   453→  const [creating, setCreating] = useState(false);
   454→
   455→  const reset = () => {
   456→    setTitle('');
   457→    setDescription('');
   458→    setSelected([]);
   459→  };
   460→
   461→  const handleCreate = async () => {
   462→    if (!title.trim()) {
   463→      toast.error('Group name is required');
   464→      return;
   465→    }
   466→    if (selected.length === 0) {
   467→      toast.error('Add at least one member');
   468→      return;
   469→    }
   470→    setCreating(true);
   471→    try {
   472→      const res = await api.post('/conversations', {
   473→        type: 'group',
   474→        title: title.trim(),
   475→        description: description.trim() || undefined,
   476→        memberIds: selected.map((u) => u.id),
   477→      });
   478→      if (res.success) {
   479→        toast.success('Group created');
   480→        onCreated(res.data as Conversation);
   481→        reset();
   482→        onOpenChange(false);
   483→      } else {
   484→        toast.error(res.error || 'Failed to create group');
   485→      }
   486→    } finally {
   487→      setCreating(false);
   488→    }
   489→  };
   490→
   491→  return (
   492→    <Dialog
   493→      open={open}
   494→      onOpenChange={(o) => {
   495→        if (!o) reset();
   496→        onOpenChange(o);
   497→      }}
   498→    >
   499→      <DialogContent className="sm:max-w-md">
   500→        <DialogHeader>
   501→          <DialogTitle>New Group</DialogTitle>
   502→          <DialogDescription>
   503→            Create a group conversation with other users.
   504→          </DialogDescription>
   505→        </DialogHeader>
   506→        <div className="space-y-3 py-2">
   507→          <div className="space-y-1.5">
   508→            <label className="text-xs font-medium text-muted-foreground">
   509→              Group name *
   510→            </label>
   511→            <Input
   512→              placeholder="e.g. Project Alpha Team"
   513→              value={title}
   514→              onChange={(e) => setTitle(e.target.value)}
   515→              className="bg-muted/50"
   516→            />
   517→          </div>
   518→          <div className="space-y-1.5">
   519→            <label className="text-xs font-medium text-muted-foreground">
   520→              Description (optional)
   521→            </label>
   522→            <Textarea
   523→              placeholder="What's this group about?"
   524→              value={description}
   525→              onChange={(e) => setDescription(e.target.value)}
   526→              rows={2}
   527→              className="bg-muted/50 resize-none"
   528→            />
   529→          </div>
   530→          <div className="space-y-1.5">
   531→            <label className="text-xs font-medium text-muted-foreground">
   532→              Add members *
   533→            </label>
   534→            <MemberPicker
   535→              excludedIds={[currentUserId, ...selected.map((u) => u.id)]}
   536→              onSelect={(u) => setSelected((prev) => [...prev, u])}
   537→            />
   538→            {selected.length > 0 && (
   539→              <div className="flex flex-wrap gap-1.5 pt-1">
   540→                {selected.map((u) => (
   541→                  <span
   542→                    key={u.id}
   543→                    className="inline-flex items-center gap-1.5 bg-primary/10 text-primary text-xs px-2 py-1 rounded-full"
   544→                  >
   545→                    {u.profile?.fullName || u.email}
   546→                    <button
   547→                      onClick={() =>
   548→                        setSelected((prev) =>
   549→                          prev.filter((x) => x.id !== u.id)
   550→                        )
   551→                      }
   552→                    >
   553→                      <X className="h-3 w-3" />
   554→                    </button>
   555→                  </span>
   556→                ))}
   557→              </div>
   558→            )}
   559→          </div>
   560→        </div>
   561→        <DialogFooter>
   562→          <Button variant="ghost" onClick={() => onOpenChange(false)}>
   563→            Cancel
   564→          </Button>
   565→          <Button
   566→            className="gradient-emerald"
   567→            onClick={handleCreate}
   568→            disabled={creating}
   569→          >
   570→            {creating ? 'Creating…' : 'Create Group'}
   571→          </Button>
   572→        </DialogFooter>
   573→      </DialogContent>
   574→    </Dialog>
   575→  );
   576→}

   755→function ConversationListItem({
   756→  conv,
   757→  active,
   758→  index,
   759→  onlineUsers,
   760→  typingUsers,
   761→  currentUserId,
   762→  onSelect,
   763→}: {
   764→  conv: Conversation;
   765→  active: boolean;
   766→  index: number;
   767→  onlineUsers: Set<string>;
   768→  typingUsers: string[];
   769→  currentUserId: string;
   770→  onSelect: (conv: Conversation) => void;
   771→}) {
   772→  const myMembership = conv.members?.find((m) => m.userId === currentUserId);
   773→  const pinned = !!(myMembership?.pinned || (conv as Conversation & { pinned?: boolean }).pinned);
   774→  const muted = !!myMembership?.muted;
   775→  const unread = conv.unreadCount || 0;
   776→
   777→  // Title and avatar resolution
   778→  let title = conv.title || '';
   779→  let photoUrl: string | undefined;
   780→  let otherUserId: string | undefined;
   781→  if (conv.type === 'direct') {
   782→    const other = conv.members?.find((m) => m.userId !== currentUserId);
   783→    title = other?.user.profile?.fullName || other?.user.email || 'Direct Message';
   784→    photoUrl = other?.user.profile?.profilePhoto;
   785→    otherUserId = other?.userId;
   786→  } else if (!title) {
   787→    title = conv.type === 'channel' ? 'Channel' : 'Untitled Group';
   788→  }
   789→
   790→  // Last message preview
   791→  const lastMsg = conv.messages?.[0];
   792→  const typingOthers = typingUsers.filter((u) => u !== currentUserId);
   793→  let previewText = '';
   794→  let previewTime = conv.updatedAt;
   795→  if (typingOthers.length > 0) {
   796→    previewText = 'typing…';
   797→  } else if (lastMsg) {
   798→    const senderName =
   799→      lastMsg.userId === currentUserId
   800→        ? 'You'
   801→        : lastMsg.user?.profile?.fullName?.split(' ')[0] || 'User';
   802→    previewText =
   803→      conv.type === 'direct' ? lastMsg.content : `${senderName}: ${lastMsg.content}`;
   804→    previewTime = lastMsg.createdAt;
   805→  } else {
   806→    previewText = 'No messages yet';
   807→  }
   808→
   809→  const isOnline = otherUserId ? onlineUsers.has(otherUserId) : false;
   810→
   811→  return (
   812→    <button
   813→ onClick={() => onSelect(conv)}
   814→ className={`w-full p-3 text-left transition-colors duration-150 border-b border-border/20 group relative ${
   815→ active ? 'bg-primary/10' : 'hover:bg-muted/50'
   816→ } animate-[fadeIn_0.3s_ease-out]`}
   817→ >
   818→      {active && (
   819→        <div
   820→ layoutId="activeConvBar"
   821→ className="absolute left-0 top-2 bottom-2 w-[3px] rounded-r-full gradient-emerald"
   822→ />
   823→      )}
   824→      {pinned && !active && (
   825→        <Pin className="absolute top-2 right-2 h-3 w-3 text-muted-foreground/60" />
   826→      )}
   827→      <div className="flex items-start gap-3">
   828→        <div className="relative flex-shrink-0">
   829→          {conv.type === 'direct' ? (
   830→            <UserAvatar
   831→              userId={otherUserId || conv.id}
   832→              name={title}
   833→              photoUrl={photoUrl}
   834→              size={44}
   835→              rounded="rounded-full"
   836→            />
   837→          ) : (
   838→            <GroupAvatar
   839→              seed={conv.id}
   840→              title={title}
   841→              avatarUrl={conv.avatarUrl}
   842→              size={44}
   843→              rounded="rounded-full"
   844→            />
   845→          )}
   846→          {isOnline && (
   847→            <span className="absolute -bottom-0.5 -right-0.5 w-3 h-3 bg-emerald-500 rounded-full border-2 border-card" />
   848→          )}
   849→        </div>
   850→        <div className="min-w-0 flex-1">
   851→          <div className="flex items-center justify-between gap-2">
   852→            <p
   853→              className={`text-sm truncate flex-1 flex items-center gap-1 ${
   854→                active
   855→                  ? 'font-semibold'
   856→                  : unread > 0
   857→                    ? 'font-semibold'
   858→                    : 'font-medium'
   859→              }`}
   860→            >
   861→              <span className="truncate">{title}</span>
   862→              {muted && (
   863→                <BellOff className="h-3 w-3 text-muted-foreground/60 flex-shrink-0" />
   864→              )}
   865→            </p>
   866→            <div className="flex items-center gap-1.5 flex-shrink-0">
   867→              {unread > 0 && (
   868→                <span
   869→ className="min-w-[18px] h-[18px] px-1 rounded-full gradient-emerald text-white text-[10px] font-bold flex items-center justify-center premium-shadow animate-[fadeIn_0.3s_ease-out]"
   870→ >
   871→                  {unread > 99 ? '99+' : unread}
   872→                </span>
   873→              )}
   874→              <span className="text-[10px] text-muted-foreground/60 flex-shrink-0">
   875→                {previewTime ? formatRelativeTime(previewTime) : ''}
   876→              </span>
   877→            </div>
   878→          </div>
   879→          <p
   880→            className={`text-[11px] truncate mt-0.5 ${
   881→              typingOthers.length > 0
   882→                ? 'text-primary font-medium'
   883→                : 'text-muted-foreground'
   884→            }`}
   885→          >
   886→            {truncate(previewText, 48)}
   887→          </p>
   888→        </div>
   889→      </div>
   890→    </button>
   891→  );
   892→}

   898→function QuickReactButton({ onReact }: { onReact: (emoji: string) => void }) {
   899→  const [open, setOpen] = useState(false);
   900→  const ref = useRef<HTMLDivElement>(null);
   901→
   902→  useEffect(() => {
   903→    if (!open) return;
   904→    const handler = (e: MouseEvent) => {
   905→      if (ref.current && !ref.current.contains(e.target as Node)) {
   906→        setOpen(false);
   907→      }
   908→    };
   909→    document.addEventListener('mousedown', handler);
   910→    return () => document.removeEventListener('mousedown', handler);
   911→  }, [open]);
   912→
   913→  return (
   914→    <div ref={ref} className="relative">
   915→      <button
   916→        onClick={() => setOpen((o) => !o)}
   917→        className="inline-flex items-center justify-center w-6 h-6 rounded-full bg-muted/60 hover:bg-muted text-muted-foreground transition-colors"
   918→        title="Add reaction"
   919→      >
   920→        <Plus className="h-3 w-3" />
   921→      </button>
   922→      {open && (
   923→          <div
   924→ className="absolute z-50 bottom-8 right-0 bg-card border border-border rounded-full shadow-lg p-1 flex items-center gap-0.5 animate-[fadeIn_0.3s_ease-out]"
   925→ >
   926→            {QUICK_REACT_EMOJIS.map((e) => (
   927→              <button
   928→                key={e}
   929→                onClick={() => {
   930→                  onReact(e);
   931→                  setOpen(false);
   932→                }}
   933→                className="w-7 h-7 rounded-full hover:bg-muted flex items-center justify-center text-sm transition-transform hover:scale-125"
   934→              >
   935→                {e}
   936→              </button>
   937→            ))}
   938→          </div>
   939→        )}
   940→</div>
   941→  );

   944→function MessageReactions({
   945→  reactions,
   946→  currentUserId,
   947→  onReact,
   948→}: {
   949→  reactions: MessageReactionItem[];
   950→  currentUserId: string;
   951→  onReact: (emoji: string) => void;
   952→}) {
   953→  const grouped = useMemo(() => {
   954→    const map = new Map<
   955→      string,
   956→      { count: number; mine: boolean; users: string[] }
   957→    >();
   958→    for (const r of reactions) {
   959→      const entry = map.get(r.emoji) || { count: 0, mine: false, users: [] };
   960→      entry.count += 1;
   961→      if (r.userId === currentUserId) entry.mine = true;
   962→      entry.users.push(r.user?.profile?.fullName || 'User');
   963→      map.set(r.emoji, entry);
   964→    }
   965→    return Array.from(map.entries());
   966→  }, [reactions, currentUserId]);
   967→
   968→  return (
   969→    <div className="flex flex-wrap items-center gap-1 mt-1">
   970→      {grouped.map(([emoji, info]) => (
   971→        <button
   972→          key={emoji}
   973→          onClick={() => onReact(emoji)}
   974→          className={`inline-flex items-center gap-1 px-1.5 h-6 rounded-full text-[11px] font-medium transition-all hover:scale-105 ${
   975→            info.mine
   976→              ? 'gradient-emerald text-white premium-shadow'
   977→              : 'bg-muted text-foreground hover:bg-muted/70'
   978→          }`}
   979→          title={info.users.join(', ')}
   980→        >
   981→          <span>{emoji}</span>
   982→          <span>{info.count}</span>
   983→        </button>
   984→      ))}
   985→      <QuickReactButton onReact={onReact} />
   986→    </div>
   987→  );
   988→}
