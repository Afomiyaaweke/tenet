   200→   200→function formatAIResultToHTML(data: Record<string, unknown>): string {
   201→   201→  let html = '';
   202→   202→  for (const [key, value] of Object.entries(data)) {
   203→   203→    const title = key.replace(/([A-Z])/g, ' $1').replace(/^./, s => s.toUpperCase());
   204→   204→    const content = typeof value === 'string' ? value : JSON.stringify(value, null, 2);
   205→   205→    const formattedContent = content
   206→   206→      .replace(/&/g, '&amp;')
   207→   207→      .replace(/</g, '&lt;')
   208→   208→      .replace(/>/g, '&gt;')
   209→   209→      .replace(/\*\*(.+?)\*\*/g, '<strong>$1</strong>')
   210→   210→      .replace(/^[-*•]\s+(.*)/gm, '<li>$1</li>')
   211→   211→      .replace(/^(\d+)[.)]\s+(.*)/gm, '<li>$2</li>')
   212→   212→      .replace(/^#{1,3}\s+(.*)/gm, '<strong>$1</strong>')
   213→   213→      .replace(/\n\n/g, '</p><p>')
   214→   214→      .replace(/\n/g, '<br/>');
   215→   215→    html += `<h2 style="font-size:16px;font-weight:700;margin:16px 0 8px;color:#059669;border-bottom:1px solid #d1fae5;padding-bottom:4px;">${title}</h2>`;
   216→   216→    html += `<p>${formattedContent}</p>`;
   217→   217→  }
   218→   218→  return html;
   219→   219→}

   241→   241→function SkillTagSelector({ selected, onChange }: { selected: string[]; onChange: (s: string[]) => void }) {
   242→   242→  const toggle = (skill: string) => {
   243→   243→    if (selected.includes(skill)) onChange(selected.filter(s => s !== skill));
   244→   244→    else onChange([...selected, skill]);
   245→   245→  };
   246→   246→  return (
   247→   247→    <div className="flex flex-wrap gap-1">
   248→   248→      {COMMON_SKILLS.map(skill => (
   249→   249→        <button key={skill} type="button" onClick={() => toggle(skill)}
   250→   250→          className={`px-2 py-0.5 rounded-full text-[10px] font-medium transition-all border ${
   251→   251→            selected.includes(skill)
   252→   252→              ? 'bg-emerald-50 text-emerald-700 border-emerald-300'
   253→   253→              : 'bg-muted/50 text-muted-foreground border-border hover:border-emerald-300'
   254→   254→          }`}>
   255→   255→          {skill}
   256→   256→        </button>
   257→   257→      ))}
   258→   258→    </div>
   259→   259→  );
   260→   260→}
