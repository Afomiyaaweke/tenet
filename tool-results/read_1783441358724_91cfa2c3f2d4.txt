   107→function formatETB(amount: number): string {
   108→  if (amount >= 1_000_000) return `ETB ${(amount / 1_000_000).toFixed(1)}M`;
   109→  if (amount >= 1_000) return `ETB ${(amount / 1_000).toFixed(0)}K`;
   110→  return `ETB ${amount.toLocaleString()}`;
   111→}
   112→
   113→function timeAgo(dateStr: string): string {
   114→  const diff = Date.now() - new Date(dateStr).getTime();
   115→  const mins = Math.floor(diff / 60000);
   116→  if (mins < 1) return 'just now';
   117→  if (mins < 60) return `${mins}m ago`;
   118→  const hrs = Math.floor(mins / 60);
   119→  if (hrs < 24) return `${hrs}h ago`;
   120→  const days = Math.floor(hrs / 24);
   121→  return `${days}d ago`;
   122→}
   123→
   124→function daysUntil(dateStr: string): number {
   125→  return Math.ceil((new Date(dateStr).getTime() - Date.now()) / 86400000);
   126→}

   128→// ─── Sparkline SVG Component ────────────────────────────────────────
   129→function SparklineBars({ data, color = '#10b981', height = 32 }: { data: number[]; color?: string; height?: number }) {
   130→  if (!data.length) return null;
   131→  const max = Math.max(...data, 1);
   132→  const barW = 4;
   133→  const gap = 3;
   134→  const totalW = data.length * (barW + gap) - gap;
   135→
   136→  return (
   137→    <svg width={totalW} height={height} className="flex-shrink-0">
   138→      {data.map((v, i) => {
   139→        const barH = Math.max((v / max) * (height - 4), 3);
   140→        return (
   141→          <rect
   142→            key={i}
   143→            x={i * (barW + gap)}
   144→            y={height - barH}
   145→            width={barW}
   146→            height={barH}
   147→            rx={2}
   148→            fill={color}
   149→            opacity={0.35 + (i / data.length) * 0.45}
   150→          />
   151→        );
   152→      })}
   153→    </svg>
   154→  );
   155→}
